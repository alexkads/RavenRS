1//! Charts embedded in a document (ODF 1.3 §10.4.3 `draw:object` whose sub-document is an
2//! `office:chart`, Part 3 §11): read into WordCraft's editable chart model when the chart is one
3//! WordCraft can edit as it is (one plot type in two dimensions, its data in the chart's own
4//! table). Anything else returns `None`, and the frame keeps its replacement picture.
5//!
6//! The data comes from the chart's `table:table` (§11.1): the series' and categories' cell range
7//! addresses (§19.583, `local-table.$B$2:.$B$5`) point into it, counting its header rows and
8//! columns like any other.
9
10use std::collections::HashMap;
11
12use crate::doc::chart::{ChartSeries, ChartSpec, ChartType, LegendPos, MAX_CELLS, MAX_POINTS, MAX_SERIES, MAX_TEXT};
13
14use super::xml::Attrs;
15use super::{Ev, events, get};

17/// Most XML events read from a chart sub-document.
18const MAX_EVENTS: usize = 2_000_000;
19/// Most rows read from a chart's table: a header row and a row per category.
20const MAX_ROWS: usize = MAX_POINTS + 1;
21/// Most columns read from it: a label column and two per series (a scatter series' x values).
22const MAX_COLS: usize = MAX_SERIES * 2 + 1;
23
24/// One cell of the chart's table: its text, and its number when it holds one (`Some(None)`: a
25/// number cell without a usable value, LibreOffice's `NaN` gap).
26#[derive(Clone, Default)]
27struct DataCell {
28    text: String,
29    num: Option<Option<f64>>,
30}
31
32/// A `chart:series` element as read.
33#[derive(Default)]
34struct RawSeries {
35    values: Option<String>,
36    label: Option<String>,
37    class: Option<String>,
38    style: Option<String>,
39    domain: Option<String>,
40    secondary: bool,
41}
42
43#[derive(Default)]
44struct Reader {
45    /// Chart styles' `style:chart-properties`, by style name.
46    props: HashMap<String, Attrs>,
47    style: Option<String>,
48    class: Option<String>,
49    plot_style: Option<String>,
50    title: Option<String>,
51    in_title: bool,
52    /// The legend's position, when there is a legend.
53    legend: Option<String>,
54    categories: Option<String>,
55    series: Vec<RawSeries>,
56    in_series: bool,
57    /// The chart's table: its rows of cells, and how deep in tables the reader is.
58    rows: Vec<Vec<DataCell>>,
59    table_depth: usize,
60    /// How many times the open row and cell repeat.
61    row_repeat: usize,
62    cell_repeat: usize,
63    cell: Option<DataCell>,
64    cells: usize,
65    in_chart: bool,
66}
67
68/// A count attribute (`table:number-rows-repeated`…): 1 when absent or unusable, at most `max`.
69fn repeat(a: &Attrs, k: &str, max: usize) -> usize {
70    get(a, k).and_then(|v| v.trim().parse::<usize>().ok()).unwrap_or(1).clamp(1, max.max(1))
71}
72
73impl Reader {
74    fn start(&mut self, q: &str, a: &Attrs) {
75        match q {
76            "style:style" => self.style = get(a, "style:name").map(str::to_string),
77            "style:chart-properties" => {
78                if let Some(name) = self.style.clone() {
79                    self.props.insert(name, a.clone());
80                }
81            }
82            "chart:chart" if !self.in_chart => {
83                self.in_chart = true;
84                self.class = get(a, "chart:class").map(str::to_string);
85            }
86            "chart:title" => self.in_title = true,
87            "text:p" if self.in_title => {
88                let t = self.title.get_or_insert_with(String::new);
89                if !t.is_empty() {
90                    t.push(' ');
91                }
92            }
93            "chart:legend" => self.legend = Some(get(a, "chart:legend-position").unwrap_or("end").to_string()),
94            "chart:plot-area" => self.plot_style = get(a, "chart:style-name").map(str::to_string),
95            "chart:categories" if self.categories.is_none() => self.categories = get(a, "table:cell-range-address").map(str::to_string),
96            "chart:series" => {
97                self.in_series = true;
98                if self.series.len() < MAX_SERIES {
99                    self.series.push(RawSeries {
100                        values: get(a, "chart:values-cell-range-address").map(str::to_string),
101                        label: get(a, "chart:label-cell-address").map(str::to_string),
102                        class: get(a, "chart:class").map(str::to_string),
103                        style: get(a, "chart:style-name").map(str::to_string),
104                        domain: None,
105                        secondary: get(a, "chart:attached-axis").is_some_and(|v| v.starts_with("secondary")),
106                    });
107                }
108            }
109            "chart:domain" if self.in_series => {
110                if let Some(s) = self.series.last_mut().filter(|s| s.domain.is_none()) {
111                    s.domain = get(a, "table:cell-range-address").map(str::to_string);
112                }
113            }
114            "table:table" => self.table_depth += 1,
115            "table:table-row" if self.table_depth == 1 => {
116                self.row_repeat = repeat(a, "table:number-rows-repeated", MAX_ROWS);
117                if self.rows.len() < MAX_ROWS {
118                    self.rows.push(Vec::new());
119                }
120            }
121            "table:table-cell" | "table:covered-table-cell" if self.table_depth == 1 => {
122                self.cell_repeat = repeat(a, "table:number-columns-repeated", MAX_COLS);
123                let num = (get(a, "office:value-type") == Some("float") || get(a, "office:value-type") == Some("percentage"))
124                    .then(|| get(a, "office:value").and_then(|v| v.trim().parse::<f64>().ok()).filter(|v| v.is_finite()));
125                self.cell = Some(DataCell { text: String::new(), num });
126            }
127            _ => {}
128        }
129    }
130
131    fn end(&mut self, q: &str) {
132        match q {
133            "style:style" => self.style = None,
134            "chart:title" => self.in_title = false,
135            "chart:series" => self.in_series = false,
136            "table:table" => self.table_depth = self.table_depth.saturating_sub(1),
137            "table:table-cell" | "table:covered-table-cell" if self.table_depth == 1 => {
138                if let Some(cell) = self.cell.take()
139                    && let Some(row) = self.rows.last_mut()
140                {
141                    let n = self.cell_repeat.min(MAX_COLS.saturating_sub(row.len())).min(MAX_CELLS.saturating_sub(self.cells));
142                    self.cells += n;
143                    row.extend(std::iter::repeat_n(cell, n));
144                }
145            }
146            "table:table-row" if self.table_depth == 1 => {
147                // A repeated row: its copies, within the table's and the chart's limits.
148                if let Some(row) = self.rows.last().cloned() {
149                    let room = MAX_ROWS.saturating_sub(self.rows.len());
150                    let by_cells = MAX_CELLS.saturating_sub(self.cells).checked_div(row.len()).unwrap_or(room);
151                    let n = self.row_repeat.saturating_sub(1).min(room).min(by_cells);
152                    self.cells += n.saturating_mul(row.len());
153                    self.rows.extend(std::iter::repeat_n(row, n));
154                }
155                self.row_repeat = 1;
156            }
157            _ => {}
158        }
159    }
160
161    fn text(&mut self, t: &str) {
162        if self.in_title
163            && let Some(title) = &mut self.title
164            && title.len() < MAX_TEXT * 4
165        {
166            title.push_str(t);
167        } else if let Some(cell) = &mut self.cell
168            && cell.text.len() < MAX_TEXT * 4
169        {
170            cell.text.push_str(t);
171        }
172    }
173
174    /// The chart properties of style `name`, attribute `k`.
175    fn prop(&self, name: Option<&str>, k: &str) -> Option<&str> {
176        name.and_then(|n| self.props.get(n)).and_then(|a| get(a, k))
177    }
178
179    fn cell(&self, (col, row): (usize, usize)) -> Option<&DataCell> {
180        self.rows.get(row).and_then(|r| r.get(col))
181    }
182
183    /// The cells a range address covers, in order, at most [`MAX_POINTS`].
184    fn range(&self, addr: &str) -> Vec<Option<&DataCell>> {
185        cells_of(addr).into_iter().map(|at| self.cell(at)).collect()
186    }
187
188    fn chart(&self) -> Option<ChartSpec> {
189        let class = self.class.as_deref().map(|c| c.trim_start_matches("chart:"))?;
190        let plot = self.plot_style.as_deref();
191        let on = |k: &str| self.prop(plot, k) == Some("true");
192        if on("chart:three-dimensional") || on("chart:percentage") || on("chart:deep") {
193            return None;
194        }
195        let stacked = on("chart:stacked");
196        let kind = match class {
197            "bar" => match (on("chart:vertical"), stacked) {
198                (false, false) => ChartType::Column,
199                (false, true) => ChartType::ColumnStacked,
200                (true, false) => ChartType::Bar,
201                (true, true) => ChartType::BarStacked,
202            },
203            "line" if !stacked => ChartType::Line,
204            "area" if !stacked => ChartType::Area,
205            "circle" => ChartType::Pie,
206            "ring" => ChartType::Doughnut,
207            "scatter" => ChartType::Scatter,
208            _ => return None,
209        };
210        // One plot type on one value axis: a series of another type, or on a second axis, is
211        // a combination chart WordCraft can't edit.
212        let own = |c: &str| {
213            let c = c.trim_start_matches("chart:");
214            c == class || (kind == ChartType::Doughnut && c == "circle")
215        };
216        if self.series.is_empty() || self.series.iter().any(|s| s.secondary || !s.class.as_deref().is_none_or(own)) {
217            return None;
218        }
219        let label = |c: Option<&DataCell>| c.map(|c| c.text.trim().to_string()).unwrap_or_default();
220        let series = self
221            .series
222            .iter()
223            .enumerate()
224            .map(|(i, s)| {
225                // A series without its own range takes the table's next column.
226                let values = match &s.values {
227                    Some(addr) => self.range(addr).into_iter().map(|c| c.and_then(value)).collect(),
228                    None => self.rows.iter().skip(1).take(MAX_POINTS).map(|r| r.get(i + 1).and_then(value)).collect(),
229                };
230                let name = s.label.as_deref().and_then(|a| cells_of(a).first().copied()).map(|at| label(self.cell(at))).unwrap_or_default();
231                let name = if name.is_empty() { format!("Series {}", i + 1) } else { name };
232                ChartSeries { name, values }
233            })
234            .collect();
235        // A scatter chart's x values: its first series' domain.
236        let domain = self.series.first().and_then(|s| s.domain.as_deref()).filter(|_| kind == ChartType::Scatter);
237        let categories = match domain.or(self.categories.as_deref()) {
238            Some(addr) => self.range(addr).into_iter().map(label).collect(),
239            None => Vec::new(),
240        };
241        let data_labels = std::iter::once(plot)
242            .chain(self.series.iter().map(|s| s.style.as_deref()))
243            .any(|st| matches!(self.prop(st, "chart:data-label-number"), Some("value" | "percentage" | "value-and-percentage")));
244        let legend = self.legend.as_deref().map(|p| match p {
245            "start" => LegendPos::Left,
246            p if p.starts_with("top") => LegendPos::Top,
247            p if p.starts_with("bottom") => LegendPos::Bottom,
248            _ => LegendPos::Right,
249        });
250        let title = self.title.as_deref().map(str::trim).filter(|t| !t.is_empty()).map(str::to_string);
251        Some(ChartSpec { kind, categories, series, title, legend, data_labels }.sanitized())
252    }
253}

255/// A cell's number: its value, or its text read as a number.
256fn value(c: &DataCell) -> Option<f64> {
257    match c.num {
258        Some(v) => v,
259        None => c.text.trim().parse::<f64>().ok().filter(|v| v.is_finite()),
260    }
261}

263/// A cell address (`$B$2`, `local-table.B2`, `'my table'.$B$2`): its column and row, from 0.
264fn cell_at(s: &str) -> Option<(usize, usize)> {
265    // The table name, when there is one, ends at the last dot.
266    let s = s.rsplit('.').next().unwrap_or(s).trim().trim_start_matches('$');
267    let letters = s.bytes().take_while(u8::is_ascii_alphabetic).count();
268    if letters == 0 || letters > 3 {
269        return None;
270    }
271    let (col, rest) = s.split_at_checked(letters)?;
272    let col = col.bytes().try_fold(0usize, |n, b| n.checked_mul(26)?.checked_add(usize::from(b.to_ascii_uppercase() - b'A') + 1))?;
273    let row: usize = rest.trim_start_matches('$').parse().ok()?;
274    Some((col.checked_sub(1)?, row.checked_sub(1)?))
275}

277/// The cells of a range address's first range (`local-table.$B$2:.$B$5`), in reading order: down
278/// a column, or along a row (a block: down its first column), at most [`MAX_POINTS`].
279fn cells_of(addr: &str) -> Vec<(usize, usize)> {
280    let first = addr.split_whitespace().next().unwrap_or_default();
281    let mut ends = first.splitn(2, ':');
282    let Some(a) = ends.next().and_then(cell_at) else { return Vec::new() };
283    let b = match ends.next() {
284        Some(e) => match cell_at(e) {
285            Some(b) => b,
286            None => return Vec::new(),
287        },
288        None => a,
289    };
290    let (c0, c1) = (a.0.min(b.0), a.0.max(b.0));
291    let (r0, r1) = (a.1.min(b.1), a.1.max(b.1));
292    if r0 == r1 && c0 != c1 { (c0..=c1).take(MAX_POINTS).map(|c| (c, r0)).collect() } else { (r0..=r1).take(MAX_POINTS).map(|r| (c0, r)).collect() }
293}

295/// The chart a chart sub-document's `content.xml` holds, when WordCraft can edit it.
296pub(in crate::formats::odt) fn read_chart(xml: &[u8]) -> Option<ChartSpec> {
297    let mut r = Reader::default();
298    let mut count = 0usize;
299    let mut done = false;
300    events(xml, |ev| {
301        count += 1;
302        if done || count > MAX_EVENTS {
303            return;
304        }
305        match ev {
306            Ev::Start(q, a) => r.start(&q, &a),
307            Ev::End(q) => {
308                if q == "chart:chart" && r.in_chart {
309                    done = true;
310                }
311                r.end(&q);
312            }
313            Ev::Text(t) => r.text(&t),
314        }
315    });
316    // A sub-document cut off before its chart ended is no chart.
317    if !done {
318        return None;
319    }
320    r.chart()
321}
