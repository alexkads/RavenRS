1//! Date and time styles (ODF 1.3 §16.27 `number:date-style`, `number:time-style`) and Word's
2//! date-time pictures (`DATE \@ "dddd, MMMM d, yyyy"`), both ways: the parts — day, month,
3//! year, weekday, hours, minutes, seconds, AM/PM — and the text between them.
4
5use super::writer::x;

7/// Longest picture kept, characters.
8const MAX_PICTURE: usize = 200;
9
10/// A part of a date or time.
11#[derive(Clone, Debug, PartialEq)]
12pub(super) enum Part {
13    /// Day of the month (long: two digits).
14    Day(bool),
15    /// Month: long, and by name.
16    Month(bool, bool),
17    /// Year (long: four digits).
18    Year(bool),
19    /// Day of the week by name (long: in full).
20    Weekday(bool),
21    Hours(bool),
22    Minutes(bool),
23    Seconds(bool),
24    AmPm,
25    Text(String),
26}
27
28/// The part element `name` (local, in the number namespace) stands for; its text, if any,
29/// comes after.
30pub(super) fn part(name: &str, long: bool, textual: bool) -> Option<Part> {
31    Some(match name {
32        "day" => Part::Day(long),
33        "month" => Part::Month(long, textual),
34        "year" => Part::Year(long),
35        "day-of-week" => Part::Weekday(long),
36        "hours" => Part::Hours(long),
37        "minutes" => Part::Minutes(long),
38        "seconds" => Part::Seconds(long),
39        "am-pm" => Part::AmPm,
40        _ => return None,
41    })
42}

44/// The Word picture for `parts`: 12-hour hours when an AM/PM shows, text in single quotes when
45/// it holds letters (a quote doubled); quotes and backslashes, which would end the field's
46/// argument, left out.
47pub(super) fn picture(parts: &[Part]) -> String {
48    let ampm = parts.contains(&Part::AmPm);
49    let pick = |long: bool, a: &'static str, b: &'static str| if long { a } else { b };
50    let mut s = String::new();
51    for p in parts {
52        match p {
53            Part::Day(l) => s.push_str(pick(*l, "dd", "d")),
54            Part::Month(l, true) => s.push_str(pick(*l, "MMMM", "MMM")),
55            Part::Month(l, false) => s.push_str(pick(*l, "MM", "M")),
56            Part::Year(l) => s.push_str(pick(*l, "yyyy", "yy")),
57            Part::Weekday(l) => s.push_str(pick(*l, "dddd", "ddd")),
58            Part::Hours(l) if ampm => s.push_str(pick(*l, "hh", "h")),
59            Part::Hours(l) => s.push_str(pick(*l, "HH", "H")),
60            Part::Minutes(l) => s.push_str(pick(*l, "mm", "m")),
61            Part::Seconds(l) => s.push_str(pick(*l, "ss", "s")),
62            Part::AmPm => s.push_str("AM/PM"),
63            Part::Text(t) => {
64                let t: String = t.chars().filter(|c| !matches!(c, '"' | '\\')).collect();
65                // Letters are quoted; an apostrophe is two, inside quoted text or out of it.
66                if t.chars().any(char::is_alphabetic) {
67                    s.push('\'');
68                    s.push_str(&t.replace('\'', "''"));
69                    s.push('\'');
70                } else {
71                    s.push_str(&t.replace('\'', "''"));
72                }
73            }
74        }
75    }
76    s.chars().take(MAX_PICTURE).collect()
77}

79/// The parts of Word picture `pic` (the reverse of [`picture`]).
80fn parts_of(pic: &str) -> Vec<Part> {
81    let chars: Vec<char> = pic.chars().take(MAX_PICTURE).collect();
82    let mut out: Vec<Part> = Vec::new();
83    let mut text = String::new();
84    let mut i = 0;
85    let flush = |text: &mut String, out: &mut Vec<Part>| {
86        if !text.is_empty() {
87            out.push(Part::Text(std::mem::take(text)));
88        }
89    };
90    while let Some(&c) = chars.get(i) {
91        // Two quotes outside quoted text are an apostrophe, as inside it.
92        if c == '\'' && chars.get(i + 1) == Some(&'\'') {
93            text.push('\'');
94            i += 2;
95            continue;
96        }
97        if c == '\'' {
98            i += 1;
99            while let Some(&q) = chars.get(i) {
100                i += 1;
101                if q != '\'' {
102                    text.push(q);
103                } else if chars.get(i) == Some(&'\'') {
104                    text.push('\'');
105                    i += 1;
106                } else {
107                    break;
108                }
109            }
110            continue;
111        }
112        if chars.get(i..i + 5).is_some_and(|w| w.iter().collect::<String>().eq_ignore_ascii_case("am/pm")) {
113            flush(&mut text, &mut out);
114            out.push(Part::AmPm);
115            i += 5;
116            continue;
117        }
118        let mut n = 1;
119        while chars.get(i + n) == Some(&c) {
120            n += 1;
121        }
122        let p = match (c, n) {
123            ('y' | 'Y', 4..) => Some(Part::Year(true)),
124            ('y' | 'Y', _) => Some(Part::Year(false)),
125            ('M', 4..) => Some(Part::Month(true, true)),
126            ('M', 3) => Some(Part::Month(false, true)),
127            ('M', 2) => Some(Part::Month(true, false)),
128            ('M', _) => Some(Part::Month(false, false)),
129            ('d' | 'D', 4..) => Some(Part::Weekday(true)),
130            ('d' | 'D', 3) => Some(Part::Weekday(false)),
131            ('d' | 'D', 2) => Some(Part::Day(true)),
132            ('d' | 'D', _) => Some(Part::Day(false)),
133            ('h' | 'H', n) => Some(Part::Hours(n >= 2)),
134            ('m', n) => Some(Part::Minutes(n >= 2)),
135            ('s', n) => Some(Part::Seconds(n >= 2)),
136            _ => None,
137        };
138        match p {
139            Some(p) => {
140                flush(&mut text, &mut out);
141                out.push(p);
142            }
143            None => {
144                for _ in 0..n {
145                    text.push(c);
146                }
147            }
148        }
149        i += n;
150    }
151    flush(&mut text, &mut out);
152    out
153}

155/// A data style named `name` for Word picture `pic`: a date style when it shows any part of a
156/// date, a time style otherwise. `None` for a picture with no part.
157pub(super) fn style_xml(name: &str, pic: &str) -> Option<String> {
158    let parts = parts_of(pic);
159    if !parts.iter().any(|p| !matches!(p, Part::Text(_))) {
160        return None;
161    }
162    let dated = parts.iter().any(|p| matches!(p, Part::Day(_) | Part::Month(..) | Part::Year(_) | Part::Weekday(_)));
163    let long = |l: bool| if l { " number:style=\"long\"" } else { "" };
164    let body: String = parts
165        .iter()
166        .map(|p| match p {
167            Part::Day(l) => format!("<number:day{}/>", long(*l)),
168            Part::Month(l, t) => format!("<number:month{}{}/>", long(*l), if *t { " number:textual=\"true\"" } else { "" }),
169            Part::Year(l) => format!("<number:year{}/>", long(*l)),
170            Part::Weekday(l) => format!("<number:day-of-week{}/>", long(*l)),
171            Part::Hours(l) => format!("<number:hours{}/>", long(*l)),
172            Part::Minutes(l) => format!("<number:minutes{}/>", long(*l)),
173            Part::Seconds(l) => format!("<number:seconds{}/>", long(*l)),
174            Part::AmPm => "<number:am-pm/>".to_string(),
175            Part::Text(t) => format!("<number:text>{}</number:text>", x(t)),
176        })
177        .collect();
178    let el = if dated { "number:date-style" } else { "number:time-style" };
179    Some(format!("<{el} style:name=\"{name}\">{body}</{el}>"))
180}

182/// The `\@` picture of Word field code `instr`, if it has one.
183pub(super) fn picture_of(instr: &str) -> Option<String> {
184    let at = instr.find("\\@")?;
185    let rest = instr.get(at + 2..)?.trim_start();
186    let pic = match rest.strip_prefix('"') {
187        Some(q) => q.split('"').next()?,
188        None => rest.split_whitespace().next()?,
189    };
190    (!pic.is_empty()).then(|| pic.chars().take(MAX_PICTURE).collect())
191}

193/// A date's parts in the order and with the separators the locale gives them, for a date style
194/// with `number:automatic-order` (§19.341): its weekday, month, day and year as US English
195/// orders them (LibreOffice's "November 2, 2026", or "11/2/2026" with a numeric month). Its own
196/// text between them is left out; a time's parts stay as they are.
197pub(super) fn auto_ordered(parts: &[Part]) -> Vec<Part> {
198    let find = |f: fn(&Part) -> bool| parts.iter().find(|p| f(p)).cloned();
199    let weekday = find(|p| matches!(p, Part::Weekday(_)));
200    let month = find(|p| matches!(p, Part::Month(..)));
201    let day = find(|p| matches!(p, Part::Day(_)));
202    let year = find(|p| matches!(p, Part::Year(_)));
203    let Some(month) = month else { return parts.to_vec() };
204    let textual = matches!(month, Part::Month(_, true));
205    let text = |s: &str| Part::Text(s.to_string());
206    let mut out = Vec::new();
207    if let Some(w) = weekday {
208        out.extend([w, text(", ")]);
209    }
210    out.push(month);
211    if let Some(d) = day {
212        out.extend([text(if textual { " " } else { "/" }), d]);
213    }
214    if let Some(y) = year {
215        out.extend([text(if textual { ", " } else { "/" }), y]);
216    }
217    // A time with it keeps its own parts, after.
218    let time: Vec<Part> = parts.iter().filter(|p| matches!(p, Part::Hours(_) | Part::Minutes(_) | Part::Seconds(_) | Part::AmPm)).cloned().collect();
219    if !time.is_empty() {
220        out.push(text(" "));
221        for (i, p) in time.into_iter().enumerate() {
222            if i > 0 && !matches!(p, Part::AmPm) {
223                out.push(text(":"));
224            } else if matches!(p, Part::AmPm) {
225                out.push(text(" "));
226            }
227            out.push(p);
228        }
229    }
230    out
231}

233/// English month names, in order; a month shown is its name or the start of it (three letters
234/// or more).
235const MONTHS: [&str; 12] = ["january", "february", "march", "april", "may", "june", "july", "august", "september", "october", "november", "december"];
236
237/// The value a date or time shown as `shown` under Word picture `pic` stands for, as a fixed
238/// field's `text:date-value` / `text:time-value` (§19.871, §19.880) state it: `YYYY-MM-DD`, with
239/// `Thh:mm:ss` when it shows a time (on 1899-12-30, ODF's day 0, when it shows no date). `None`
240/// when the text doesn't read under the picture (month names in another language, say).
241pub(super) fn value_shown(pic: &str, shown: &str) -> Option<String> {
242    let parts = parts_of(pic);
243    let s: Vec<char> = shown.trim().chars().take(MAX_PICTURE).collect();
244    let mut i = 0usize;
245    let (mut day, mut month, mut year) = (None, None, None);
246    let (mut hour, mut minute, mut second, mut pm) = (None, None, 0u32, None);
247    // A run of digits (at most `max`) or of letters at `i`.
248    let digits = |i: &mut usize, max: usize| -> Option<u32> {
249        let start = *i;
250        while *i - start < max && s.get(*i).is_some_and(char::is_ascii_digit) {
251            *i += 1;
252        }
253        s.get(start..*i).filter(|d| !d.is_empty())?.iter().collect::<String>().parse().ok()
254    };
255    let word = |i: &mut usize| -> String {
256        let start = *i;
257        while s.get(*i).is_some_and(|c| c.is_alphabetic()) {
258            *i += 1;
259        }
260        s.get(start..*i).map(|w| w.iter().collect::<String>().to_lowercase()).unwrap_or_default()
261    };
262    for p in &parts {
263        match p {
264            Part::Text(t) => {
265                for c in t.chars() {
266                    match s.get(i) {
267                        Some(&got) if got.eq_ignore_ascii_case(&c) => i += 1,
268                        // Spaces are loose: one shown where the picture has none, or none where it has one.
269                        Some(&got) if got.is_whitespace() => i += 1,
270                        _ if c.is_whitespace() => {}
271                        _ => return None,
272                    }
273                }
274            }
275            Part::Day(_) => day = Some(digits(&mut i, 2)?),
276            Part::Month(_, false) => month = Some(digits(&mut i, 2)?),
277            Part::Month(_, true) => {
278                let w = word(&mut i);
279                let n = MONTHS.iter().position(|m| w.len() >= 3 && m.starts_with(w.as_str()))?;
280                month = Some(u32::try_from(n).ok()? + 1);
281            }
282            Part::Year(_) => {
283                let start = i;
284                let y = digits(&mut i, 4)?;
285                // Two digits: Word's window, 1930–2029.
286                year = Some(if i - start <= 2 { if y < 30 { 2000 + y } else { 1900 + y } } else { y });
287            }
288            Part::Weekday(_) => {
289                if word(&mut i).is_empty() {
290                    return None;
291                }
292            }
293            Part::Hours(_) => hour = Some(digits(&mut i, 2)?),
294            Part::Minutes(_) => minute = Some(digits(&mut i, 2)?),
295            Part::Seconds(_) => second = digits(&mut i, 2)?,
296            Part::AmPm => {
297                let w = word(&mut i);
298                pm = Some(match w.as_str() {
299                    "am" | "a" => false,
300                    "pm" | "p" => true,
301                    _ => return None,
302                });
303            }
304        }
305    }
306    if i < s.len() {
307        return None;
308    }
309    let time = match (hour, minute) {
310        (Some(h), m) => {
311            let h = match pm {
312                Some(true) if h < 12 => h + 12,
313                Some(false) if h == 12 => 0,
314                _ => h,
315            };
316            let m = m.unwrap_or(0);
317            (h < 24 && m < 60 && second < 60).then(|| format!("T{h:02}:{m:02}:{second:02}"))?
318        }
319        (None, Some(_)) => return None,
320        (None, None) => String::new(),
321    };
322    let date = match (year, month, day) {
323        (Some(y), Some(m), Some(d)) if (1..=12).contains(&m) && (1..=31).contains(&d) && y <= 9999 => format!("{y:04}-{m:02}-{d:02}"),
324        (None, None, None) if !time.is_empty() => "1899-12-30".to_string(),
325        _ => return None,
326    };
327    Some(format!("{date}{time}"))
328}

330#[cfg(test)]
331mod value_tests {
332    use super::value_shown;
333
334    #[test]
335    fn a_shown_date_or_time_reads_back_under_its_picture() {
336        assert_eq!(value_shown("dddd, MMMM dd, yyyy", "Sunday, November 01, 2026").as_deref(), Some("2026-11-01"));
337        assert_eq!(value_shown("dd/MM/yy", "03/11/26").as_deref(), Some("2026-11-03"));
338        assert_eq!(value_shown("hh:mm AM/PM", "04:04 PM").as_deref(), Some("1899-12-30T16:04:00"));
339        assert_eq!(value_shown("d MMM yyyy H:mm:ss", "5 Mar 1999 7:08:09").as_deref(), Some("1999-03-05T07:08:09"));
340        // Not readable: another language's month, extra text, an impossible day, junk.
341        assert_eq!(value_shown("d MMMM yyyy", "5 mars 2026"), None);
342        assert_eq!(value_shown("d MMMM yyyy", "5 March 2026 and more"), None);
343        assert_eq!(value_shown("dd/MM/yyyy", "45/11/2026"), None);
344        assert_eq!(value_shown("dd/MM/yyyy", ""), None);
345        assert_eq!(value_shown("", "anything"), None);
346    }
347
348    /// An apostrophe in a date style's text is two quotes in the picture, quoted or not, and
349    /// reads back as one.
350    #[test]
351    fn an_apostrophe_in_a_date_styles_text_round_trips() {
352        use super::{Part, parts_of, picture};
353        for text in ["'", "''", "o'clock", "'tis", " ' "] {
354            let parts = vec![Part::Text(text.to_string())];
355            let pic = picture(&parts);
356            assert_eq!(parts_of(&pic), parts, "{text:?} as {pic:?}");
357        }
358        assert_eq!(picture(&[Part::Text("'".into())]), "''");
359    }
