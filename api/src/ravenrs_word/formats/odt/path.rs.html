1//! Geometry of polygons, polylines, paths and regular polygons (ODF 1.3 §10.3.4–10.3.7, §19.1xx
2//! `draw:points`, `svg:d`, `svg:viewBox`): their points, in their view box's coordinates, into
3//! a WordCraft freeform (straight segments: curves and arcs are followed by short steps).
4
5use crate::doc::freeform::{FreePath, Freeform, MAX_PATHS, MAX_POINTS};
6
7/// Steps a curve or arc is followed by.
8const CURVE_STEPS: usize = 16;
9/// Most numbers read from one `svg:d` or `draw:points` (hostile files).
10const MAX_NUMBERS: usize = 200_000;
11/// Largest coordinate kept.
12const MAX_COORD: f64 = 1.0e6;
13
14/// A view box (`svg:viewBox`, §19.570: min x, min y, width, height).
15pub(super) fn view_box(v: Option<&str>) -> Option<[f32; 4]> {
16    let n: Vec<f32> = numbers(v?).take(4).collect();
17    let [x, y, w, h] = n.try_into().ok()?;
18    Some([x, y, w, h])
19}

21/// The numbers in `s`, separated by spaces or commas (an SVG number list), at most
22/// [`MAX_NUMBERS`]; what doesn't read as a number ends it.
23fn numbers(s: &str) -> impl Iterator<Item = f32> + '_ {
24    s.split(|c: char| c.is_whitespace() || c == ',')
25        .filter(|t| !t.is_empty())
26        .take(MAX_NUMBERS)
27        .map_while(|t| t.parse::<f32>().ok())
28        .map(|v| if v.is_finite() { v.clamp(-MAX_COORD as f32, MAX_COORD as f32) } else { 0.0 })
29}

31/// A freeform from `paths` (view box coordinates) in view box `vb` (none, or an empty one: the
32/// box around the points): points from its corner.
33fn freeform(paths: Vec<FreePath>, vb: Option<[f32; 4]>) -> Option<Freeform> {
34    let [x0, y0, w, h] = match vb {
35        Some(vb @ [_, _, w, h]) if w > 0.0 && h > 0.0 => vb,
36        _ => {
37            let all = || paths.iter().flat_map(|p| p.pts.iter());
38            let (x0, y0) = all().fold((f32::MAX, f32::MAX), |(x, y), p| (x.min(p[0]), y.min(p[1])));
39            let (x1, y1) = all().fold((f32::MIN, f32::MIN), |(x, y), p| (x.max(p[0]), y.max(p[1])));
40            if x0 > x1 { [0.0, 0.0, 0.0, 0.0] } else { [x0, y0, x1 - x0, y1 - y0] }
41        }
42    };
43    let paths: Vec<FreePath> = paths
44        .into_iter()
45        .take(MAX_PATHS)
46        .map(|p| FreePath { pts: p.pts.into_iter().take(MAX_POINTS).map(|[x, y]| [x - x0, y - y0]).collect(), closed: p.closed })
47        .filter(|p| !p.pts.is_empty())
48        .collect();
49    (!paths.is_empty()).then(|| Freeform { w: w.max(0.0), h: h.max(0.0), paths, ..Default::default() }.sanitized())
50}

52/// A polygon's or polyline's points (`draw:points`: "x,y x,y …"), closed or not.
53pub(super) fn points(v: &str, closed: bool, vb: Option<[f32; 4]>) -> Option<Freeform> {
54    let n: Vec<f32> = numbers(v).collect();
55    let pts = n.as_chunks::<2>().0.iter().map(|c| [c[0], c[1]]).take(MAX_POINTS).collect();
56    freeform(vec![FreePath { pts, closed }], vb)
57}

59/// A regular polygon (§10.3.6) of `corners` corners in a `w` × `h` box, its first corner at the
60/// top middle; concave (a star), its inner corners `sharpness` (0–1) of the way in.
61pub(super) fn regular(corners: u32, concave: bool, sharpness: f32, w: f32, h: f32) -> Option<Freeform> {
62    let n = corners.clamp(3, 1000) as usize;
63    let (rx, ry) = (f64::from(w) / 2.0, f64::from(h) / 2.0);
64    let inner = 1.0 - f64::from(if sharpness.is_finite() { sharpness.clamp(0.0, 1.0) } else { 0.0 });
65    let step = std::f64::consts::TAU / n as f64;
66    let mut pts = Vec::with_capacity(n * 2);
67    for i in 0..n {
68        let a = -std::f64::consts::FRAC_PI_2 + step * i as f64;
69        pts.push([(rx + rx * a.cos()) as f32, (ry + ry * a.sin()) as f32]);
70        if concave {
71            let b = a + step / 2.0;
72            pts.push([(rx + rx * inner * b.cos()) as f32, (ry + ry * inner * b.sin()) as f32]);
73        }
74    }
75    freeform(vec![FreePath { pts, closed: true }], Some([0.0, 0.0, w, h]))
76}

78/// A path's SVG path data (`svg:d`, §19.523: moves, lines, cubic and quadratic curves and
79/// elliptical arcs, absolute and relative), each subpath a freeform path.
80pub(super) fn path(d: &str, vb: Option<[f32; 4]>) -> Option<Freeform> {
81    let mut p = PathReader::default();
82    let mut tokens = Tokens { s: d, at: 0, read: 0, peeked: None };
83    let mut cmd = None;
84    while let Some(t) = tokens.next() {
85        let c = match t {
86            Token::Cmd(c) => c,
87            // More numbers repeat the command (a move's are lines).
88            Token::Num(_) => match cmd {
89                Some('M') => 'L',
90                Some('m') => 'l',
91                // A close takes no numbers: a number after one ends the data (it would be read
92                // again for ever otherwise).
93                Some('Z' | 'z') | None => break,
94                Some(c) => c,
95            },
96        };
97        if let Token::Num(_) = t {
98            tokens.peeked = Some(t);
99        }
100        cmd = Some(c);
101        if !p.command(c, &mut tokens) {
102            break;
103        }
104    }
105    p.end(false);
106    freeform(p.paths, vb)
107}

109#[derive(Clone, Copy)]
110enum Token {
111    Cmd(char),
112    Num(f64),
113}
114
115/// The commands and numbers of path data.
116struct Tokens<'a> {
117    s: &'a str,
118    at: usize,
119    read: usize,
120    /// A number read as the start of a repeated command, to be read again as its first.
121    peeked: Option<Token>,
122}
123
124impl Tokens<'_> {
125    fn next(&mut self) -> Option<Token> {
126        if let Some(t) = self.peeked.take() {
127            return Some(t);
128        }
129        self.read += 1;
130        if self.read > MAX_NUMBERS {
131            return None;
132        }
133        let rest = self.s.get(self.at..)?;
134        let skip = rest.len() - rest.trim_start_matches(|c: char| c.is_whitespace() || c == ',').len();
135        self.at += skip;
136        let rest = self.s.get(self.at..)?;
137        let c = rest.chars().next()?;
138        if c.is_ascii_alphabetic() && c != 'e' && c != 'E' {
139            self.at += c.len_utf8();
140            return Some(Token::Cmd(c));
141        }
142        // A number: sign, digits, a point, an exponent; a second point or sign starts the next.
143        let b = rest.as_bytes();
144        let mut end = 0;
145        let (mut dot, mut exp) = (false, false);
146        while let Some(&ch) = b.get(end) {
147            let ok = match ch {
148                b'0'..=b'9' => true,
149                b'+' | b'-' => end == 0 || matches!(b.get(end - 1), Some(b'e' | b'E')),
150                b'.' if !dot && !exp => {
151                    dot = true;
152                    true
153                }
154                b'e' | b'E' if !exp && end > 0 => {
155                    exp = true;
156                    true
157                }
158                _ => false,
159            };
160            if !ok {
161                break;
162            }
163            end += 1;
164        }
165        if end == 0 {
166            // Not a number: stop.
167            self.at = self.s.len();
168            return None;
169        }
170        let v = rest.get(..end).and_then(|n| n.parse::<f64>().ok()).filter(|v| v.is_finite()).unwrap_or(0.0);
171        self.at += end;
172        Some(Token::Num(v.clamp(-MAX_COORD, MAX_COORD)))
173    }
174
175    fn num(&mut self) -> Option<f64> {
176        match self.next() {
177            Some(Token::Num(v)) => Some(v),
178            Some(Token::Cmd(_)) => {
179                // A command where a number belongs: the data is malformed; stop.
180                self.at = self.s.len();
181                None
182            }
183            None => None,
184        }
185    }
186}

188#[derive(Default)]
189struct PathReader {
190    paths: Vec<FreePath>,
191    cur: Vec<[f32; 2]>,
192    at: (f64, f64),
193    start: (f64, f64),
194    /// The last curve's second control point, for a smooth curve's reflection.
195    last_ctrl: Option<(f64, f64, bool)>,
196}
197
198impl PathReader {
199    fn push(&mut self, p: (f64, f64)) {
200        if self.cur.len() < MAX_POINTS {
201            let c = |v: f64| v.clamp(-MAX_COORD, MAX_COORD) as f32;
202            self.cur.push([c(p.0), c(p.1)]);
203        }
204        self.at = p;
205    }
206
207    /// A drawing command without a move before it starts its subpath where the pen is.
208    fn begin(&mut self) {
209        if self.cur.is_empty() {
210            self.push(self.at);
211        }
212    }
213
214    fn end(&mut self, closed: bool) {
215        let pts = std::mem::take(&mut self.cur);
216        if !pts.is_empty() && self.paths.len() < MAX_PATHS {
217            self.paths.push(FreePath { pts, closed });
218        }
219    }
220
221    /// Command `c` with its numbers from `t`; false when the data runs out or is malformed.
222    fn command(&mut self, c: char, t: &mut Tokens<'_>) -> bool {
223        let rel = c.is_ascii_lowercase();
224        let (ox, oy) = if rel { self.at } else { (0.0, 0.0) };
225        let pt = |t: &mut Tokens<'_>| Some((t.num()? + ox, t.num()? + oy));
226        let mut quad = false;
227        match c.to_ascii_uppercase() {
228            'M' => {
229                let Some(p) = pt(t) else { return false };
230                self.end(false);
231                self.start = p;
232                self.push(p);
233            }
234            'L' => {
235                let Some(p) = pt(t) else { return false };
236                self.begin();
237                self.push(p);
238            }
239            'H' => {
240                let Some(x) = t.num() else { return false };
241                let x = if rel { x + self.at.0 } else { x };
242                self.begin();
243                self.push((x, self.at.1));
244            }
245            'V' => {
246                let Some(y) = t.num() else { return false };
247                let y = if rel { y + self.at.1 } else { y };
248                self.begin();
249                self.push((self.at.0, y));
250            }
251            'C' | 'S' => {
252                let c1 = if c.eq_ignore_ascii_case(&'S') { self.reflected(false) } else { pt(t) };
253                let (Some(c1), Some(c2), Some(e)) = (c1, pt(t), pt(t)) else { return false };
254                self.cubic(c1, c2, e);
255                self.last_ctrl = Some((c2.0, c2.1, false));
256                return true;
257            }
258            'Q' | 'T' => {
259                let q = if c.eq_ignore_ascii_case(&'T') { self.reflected(true) } else { pt(t) };
260                let (Some(q), Some(e)) = (q, pt(t)) else { return false };
261                // A quadratic curve is the cubic with controls two thirds of the way to its own.
262                let s = self.at;
263                let c1 = (s.0 + 2.0 / 3.0 * (q.0 - s.0), s.1 + 2.0 / 3.0 * (q.1 - s.1));
264                let c2 = (e.0 + 2.0 / 3.0 * (q.0 - e.0), e.1 + 2.0 / 3.0 * (q.1 - e.1));
265                self.cubic(c1, c2, e);
266                self.last_ctrl = Some((q.0, q.1, true));
267                quad = true;
268            }
269            'A' => {
270                let (Some(rx), Some(ry), Some(rot), Some(large), Some(sweep), Some(e)) = (t.num(), t.num(), t.num(), t.num(), t.num(), pt(t)) else {
271                    return false;
272                };
273                self.arc((rx, ry), rot, large != 0.0, sweep != 0.0, e);
274            }
275            'Z' => {
276                self.end(true);
277                // A new subpath after it starts where this one did, unless it moves.
278                self.at = self.start;
279            }
280            _ => return false,
281        }
282        if !quad {
283            self.last_ctrl = None;
284        }
285        true
286    }
287
288    /// The first control point of a smooth curve: the last one's reflected (`quad`: of a
289    /// quadratic curve), or the current point.
290    fn reflected(&self, quad: bool) -> Option<(f64, f64)> {
291        Some(match self.last_ctrl {
292            Some((x, y, q)) if q == quad => (2.0 * self.at.0 - x, 2.0 * self.at.1 - y),
293            _ => self.at,
294        })
295    }
296
297    fn cubic(&mut self, c1: (f64, f64), c2: (f64, f64), e: (f64, f64)) {
298        let s = self.at;
299        self.begin();
300        for i in 1..CURVE_STEPS {
301            let t = i as f64 / CURVE_STEPS as f64;
302            let u = 1.0 - t;
303            let (a, b, c, d) = (u * u * u, 3.0 * u * u * t, 3.0 * u * t * t, t * t * t);
304            self.push((a * s.0 + b * c1.0 + c * c2.0 + d * e.0, a * s.1 + b * c1.1 + c * c2.1 + d * e.1));
305        }
306        self.push(e);
307    }
308
309    /// An elliptical arc to `e` (SVG's endpoint form, turned into its centre form).
310    fn arc(&mut self, r: (f64, f64), rot_deg: f64, large: bool, sweep: bool, e: (f64, f64)) {
311        let s = self.at;
312        self.begin();
313        let (mut rx, mut ry) = (r.0.abs(), r.1.abs());
314        if rx < 1e-9 || ry < 1e-9 || (s.0 - e.0).abs() + (s.1 - e.1).abs() < 1e-9 {
315            self.push(e);
316            return;
317        }
318        let (sin, cos) = (rot_deg % 360.0).to_radians().sin_cos();
319        // The start relative to the chord's middle, in the ellipse's own axes.
320        let (dx, dy) = ((s.0 - e.0) / 2.0, (s.1 - e.1) / 2.0);
321        let (x1, y1) = (cos * dx + sin * dy, -sin * dx + cos * dy);
322        // Radii too small to reach are scaled up just enough.
323        let k = (x1 * x1) / (rx * rx) + (y1 * y1) / (ry * ry);
324        if k > 1.0 {
325            (rx, ry) = (rx * k.sqrt(), ry * k.sqrt());
326        }
327        let num = (rx * rx * ry * ry - rx * rx * y1 * y1 - ry * ry * x1 * x1).max(0.0);
328        let den = rx * rx * y1 * y1 + ry * ry * x1 * x1;
329        let mut f = if den > 0.0 { (num / den).sqrt() } else { 0.0 };
330        if large == sweep {
331            f = -f;
332        }
333        let (cx1, cy1) = (f * rx * y1 / ry, -f * ry * x1 / rx);
334        let (cx, cy) = (cos * cx1 - sin * cy1 + (s.0 + e.0) / 2.0, sin * cx1 + cos * cy1 + (s.1 + e.1) / 2.0);
335        let angle = |ux: f64, uy: f64| uy.atan2(ux);
336        let a0 = angle((x1 - cx1) / rx, (y1 - cy1) / ry);
337        let a1 = angle((-x1 - cx1) / rx, (-y1 - cy1) / ry);
338        let mut da = a1 - a0;
339        if sweep && da < 0.0 {
340            da += std::f64::consts::TAU;
341        } else if !sweep && da > 0.0 {
342            da -= std::f64::consts::TAU;
343        }
344        if !da.is_finite() || !cx.is_finite() || !cy.is_finite() {
345            self.push(e);
346            return;
347        }
348        for i in 1..CURVE_STEPS {
349            let a = a0 + da * i as f64 / CURVE_STEPS as f64;
350            let (px, py) = (rx * a.cos(), ry * a.sin());
351            self.push((cos * px - sin * py + cx, sin * px + cos * py + cy));
352        }
353        self.push(e);
354    }
355}

357/// `freeform`'s paths as SVG path data in a view box `vw` × `vh` (whole numbers, as ODF's view
358/// box is): each path moved to, lined through, and closed when it is.
359pub(super) fn path_data(f: &Freeform, vw: u32, vh: u32) -> String {
360    path_text(f, vw, vh, false)
361}

363/// Whether `freeform` has both closed and open paths: drawn filled and unfilled.
364pub(super) fn is_mixed(f: &Freeform) -> bool {
365    f.paths.iter().any(|p| p.closed) && f.paths.iter().any(|p| !p.closed)
366}

368/// `freeform`'s paths as a custom shape's path (§19.145 `draw:enhanced-path`) in a view box
369/// `vw` × `vh`: each path ended (`N`), open ones unfilled (`F`).
370pub(super) fn enhanced_path_data(f: &Freeform, vw: u32, vh: u32) -> String {
371    path_text(f, vw, vh, true)
372}
373
374fn path_text(f: &Freeform, vw: u32, vh: u32, enhanced: bool) -> String {
375    let sx = if f.w > 0.0 { vw as f32 / f.w } else { 1.0 };
376    let sy = if f.h > 0.0 { vh as f32 / f.h } else { 1.0 };
377    let n = |v: f32| {
378        let v = if v.is_finite() { v.clamp(-MAX_COORD as f32, MAX_COORD as f32) } else { 0.0 };
379        format!("{}", v.round() as i64)
380    };
381    let mut d = String::new();
382    for p in f.paths.iter().take(MAX_PATHS) {
383        for (i, [x, y]) in p.pts.iter().take(MAX_POINTS).enumerate() {
384            d.push_str(if i == 0 { "M" } else { " L" });
385            if enhanced {
386                d.push(' ');
387            }
388            d.push_str(&format!("{} {}", n(x * sx), n(y * sy)));
389        }
390        if p.closed {
391            d.push_str(" Z");
392        }
393        if enhanced {
394            d.push_str(if p.closed { " N" } else { " F N" });
395        }
396        d.push(' ');
397    }
398    d.trim_end().to_string()
399}

404    /// hostile or malformed `svg:d`).
405    #[test]
406    fn a_number_after_a_close_ends_the_path() {
407        let f = super::path("M 0 0 L 10 0 L 10 10 Z 5 5", Some([0.0, 0.0, 10.0, 10.0]));
408        let paths = f.map(|f| f.paths).unwrap_or_default();
409        assert_eq!(paths.len(), 1, "{paths:?}");

411        assert_eq!(paths[0].pts.len(), 3);

413        assert!(super::path("m 0 0 l 10 0 l 0 10 z 1", None).is_some());
