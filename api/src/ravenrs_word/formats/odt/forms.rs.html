1//! Form controls (ODF 1.3 §13: `office:forms` at the start of the text, each control placed by
2//! a `draw:control` in a paragraph) read as WordCraft content controls and written back:
3//! text fields (`form:text`, `form:textarea`) as plain text controls, check boxes and radio
4//! buttons as check boxes (a radio button with round symbols), list boxes as drop-down lists,
5//! combo boxes as combo boxes, date fields as date controls. A control's name is its tag, a
6//! check box's label its title. Buttons, images and other controls aren't read.
7

15/// Most controls read, list entries in one, and characters of a value.
16const MAX_CONTROLS: usize = 10_000;
17const MAX_ITEMS: usize = 1000;
18const MAX_VALUE: usize = 10_000;

20/// A radio button's symbols: a selected and an empty circle.
21pub(in crate::formats::odt) const RADIO_ON: char = '\u{25C9}';
22pub(in crate::formats::odt) const RADIO_OFF: char = '\u{25CB}';

24/// Form controls WordCraft has no kind for, kept verbatim (see [`ContentControl::form_xml`]):
25/// every control of ODF 1.3 §13.5 but those it reads (text fields and areas, check boxes, option
26/// buttons, list and combo boxes, date fields) and hidden ones (kept apart).
27const KEPT: [&str; 12] = [
28    "form:button",
29    "form:image",
30    "form:fixed-text",
31    "form:file",
32    "form:value-range",
33    "form:frame",
34    "form:image-frame",
35    "form:generic-control",
36    "form:password",
37    "form:number",
38    "form:time",
39    "form:grid",
40];
41/// Most bytes of a kept control's XML.
42const MAX_KEPT: usize = 64 * 1024;
43/// The namespace prefixes a kept control may use: those the content WordCraft writes declares.
44const PREFIXES: [&str; 19] = [
45    "office", "style", "text", "table", "draw", "fo", "xlink", "dc", "meta", "svg", "number", "form", "ooo", "ooow", "oooc", "script", "dom",
46    "loext", "xml",
47];

49/// The `text:placeholder-type` (§19.851) of a placeholder: what it stands for.
50pub(in crate::formats::odt) fn placeholder_type(t: Option<&str>) -> &'static str {
51    match t.map(str::trim) {
52        Some("table") => "table",
53        Some("text-box") => "text-box",
54        Some("image") => "image",
55        Some("object") => "object",
56        _ => "text",
57    }
58}

78/// `office:forms`, read as it streams past.
79#[derive(Default)]
80pub(in crate::formats::odt) struct FormsReader {
81    /// How deep inside `office:forms` the reader is (0: outside).
82    depth: usize,
83    /// The control being read: its id, control, selected entries.
84    open: Option<(String, ContentControl, Vec<String>)>,
85    /// The controls read, by `form:id` (or `xml:id`).
86    pub(in crate::formats::odt) controls: HashMap<String, Read>,
87    /// Hidden controls (§13.5.15 `form:hidden`: a name and a value the form sends, not shown),
88    /// kept verbatim without their ids.
89    pub(in crate::formats::odt) hidden: Vec<String>,
90    /// A control WordCraft has no kind for being kept: its id, name, label, its XML so far (its
91    /// id left out) and how many of its elements are open.
92    kept: Option<Kept>,
93}
94
95struct Kept {
96    id: String,
97    name: String,
98    label: String,
99    xml: String,
100    open: usize,
101    /// It uses a prefix the content WordCraft writes doesn't declare: not kept.
102    foreign: bool,
103}
104
105/// Whether name `q` has a prefix the content WordCraft writes declares.
106fn known(q: &str) -> bool {
107    q.split_once(':').is_some_and(|(p, _)| PREFIXES.contains(&p))
108}
109
110/// Whether element `q` with attributes `a` uses only declared prefixes.
111fn declared(q: &str, a: &Attrs) -> bool {
112    known(q) && a.iter().all(|(k, _)| known(k))
113}
114
115/// Element `q` with attributes `a` as XML, its ids (`form:id`, `xml:id`) left out.
116fn element_xml(q: &str, a: &Attrs) -> String {
117    let mut s = format!("<{q}");
118    for (k, v) in a.iter().filter(|(k, _)| k != "form:id" && k != "xml:id") {
119        s.push_str(&format!(" {k}=\"{}\"", x(v)));
120    }
121    s.push('>');
122    s
123}

125impl FormsReader {
126    /// Element `q` starts: whether it is part of `office:forms` (and so read here).
127    pub(in crate::formats::odt) fn start(&mut self, q: &str, a: &Attrs) -> bool {
128        if self.depth == 0 {
129            if q != "office:forms" {
130                return false;
131            }
132            self.depth = 1;
133            return true;
134        }
135        self.depth += 1;
136        if let Some(k) = &mut self.kept {
137            if k.xml.len() < MAX_KEPT {
138                k.xml.push_str(&element_xml(q, a));
139            }
140            k.foreign |= !declared(q, a);
141            k.open += 1;
142            return true;
143        }
144        let id = get(a, "form:id").or_else(|| get(a, "xml:id")).unwrap_or_default().to_string();
145        // A hidden control: kept as it is, its properties with it (its id is given anew).
146        if q == "form:hidden" && self.hidden.len() < MAX_CONTROLS {
147            self.kept = Some(Kept {
148                id: String::new(),
149                name: String::new(),
150                label: String::new(),
151                xml: element_xml(q, a),
152                open: 1,
153                foreign: !declared(q, a),
154            });
155            return true;
156        }
157        if KEPT.contains(&q) && !id.is_empty() && self.controls.len() < MAX_CONTROLS {
158            let name = get(a, "form:name").map(capped).unwrap_or_default();
159            let label = get(a, "form:label").map(capped).filter(|l| !l.is_empty()).unwrap_or_else(|| name.clone());
160            self.kept = Some(Kept { id, name, label, xml: element_xml(q, a), open: 1, foreign: !declared(q, a) });
161            return true;
162        }
163        let name = get(a, "form:name").map(capped).unwrap_or_default();
164        let value = || get(a, "form:current-value").or_else(|| get(a, "form:value")).map(capped).unwrap_or_default();
165        let kind = match q {
166            "form:text" | "form:formatted-text" => Some((ControlKind::Text { multi_line: false }, value())),
167            "form:textarea" => Some((ControlKind::Text { multi_line: true }, value())),
168            "form:checkbox" | "form:radio" => {
169                let radio = q == "form:radio";
170                let on = if radio {
171                    get(a, "form:current-selected").or_else(|| get(a, "form:selected")) == Some("true")
172                } else {
173                    get(a, "form:current-state").or_else(|| get(a, "form:state")) == Some("checked")
174                };
175                let (checked_char, unchecked_char) =
176                    if radio { (RADIO_ON, RADIO_OFF) } else { (crate::doc::control::CHECKED, crate::doc::control::UNCHECKED) };
177                let symbol = if on { checked_char } else { unchecked_char };
178                let kind =
179                    ControlKind::CheckBox { checked: on, checked_char, checked_font: String::new(), unchecked_char, unchecked_font: String::new() };
180                Some((kind, symbol.to_string()))
181            }
182            "form:listbox" => Some((ControlKind::DropDown { items: Vec::new(), last_value: String::new() }, String::new())),
183            "form:combobox" => Some((ControlKind::ComboBox { items: Vec::new(), last_value: String::new() }, value())),
184            "form:date" => {
185                let v = value();
186                // An xsd:date (`2026-11-01`).
187                let full_date =
188                    if v.len() >= 10 && v.is_char_boundary(10) { format!("{}T00:00:00Z", v.get(..10).unwrap_or_default()) } else { String::new() };
189                Some((
190                    ControlKind::Date {
191                        full_date,
192                        format: "yyyy-MM-dd".into(),
193                        lid: String::new(),
194                        calendar: String::new(),
195                        store_as: String::new(),
196                    },
197                    v,
198                ))
199            }
200            "form:option" | "form:item" => {
201                if let Some((_, c, selected)) = &mut self.open
202                    && let ControlKind::DropDown { items, .. } | ControlKind::ComboBox { items, .. } = &mut c.kind
203                    && items.len() < MAX_ITEMS
204                {
205                    let display = get(a, "form:label").map(capped).unwrap_or_default();
206                    let value = get(a, "form:value").map(capped).filter(|v| *v != display).unwrap_or_default();
207                    if get(a, "form:current-selected").or_else(|| get(a, "form:selected")) == Some("true") {
208                        selected.push(display.clone());
209                    }
210                    items.push(ListItem { display, value });
211                }
212                None
213            }
214            _ => None,
215        };
216        if let Some((kind, text)) = kind
217            && !id.is_empty()
218            && self.controls.len() < MAX_CONTROLS
219        {
220            let mut c = ContentControl::new(kind);
221            c.tag = name;
222            c.title = get(a, "form:label").map(capped).unwrap_or_default();
223            self.open = Some((id, c, vec![text]));
224        }
225        true
226    }
227
228    /// Element `q` ends: whether it was part of `office:forms`.
229    pub(in crate::formats::odt) fn end(&mut self, q: &str) -> bool {
230        if self.depth == 0 {
231            return false;
232        }
233        self.depth -= 1;
234        if let Some(k) = &mut self.kept {
235            k.xml.push_str(&format!("</{q}>"));
236            k.open = k.open.saturating_sub(1);
237            if k.open == 0
238                && let Some(k) = self.kept.take()
239            {
240                // Too long to keep whole, or naming what WordCraft can't write: not kept.
241                if k.xml.len() <= MAX_KEPT && !k.foreign && k.id.is_empty() {
242                    self.hidden.push(k.xml);
243                } else if k.xml.len() <= MAX_KEPT && !k.foreign {
244                    let mut c = ContentControl::new(ControlKind::Text { multi_line: false });
245                    c.tag = k.name;
246                    c.lock = crate::doc::control::ControlLock::ContentLocked;
247                    c.form_xml = k.xml;
248                    self.controls.insert(k.id, (c, k.label));
249                }
250            }
251            return true;
252        }
253        if matches!(
254            q,
255            "form:text" | "form:formatted-text" | "form:textarea" | "form:checkbox" | "form:radio" | "form:listbox" | "form:combobox" | "form:date"
256        ) && let Some((id, mut c, texts)) = self.open.take()
257        {
258            // The text: a list box's selected entry, else the value read with the control.
259            let mut text = match &mut c.kind {
260                ControlKind::DropDown { last_value, .. } => {
261                    let sel = texts.get(1).cloned().unwrap_or_default();
262                    last_value.clone_from(&sel);
263                    sel
264                }
265                ControlKind::ComboBox { last_value, .. } => {
266                    let v = texts.first().cloned().unwrap_or_default();
267                    last_value.clone_from(&v);
268                    v
269                }
270                _ => texts.first().cloned().unwrap_or_default(),
271            };
272            // An empty text control or list shows its placeholder.
273            if text.is_empty() && !matches!(c.kind, ControlKind::CheckBox { .. }) {
274                c.showing_placeholder = true;
275                text = c.placeholder_shown().to_string();
276            }
277            self.controls.insert(id, (c, text));
278        }
279        true
280    }
281}

283/// The text of a control's selected value, unless it shows its placeholder.
284fn value(c: &ContentControl, text: &str) -> String {
285    if c.showing_placeholder { String::new() } else { text.to_string() }

288/// A content control as a form control in `office:forms`, with id `id`: the element, and the
289/// size of its `draw:control` (points). `None` for kinds a form has no control for.
290pub(in crate::formats::odt) fn form_xml(c: &ContentControl, text: &str, id: &str, n: usize) -> Option<(String, f32, f32)> {
291    let size = |(xml, w, h): (String, f32, f32)| {
292        let fit = |v: f32| if v.is_finite() { v.clamp(1.0, 1584.0) } else { 16.0 };
293        match c.form_size {
294            Some([fw, fh]) => (xml, fit(fw), fit(fh)),
295            None => (xml, w, h),
296        }
297    };
298    // A control kept verbatim, with its new id, while it is still the control it was read as
299    // (made another kind, it is written as that).
300    if !c.form_xml.is_empty() && c.kind == (ControlKind::Text { multi_line: false }) && !is_placeholder(c) {
301        let end = c.form_xml.find(|ch: char| ch.is_whitespace() || ch == '>')?;
302        let (head, rest) = (c.form_xml.get(..end)?, c.form_xml.get(end..)?);
303        return Some(size((format!("{head} xml:id=\"{id}\" form:id=\"{id}\"{rest}"), 72.0, 20.0)));
304    }
305    let name = if c.tag.is_empty() { format!("Control {n}") } else { c.tag.clone() };
306    let common = format!(" form:name=\"{}\" xml:id=\"{id}\" form:id=\"{id}\"", x(&name));
307    let label = if c.title.is_empty() { String::new() } else { format!(" form:label=\"{}\"", x(&c.title)) };
308    let v = value(c, text);
309    // Wide enough for its text (at least a short word), one line high.
310    let width = |s: &str| (s.chars().count().clamp(8, 60) as f32 * 6.0 + 16.0).min(432.0);
311    // No `form:control-implementation`: the element says what the control is.
312    let impl_ = |_: &str| "";
313    let out = match &c.kind {
314        ControlKind::Text { multi_line: false } => {
315            (format!("<form:text{common}{}{} form:current-value=\"{}\"/>", impl_("TextField"), label, x(&v)), width(&v), 16.0)
316        }
317        ControlKind::Text { multi_line: true } => {
318            // Several lines (LibreOffice's `MultiLine` property): without it LibreOffice shows
319            // one line, centred in the box.
320            (
321                format!(
322                    "<form:textarea{common}{}{} form:current-value=\"{}\"><form:properties><form:property form:property-name=\"MultiLine\" office:value-type=\"boolean\" office:boolean-value=\"true\"/></form:properties></form:textarea>",
323                    impl_("TextField"),
324                    label,
325                    x(&v)
326                ),
327                216.0,
328                48.0,
329            )
330        }
331        ControlKind::CheckBox { checked, checked_char, .. } if *checked_char == RADIO_ON => (
332            format!("<form:radio{common}{}{label} form:current-selected=\"{checked}\" form:selected=\"{checked}\"/>", impl_("RadioButton")),
333            width(&c.title).min(18.0 + c.title.chars().count() as f32 * 6.0),
334            16.0,
335        ),
336        ControlKind::CheckBox { checked, .. } => {
337            let state = if *checked { "checked" } else { "unchecked" };
338            (
339                format!("<form:checkbox{common}{}{label} form:current-state=\"{state}\" form:state=\"{state}\"/>", impl_("CheckBox")),
340                18.0 + c.title.chars().count() as f32 * 6.0,
341                16.0,
342            )
343        }
344        ControlKind::DropDown { items, last_value } => {
345            let chosen = if last_value.is_empty() { v.as_str() } else { last_value.as_str() };
346            let options: String = items
347                .iter()
348                .map(|i| {
349                    let value = if i.value.is_empty() { String::new() } else { format!(" form:value=\"{}\"", x(&i.value)) };
350                    let sel = if !chosen.is_empty() && (i.display == chosen || i.value() == chosen) {
351                        " form:current-selected=\"true\" form:selected=\"true\""
352                    } else {
353                        ""
354                    };
355                    format!("<form:option form:label=\"{}\"{value}{sel}/>", x(&i.display))
356                })
357                .collect();
358            let widest = items.iter().map(|i| i.display.as_str()).max_by_key(|s| s.chars().count()).unwrap_or("");
359            (format!("<form:listbox{common}{} form:dropdown=\"true\">{options}</form:listbox>", impl_("ListBox")), width(widest) + 14.0, 16.0)
360        }
361        ControlKind::ComboBox { items, .. } => {
362            let entries: String = items.iter().map(|i| format!("<form:item form:label=\"{}\"/>", x(&i.display))).collect();
363            let widest = items.iter().map(|i| i.display.as_str()).chain([v.as_str()]).max_by_key(|s| s.chars().count()).unwrap_or("");
364            (
365                format!(
366                    "<form:combobox{common}{} form:dropdown=\"true\" form:current-value=\"{}\">{entries}</form:combobox>",
367                    impl_("ComboBox"),
368                    x(&v)
369                ),
370                width(widest) + 14.0,
371                16.0,
372            )
373        }
374        ControlKind::Date { full_date, .. } => {
375            let date = full_date
376                .get(..10)
377                .filter(|d| d.len() == 10 && d.as_bytes().get(4) == Some(&b'-') && d.as_bytes().get(7) == Some(&b'-'))
378                .unwrap_or("");
379            let current = if date.is_empty() || c.showing_placeholder { String::new() } else { format!(" form:current-value=\"{date}\"") };
380            (format!("<form:date{common}{}{current}/>", impl_("DateField")), 96.0, 16.0)
381        }
382        _ => return None,
383    };
384    Some(size(out))
385}

387/// The `office:forms` holding the controls `xml` (empty when there are none).
388pub(in crate::formats::odt) fn forms_xml(xml: &[String], hidden: &[String]) -> String {
389    if xml.is_empty() && hidden.is_empty() {
390        return String::new();
391    }
392    // Hidden controls, each with an id of its own.
393    let hidden: String = hidden
394        .iter()
395        .enumerate()
396        .filter_map(|(i, h)| {
397            let end = h.find(|ch: char| ch.is_whitespace() || ch == '>')?;
398            Some(format!("{} xml:id=\"hidden{i}\" form:id=\"hidden{i}\"{}", h.get(..end)?, h.get(end..)?))
399        })
400        .collect();
401    format!(
402        "<office:forms form:automatic-focus=\"false\" form:apply-design-mode=\"false\"><form:form form:name=\"Form\">{}{hidden}</form:form></office:forms>",
403        xml.concat()
404    )
405}
