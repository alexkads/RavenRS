1//! Markup kept from the file an ODT document was opened from (see `crate::doc::kept`): the
2//! definitions kept paragraphs refer to, read here, and written back with them.
3//!
4//! A kept paragraph names its automatic styles (`P12`, `T5`), and through them common styles,
5//! list styles, data styles and font faces. They are kept as read, by name: `auto:` for
6//! `content.xml`'s automatic styles, `common:` for `styles.xml`'s common styles, `font:` for
7//! font faces. On save, the ones a written paragraph needs (and what they need in turn) go back
8//! in; WordCraft's own automatic styles take another prefix, so names can't meet.
9

11use std::sync::Arc;

13use super::xml::{Ev, events_at, get};

15/// Most definitions kept from one part (hostile input).
16const MAX_DEFS: usize = 100_000;
17
18/// The definitions in `xml` (a `content.xml` or `styles.xml`), each by the key it is kept
19/// under: a child of `office:automatic-styles` as `auto:<name>` in `content.xml` (`content`
20/// true), `sauto:<name>` in `styles.xml`; of `office:styles` as `common:<name>`; of
21/// `office:font-face-decls` as `font:<name>`; of `office:master-styles` as `master:<name>`; the
22/// namespaces the root declares as `ns:<prefix>` (the URI); and in `content.xml`, the document's
23/// `office:scripts` as `top:office:scripts` and the declarations opening the text
24/// (`text:variable-decls`…) as `decls:<element>`.
25pub(super) fn defs(xml: &[u8], content: bool) -> Vec<(String, Arc<str>)> {
26    let mut out = Vec::new();
27    let mut path: Vec<String> = Vec::new();
28    let mut open: Option<(String, usize, usize)> = None;
29    let mut before_1_3 = false;
30    let mut prefixes = std::collections::BTreeSet::new();
31    events_at(xml, |ev, at| match ev {
32        Ev::Start(q, a) => {
33            path.push(q.clone());
34            let depth = path.len();
35            // The namespaces the root declares, which kept markup may use.
36            if depth == 1 {
37                // Older files' unitless gradient angles are tenths of a degree (see `to_1_3`).
38                before_1_3 = super::strokes::legacy_angles(get(&a, "office:version"));
39                for (k, v) in &a {
40                    if let Some(p) = k.strip_prefix("xmlns:").filter(|p| !p.is_empty() && p.len() <= 64)
41                        && out.len() < MAX_DEFS
42                    {
43                        prefixes.insert(p.to_string());
44                        out.push((format!("ns:{p}"), Arc::from(v.as_str())));
45                    }
46                }
47                return;
48            }
49            // Those a part's container declares too: what is kept of it leaves the container behind.
50            if depth == 2 {
51                for (k, v) in &a {
52                    if let Some(p) = k.strip_prefix("xmlns:").filter(|p| !p.is_empty() && p.len() <= 64)
53                        && out.len() < MAX_DEFS
54                        && prefixes.insert(p.to_string())
55                    {
56                        out.push((format!("ns:{p}"), Arc::from(v.as_str())));
57                    }
58                }
59            }
60            // A reference mark (§6.6.2), which `text:reference-ref` names: WordCraft reads it as a
61            // bookmark, and writes it as both.
62            if content
63                && matches!(q.as_str(), "text:reference-mark" | "text:reference-mark-start")
64                && let Some(n) = get(&a, "text:name").filter(|n| !n.is_empty() && n.len() <= 256)
65                && out.len() < MAX_DEFS
66            {
67                out.push((format!("refmark:{n}"), Arc::from("")));
68            }
69            if open.is_some() || out.len() >= MAX_DEFS {
70                return;
71            }
72            let parent = path.get(depth.wrapping_sub(2)).map(String::as_str).unwrap_or("");
73            let key = match (depth, parent) {
74                (2, _) if content && q == "office:scripts" => Some("top:office:scripts".to_string()),
75                // Whether the text holds soft page breaks (§19.857), which markup kept may.
76                (3, "office:body")
77                    if content
78                        && q == "office:text"
79                        && get(&a, "text:use-soft-page-breaks").is_some_and(|v| v.trim() == "true")
80                        && out.len() < MAX_DEFS =>
81                {
82                    out.push(("soft-page-breaks".to_string(), Arc::from("true")));
83                    None
84                }
85                (3, "office:automatic-styles") => child_key(if content { "auto" } else { "sauto" }, &q, &a),
86                (3, "office:styles") => child_key("common", &q, &a),
87                (3, "office:font-face-decls") => child_key("font", &q, &a),
88                (3, "office:master-styles") => child_key("master", &q, &a),
89                (4, "office:text") if content && q.starts_with("text:") && q.ends_with("-decls") => Some(format!("decls:{q}")),
90                (4, "office:text") if content && q == "office:forms" => Some("forms".to_string()),
91                // A tracked change (§5.5.2), by its id: a paragraph written as read names it.
92                (5, "text:tracked-changes") if content && q == "text:changed-region" => {
93                    get(&a, "text:id").or_else(|| get(&a, "xml:id")).filter(|i| !i.is_empty() && i.len() <= 256).map(|i| format!("change:{i}"))
94                }
95                _ => None,
96            };
97            if let Some(key) = key {
98                open = Some((key, at.0, depth));
99            }
100        }
101        Ev::End(_) => {
102            if let Some((_, from, d)) = &open
103                && *d == path.len()
104            {
105                let from = *from;
106                if let Some((key, ..)) = open.take()
107                    && let Some(s) = xml.get(from..at.1).and_then(|b| std::str::from_utf8(b).ok())
108                {
109                    let mut s = if before_1_3 { to_1_3(s) } else { s.to_string() };
110                    // ODF 1.3 names a changed region by `xml:id` (§5.5.2); an older file's by
111                    // `text:id` alone.
112                    if let Some(id) = key.strip_prefix("change:")
113                        && s.split('>').next().is_some_and(|h| !h.contains("xml:id="))
114                    {
115                        s.insert_str("<text:changed-region".len(), &format!(" xml:id=\"{}\"", super::writer::x(id)));
116                    }
117                    out.push((key, Arc::from(s)));
118                }
119            }
120            path.pop();
121        }
122        Ev::Text(_) => {}
123    });
124    out
125}

127/// Markup kept from a file older than ODF 1.3, written as ODF 1.3: a gradient's, transparency
128/// gradient's or hatch's angle without a unit is given its unit, as the file meant it — tenths
129/// of a degree (`300` → `30deg`). That reading deviates from the standard out of necessity
130/// (§18.3.1 says degrees): see `strokes::legacy_angles`.
131fn to_1_3(xml: &str) -> String {
132    if !(xml.contains("<draw:gradient") || xml.contains("<draw:opacity") || xml.contains("<draw:hatch")) {
133        return xml.to_string();
134    }
135    let mut out = String::with_capacity(xml.len() + 16);
136    let mut rest = xml;
137    while let Some(i) = rest.find(['<']) {
138        let (head, tail) = rest.split_at(i);
139        out.push_str(head);
140        let end = tail.find('>').map_or(tail.len(), |e| e + 1);
141        let (tag, after) = tail.split_at(end);
142        if tag.starts_with("<draw:gradient") || tag.starts_with("<draw:opacity") || tag.starts_with("<draw:hatch") {
143            let mut t = tag.to_string();
144            for attr in ["draw:angle=\"", "draw:rotation=\""] {
145                if let Some(a) = t.find(attr).map(|a| a + attr.len())
146                    && let Some(len) = t.get(a..).and_then(|v| v.find('"'))
147                    && let Some(v) = t.get(a..a + len).and_then(|v| v.trim().parse::<f64>().ok()).filter(|v| v.is_finite())
148                {
149                    t.replace_range(a..a + len, &format!("{}deg", v / 10.0));
150                }
151            }
152            out.push_str(&t);
153        } else {
154            out.push_str(tag);
155        }
156        rest = after;
157    }
158    out.push_str(rest);
159    out
160}

162/// The key a definition (element `q` with attributes `a`, a child of group `g`) is kept under:
163/// `<g>:<name>` (its `style:name`, or a drawing definition's `draw:name`); an unnamed one (a
164/// default style, the outline style, a note or line numbering configuration, a layer set) as
165/// `<g>~<element>~<its family or note class>`.
166fn child_key(g: &str, q: &str, a: &super::xml::Attrs) -> Option<String> {
167    // Drawing definitions (dashes, gradients, arrowheads…) are named by `draw:name`.
168    match get(a, "style:name").or_else(|| get(a, "draw:name")).filter(|n| !n.is_empty()) {
169        Some(name) => Some(format!("{g}:{name}{}", space(q, get(a, "style:family").unwrap_or("")))),
170        None => {
171            let which = get(a, "style:family").or_else(|| get(a, "text:note-class")).unwrap_or("");
172            Some(format!("{g}~{q}~{which}"))
173        }
174    }
175}

177/// The name space a definition (element `q`, of style family `family`) belongs to, as a key's
178/// suffix: names are unique only within one (§16.2: a style's within its family; a list style,
179/// a data style, a gradient may share a paragraph style's name, as a graphic style may:
180/// LibreOffice's "Marginalia" is both). Paragraph styles, font faces and master pages have none;
181/// another family's style is `|<family>`, a list style `|list`, a data style `|data`, another
182/// definition `|<element>`.
183fn space(q: &str, family: &str) -> String {
184    match q {
185        "style:style" if family.trim() == "paragraph" || family.is_empty() => String::new(),
186        "style:style" => format!("|{}", family.trim()),
187        "style:font-face" | "style:master-page" => String::new(),
188        "text:list-style" => "|list".to_string(),
189        q if q.starts_with("number:") => "|data".to_string(),
190        q => format!("|{}", q.split_once(':').map_or(q, |(_, l)| l)),
191    }
192}

194/// Whether `xml` (a part WordCraft wrote) already defines what `def` (a kept definition) does:
195/// an element of its kind by its name (a list style and a paragraph style may share one).
196pub(super) fn defined(xml: &str, def: &str) -> bool {
197    let head = def.split('>').next().unwrap_or("");
198    let el = head.trim_start_matches('<').split([' ', '/', '\t', '\n', '\r']).next().unwrap_or("");
199    let name = ["style:name", "draw:name"].into_iter().find_map(|a| Some((a, raw_attr(head, a)?)));
200    let (Some((attr, name)), false) = (name, el.is_empty()) else { return false };
201    let open = format!("<{el} ");
202    xml.match_indices(&open).any(|(i, _)| {
203        xml.get(i..).and_then(|t| t.split('>').next()).is_some_and(|tag| raw_attr(tag, attr).is_some_and(|n| attr_text(n) == attr_text(name)))
204    })
205}

212/// Attribute `name`'s value as `tag` (a start tag) writes it, spaced and quoted however XML allows.
213pub(super) fn raw_attr<'a>(tag: &'a str, name: &str) -> Option<&'a str> {
214    let past_space = |at: usize| Some(at + tag.get(at..)?.find(|c: char| !c.is_ascii_whitespace())?);
215    let mut at = tag.find(|c: char| c.is_ascii_whitespace())?;
216    loop {
217        let name_at = past_space(at)?;
218        let eq = name_at + tag.get(name_at..)?.find('=')?;
219        let value_at = past_space(eq + 1)?;
220        let rest = tag.get(value_at..)?;
221        let quote = rest.chars().next().filter(|c| matches!(c, '"' | '\''))?;
222        let len = rest.get(1..)?.find(quote)?;
223        if tag.get(name_at..eq)?.trim_end() == name {
224            return rest.get(1..1 + len);
225        }
226        at = value_at + 1 + len + 1;
227    }
228}

230/// The children of `container` (a child of the root) in `xml` (a part WordCraft wrote): each
231/// one's key in `group` (as [`defs`] gives it) and its markup's byte range; and the range of
232/// the container's content (`None` when there is no such container).
233pub(super) fn children(xml: &str, container: &str, group: &str) -> (Vec<(String, (usize, usize))>, Option<(usize, usize)>) {
234    let mut out = Vec::new();
235    let (mut depth, mut inside) = (0usize, false);
236    let mut open: Option<(String, usize)> = None;
237    let mut content: Option<(usize, usize)> = None;
238    events_at(xml.as_bytes(), |ev, at| match ev {
239        Ev::Start(q, a) => {
240            depth += 1;
241            if depth == 2 {
242                inside = q == container;
243                if inside {
244                    content = Some((at.1, at.1));
245                }
246            } else if depth == 3
247                && inside
248                && let Some(key) = child_key(group, &q, &a)
249            {
250                open = Some((key, at.0));
251            }
252        }
253        Ev::End(_) => {
254            if depth == 3
255                && let Some((key, from)) = open.take()
256            {
257                out.push((key, (from, at.1)));
258            }
259            if depth == 2 && inside {
260                // An empty element (`<x/>`) has no content to replace: none.
261                content = content.filter(|c| c.0 < at.1).map(|c| (c.0, at.0.max(c.0)));
262                inside = false;
263            }
264            depth = depth.saturating_sub(1);
265        }
266        Ev::Text(_) => {}
267    });
268    (out, content)
269}

271/// `xml` (a part WordCraft wrote) with `container`'s children as read: the file's (those of
272/// `group` in `defs`, in `order`; `pick` may choose WordCraft's instead, or neither), then
273/// WordCraft's own the file had none of by that key.
274fn with_as_read(
275    xml: &str,
276    container: &str,
277    group: &str,
278    defs: &BTreeMap<String, Arc<str>>,
279    order: &[String],
280    pick: &dyn Fn(&str, &str, Option<&str>) -> Option<String>,
281) -> String {
282    let (kids, content) = children(xml, container, group);
283    let Some((from, to)) = content else { return xml.to_string() };
284    let ours = |k: &str| kids.iter().find(|(key, _)| key == k).and_then(|(_, (a, b))| xml.get(*a..*b));
285    let prefix = (format!("{group}:"), format!("{group}~"));
286    let mut body = String::new();
287    for k in order.iter().filter(|k| k.starts_with(&prefix.0) || k.starts_with(&prefix.1)) {
288        if let Some(d) = defs.get(k)
289            && let Some(chosen) = pick(k, d, ours(k))
290        {
291            body.push_str(&chosen);
292        }
293    }
294    for (k, (a, b)) in &kids {
295        if !defs.contains_key(k)
296            && let Some(x) = xml.get(*a..*b)
297        {
298            body.push_str(x);
299        }
300    }
301    let (Some(head), Some(tail)) = (xml.get(..from), xml.get(to..)) else { return xml.to_string() };
302    format!("{head}{body}{tail}")
303}

358/// The package files `defs` refer to (`xlink:href` to a path inside the package, as a style's
359/// fill picture; to a folder, as an embedded object in a header: the files in it), read with
360/// `fetch` (`entries`: the package's paths): a definition written as read needs them.
361/// Also the paths linked to that are in the package but couldn't be read (over the picture
362/// budget, unreadable): markup linking to one can't be written as read.
363pub(super) fn files(
364    defs: &[(String, Arc<str>)],
365    entries: &[String],
366    fetch: &mut dyn FnMut(&str) -> Option<Vec<u8>>,
367) -> (Vec<(String, Vec<u8>)>, BTreeSet<String>) {
368    let mut paths = BTreeSet::new();
369    for (_, xml) in defs {
370        hrefs(xml, &mut paths);
371    }
372    let mut out = Vec::new();
373    let mut missing = BTreeSet::new();
374    for p in paths {
375        if entries.contains(&p) {
376            match fetch(&p) {
377                Some(b) => out.push((p, b)),
378                None => {
379                    missing.insert(p);
380                }
381            }
382        } else {
383            // A folder (an object): the files in it.
384            let dir = format!("{}/", p.trim_end_matches('/'));
385            for e in entries.iter().filter(|e| e.starts_with(&dir) && !e.ends_with('/')).take(MAX_FILES) {
386                match fetch(e) {
387                    Some(b) => out.push((e.clone(), b)),
388                    None => {
389                        missing.insert(p.clone());
390                    }
391                }
392            }
393        }
394    }
395    (out, missing)
396}

398/// Whether markup `xml` links to one of the package files in `missing` (see [`files`]).
399pub(super) fn links_to(xml: &str, missing: &BTreeSet<String>) -> bool {
400    if missing.is_empty() {
401        return false;
402    }
403    let mut paths = BTreeSet::new();
404    hrefs(xml, &mut paths);
405    paths.iter().any(|p| missing.contains(p))
406}

408/// The package files `xml` links to (`xlink:href` to a path inside the package), added to
409/// `paths` (at most [`MAX_FILES`]).
410pub(super) fn hrefs(xml: &str, paths: &mut BTreeSet<String>) {
411    if !xml.contains("xlink:href") {
412        return;
413    }
414    super::xml::events(xml.as_bytes(), |ev| {
415        if let Ev::Start(_, a) = ev
416            && let Some(h) = get(&a, "xlink:href")
417            && inside_package(h)
418            && paths.len() < MAX_FILES
419        {
420            paths.insert(h.trim_start_matches("./").to_string());
421        }
422    });
423}

425/// Whether link `h` names a file inside the package (a relative path, not a URL or a fragment).
426fn inside_package(h: &str) -> bool {
427    !h.is_empty() && !h.contains(':') && !h.starts_with(['#', '/']) && !h.contains("..")
428}

445/// The children of `meta.xml`'s `office:meta` WordCraft doesn't model (the template, the
446/// statistics, the editing time, the printing…), as read, in order.
447pub(super) fn meta_defs(xml: &[u8]) -> Vec<(String, Arc<str>)> {
448    let mut out = Vec::new();
449    let mut depth = 0usize;
450    let mut open: Option<(String, usize)> = None;
451    events_at(xml, |ev, at| match ev {
452        Ev::Start(q, _) => {
453            depth += 1;
454            if depth == 3 && open.is_none() && !META_MODELLED.contains(&q.as_str()) && out.len() < MAX_DEFS {
455                open = Some((format!("meta~{}~{q}", out.len()), at.0));
456            }
457        }
458        Ev::End(_) => {
459            if depth == 3
460                && let Some((key, from)) = open.take()
461                && let Some(s) = xml.get(from..at.1).and_then(|b| std::str::from_utf8(b).ok())
462            {
463                out.push((key, Arc::from(s)));
464            }
465            depth = depth.saturating_sub(1);
466        }
467        Ev::Text(_) => {}
468    });
469    out
470}

472/// `meta` (a `meta.xml` WordCraft wrote) with the file's document properties WordCraft doesn't
473/// model (`defs`' `meta~` ones) added, in their order.
474pub(super) fn with_kept_meta(meta: String, defs: &BTreeMap<String, Arc<str>>, order: &[String]) -> String {
475    let kept: String = order.iter().filter(|k| k.starts_with("meta~")).filter_map(|k| defs.get(k)).map(|d| &**d).collect();
476    match meta.rfind("</office:meta>") {
477        Some(at) if !kept.is_empty() => format!("{}{kept}{}", meta.get(..at).unwrap_or(""), meta.get(at..).unwrap_or("")),
478        _ => meta,
479    }

488/// The children `depth` deep in `xml` (the root is 1): each one's element, `form:name` and
489/// markup.
490fn children_at(xml: &str, at_depth: usize) -> Vec<(String, Option<String>, &str)> {
491    let mut out = Vec::new();
492    let mut depth = 0usize;
493    let mut open: Option<(String, Option<String>, usize)> = None;
494    events_at(xml.as_bytes(), |ev, at| match ev {
495        Ev::Start(q, a) => {
496            depth += 1;
497            if depth == at_depth && open.is_none() {
498                open = Some((q, get(&a, "form:name").map(str::to_string), at.0));
499            }
500        }
501        Ev::End(_) => {
502            if depth == at_depth
503                && let Some((q, name, from)) = open.take()
504                && let Some(x) = xml.get(from..at.1)
505            {
506                out.push((q, name, x));
507            }
508            depth = depth.saturating_sub(1);
509        }
510        Ev::Text(_) => {}
511    });
512    out
513}

515/// Control `ours` (as WordCraft writes it) with what the file's control `theirs` had that
516/// WordCraft doesn't model and so can't have changed: its other attributes (not its ids) and its
517/// properties and event listeners.
518fn with_unmodelled(ours: &str, theirs: &str) -> String {
519    let attrs_of = |xml: &str| {
520        let mut out: Vec<(String, String)> = Vec::new();
521        let mut first = true;
522        super::xml::events(xml.as_bytes(), |ev| {
523            if let Ev::Start(_, a) = ev
524                && std::mem::take(&mut first)
525            {
526                out = a;
527            }
528        });
529        out
530    };
531    let have: BTreeSet<String> = attrs_of(ours).into_iter().map(|(k, _)| k).collect();
532    let extra: String = attrs_of(theirs)
533        .into_iter()
534        .filter(|(k, _)| !have.contains(k) && !matches!(k.as_str(), "form:id" | "xml:id" | "form:name") && !k.starts_with("xmlns"))
535        .map(|(k, v)| format!(" {k}=\"{}\"", super::writer::x(&v)))
536        .collect();
537    let kids: String = children_at(theirs, 2)
538        .into_iter()
539        .filter(|(q, ..)| matches!(q.as_str(), "form:properties" | "office:event-listeners"))
540        .filter(|(q, ..)| !ours.contains(&format!("<{q}")))
541        .map(|(.., x)| x)
542        .collect();
543    // Into its start tag, and before its end (an empty one opened for them).
544    let Some(end) = ours.find('>') else { return ours.to_string() };
545    let (head, rest) = ours.split_at(end);
546    let empty = head.ends_with('/');
547    let head = head.trim_end_matches('/');
548    let name = head.trim_start_matches('<').split(char::is_whitespace).next().unwrap_or("");
549    if empty {
550        if kids.is_empty() { format!("{head}{extra}/>") } else { format!("{head}{extra}>{kids}</{name}>") }
551    } else {
552        let close = format!("</{name}>");
553        let body = rest.get(1..).unwrap_or("");
554        match body.rfind(&close) {
555            Some(at) => format!("{head}{extra}>{}{kids}{}", body.get(..at).unwrap_or(""), body.get(at..).unwrap_or("")),
556            None => format!("{head}{extra}>{body}"),
557        }
558    }
559}

561/// The document's forms (`office:forms`, §13.2) as the file had them (`orig`), with the
562/// controls WordCraft writes (`ours`, its own `office:forms`): the file's forms element and each
563/// form's own start tag, properties and event listeners as read; the controls the document has
564/// in the first form (the file's controls are the document's, read into it and written back);
565/// `ours` alone when the file had none, its own form when the file had none.
566pub(super) fn merge_forms(orig: Option<&str>, ours: String) -> String {
567    let Some(orig) = orig else { return ours };
568    // `ours`: the controls (children of its one form).
569    let ours_controls = children_at(&ours, 3);
570    let orig_controls: Vec<(String, Option<String>, &str)> = children_at(orig, 3).into_iter().filter(|(q, ..)| is_control(q)).collect();
571    // Each of ours with what the file's control by that name (else in that place) had and
572    // WordCraft doesn't model.
573    let mut used = vec![false; orig_controls.len()];
574    let mut controls = String::new();
575    for (i, (q, name, xml)) in ours_controls.iter().enumerate() {
576        let same = |(k, (oq, oname, _)): &(usize, &(String, Option<String>, &str))| !used[*k] && oq == q && oname == name;
577        let found = orig_controls.iter().enumerate().find(|c| same(c)).map(|(k, _)| k).or_else(|| {
578            let nth = ours_controls.iter().take(i).filter(|(oq, ..)| oq == q).count();
579            orig_controls.iter().enumerate().filter(|(k, (oq, ..))| !used[*k] && oq == q).nth(nth).map(|(k, _)| k)
580        });
581        match found.and_then(|k| orig_controls.get(k).map(|c| (k, c.2))) {
582            Some((k, theirs)) => {
583                used[k] = true;
584                controls.push_str(&with_unmodelled(xml, theirs));
585            }
586            None => controls.push_str(xml),
587        }
588    }
589    // `orig`: its forms element, each form's start tag and non-control children.
590    let mut head: Option<String> = None;
591    let mut forms: Vec<(String, String)> = Vec::new();
592    let mut child: Option<(usize, bool)> = None;
593    let mut depth = 0usize;
594    events_at(orig.as_bytes(), |ev, at| match ev {
595        Ev::Start(q, _) => {
596            depth += 1;
597            let tag = orig.get(at.0..at.1).unwrap_or("");
598            let opened = || tag.strip_suffix("/>").map_or_else(|| tag.to_string(), |t| format!("{}>", t.trim_end()));
599            match depth {
600                1 => head = Some(opened()),
601                2 if q == "form:form" => forms.push((opened(), String::new())),
602                3 if child.is_none() => child = Some((at.0, is_control(&q))),
603                _ => {}
604            }
605        }
606        Ev::End(_) => {
607            if depth == 3
608                && let Some((from, control)) = child.take()
609                && !control
610                && let (Some(f), Some(x)) = (forms.last_mut(), orig.get(from..at.1))
611            {
612                f.1.push_str(x);
613            }
614            depth = depth.saturating_sub(1);
615        }
616        Ev::Text(_) => {}
617    });
618    let Some(head) = head else { return ours };
619    let mut out = head;
620    if forms.is_empty() && !controls.is_empty() {
621        out.push_str(&format!("<form:form form:name=\"Form\">{controls}</form:form>"));
622    }
623    for (i, (start, kids)) in forms.iter().enumerate() {
624        out.push_str(start);
625        out.push_str(kids);
626        if i == 0 {
627            out.push_str(&controls);
628        }
629        out.push_str("</form:form>");
630    }
631    out.push_str("</office:forms>");
632    out
633}

635/// A list of declarations as the file had it (`orig`, a `text:variable-decls` or the like),
636/// with those of WordCraft's (`ours`, the same list) it doesn't have by name; `ours` alone when
637/// the file had none.
638pub(super) fn merge_decls(orig: Option<&str>, ours: String) -> String {
639    let Some(orig) = orig else { return ours };
640    let named = |xml: &str| -> Vec<(String, (usize, usize))> {
641        let mut out = Vec::new();
642        let mut depth = 0usize;
643        events_at(xml.as_bytes(), |ev, at| match ev {
644            Ev::Start(_, a) => {
645                depth += 1;
646                if depth == 2
647                    && let Some(n) = get(&a, "text:name")
648                {
649                    out.push((n.to_string(), at));
650                }
651            }
652            Ev::End(_) => {
653                if depth == 2
654                    && let Some(last) = out.last_mut()
655                {
656                    last.1.1 = at.1;
657                }
658                depth = depth.saturating_sub(1);
659            }
660            Ev::Text(_) => {}
661        });
662        out
663    };
664    let have: BTreeSet<String> = named(orig).into_iter().map(|(n, _)| n).collect();
665    let extra: String = named(&ours).into_iter().filter(|(n, _)| !have.contains(n)).filter_map(|(_, (a, b))| ours.get(a..b)).collect();
666    if extra.is_empty() {
667        return orig.to_string();
668    }
669    match orig.rfind("</") {
670        Some(at) if orig.ends_with('>') && !orig.trim_end().ends_with("/>") => {
671            format!("{}{extra}{}", orig.get(..at).unwrap_or(""), orig.get(at..).unwrap_or(""))
672        }
673        // `<text:variable-decls/>`: opened and closed around them.
674        _ => {
675            let el = orig.trim_start_matches('<').split([' ', '/', '>']).next().unwrap_or("");
676            format!("{}>{extra}</{el}>", orig.trim_end().trim_end_matches("/>"))
677        }
678    }
679}

681/// The tracked changes (§5.5.7 `text:change-id`) `xml` marks, by id.
682pub(super) fn changes(xml: &str) -> BTreeSet<String> {
683    let mut out = BTreeSet::new();
684    if xml.contains("text:change-id") {
685        super::xml::events(xml.as_bytes(), |ev| {
686            if let Ev::Start(_, a) = ev
687                && let Some(id) = get(&a, "text:change-id").filter(|i| !i.is_empty() && i.len() <= 256)
688                && out.len() < MAX_DEFS
689            {
690                out.insert(id.to_string());
691            }
692        });
693    }
694    out
695}

697/// The names of the bookmarks (§6.2.1) `xml` begins (at a point too) and ends.
698pub(super) fn bookmarks(xml: &str) -> (BTreeSet<String>, BTreeSet<String>) {
699    let (mut starts, mut ends) = (BTreeSet::new(), BTreeSet::new());
700    if xml.contains("<text:bookmark") {
701        super::xml::events(xml.as_bytes(), |ev| {
702            if let Ev::Start(q, a) = ev
703                && let Some(n) = get(&a, "text:name")
704                && starts.len() + ends.len() < MAX_DEFS
705            {
706                match q.as_str() {
707                    "text:bookmark" => {
708                        starts.insert(n.to_string());
709                        ends.insert(n.to_string());
710                    }
711                    "text:bookmark-start" => {
712                        starts.insert(n.to_string());
713                    }
714                    "text:bookmark-end" => {
715                        ends.insert(n.to_string());
716                    }
717                    _ => {}
718                }
719            }
720        });
721    }
722    (starts, ends)
723}

733            {
734                out.insert(n.to_string());
735            }
736        });
737    }

741/// The kept definitions markup naming `start` needs, each once, what they name included:
742/// `(automatic, common, font faces)`, by name (with its name space's suffix, see `space`), as kept. `None` when one of them makes a page
743/// start (`style:master-page-name`): WordCraft names its master pages afresh, so such markup is
744/// written from the model instead.
745pub(super) type Needed = (BTreeMap<String, Arc<str>>, BTreeMap<String, Arc<str>>, BTreeMap<String, Arc<str>>);
746
747pub(super) fn needed(defs: &BTreeMap<String, Arc<str>>, start: BTreeSet<String>, masters: bool) -> Option<Needed> {
748    let (mut auto, mut common, mut fonts) = (BTreeMap::new(), BTreeMap::new(), BTreeMap::new());
749    let mut todo: Vec<String> = start.into_iter().collect();
750    let mut seen = BTreeSet::new();
751    while let Some(n) = todo.pop() {
752        if !seen.insert(n.clone()) || seen.len() > MAX_DEFS {
753            continue;
754        }
755        // Every definition by that name (in any name space: see `space`), from the first group
756        // that has one: automatic styles, then common ones, then font faces.
757        let mut found = false;
758        for (group, into) in [("auto", &mut auto), ("common", &mut common), ("font", &mut fonts)] {
759            let base = format!("{group}:{n}");
760            for (k, def) in defs.range(base.clone()..).take_while(|(k, _)| k.starts_with(&base)) {
761                let Some(rest) = k.strip_prefix(&base).filter(|r| r.is_empty() || r.starts_with('|')) else { continue };
762                if !masters && def.contains("style:master-page-name=\"") && !def.contains("style:master-page-name=\"\"") {
763                    return None;
764                }
765                into.insert(format!("{n}{rest}"), def.clone());
766                todo.extend(names(def));
767                found = true;
768            }
769            if found {
770                break;
771            }
772        }
773    }
774    Some((auto, common, fonts))
775}
