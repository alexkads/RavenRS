1//! Print Markup for the text formats: HTML, Markdown and plain text copies end with the
2//! document's comments, numbered in document order, each with its author, the text it comments
3//! on, what it says and its replies (like the notes at the end of a printed document).
4
5use std::collections::{BTreeMap, HashMap, HashSet};
6
7use crate::doc::para::{InlineObject, OBJ};
8use crate::doc::{Document, StoryRef};
9
10/// Longest quoted anchor text (characters).
11const MAX_QUOTE: usize = 200;
12const MAX_COMMENTS: usize = 5_000;
13const MAX_REPLIES: usize = 200;
14const MAX_PARENT_CHAIN: usize = 64;
15
16/// One comment as written out.
17#[derive(Clone, Debug, PartialEq, Eq)]
18pub struct Note {
19    pub author: String,
20    /// The text it comments on (shortened), or empty.
21    pub quote: String,
22    pub text: String,
23    /// (author, text) of each reply, oldest first.
24    pub replies: Vec<(String, String)>,
25}
26
27fn root(doc: &Document, id: u32) -> u32 {
28    let mut cur = id;
29    for _ in 0..MAX_PARENT_CHAIN {
30        match doc.comments.get(&cur).and_then(|c| c.parent) {
31            Some(p) if p != cur && doc.comments.contains_key(&p) => cur = p,
32            _ => break,
33        }
34    }
35    cur
36}
37
38fn author(a: &str) -> String {
39    if a.trim().is_empty() { "Author".to_string() } else { a.trim().to_string() }
40}
41
42fn text_of(doc: &Document, part: u32) -> String {
43    doc.plain_text(StoryRef::Part(part)).split_whitespace().collect::<Vec<_>>().join(" ")
44}

46/// The document's top-level comments: those anchored in the body in document order, then the
47/// rest; replies go under the comment they answer.
48pub fn notes(doc: &Document) -> Vec<Note> {
49    if doc.comments.is_empty() {
50        return Vec::new();
51    }
52    // Anchor order and the text between each comment's start and end.
53    let mut order: Vec<u32> = Vec::new();
54    let mut seen = HashSet::new();
55    let mut quotes: HashMap<u32, String> = HashMap::new();
56    let mut open: Vec<u32> = Vec::new();
57    for path in doc.para_paths(StoryRef::Body) {
58        let Some(p) = doc.para(StoryRef::Body, &path) else { continue };
59        let mut objects = p.objects.iter();
60        for c in p.text.chars() {
61            crate::doc::work::steps(1);
62            if c == OBJ {
63                match objects.next() {
64                    Some(InlineObject::CommentStart { id }) => {
65                        if seen.insert(*id) {
66                            order.push(*id);
67                        }
68                        if !open.contains(id) && open.len() < 64 {
69                            open.push(*id);
70                        }
71                    }
72                    Some(InlineObject::CommentEnd { id }) => open.retain(|x| x != id),
73                    _ => {}
74                }
75                continue;
76            }
77            for id in &open {
78                let q = quotes.entry(*id).or_default();
79                if q.chars().count() <= MAX_QUOTE {
80                    q.push(c);
81                }
82            }
83        }
84        for id in &open {
85            quotes.entry(*id).or_default().push(' ');
86        }
87    }
88    for id in doc.comments.keys() {
89        if seen.insert(*id) {
90            order.push(*id);
91        }
92    }
93    let mut replies: BTreeMap<u32, Vec<u32>> = BTreeMap::new();
94    for id in doc.comments.keys() {
95        let r = root(doc, *id);
96        if r != *id {
97            let v = replies.entry(r).or_default();
98            if v.len() < MAX_REPLIES {
99                v.push(*id);
100            }
101        }
102    }
103    let mut out = Vec::new();
104    for id in order {
105        let Some(c) = doc.comments.get(&id) else { continue };
106        if root(doc, id) != id {
107            continue;
108        }
109        let quote = quotes.get(&id).map(|q| q.split_whitespace().collect::<Vec<_>>().join(" ")).unwrap_or_default();
110        let quote = if quote.chars().count() > MAX_QUOTE { format!("{}…", quote.chars().take(MAX_QUOTE).collect::<String>()) } else { quote };
111        let replies =
112            replies.get(&id).into_iter().flatten().filter_map(|r| doc.comments.get(r)).map(|r| (author(&r.author), text_of(doc, r.part))).collect();
113        out.push(Note { author: author(&c.author), quote, text: text_of(doc, c.part), replies });
114        if out.len() >= MAX_COMMENTS {
115            break;
116        }
117    }
118    out
119}

121/// `bytes` (a copy in format `ext`) with the comments written at its end.
122pub fn append(ext: &str, doc: &Document, bytes: Vec<u8>) -> Vec<u8> {
123    let notes = notes(doc);
124    if notes.is_empty() {
125        return bytes;
126    }
127    match ext {
128        "html" | "htm" | "xhtml" => {
129            let mut s = match String::from_utf8(bytes) {
130                Ok(s) => s,
131                Err(e) => return e.into_bytes(),
132            };
133            let at = s.rfind("</body>").unwrap_or(s.len());
134            s.insert_str(at, &html(&notes));
135            s.into_bytes()
136        }
137        "md" | "markdown" => {
138            let mut s = bytes;
139            s.extend_from_slice(markdown(&notes).as_bytes());
140            s
141        }
142        "txt" | "text" => {
143            let mut s = bytes;
144            s.extend_from_slice(text(&notes).as_bytes());
145            s
146        }
147        _ => bytes,
148    }
149}

151fn html(notes: &[Note]) -> String {
152    use crate::formats::html::esc;
153    let mut s = String::from("<section class=\"comments\">\n<h2>Comments</h2>\n<ol>\n");
154    for (i, n) in notes.iter().enumerate() {
155        s.push_str(&format!("<li id=\"comment-{}\"><p><strong>{}</strong>", i + 1, esc(&n.author)));
156        if !n.quote.is_empty() {
157            s.push_str(&format!(" on <q>{}</q>", esc(&n.quote)));
158        }
159        s.push_str(&format!(": {}</p>", esc(&n.text)));
160        if !n.replies.is_empty() {
161            s.push_str("<ul>");
162            for (a, t) in &n.replies {
163                s.push_str(&format!("<li><strong>{}</strong>: {}</li>", esc(a), esc(t)));
164            }
165            s.push_str("</ul>");
166        }
167        s.push_str("</li>\n");
168    }
169    s.push_str("</ol>\n</section>\n");
170    s
171}

173fn markdown(notes: &[Note]) -> String {
174    use crate::formats::markdown::escape;
175    let mut s = String::from("\n\n## Comments\n\n");
176    for (i, n) in notes.iter().enumerate() {
177        s.push_str(&format!("{}. **{}**", i + 1, escape(&n.author)));
178        if !n.quote.is_empty() {
179            s.push_str(&format!(" on “{}”", escape(&n.quote)));
180        }
181        s.push_str(&format!(": {}\n", escape(&n.text)));
182        for (a, t) in &n.replies {
183            s.push_str(&format!("    - **{}**: {}\n", escape(a), escape(t)));
184        }
185    }
186    s
187}
188
189fn text(notes: &[Note]) -> String {
190    let mut s = String::from("\r\n\r\nComments\r\n");
191    for (i, n) in notes.iter().enumerate() {
192        s.push_str(&format!("\r\n[{}] {}", i + 1, n.author));
193        if !n.quote.is_empty() {
194            s.push_str(&format!(" on “{}”", n.quote));
195        }
196        s.push_str(&format!(": {}\r\n", n.text));
197        for (a, t) in &n.replies {
198            s.push_str(&format!("    {a}: {t}\r\n"));
199        }
200    }
201    s
202}
