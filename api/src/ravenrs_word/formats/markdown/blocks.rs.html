26/// A label as CommonMark compares them: case-folded, and any run of white space one space
27/// (§4.7, "matches"). `[FOO bar]` and `[foo   BAR]` are the same label.
28pub(super) fn ref_label(raw: &str) -> String {
29    raw.split_whitespace().collect::<Vec<_>>().join(" ").to_lowercase()
30}

32/// A link reference definition on its own line: `[label]: destination "optional title"`. The
33/// destination may be wrapped in `<...>`. Returns the label and the destination.
34///
35/// CommonMark allows the title on the following line and the label to span lines; a definition
36/// written across lines is left as text here, which is what it was before these were read at all.
37fn link_definition(line: &str) -> Option<(String, String)> {
38    let t = line.trim_start();
39    // Up to three spaces of indentation; four makes it indented code.
40    if line.len() - t.len() > 3 || !t.starts_with('[') {
41        return None;
42    }
43    let bytes = t.as_bytes();
44    // The label ends at the first unescaped `]`, and cannot be empty or hold an unescaped `[`.
45    let mut i = 1;
46    let mut close = None;
47    while i < bytes.len() {
48        match bytes[i] {
49            b'\\' => i += 1,
50            b'[' => return None,
51            b']' => {
52                close = Some(i);
53                break;
54            }
55            _ => {}
56        }
57        i += 1;
58    }
59    let close = close?;
60    let label = ref_label(t.get(1..close)?);
61    // `[^label]: ...` is a footnote definition, which this importer leaves as text.
62    if label.starts_with('^') {
63        return None;
64    }
65    if label.is_empty() || t.as_bytes().get(close + 1) != Some(&b':') {
66        return None;
67    }
68    let rest = t.get(close + 2..)?.trim_start();
69    if rest.is_empty() {
70        return None;
71    }
72    // The destination: `<...>`, else up to the next space. What follows may only be a title.
73    let (dest, after) = match rest.strip_prefix('<') {
74        Some(inner) => {
75            let end = inner.find('>')?;
76            (inner.get(..end)?.to_string(), inner.get(end + 1..)?)
77        }
78        None => {
79            let end = rest.find(char::is_whitespace).unwrap_or(rest.len());
80            (rest.get(..end)?.to_string(), rest.get(end..)?)
81        }
82    };
83    if dest.is_empty() {
84        return None;
85    }
86    let title = after.trim();
87    let titled =
88        title.is_empty() || (title.len() >= 2 && matches!((title.chars().next()?, title.chars().last()?), ('"', '"') | ('\'', '\'') | ('(', ')')));
89    titled.then_some((label, dest))
90}

92/// Take the link reference definitions out of `lines`, leaving the rest. Lines inside a fenced
93/// code block are left alone: a definition there is code, not a definition. Fences open and
94/// close by the same rules [`parse_blocks`] applies, or a line it reads as code would be read
95/// here as a definition and deleted.
96///
97/// A definition has to start a block: CommonMark §4.7 forbids one from interrupting a paragraph,
98/// and a line like `[Warning]: unstable` in the middle of one is the author's text. This pass
99/// sees lines without the block structure around them, so it takes a definition only after a
100/// blank line, at the start of the document, after a heading or thematic break, or after another
101/// definition. A definition anywhere else stays in the document and reads as text, which is what
102/// it did before definitions were understood at all; deleting a line that was part of a
103/// paragraph would lose it silently.
104///
105/// Returns the parsed references and the 0-based indexes of the removed lines.
106fn take_definitions(lines: &mut Vec<String>) -> (Refs, Vec<usize>) {
107    let mut refs = Refs::new();
108    let mut open: Option<(char, usize)> = None;
109    // The start of the document begins a block.
110    let mut starts_block = true;
111    let mut removed = Vec::new();
112    let mut at = 0usize;
113    lines.retain(|line| {
114        let this = at;
115        at += 1;
116        let t = line.trim_start_matches([' ', '\t']);
117        // Four spaces of indentation is code, not a fence (`parse_blocks`), on both sides.
118        let marker = (indent(line) < 4).then(|| fence(t)).flatten();
119        match open {
120            Some((c, n)) => {
121                // A closing fence holds nothing but its own run of fence characters.
122                if marker.is_some_and(|(fc, fl)| fc == c && fl >= n) && t.trim_start_matches(c).trim().is_empty() {
123                    open = None;
124                    starts_block = true;
125                }
126                return true;
127            }
128            None => {
129                if let Some(f) = marker {
130                    open = Some(f);
131                    return true;
132                }
133            }
134        }
135        if starts_block && let Some((label, dest)) = link_definition(line) {
136            // The first definition of a label wins, as CommonMark says, and definitions may
137            // follow one another.
138            refs.entry(label).or_insert(dest.into());
139            removed.push(this);
140            return false;
141        }
142        starts_block = is_blank(line) || atx_heading(t).is_some() || thematic_break(t);
143        true
144    });
145    (refs, removed)
146}

148/// Maps monotonically non-decreasing line indexes in lines filtered of link definitions back to
149/// their original source lines in a single linear pass.
150pub(super) fn remap_original_lines(block_lines: &mut [usize], removed: &[usize]) {
151    let mut rem_iter = removed.iter().copied().peekable();
152    let mut shift = 0;
153    for l in block_lines.iter_mut() {
154        while let Some(&r) = rem_iter.peek() {
155            if r <= *l + shift {
156                shift += 1;
157                rem_iter.next();
158            } else {
159                break;
160            }
161        }
162        *l += shift;
163    }
164}

166/// Context for block parsing: depth, nesting, reference definitions, cell budget, and line recording.
167struct BlockCtx<'a> {
168    depth: usize,
169    level: u8,
170    refs: &'a Refs,
171    budget: &'a std::cell::Cell<usize>,
172    lines_out: Option<&'a mut Vec<usize>>,
173}
174
175impl BlockCtx<'_> {
176    /// Records the starting line of a top-level block if line tracking is active.
177    fn record_line(&mut self, line: usize) {
178        if self.depth == 0
179            && let Some(lo) = self.lines_out.as_mut()
180        {
181            lo.push(line);
182        }
183    }
184
185    /// Records `count` blocks originating at `line` if line tracking is active.
186    fn record_lines(&mut self, line: usize, count: usize) {
187        if self.depth == 0
188            && let Some(lo) = self.lines_out.as_mut()
189        {
190            for _ in 0..count {
191                lo.push(line);
192            }
193        }
194    }
195}

202/// [`parse`] returning the flow model and the 0-based starting line of each top-level block.
203pub fn parse_with_lines(text: &str) -> (model::Flow, Vec<usize>) {
204    let text = text.strip_prefix('\u{FEFF}').unwrap_or(text);
205    let mut lines: Vec<String> = text.split('\n').map(|l| expand_tabs(l.trim_end_matches('\r'))).collect();
206    let (refs, removed) = take_definitions(&mut lines);
207    let budget = std::cell::Cell::new(super::MAX_TABLE_CELLS);
208    let mut block_lines = Vec::new();
209    let mut ctx = BlockCtx { depth: 0, level: 0, refs: &refs, budget: &budget, lines_out: Some(&mut block_lines) };
210    let mut blocks = parse_blocks(&lines, &mut ctx);
211    // A leading "Title" from front matter isn't supported; a first-level heading stays a heading.
212    if blocks.is_empty() {
213        blocks.push(FBlock::Para(Para::default()));
214        block_lines.push(0);
215    }
216    remap_original_lines(&mut block_lines, &removed);
217    (model::Flow { blocks, meta: Default::default() }, block_lines)
218}

221pub fn import_on_with_lines(text: &str, page: &crate::doc::SectionProps) -> (Document, Vec<usize>) {
222    model::begin_import();
223    let (flow, mut lines) = parse_with_lines(text);
224    let doc = model::to_doc_in(&flow, page.clone());

232/// Leading tabs → spaces (tab stops of 4); the rest of the line is kept.
233fn expand_tabs(l: &str) -> String {
234    if !l.starts_with([' ', '\t']) || !l.contains('\t') {
235        return l.to_string();
236    }
237    let mut out = String::with_capacity(l.len() + 8);
238    let mut col = 0usize;
239    let mut rest = l;
240    while let Some(c) = rest.chars().next() {
241        match c {
242            ' ' => {
243                out.push(' ');
244                col += 1;
245            }
246            '\t' => {
247                let n = 4 - col % 4;
248                out.push_str(&" ".repeat(n));
249                col += n;
250            }
251            _ => break,
252        }
253        rest = &rest[c.len_utf8()..];
254    }
255    out.push_str(rest);
256    out
257}

273fn atx_heading(t: &str) -> Option<(u8, &str)> {
274    let n = t.chars().take_while(|c| *c == '#').count();
275    if !(1..=6).contains(&n) {
276        return None;
277    }
278    let rest = t.get(n..)?;
279    if !rest.is_empty() && !rest.starts_with([' ', '\t']) {
280        return None;
281    }
282    let mut body = rest.trim();
283    // Optional closing sequence.
284    let trimmed = body.trim_end_matches('#');
285    if trimmed.len() != body.len() && (trimmed.is_empty() || trimmed.ends_with(' ')) {
286        body = trimmed.trim_end();
287    }
288    Some((n as u8, body))
289}

291pub(super) fn thematic_break(t: &str) -> bool {
292    let mut ch = None;
293    let mut n = 0;
294    for c in t.chars() {
295        match c {
296            ' ' | '\t' => {}
297            '*' | '-' | '_' => {
298                if ch.is_some_and(|x| x != c) {
299                    return false;
300                }
301                ch = Some(c);
302                n += 1;
303            }
304            _ => return false,
305        }
306    }
307    n >= 3
308}

310fn fence(t: &str) -> Option<(char, usize)> {
311    let c = t.chars().next().filter(|c| *c == '`' || *c == '~')?;
312    let n = t.chars().take_while(|x| *x == c).count();
313    if n < 3 {
314        return None;
315    }
316    if c == '`' && t.get(n..).is_some_and(|info| info.contains('`')) {
317        return None;
318    }
319    Some((c, n))
320}

322#[derive(Clone, Copy, PartialEq)]
323struct Marker {
324    ordered: bool,
325    /// Bullet char or ordered delimiter.
326    ch: char,
327    /// Bytes of the marker itself (after the indent).
328    len: usize,
329    start: u32,
330}
331
332fn list_marker(t: &str) -> Option<Marker> {
333    let first = t.chars().next()?;
334    let (m, len) = if matches!(first, '-' | '+' | '*') {
335        (Marker { ordered: false, ch: first, len: 1, start: 1 }, 1)
336    } else {
337        let digits = t.chars().take_while(char::is_ascii_digit).count();
338        if !(1..=9).contains(&digits) {
339            return None;
340        }
341        let d = t.get(digits..)?.chars().next()?;
342        if d != '.' && d != ')' {
343            return None;
344        }
345        let start = t.get(..digits)?.parse::<u32>().unwrap_or(1);
346        (Marker { ordered: true, ch: d, len: digits + 1, start }, digits + 1)
347    };
348    let rest = t.get(len..)?;
349    if rest.is_empty() || rest.starts_with([' ', '\t']) { Some(m) } else { None }
350}

352/// A table row without its leading and trailing pipes (an escaped last pipe stays).
353fn row_body(l: &str) -> &str {
354    let t = l.trim();
355    let t = t.strip_prefix('|').unwrap_or(t);
356    if let Some(before) = t.strip_suffix('|') {
357        let num_bs = before.chars().rev().take_while(|&c| c == '\\').count();
358        if num_bs % 2 == 0 { before } else { t }
359    } else {
360        t
361    }
362}

364/// The first `keep` cells of a table row, and how many cells it has: the rest are only counted.
365pub(super) fn split_row(l: &str, keep: usize) -> (Vec<String>, usize) {
366    let mut cells = Vec::new();
367    let mut count = 0usize;
368    let mut cur = String::new();
369    let mut chars = row_body(l).chars().peekable();
370    while let Some(c) = chars.next() {
371        let full = cells.len() >= keep;
372        match c {
373            '\\' => match chars.peek() {
374                Some(&'|') => {
375                    if !full {
376                        cur.push('|');
377                    }
378                    chars.next();
379                }
380                Some(&'\\') => {
381                    if !full {
382                        cur.push_str("\\\\");
383                    }
384                    chars.next();
385                }
386                _ if !full => cur.push('\\'),
387                _ => {}
388            },
389            '|' => {
390                if !full {
391                    cells.push(std::mem::take(&mut cur).trim().to_string());
392                }
393                count += 1;
394            }
395            _ if !full => cur.push(c),
396            _ => {}
397        }
398    }
399    if cells.len() < keep {
400        cells.push(cur.trim().to_string());
401    }
402    (cells, count + 1)
403}

405/// The alignments of the first `keep` cells of a delimiter row, and how many cells it has; every
406/// cell is checked, so a bad one past `keep` still means it isn't a delimiter row.
407fn delimiter_row(l: &str, keep: usize) -> Option<(Vec<Option<Align>>, usize)> {
408    if !l.contains('-') {
409        return None;
410    }
411    let mut out = Vec::new();
412    let mut count = 0usize;
413    // A valid cell is only dashes, colons and spaces, so an escaped pipe can't be in one.
414    for c in row_body(l).split('|') {
415        let c = c.trim();
416        let left = c.starts_with(':');
417        let right = c.ends_with(':');
418        let core = c.trim_matches(':');
419        if core.is_empty() || !core.chars().all(|x| x == '-') {
420            return None;
421        }
422        if out.len() < keep {
423            out.push(match (left, right) {
424                (true, true) => Some(Align::Center),
425                (false, true) => Some(Align::Right),
426                (true, false) => Some(Align::Left),
427                _ => None,
428            });
429        }
430        count += 1;
431    }
432    Some((out, count))
433}

435/// Does this line start a block that interrupts a paragraph?
436fn interrupts(l: &str) -> bool {
437    if indent(l) >= 4 {
438        return false;
439    }
440    let t = l.trim_start_matches([' ', '\t']);
441    atx_heading(t).is_some()
442        || thematic_break(t)
443        || fence(t).is_some()
444        || t.starts_with('>')
445        || t.starts_with("<!--")
446        || list_marker(t).is_some_and(|m| (!m.ordered || m.start == 1) && !is_blank(t.get(m.len..).unwrap_or("")))
447}

449/// Flushes accumulated paragraph text into `out`, recording its starting line if tracked.
450fn flush_para(lines: &mut Vec<String>, start_line: &mut Option<usize>, out: &mut Vec<FBlock>, ctx: &mut BlockCtx<'_>) {
451    if lines.is_empty() {
452        *start_line = None;
453        return;
454    }
455    let line_no = start_line.take().unwrap_or(0);
456    let text = join_lines(lines);
457    lines.clear();
458    let inlines = parse_inlines(&text, &Fmt::default(), ctx.depth, ctx.refs);
459    if inlines.is_empty() {
460        return;
461    }
462    let p = Para { inlines, ..Default::default() };
463    out.push(FBlock::Para(p));
464    ctx.record_line(line_no);
465}

467/// Paragraph lines → text with `\n` for hard breaks and spaces for soft ones.
468fn join_lines(lines: &[String]) -> String {
469    let mut s = String::new();
470    let n = lines.len();
471    for (i, l) in lines.iter().enumerate() {
472        let l = l.trim_start_matches([' ', '\t']);
473        if i + 1 == n {
474            s.push_str(l.trim_end());
475        } else if l.ends_with("  ") {
476            s.push_str(l.trim_end());
477            s.push('\n');
478        } else if l.ends_with('\\') && !l.ends_with("\\\\") {
479            s.push_str(&l[..l.len() - 1]);
480            s.push('\n');
481        } else {
482            s.push_str(l.trim_end());
483            s.push(' ');
484        }
485    }
486    s
487}

496/// Attempts to parse a table starting at `lines[*i]`. Returns `true` if a table was parsed.
497fn parse_table(lines: &[String], i: &mut usize, t: &str, ctx: &mut BlockCtx<'_>, out: &mut Vec<FBlock>) -> bool {
498    let Some((aligns, width)) = lines.get(*i + 1).and_then(|l| delimiter_row(l, MAX_COLS)) else { return false };
499    let (head, head_width) = split_row(t, MAX_COLS);
500    if head_width != width {
501        return false;
502    }
503    let table_start = *i;
504    let mut table = FTable::default();
505    let cell = |s: &str, header: bool, a: Option<Align>| {
506        let p = Para { align: a, inlines: parse_inlines(s, &Fmt::default(), ctx.depth, ctx.refs), ..Default::default() };
507        Cell { blocks: vec![FBlock::Para(p)], header, ..Default::default() }
508    };
509    let cols = aligns.len();
510    let fits = |rows: usize| rows < MAX_ROWS && ctx.budget.get() >= cols;
511    let mut cut = head_width > cols;
512    if fits(0) {
513        ctx.budget.set(ctx.budget.get() - cols);
514        table.rows.push(head.iter().zip(&aligns).map(|(s, a)| cell(s, true, *a)).collect());
515    } else {
516        cut = true;
517    }
518    *i += 2;
519    while let Some(l) = lines.get(*i) {
520        if is_blank(l) || !l.contains('|') || interrupts(l) {
521            break;
522        }
523        if !table.rows.is_empty() && fits(table.rows.len()) {
524            ctx.budget.set(ctx.budget.get() - cols);
525            let (mut cells, _) = split_row(l, cols);
526            cells.resize(cols, String::new());
527            table.rows.push(cells.iter().zip(&aligns).map(|(s, a)| cell(s, false, *a)).collect());
528        } else {
529            cut = true;
530        }
531        *i += 1;
532    }
533    if cut {
534        model::dropped(model::Dropped::TableCells);
535    }
536    if !table.rows.is_empty() {
537        out.push(FBlock::Table(table));
538        ctx.record_line(table_start);
539    }
540    true
541}

543/// Parse block structure.
544fn parse_blocks(lines: &[String], ctx: &mut BlockCtx<'_>) -> Vec<FBlock> {
545    let mut out = Vec::new();
546    let mut para: Vec<String> = Vec::new();
547    let mut para_start = None;
548    let mut i = 0;
549    while let Some(line) = lines.get(i) {
550        if is_blank(line) {
551            flush_para(&mut para, &mut para_start, &mut out, ctx);
552            i += 1;
553            continue;
554        }
555        let ind = indent(line);
556        if ind >= 4 {
557            if !para.is_empty() {
558                if para_start.is_none() {
559                    para_start = Some(i);
560                }
561                para.push(line.clone());
562                i += 1;
563                continue;
564            }
565            parse_indented_code(lines, &mut i, &mut out, ctx);
566            continue;
567        }
568        let t = line.trim_start_matches([' ', '\t']);
569        // Setext heading underline.
570        if !para.is_empty() {
571            let u = t.trim_end();
572            if !u.is_empty() && (u.chars().all(|c| c == '=') || (u.chars().all(|c| c == '-'))) {
573                let lvl = if u.starts_with('=') { 1 } else { 2 };
574                let line_no = para_start.take().unwrap_or(i);
575                let text = join_lines(&para);
576                para.clear();
577                let mut p = Para::new(Kind::Heading(lvl));
578                p.inlines = parse_inlines(&text, &Fmt::default(), ctx.depth, ctx.refs);
579                out.push(FBlock::Para(p));
580                ctx.record_line(line_no);
581                i += 1;
582                continue;
583            }
584        }
585        if let Some((n, body)) = atx_heading(t) {
586            flush_para(&mut para, &mut para_start, &mut out, ctx);
587            let mut p = Para::new(Kind::Heading(n));
588            p.inlines = parse_inlines(body, &Fmt::default(), ctx.depth, ctx.refs);
589            out.push(FBlock::Para(p));
590            ctx.record_line(i);
591            i += 1;
592            continue;
593        }
594        if thematic_break(t) {
595            flush_para(&mut para, &mut para_start, &mut out, ctx);
596            out.push(FBlock::Para(Para::new(Kind::Rule)));
597            ctx.record_line(i);
598            i += 1;
599            continue;
600        }
601        if let Some((fc, fl)) = fence(t) {
602            flush_para(&mut para, &mut para_start, &mut out, ctx);
603            i += 1;
604            while let Some(l) = lines.get(i) {
605                let cur = i;
606                i += 1;
607                let lt = l.trim_start_matches([' ', '\t']);
608                if indent(l) < 4 && fence(lt).is_some_and(|(c, n)| c == fc && n >= fl) && lt.trim_start_matches(fc).trim().is_empty() {
609                    break;
610                }
611                out.push(code_para(strip_indent(l, ind)));
612                ctx.record_line(cur);
613            }
614            continue;
615        }
616        if t.starts_with("<!--") {
617            flush_para(&mut para, &mut para_start, &mut out, ctx);
618            while let Some(l) = lines.get(i) {
619                i += 1;
620                if l.contains("-->") {
621                    break;
622                }
623            }
624            continue;
625        }
626        if t.starts_with('>') {
627            flush_para(&mut para, &mut para_start, &mut out, ctx);
628            let quote_start = i;
629            let before = out.len();
630            parse_quote(lines, &mut i, ctx, &mut out);
631            ctx.record_lines(quote_start, out.len() - before);
632            continue;
633        }
634        if para.is_empty() && t.contains('|') && parse_table(lines, &mut i, t, ctx, &mut out) {
635            continue;
636        }
637        if let Some(m) = list_marker(t)
638            && (para.is_empty() || ((!m.ordered || m.start == 1) && !is_blank(t.get(m.len..).unwrap_or(""))))
639        {
640            flush_para(&mut para, &mut para_start, &mut out, ctx);
641            i = parse_list(lines, i, ctx, &mut out);
642            continue;
643        }
644        if para_start.is_none() {
645            para_start = Some(i);
646        }
647        para.push(line.clone());
648        i += 1;
649    }
650    flush_para(&mut para, &mut para_start, &mut out, ctx);
651    out
652}

654/// Parses indented code block starting at `lines[*i]`.
655fn parse_indented_code(lines: &[String], i: &mut usize, out: &mut Vec<FBlock>, ctx: &mut BlockCtx<'_>) {
656    let mut code: Vec<(usize, &str)> = Vec::new();
657    while let Some(line) = lines.get(*i) {
658        let cur = *i;
659        if is_blank(line) {
660            code.push((cur, ""));
661        } else if indent(line) >= 4 {
662            code.push((cur, strip_indent(line, 4)));
663        } else {
664            break;
665        }
666        *i += 1;
667    }
668    while code.last().is_some_and(|(_, line)| line.is_empty()) {
669        code.pop();
670    }
671    for (line_idx, text) in code {
672        out.push(code_para(text));
673        ctx.record_line(line_idx);
674    }
675}

677/// Parses block quote starting at `lines[*i]`.
678fn parse_quote(lines: &[String], i: &mut usize, ctx: &BlockCtx<'_>, out: &mut Vec<FBlock>) {
679    let mut inner = Vec::new();
680    while let Some(line) = lines.get(*i) {
681        let trimmed = line.trim_start_matches([' ', '\t']);
682        if indent(line) < 4 && trimmed.starts_with('>') {
683            let content = &trimmed[1..];
684            inner.push(content.strip_prefix(' ').unwrap_or(content).to_string());
685        } else if !is_blank(line) && !interrupts(line) && inner.last().is_some_and(|line| !is_blank(line)) {
686            inner.push(line.clone());
687        } else {
688            break;
689        }
690        *i += 1;
691    }
692    if ctx.depth >= MAX_NEST {
693        let mut p = Para::new(Kind::Quote);
694        p.push_text(&inner.join(" "), &Fmt::default());
695        out.push(FBlock::Para(p));
696        return;
697    }
698    let mut child = BlockCtx { depth: ctx.depth + 1, level: ctx.level, refs: ctx.refs, budget: ctx.budget, lines_out: None };
699    for mut block in parse_blocks(&inner, &mut child) {
700        if let FBlock::Para(p) = &mut block
701            && p.kind == Kind::Normal
702        {
703            p.kind = Kind::Quote;
704        }
705        out.push(block);
706    }
707}

709/// Computes columns and byte count for spaces following a list item marker.
710fn marker_space(ind: usize, marker_len: usize, after: &str) -> (usize, usize) {
711    let mut cols = 0;
712    let mut bytes = 0;
713    let mut rest = after;
714    while let Some(c) = rest.chars().next() {
715        match c {
716            ' ' => {
717                cols += 1;
718                bytes += 1;
719                rest = &rest[1..];
720                if cols >= 4 {
721                    break;
722                }
723            }
724            '\t' => {
725                let col = ind + marker_len + cols;
726                let n = 4 - col % 4;
727                cols += n;
728                bytes += 1;
729                break;
730            }
731            _ => break,
732        }
733    }
734    (cols, bytes)
735}

737/// Parse a list starting at `lines[i]`; returns the index after it.
738fn parse_list(lines: &[String], mut i: usize, ctx: &mut BlockCtx<'_>, out: &mut Vec<FBlock>) -> usize {
739    let Some(first) = lines.get(i).and_then(|l| list_marker(l.trim_start_matches([' ', '\t']))) else { return i + 1 };
740    while let Some(line) = lines.get(i) {
741        let ind = indent(line);
742        if ind >= 4 {
743            break;
744        }
745        let t = line.trim_start_matches([' ', '\t']);
746        let Some(m) = list_marker(t) else { break };
747        if m.ordered != first.ordered || m.ch != first.ch {
748            break;
749        }
750        let item_start = i;
751        let after = t.get(m.len..).unwrap_or("");
752        let (spaces, skip) = if indent(after) > 4 {
753            (1, 1)
754        } else if after.starts_with([' ', '\t']) {
755            marker_space(ind, m.len, after)
756        } else {
757            (0, 0)
758        };
759        let content_indent = if is_blank(after) { ind + m.len + 1 } else { ind + m.len + spaces };
760        let item0 = if is_blank(after) { String::new() } else { after.get(skip.min(after.len())..).unwrap_or("").to_string() };
761        let mut item: Vec<String> = vec![item0];
762        i += 1;
763        while let Some(l) = lines.get(i) {
764            if is_blank(l) {
765                item.push(String::new());
766            } else if indent(l) >= content_indent {
767                item.push(strip_indent(l, content_indent).to_string());
768            } else if item.last().is_some_and(|x| !is_blank(x)) && !interrupts(l) && list_marker(l.trim_start_matches([' ', '\t'])).is_none() {
769                // Lazy continuation of the item's paragraph.
770                item.push(l.trim_start_matches([' ', '\t']).to_string());
771            } else {
772                break;
773            }
774            i += 1;
775        }
776        // Blank lines at the end of an item belong between items.
777        while item.len() > 1 && item.last().is_some_and(String::is_empty) {
778            item.pop();
779        }
780        let li = ListInfo { ordered: first.ordered, level: ctx.level.min(8), start: first.start, ..Default::default() };
781        let inner = if ctx.depth >= MAX_NEST {
782            let mut p = Para::default();
783            p.push_text(&item.join(" "), &Fmt::default());
784            vec![FBlock::Para(p)]
785        } else {
786            let mut child =
787                BlockCtx { depth: ctx.depth + 1, level: ctx.level.saturating_add(1).min(8), refs: ctx.refs, budget: ctx.budget, lines_out: None };
788            parse_blocks(&item, &mut child)
789        };
790        let before_count = out.len();
791        let mut placed = false;
792        if inner.is_empty() {
793            out.push(FBlock::Para(Para { list: Some(li), ..Default::default() }));
794            placed = true;
795        }
796        for b in inner {
797            match b {
798                FBlock::Para(mut p) if p.list.is_none() => {
799                    if !placed {
800                        p.list = Some(li);
801                        placed = true;
802                        out.push(FBlock::Para(p));
803                    } else if let Some(FBlock::Para(prev)) = out.last_mut()
804                        && prev.list.is_some_and(|x| x.level == li.level)
805                    {
806                        // A continuation paragraph joins the item with a line break.
807                        prev.push_text("\n", &Fmt::default());
808                        prev.inlines.extend(p.inlines);
809                    } else {
810                        p.list = Some(li);
811                        out.push(FBlock::Para(p));
812                    }
813                }
814                FBlock::Para(p) => {
815                    if !placed {
816                        // The item starts with a nested list: give it an empty first line.
817                        out.push(FBlock::Para(Para { list: Some(li), ..Default::default() }));
818                        placed = true;
819                    }
820                    out.push(FBlock::Para(p));
821                }
822                other => {
823                    placed = true;
824                    out.push(other);
825                }
826            }
827        }
828        ctx.record_lines(item_start, out.len() - before_count);
829        // Blank lines between items.
830        while lines.get(i).is_some_and(|l| is_blank(l)) {
831            let next_is_item = lines.get(i + 1).is_some_and(|l| {
832                indent(l) < 4 && list_marker(l.trim_start_matches([' ', '\t'])).is_some_and(|m| m.ordered == first.ordered && m.ch == first.ch)
833            });
834            if !next_is_item {
835                return i;
836            }
837            i += 1;
838        }
839    }
840    i
841}
