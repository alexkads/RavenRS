29fn push_text(nodes: &mut Vec<Node>, s: &str, f: &Fmt) {
30    if s.is_empty() {
31        return;
32    }
33    if let Some(Node::Text(t, pf)) = nodes.last_mut()
34        && pf == f
35    {
36        t.push_str(s);
37        return;
38    }
39    nodes.push(Node::Text(s.to_string(), f.clone()));
40}

42/// The closing `]` matching the `[` at byte `open` (escapes and code spans respected). Gives up
43/// after looking `max` bytes ahead. Records in `known` every bracket pair it passes, and every `[`
44/// still open at the end of the text: a scan from one of those would take the same path.
45fn matching_bracket(s: &str, open: usize, max: usize, ticks: &Backticks, known: &mut HashMap<usize, Option<usize>>) -> Option<usize> {
46    let b = s.as_bytes();
47    let mut opens: Vec<usize> = Vec::new();
48    let mut i = open;
49    while let Some(&c) = b.get(i) {
50        if i - open > max {
51            return None;
52        }
53        match c {
54            b'\\' => i += 1,
55            b'`' => {
56                let n = b.get(i..).map(|r| r.iter().take_while(|x| **x == b'`').count()).unwrap_or(1);
57                if let Some(end) = ticks.find(i + n, n) {
58                    i = end + n - 1;
59                } else {
60                    i += n - 1;
61                }
62            }
63            b'[' => opens.push(i),
64            b']' => {
65                if let Some(o) = opens.pop() {
66                    known.insert(o, Some(i));
67                }
68                if opens.is_empty() {
69                    return Some(i);
70                }
71            }
72            _ => {}
73        }
74        i += 1;
75    }
76    for o in opens {
77        known.insert(o, None);
78    }
79    None
80}

82/// Where a text's runs of backticks start, by run length. Finding the run that closes a code span
83/// is then a lookup: scanning the rest of the text for it, again for every run nothing closes
84/// (each of a different length) and from every link bracket over them, was superlinear.
85struct Backticks(HashMap<usize, Vec<usize>>);
86
87impl Backticks {
88    fn new(s: &str) -> Self {
89        let b = s.as_bytes();
90        let mut runs: HashMap<usize, Vec<usize>> = HashMap::new();
91        let mut i = 0;
92        while let Some(&c) = b.get(i) {
93            if c == b'`' {
94                let k = b.get(i..).map(|r| r.iter().take_while(|x| **x == b'`').count()).unwrap_or(1);
95                runs.entry(k).or_default().push(i);
96                i += k;
97            } else {
98                i += 1;
99            }
100        }
101        Self(runs)
102    }
103
104    /// Byte index of a run of exactly `n` backticks at or after `from` (which is not inside a run).
105    fn find(&self, from: usize, n: usize) -> Option<usize> {
106        let starts = self.0.get(&n)?;
107        starts.get(starts.partition_point(|&p| p < from)).copied()
108    }
109}

111struct LinkScan<'a> {
112    text: &'a str,
113    refs: &'a Refs,
114    ticks: &'a Backticks,
115    known: HashMap<usize, Option<usize>>,
116    budget: usize,
117}
118
119impl<'a> LinkScan<'a> {
120    fn parse_at(&mut self, open: usize) -> Option<(&'a str, Arc<str>, usize)> {
121        let bytes = self.text.as_bytes();
122        let close = match self.known.get(&open).copied() {
123            Some(found) => found,
124            None if self.budget > 0 => {
125                let found = matching_bracket(self.text, open, self.budget, self.ticks, &mut self.known);
126                self.budget = self.budget.saturating_sub(found.unwrap_or(self.text.len()).saturating_sub(open));
127                found
128            }
129            None => None,
130        }?;
131        if self.budget == 0 {
132            return None;
133        }
134        let label_text = self.text.get(open + 1..close)?;
135        if bytes.get(close + 1) == Some(&b'(') {
136            let destination = link_dest(self.text, close + 1);
137            self.budget = self.budget.saturating_sub(destination.as_ref().map_or(self.text.len(), |d| d.1).saturating_sub(close));
138            let (destination, end) = destination?;
139            return Some((label_text, Arc::from(destination), end));
140        }
141        // Full and shortcut reference links use definitions collected before inline parsing.
142        if self.refs.is_empty() {
143            return None;
144        }
145        let (label, end) = match self.text.get(close + 1..).and_then(|rest| rest.strip_prefix('[')) {
146            Some(rest) => {
147                let ahead = within(rest, self.budget);
148                let shut = ahead.find(']');
149                self.budget = self.budget.saturating_sub(shut.map_or(ahead.len(), |n| n + 1));
150                let shut = shut?;
151                let inner = rest.get(..shut)?;
152                let label = if inner.trim().is_empty() { label_text } else { inner };
153                (label, close + 2 + shut + 1)
154            }
155            None => (label_text, close + 1),
156        };
157        // A cached bracket pair still costs a pass over its label. Without charging it,
158        // nested undefined references normalize every suffix again.
159        if label.len() > self.budget {
160            self.budget = 0;
161            return None;
162        }
163        self.budget -= label.len();
164        Some((label_text, self.refs.get(&ref_label(label))?.clone(), end))
165    }
166}

168/// Parse `(dest "title")` at byte `at` (which holds `(`): (destination, index after `)`).
169fn link_dest(s: &str, at: usize) -> Option<(String, usize)> {
170    let rest = s.get(at..)?.strip_prefix('(')?;
171    let mut i = at + 1;
172    let ws = rest.len() - rest.trim_start().len();
173    i += ws;
174    let r = s.get(i..)?;
175    let (dest, used) = if let Some(inner) = r.strip_prefix('<') {
176        // The first `>` not escaped by a backslash ends it; a line break first means no link.
177        let b = inner.as_bytes();
178        let mut k = 0;
179        let end = loop {
180            match b.get(k)? {
181                b'\\' => k += 1,
182                b'>' => break k,
183                b'\n' => return None,
184                _ => {}
185            }
186            k += 1;
187        };
188        (inner.get(..end)?.to_string(), end + 2)
189    } else {
190        let mut depth = 0i32;
191        let mut end = r.len();
192        let mut chars = r.char_indices().peekable();
193        while let Some((k, c)) = chars.next() {
194            match c {
195                '\\' => {
196                    chars.next();
197                }
198                '(' => depth += 1,
199                ')' if depth == 0 => {
200                    end = k;
201                    break;
202                }
203                ')' => depth -= 1,
204                c if c.is_whitespace() => {
205                    end = k;
206                    break;
207                }
208                _ => {}
209            }
210        }
211        (r.get(..end)?.to_string(), end)
212    };
213    i += used;
214    let r = s.get(i..)?;
215    let ws = r.len() - r.trim_start().len();
216    i += ws;
217    let r = s.get(i..)?;
218    if let Some(q) = r.chars().next().filter(|c| matches!(c, '"' | '\'' | '(')) {
219        let close = if q == '(' { ')' } else { q };
220        let end = r.get(1..)?.find(close)?;
221        i += end + 2;
222        let r = s.get(i..)?;
223        i += r.len() - r.trim_start().len();
224    }
225    if s.get(i..)?.starts_with(')') { Some((unescape(&dest), i + 1)) } else { None }
226}

228fn unescape(s: &str) -> String {
229    let mut out = String::with_capacity(s.len());
230    let mut chars = s.chars().peekable();
231    while let Some(c) = chars.next() {
232        if c == '\\'
233            && let Some(n) = chars.peek().copied().filter(char::is_ascii_punctuation)
234        {
235            out.push(n);
236            chars.next();
237        } else {
238            out.push(c);
239        }
240    }
241    out
242}

244/// The value of attribute `name` in an HTML tag's text.
245fn tag_attr(tag: &str, name: &str) -> Option<String> {
246    let lower = tag.to_ascii_lowercase();
247    let mut from = 0;
248    while let Some(k) = lower.get(from..)?.find(name) {
249        let at = from + k;
250        let before = lower.get(..at)?.chars().next_back();
251        let after = lower.get(at + name.len()..)?.trim_start();
252        if before.is_some_and(char::is_whitespace) && after.starts_with('=') {
253            let v = after.get(1..)?.trim_start();
254            let q = v.chars().next()?;
255            return Some(if q == '"' || q == '\'' {
256                let inner = v.get(1..)?;
257                let off = tag.len() - inner.len();
258                let end = inner.find(q)?;
259                tag.get(off..off + end)?.to_string()
260            } else {
261                let end = v.find(|c: char| c.is_whitespace() || c == '>').unwrap_or(v.len());
262                let off = tag.len() - v.len();
263                tag.get(off..off + end)?.to_string()
264            });
265        }
266        from = at + name.len();
267    }
268    None
269}

271fn tokenize(s: &str, base: &Fmt, depth: usize, refs: &Refs) -> Vec<Node> {
272    let mut nodes: Vec<Node> = Vec::new();
273    // Bytes the link/image scans and reference-label lookups may read in total. Unmatched
274    // brackets and nested undefined references would otherwise rescan the rest of the text,
275    // which is quadratic on hostile input. Once the budget is gone, further brackets are plain
276    // text. A scan pays for what it reads; earlier bracket matches are looked up in `known`.
277    let ticks = Backticks::new(s);
278    let mut links = LinkScan { text: s, refs, ticks: &ticks, known: HashMap::new(), budget: s.len().saturating_mul(8).saturating_add(8192) };
279    let b = s.as_bytes();
280    let mut i = 0usize;
281    let mut text_start = 0usize;
282    macro_rules! flush {
283        () => {
284            if text_start < i {
285                push_text(&mut nodes, s.get(text_start..i).unwrap_or(""), base);
286            }
287        };
288    }
289    while let Some(&c) = b.get(i) {
290        match c {
291            b'\\' => {
292                flush!();
293                match s.get(i + 1..).and_then(|r| r.chars().next()) {
294                    Some(n) if n.is_ascii_punctuation() => {
295                        push_text(&mut nodes, &n.to_string(), base);
296                        i += 2;
297                    }
298                    _ => {
299                        push_text(&mut nodes, "\\", base);
300                        i += 1;
301                    }
302                }
303                text_start = i;
304            }
305            b'`' => {
306                flush!();
307                let (code, fmt, end) = code_span(s, i, base, &ticks);
308                push_text(&mut nodes, &code, &fmt);
309                i = end;
310                text_start = i;
311            }
312            b'*' | b'_' | b'~' => {
313                flush!();
314                let ch = c as char;
315                let n = b.get(i..).map(|r| r.iter().take_while(|x| **x == c).count()).unwrap_or(1);
316                let before = s.get(..i).and_then(|r| r.chars().next_back()).unwrap_or(' ');
317                let after = s.get(i + n..).and_then(|r| r.chars().next()).unwrap_or(' ');
318                let lf = !after.is_whitespace() && (!is_punct(after) || before.is_whitespace() || is_punct(before));
319                let rf = !before.is_whitespace() && (!is_punct(before) || after.is_whitespace() || is_punct(after));
320                let (open, close) = match ch {
321                    '_' => (lf && (!rf || is_punct(before)), rf && (!lf || is_punct(after))),
322                    '~' if n != 2 => (false, false),
323                    _ => (lf, rf),
324                };
325                if open || close {
326                    nodes.push(Node::Delim { ch, orig: n, left: n, open, close, opens: Vec::new(), closes: Vec::new() });
327                } else {
328                    push_text(&mut nodes, s.get(i..i + n).unwrap_or(""), base);
329                }
330                i += n;
331                text_start = i;
332            }
333            b'!' | b'[' if depth < MAX_NEST => {
334                let img = c == b'!';
335                let open = if img { i + 1 } else { i };
336                let parsed = if b.get(open) == Some(&b'[') { links.parse_at(open) } else { None };
337                match parsed {
338                    Some((inner, dest, end)) if img => {
339                        flush!();
340                        let alt: String = parse_inlines(inner, &Fmt::default(), depth + 1, refs)
341                            .iter()
342                            .filter_map(|x| if let Inline::Text(t, _) = x { Some(t.as_str()) } else { None })
343                            .collect();
344                        match model::data_uri(&dest).and_then(|d| make_img(d, None, None, &alt)) {
345                            Some(im) => nodes.push(Node::Inl(Inline::Image(im))),
346                            None => push_text(&mut nodes, &alt, base),
347                        }
348                        i = end;
349                        text_start = i;
350                    }
351                    Some((inner, dest, end)) if base.link.is_none() => {
352                        flush!();
353                        let mut f = base.clone();
354                        // A destination too long to keep leaves its text unlinked.
355                        f.link = (dest.len() <= model::MAX_LINK).then_some(dest);
356                        if f.link.is_none() {
357                            model::dropped(model::Dropped::Link);
358                        }
359                        for x in parse_inlines(inner, &f, depth + 1, refs) {
360                            nodes.push(Node::Inl(x));
361                        }
362                        i = end;
363                        text_start = i;
364                    }
365                    _ => i += 1,
366                }
367            }
368            b'<' => {
369                if let Some((end, parsed)) = angle_inlines(s, i, base) {
370                    flush!();
371                    nodes.extend(parsed);
372                    i = end;
373                    text_start = end;
374                } else {
375                    i += 1;
376                }
377            }
378            b'&' => {
379                let rest = s.get(i..).unwrap_or("");
380                match within(rest, 33).find(';').and_then(|k| decode_entity(rest.get(1..k)?).map(|ch| (ch, k))) {
381                    Some((ch, k)) => {
382                        flush!();
383                        push_text(&mut nodes, &ch, base);
384                        i += k + 1;
385                        text_start = i;
386                    }
387                    None => i += 1,
388                }
389            }
390            _ => i += 1,
391        }
392    }
393    flush!();
394    nodes
395}

397fn angle_inlines(s: &str, start: usize, base: &Fmt) -> Option<(usize, Vec<Node>)> {
398    if let Some(pos) = s.get(start..).and_then(|rest| rest.strip_prefix("<!--")).and_then(|rest| rest.find("-->")) {
399        return Some((start + 4 + pos + 3, Vec::new()));
400    }
401    let rest = s.get(start + 1..).unwrap_or("");
402    let tag_end = within(rest, MAX_TAG).find('>')?;
403    let inner = rest.get(..tag_end)?;
404    let lower = inner.to_ascii_lowercase();
405    let end = start + inner.len() + 2;
406    let mut nodes = Vec::new();
407    if !inner.is_empty()
408        && !inner.contains([' ', '<', '\n'])
409        && (inner.contains("://") || lower.starts_with("mailto:") || (inner.contains('@') && !inner.contains('/')))
410    {
411        let mut fmt = base.clone();
412        fmt.link = Some(if inner.contains('@') && !inner.contains(':') { format!("mailto:{inner}").into() } else { inner.into() });
413        push_text(&mut nodes, inner, &fmt);
414        Some((end, nodes))
415    } else if matches!(lower.trim_end_matches('/').trim(), "br") {
416        push_text(&mut nodes, "\n", base);
417        Some((end, nodes))
418    } else if lower.starts_with("a ") && rest.get(inner.len() + 1..).is_some_and(|tail| tail.to_ascii_lowercase().starts_with("</a>")) {
419        if let Some(name) = tag_attr(inner, "id").or_else(|| tag_attr(inner, "name")) {
420            nodes.push(Node::Inl(Inline::Anchor(name)));
421        }
422        Some((end + 4, nodes))
423    } else {
424        None
425    }
426}

428fn code_span(s: &str, start: usize, base: &Fmt, ticks: &Backticks) -> (String, Fmt, usize) {
429    let bytes = s.as_bytes();
430    let count = bytes.get(start..).map(|tail| tail.iter().take_while(|byte| **byte == b'`').count()).unwrap_or(1);
431    let Some(end) = ticks.find(start + count, count) else {
432        return (s.get(start..start + count).unwrap_or("").to_string(), base.clone(), start + count);
433    };
434    let mut code = s.get(start + count..end).unwrap_or("").replace('\n', " ");
435    if code.len() >= 2 && code.starts_with(' ') && code.ends_with(' ') && !code.trim().is_empty() {
436        code = code[1..code.len() - 1].to_string();
437    }
438    let mut fmt = base.clone();
439    fmt.code = true;
440    (code, fmt, end + count)
441}

446/// The first `n` bytes of `s` (fewer if that would cut a character), so a search for a closing
447/// character stays short however long the text is.
448pub(super) fn within(s: &str, n: usize) -> &str {
449    let mut end = n.min(s.len());
450    while !s.is_char_boundary(end) {
451        end -= 1;
452    }
453    s.get(..end).unwrap_or("")
454}

456fn emph_index(ch: char) -> usize {
457    match ch {
458        '*' => 0,
459        '_' => 1,
460        _ => 2,
461    }

464/// The CommonMark "process emphasis" pass over the delimiter runs, then flattening into spans.
465fn emit(mut nodes: Vec<Node>) -> Vec<Inline> {
466    // Delimiter stack of node indexes; bottoms per (char, closer length % 3, closer can open).
467    let mut stack: Vec<usize> = Vec::new();
468    let mut bottom = [[[None::<usize>; 2]; 3]; 3];
469    for c in 0..nodes.len() {
470        let (ch, corig, copen, cclose) = match nodes.get(c) {
471            Some(Node::Delim { ch, orig, open, close, .. }) => (*ch, *orig, *open, *close),
472            _ => continue,
473        };
474        if cclose {
475            loop {
476                let cleft = match nodes.get(c) {
477                    Some(Node::Delim { left, .. }) => *left,
478                    _ => 0,
479                };
480                if cleft == 0 {
481                    break;
482                }
483                let bot = bottom.get(emph_index(ch)).and_then(|x| x.get(corig % 3)).and_then(|x| x.get(copen as usize)).copied().flatten();
484                let mut found = None;
485                for (k, &o) in stack.iter().enumerate().rev() {
486                    if bot.is_some_and(|b| o <= b) {
487                        break;
488                    }
489                    if let Some(Node::Delim { ch: och, orig: oorig, left: oleft, open: oopen, close: oclose, .. }) = nodes.get(o) {
490                        if *och != ch || !*oopen || *oleft == 0 {
491                            continue;
492                        }
493                        if ch == '~' && (*oleft < 2 || cleft < 2) {
494                            continue;
495                        }
496                        let odd = (*oclose || copen) && (oorig + corig) % 3 == 0 && !(oorig % 3 == 0 && corig % 3 == 0);
497                        if odd {
498                            continue;
499                        }
500                        found = Some((k, o, *oleft));
501                        break;
502                    }
503                }
504                let Some((k, o, oleft)) = found else {
505                    if let Some(slot) = bottom.get_mut(emph_index(ch)).and_then(|x| x.get_mut(corig % 3)).and_then(|x| x.get_mut(copen as usize)) {
506                        *slot = Some(c.saturating_sub(1));
507                    }
508                    break;
509                };
510                let (used, kind) = if ch == '~' {
511                    (2, Emph::Strike)
512                } else if oleft >= 2 && cleft >= 2 {
513                    (2, Emph::Strong)
514                } else {
515                    (1, Emph::Em)
516                };
517                if let Some(Node::Delim { left, opens, .. }) = nodes.get_mut(o) {
518                    *left -= used;
519                    opens.push(kind);
520                }
521                if let Some(Node::Delim { left, closes, .. }) = nodes.get_mut(c) {
522                    *left -= used;
523                    closes.push(kind);
524                }
525                // Delimiters between opener and closer can no longer match.
526                stack.truncate(k + 1);
527                if oleft - used == 0 {
528                    stack.truncate(k);
529                }
530            }
531        }
532        let left = match nodes.get(c) {
533            Some(Node::Delim { left, .. }) => *left,
534            _ => 0,
535        };
536        if copen && left > 0 {
537            stack.push(c);
538        }
539    }
540    // Flatten.
541    let mut out: Vec<Inline> = Vec::new();
542    let (mut em, mut strong, mut strike) = (0usize, 0usize, 0usize);
543    let push = |out: &mut Vec<Inline>, t: &str, f: &Fmt, em: usize, strong: usize, strike: usize| {
544        if t.is_empty() {
545            return;
546        }
547        let mut f = f.clone();
548        f.italic |= em > 0;
549        f.bold |= strong > 0;
550        f.strike |= strike > 0;
551        if let Some(Inline::Text(pt, pf)) = out.last_mut()
552            && *pf == f
553        {
554            pt.push_str(t);
555            return;
556        }
557        out.push(Inline::Text(t.to_string(), f));
558    };
559    // Delimiter literals take the formatting (link, …) of the text before them: the last text
560    // node, tracked here rather than searched for backwards from every delimiter (quadratic on
561    // a paragraph of delimiters).
562    let mut last_text: Option<&Fmt> = None;
563    for n in &nodes {
564        match n {
565            Node::Text(t, f) => {
566                push(&mut out, t, f, em, strong, strike);
567                last_text = Some(f);
568            }
569            Node::Inl(Inline::Text(t, f)) => push(&mut out, t, f, em, strong, strike),
570            Node::Inl(other) => out.push(other.clone()),
571            Node::Delim { ch, left, opens, closes, .. } => {
572                for e in closes {
573                    match e {
574                        Emph::Em => em = em.saturating_sub(1),
575                        Emph::Strong => strong = strong.saturating_sub(1),
576                        Emph::Strike => strike = strike.saturating_sub(1),
577                    }
578                }
579                if *left > 0 {
580                    let lit: String = std::iter::repeat_n(*ch, *left).collect();
581                    let f = last_text.map_or_else(Fmt::default, |f| Fmt { bold: false, italic: false, strike: false, ..f.clone() });
582                    push(&mut out, &lit, &f, em, strong, strike);
583                }
584                for e in opens {
585                    match e {
586                        Emph::Em => em += 1,
587                        Emph::Strong => strong += 1,
588                        Emph::Strike => strike += 1,
589                    }
590                }
591            }
592        }
593    }
594    out
595}
