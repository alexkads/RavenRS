1//! WordCraft import/export for the "other" text formats: plain text, Markdown (CommonMark subset
2//! plus GFM tables and strikethrough), HTML (and MHT web archives, read-only), RTF, OpenDocument
3//! Text and LaTeX — and, read-only, spreadsheet cells for mail-merge recipient lists ([`sheet`]:
4//! .xlsx/.xlsm/.xls/.ods).
5//!
6//! Every format maps to a small flow model ([`model::Flow`]: paragraphs with a kind, list
7//! membership, alignment and formatted spans; tables of cells), which [`model::to_doc`] and
8//! [`model::from_doc`] convert to and from the full [`Document`]. All parsers are written from
9//! the public specifications, are lenient by design and never panic on hostile input: nesting is
10//! depth-limited and sizes are capped.
11#![deny(clippy::unwrap_used, clippy::expect_used, clippy::panic, clippy::unimplemented, clippy::todo, clippy::unreachable)]

35/// Import `bytes` in the format named by `ext` (no dot, any case). `None` = not our format.
36pub fn import(ext: &str, bytes: &[u8]) -> Option<Result<Document, String>> {
37    let ext = ext.trim_start_matches('.').to_ascii_lowercase();
38    let r = match ext.as_str() {
39        "txt" | "text" => Ok(txt::import(bytes)),
40        "md" | "markdown" => Ok(markdown::import(&txt::decode(bytes))),
41        "html" | "htm" | "xhtml" => Ok(html::import_page(&html::decode(bytes), &|_| None)),
42        "mht" | "mhtml" => Ok(mht::import(bytes)),
43        "rtf" => rtf::import(bytes),
44        "odt" | "ott" => odt::import(bytes),
45        "fodt" => odt::import_flat(bytes),
46        "tex" | "latex" | "ltx" => Ok(latex::import(bytes)),
47        _ => return None,
48    };
49    Some(r.map(|mut d| {
50        d.ensure_nonempty();
51        d
52    }))
53}

55/// [`import`] onto `page` (its size and margins) for the formats whose files give no page size:
56/// plain text, Markdown, HTML and MHT start on it, and their tables size to it (#1381). The other
57/// formats keep the page their file gives.
58pub fn import_on(ext: &str, bytes: &[u8], page: &crate::doc::SectionProps) -> Option<Result<Document, String>> {
59    let ext = ext.trim_start_matches('.').to_ascii_lowercase();
60    let mut doc = match ext.as_str() {
61        "odt" | "ott" => return Some(odt::import_on(bytes, page)),
62        "fodt" => return Some(odt::import_flat_on(bytes, page)),
63        "txt" | "text" => txt::import_on(bytes, page),
64        "md" | "markdown" => markdown::import_on(&txt::decode(bytes), page),
65        "html" | "htm" | "xhtml" => html::import_on(&html::decode(bytes), &|_| None, page),
66        "mht" | "mhtml" => mht::import_on(bytes, page),
67        _ => return import(&ext, bytes),
68    };
69    doc.ensure_nonempty();
70    Some(Ok(doc))
71}

73/// Export `doc` in the format named by `ext`. `None` = not our format.
74pub fn export(ext: &str, doc: &Document) -> Option<Result<Vec<u8>, String>> {
75    let ext = ext.trim_start_matches('.').to_ascii_lowercase();
76    Some(match ext.as_str() {
77        "txt" | "text" => Ok(txt::export(doc)),
78        "md" | "markdown" => Ok(markdown::export(doc).into_bytes()),
79        "html" | "htm" | "xhtml" => Ok(html::export(doc).into_bytes()),
80        "rtf" => Ok(rtf::export(doc).into_bytes()),
81        "odt" => odt::export(doc),
82        "ott" => odt::export_template(doc),
83        "fodt" => odt::export_flat(doc),
84        "tex" | "latex" | "ltx" => Ok(latex::export(doc).into_bytes()),
85        _ => return None,
86    })
87}

89/// How [`export_with`] writes a copy.
90#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
91pub struct ExportOptions {
92    /// Print Markup: HTML, Markdown and plain text end with the document's comments (who wrote
93    /// each, the text it comments on, what it says, its replies). Other formats keep comments
94    /// their own way or not at all.
95    pub comments: bool,
96}
97
98/// [`export`] with options.
99pub fn export_with(ext: &str, doc: &Document, opts: &ExportOptions) -> Option<Result<Vec<u8>, String>> {
100    let out = export(ext, doc)?;
101    if !opts.comments {
102        return Some(out);
103    }
104    let ext = ext.trim_start_matches('.').to_ascii_lowercase();
105    Some(out.map(|bytes| comments::append(&ext, doc, bytes)))
106}
