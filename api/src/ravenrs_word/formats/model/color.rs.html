3fn hsl_to_rgb(h: f64, s: f64, l: f64) -> Rgb {
4    let h = (h / 360.0).rem_euclid(1.0);
5    let s = s.clamp(0.0, 1.0);
6    let l = l.clamp(0.0, 1.0);
7    if s == 0.0 {
8        let v = (l * 255.0).round().clamp(0.0, 255.0) as u8;
9        return Rgb(v, v, v);
10    }
11    let q = if l < 0.5 { l * (1.0 + s) } else { l + s - l * s };
12    let p = 2.0 * l - q;
13    let hue_to_rgb = |mut t: f64| -> f64 {
14        if t < 0.0 {
15            t += 1.0;
16        }
17        if t > 1.0 {
18            t -= 1.0;
19        }
20        if t < 1.0 / 6.0 {
21            return p + (q - p) * 6.0 * t;
22        }
23        if t < 1.0 / 2.0 {
24            return q;
25        }
26        if t < 2.0 / 3.0 {
27            return p + (q - p) * (2.0 / 3.0 - t) * 6.0;
28        }
29        p
30    };
31    let r = (hue_to_rgb(h + 1.0 / 3.0) * 255.0).round().clamp(0.0, 255.0) as u8;
32    let g = (hue_to_rgb(h) * 255.0).round().clamp(0.0, 255.0) as u8;
33    let b = (hue_to_rgb(h - 1.0 / 3.0) * 255.0).round().clamp(0.0, 255.0) as u8;
34    Rgb(r, g, b)
35}

37/// Composite `rgb` over the white page a paste lands on, at CSS alpha `a` (0..1): `rgba(0, 0, 0,
38/// 0.45)` reads like the rgb(140, 140, 140) it looks like on the site, instead of solid black.
39fn over_white(r: u8, g: u8, b: u8, a: f32) -> Rgb {
40    let mix = |c: u8| (c as f32 * a + 255.0 * (1.0 - a)).round().clamp(0.0, 255.0) as u8;
41    Rgb(mix(r), mix(g), mix(b))
42}

44/// The alpha of a CSS colour's `/ alpha` or 4th-value part, where present: 1.0 (opaque) otherwise.
45fn css_alpha(after: Option<&str>) -> f32 {
46    let Some(a) = after else { return 1.0 };
47    let a = a.trim();
48    // CSS Colour 4's `none` is a missing component, and a missing alpha is transparent.
49    if a.eq_ignore_ascii_case("none") {
50        return 0.0;
51    }
52    // A value that is not a finite number (`nan`, `inf`) is no alpha at all: clamp would keep
53    // NaN, and `over_white` would turn the colour black.
54    let parse = |t: &str| t.parse::<f32>().ok().filter(|f| f.is_finite());
55    if let Some(p) = a.strip_suffix('%') {
56        return parse(p.trim()).map(|f| f.clamp(0.0, 100.0) / 100.0).unwrap_or(1.0);
57    }
58    parse(a).map(|f| f.clamp(0.0, 1.0)).unwrap_or(1.0)
59}

61/// Parse a CSS / HTML colour: `#rgb`, `#rgba`, `#rrggbb`, `#rrggbbaa`, `rgb(...)`, `hsl(...)` or a basic name.
62/// A colour with alpha is composited over white: the model keeps opaque colours, and a paste lands
63/// on a white page.
64pub fn parse_color(s: &str) -> Option<Rgb> {
65    let s = s.trim().to_ascii_lowercase();
66    if let Some(h) = s.strip_prefix('#') {
67        if (h.len() == 3 || h.len() == 4) && h.is_ascii() {
68            let d = |i: usize| h.get(i..i + 1).and_then(|x| u8::from_str_radix(x, 16).ok()).map(|v| v * 17);
69            let (r, g, b) = (d(0)?, d(1)?, d(2)?);
70            // A 4th digit is the alpha, composited over the page like `rgba(...)`'s is.
71            let a = if h.len() == 4 { d(3)? as f32 / 255.0 } else { 1.0 };
72            return Some(over_white(r, g, b, a));
73        }
74        if h.len() == 8 && h.bytes().all(|b| b.is_ascii_hexdigit()) {
75            let p = |i: usize| h.get(i..i + 2).and_then(|x| u8::from_str_radix(x, 16).ok());
76            return Some(over_white(p(0)?, p(2)?, p(4)?, p(6)? as f32 / 255.0));
77        }
78        return Rgb::parse(h);
79    }
80    if let Some(inner) = s.strip_prefix("rgb(").or_else(|| s.strip_prefix("rgba(")) {
81        let inner = inner.trim_end_matches(')').trim();
82        let (rgb_part, alpha_part) = inner.split_once('/').map_or((inner, None), |(a, b)| (a, Some(b)));
83        // Legacy comma syntax carries the alpha as a 4th value (`rgba(0, 0, 0, 0.45)`).
84        let parts: Vec<&str> = if rgb_part.contains(',') { rgb_part.split(',').collect() } else { rgb_part.split_whitespace().collect() };
85        let alpha = match alpha_part {
86            Some(a) => css_alpha(Some(a)),
87            None if parts.len() >= 4 => css_alpha(Some(parts[3])),
88            _ => 1.0,
89        };
90        let v: Vec<u8> = parts
91            .iter()
92            .take(3)
93            .filter_map(|x| {
94                let x = x.trim();
95                if let Some(p) = x.strip_suffix('%') {
96                    p.trim().parse::<f32>().ok().map(|f| (f.clamp(0.0, 100.0) * 2.55).round() as u8)
97                } else {
98                    x.parse::<f32>().ok().map(|f| f.clamp(0.0, 255.0).round() as u8)
99                }
100            })
101            .collect();
102        if let [r, g, b] = v[..] {
103            return Some(over_white(r, g, b, alpha));
104        }
105        return None;
106    }
107    if let Some(inner) = s.strip_prefix("hsl(").or_else(|| s.strip_prefix("hsla(")) {
108        let inner = inner.trim_end_matches(')').trim();
109        let (hsl_part, alpha_part) = inner.split_once('/').map_or((inner, None), |(a, b)| (a, Some(b)));
110        let parts: Vec<&str> = if hsl_part.contains(',') { hsl_part.split(',').collect() } else { hsl_part.split_whitespace().collect() };
111        let alpha = match alpha_part {
112            Some(a) => css_alpha(Some(a)),
113            None if parts.len() >= 4 => css_alpha(Some(parts[3])),
114            _ => 1.0,
115        };
116        if parts.len() < 3 {
117            return None;
118        }
119        let h_str = parts[0].trim();
120        let h: f64 = if let Some(deg) = h_str.strip_suffix("deg") {
121            deg.trim().parse().ok()?
122        } else if let Some(rad) = h_str.strip_suffix("rad") {
123            rad.trim().parse::<f64>().ok()? * 180.0 / std::f64::consts::PI
124        } else if let Some(turn) = h_str.strip_suffix("turn") {
125            turn.trim().parse::<f64>().ok()? * 360.0
126        } else {
127            h_str.parse().ok()?
128        };
129        let parse_pct = |x: &str| -> Option<f64> {
130            let x = x.trim();
131            if let Some(p) = x.strip_suffix('%') {
132                Some((p.trim().parse::<f64>().ok()? / 100.0).clamp(0.0, 1.0))
133            } else {
134                Some(x.parse::<f64>().ok()?.clamp(0.0, 1.0))
135            }
136        };
137        let s = parse_pct(parts[1])?;
138        let l = parse_pct(parts[2])?;
139        let Rgb(r, g, b) = hsl_to_rgb(h, s, l);
140        return Some(over_white(r, g, b, alpha));
141    }
142    Some(match s.as_str() {
143        "black" => Rgb(0, 0, 0),
144        "white" => Rgb(255, 255, 255),
145        "red" => Rgb(255, 0, 0),
146        "green" => Rgb(0, 128, 0),
147        "lime" => Rgb(0, 255, 0),
148        "blue" => Rgb(0, 0, 255),
149        "yellow" => Rgb(255, 255, 0),
150        "cyan" | "aqua" => Rgb(0, 255, 255),
151        "magenta" | "fuchsia" => Rgb(255, 0, 255),
152        "gray" | "grey" => Rgb(128, 128, 128),
153        "silver" => Rgb(192, 192, 192),
154        "maroon" => Rgb(128, 0, 0),
155        "olive" => Rgb(128, 128, 0),
156        "navy" => Rgb(0, 0, 128),
157        "purple" => Rgb(128, 0, 128),
158        "teal" => Rgb(0, 128, 128),
159        "orange" => Rgb(255, 165, 0),
160        _ => return None,
161    })

164#[cfg(test)]
165mod tests {
166    use super::*;
167
168    #[test]
169    fn eight_digit_hex_rejects_signs() {
170        assert_eq!(parse_color("#+1+2+3+4"), None);
171        // #11223344 is alpha 0x44 over white.
172        assert_eq!(parse_color("#11223344"), Some(Rgb(192, 196, 201)));
173    }
174
175    #[test]
176    fn colors_with_alpha_composite_over_white() {
177        // The secondary text colour of dark-theme sites: solid black would read as the site's
178        // own emphasis, white is invisible on the page; the composite is what the eye saw.
179        assert_eq!(parse_color("rgba(0, 0, 0, 0.45)"), Some(Rgb(140, 140, 140)));
180        assert_eq!(parse_color("rgb(0 0 0 / 0.45)"), Some(Rgb(140, 140, 140)));
181        assert_eq!(parse_color("rgba(0, 0, 0, 0)"), Some(Rgb(255, 255, 255)));
182        assert_eq!(parse_color("rgba(255, 0, 0, 1)"), Some(Rgb(255, 0, 0)));
183        assert_eq!(parse_color("rgba(255, 0, 0, 45%)"), Some(Rgb(255, 140, 140)));
184        assert_eq!(parse_color("hsla(120, 100%, 50%, 0.5)"), Some(Rgb(128, 255, 128)));
185        // A 4-digit hex's last digit is the alpha, too: #0008 is black at 8/15 over white.
186        assert_eq!(parse_color("#0008"), Some(Rgb(119, 119, 119)));
187        // `none` is a missing alpha, and a missing alpha is transparent, not the opaque default
188        // an omitted alpha takes.
189        assert_eq!(parse_color("rgb(0 0 0 / none)"), Some(Rgb(255, 255, 255)));
190        assert_eq!(parse_color("rgb(0 0 0)"), Some(Rgb(0, 0, 0)));
191        // An alpha that is not a finite number is no alpha: the colour stays opaque, not black.
192        assert_eq!(parse_color("rgba(0, 0, 0, nan)"), Some(Rgb(0, 0, 0)));
193        assert_eq!(parse_color("rgba(0, 0, 0, inf)"), Some(Rgb(0, 0, 0)));
194        assert_eq!(parse_color("rgba(0, 0, 0, nan%)"), Some(Rgb(0, 0, 0)));
195    }
