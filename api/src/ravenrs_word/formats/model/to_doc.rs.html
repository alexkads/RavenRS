13pub fn char_props(f: &Fmt) -> CharProps {
14    let on = |b: bool| if b { Some(true) } else { None };
15    CharProps {
16        bold: on(f.bold),
17        italic: on(f.italic),
18        underline: if f.underline { Some(Underline::Single) } else { None },
19        strike: on(f.strike),
20        vert_align: if f.sup {
21            Some(VertAlign::Superscript)
22        } else if f.sub {
23            Some(VertAlign::Subscript)
24        } else {
25            None
26        },
27        link: f.link.as_deref().filter(|l| !l.is_empty()).map(str::to_string),
28        color: f.color.map(TextColor::Rgb),
29        size: f.size.filter(|s| s.is_finite() && *s >= 1.0 && *s <= 1638.0),
30        font: if f.code { Some(MONO_FONT.to_string()) } else { f.font.as_deref().filter(|s| !s.is_empty()).map(str::to_string) },
31        shading: f.background,
32        ..Default::default()
33    }
34    .with_extra(f)
35}

42impl WithExtra for CharProps {
43    fn with_extra(mut self, f: &Fmt) -> CharProps {
44        let Some(x) = f.extra.as_deref() else { return self };
45        if f.underline
46            && let Some(u) = x.underline
47        {
48            self.underline = Some(u);
49        }
50        if f.underline {
51            self.underline_color = x.underline_color;
52        }
53        // Turned off on the text, over what the style of its paragraph or span turns on.
54        if !f.bold && x.bold == Some(false) {
55            self.bold = Some(false);
56        }
57        if !f.italic && x.italic == Some(false) {
58            self.italic = Some(false);
59        }
60        if !f.strike && x.strike == Some(false) {
61            // A double strike under it goes too.
62            (self.strike, self.double_strike) = (Some(false), Some(false));
63        }
64        if !f.underline && x.underline == Some(Underline::None) {
65            self.underline = Some(Underline::None);
66        }
67        if x.double_strike == Some(true) {
68            (self.strike, self.double_strike) = (None, Some(true));
69        }
70        let finite = |v: Option<f32>, lo: f32, hi: f32| v.filter(|v| v.is_finite()).map(|v| v.clamp(lo, hi));
71        self.style = x.style.as_deref().map(clean_text).filter(|s| !s.is_empty());
72        self.caps = x.caps;
73        self.small_caps = x.small_caps;
74        self.hidden = x.hidden;
75        self.spacing = finite(x.spacing, -1584.0, 1584.0);
76        self.scale = finite(x.scale, 1.0, 600.0);
77        self.kern = finite(x.kern, 0.0, 1638.0);
78        self.emphasis = x.emphasis;
79        self.position = finite(x.position, -1584.0, 1584.0);
80        self.outline = x.outline;
81        self.shadow = x.shadow;
82        self.emboss = x.emboss;
83        self.engrave = x.engrave;
84        self.border = x.border.filter(|b| b.is_visible() && b.width.is_finite()).map(|mut b| {
85            (b.width, b.space) = (b.width.clamp(0.0, 12.0), if b.space.is_finite() { b.space.clamp(0.0, 31.0) } else { 0.0 });
86            b
87        });
88        self.lang = x.lang.as_deref().map(clean_text).filter(|l| !l.is_empty() && l.len() <= 64);
89        let name = |s: &Option<String>, max: usize| s.as_deref().map(clean_text).filter(|v| !v.is_empty() && v.len() <= max);
90        self.font_cs = name(&x.font_cs, super::types::MAX_FONT_NAME);
91        self.font_ea = name(&x.font_ea, super::types::MAX_FONT_NAME);
92        self.size_cs = x.size_cs.filter(|s| s.is_finite()).map(|s| s.clamp(1.0, 1638.0));
93        self.bold_cs = x.bold_cs;
94        self.italic_cs = x.italic_cs;
95        self.lang_bidi = name(&x.lang_bidi, 64);
96        self.lang_ea = name(&x.lang_ea, 64);
97        self
98    }
99}

118/// Append text (without object characters) or one inline object to the end of `p`.
119fn append(p: &mut Paragraph, text: &str, props: &CharProps, obj: Option<InlineObject>) {
120    let text: std::borrow::Cow<str> = if text.contains(OBJ) { text.chars().filter(|c| *c != OBJ).collect::<String>().into() } else { text.into() };
121    let piece = match obj {
122        Some(o) => {
123            p.objects.push(o);
124            OBJ.to_string()
125        }
126        None => text.into_owned(),
127    };
128    if piece.is_empty() {
129        return;
130    }
131    p.text.push_str(&piece);
132    match p.runs.last_mut() {
133        Some(last) if last.props == *props => last.len += piece.len(),
134        _ => p.runs.push(Run { len: piece.len(), props: props.clone() }),
135    }
136}

138/// The paragraph properties of a flow paragraph (its list as instance `list_num`, when given).
139fn para_props(p: &Para, list_num: Option<u32>) -> ParaProps {
140    let mut out = ParaProps {
141        style: match p.kind {
142            Kind::Heading(n) => Some(format!("Heading{}", n.clamp(1, 6))),
143            Kind::Title => Some("Title".into()),
144            Kind::Quote => Some("Quote".into()),
145            Kind::Code => Some(CODE_STYLE.into()),
146            Kind::Rule | Kind::Normal => None,
147        },
148        ..Default::default()
149    };
150    if p.kind == Kind::Rule {
151        out.borders = Some(rule_borders());
152    } else if p.borders.is_some() {
153        out.borders = p.borders;
154    }
155    if let (Some(li), Some(num)) = (p.list, list_num) {
156        if out.style.is_none() {
157            out.style = Some("ListParagraph".into());
158        }
159        out.numbering = Some(NumRef { num, level: li.level.min(8) });
160    }
161    out.align = p.align;
162    let sp = p.spacing;
163    out.space_before = sp.before;
164    out.space_after = sp.after;
165    out.line_spacing = sp.line;
166    out.indent_left = sp.left;
167    out.indent_right = sp.right;
168    out.indent_first = sp.first;
169    if sp.before_auto {
170        out.space_before_auto = Some(true);
171    }
172    if sp.after_auto {
173        out.space_after_auto = Some(true);
174    }
175    if p.page_break {
176        out.page_break_before = Some(true);
177    }
178    if let Some(st) = p.extra.as_ref().and_then(|x| x.style.clone()) {
179        out.style = Some(st);
180    }
181    if let Some(c) = p.extra.as_ref().and_then(|x| x.contextual_spacing) {
182        out.contextual_spacing = Some(c);
183    }
184    if p.extra.as_ref().and_then(|x| x.bidi) == Some(true) {
185        out.bidi = Some(true);
186    }
187    if let Some(x) = p.extra.as_ref().filter(|x| x.drop_cap.is_some_and(|l| l > 1)) {
188        (out.drop_cap, out.drop_cap_distance) =
189            (x.drop_cap.map(|l| l.min(10)), x.drop_cap_distance.filter(|d| d.is_finite()).map(|d| d.clamp(0.0, 1584.0)));
190    }
191    if let Some(bg) = p.extra.as_ref().and_then(|x| x.shading) {
192        out.shading = Some(bg);
193    }
194    if let Some(x) = &p.extra {
195        out.keep_lines = x.keep_lines.or(out.keep_lines);
196        out.widow_control = x.widow_control.or(out.widow_control);
197        out.suppress_hyphens = x.suppress_hyphens.or(out.suppress_hyphens);
198        // An outline level apart from its heading style's.
199        out.outline_level = x.outline_level.map(|l| l.min(8)).or(out.outline_level);
200    }
201    // Tab stops a format gives the paragraph itself: finite, on the page, at most 64.
202    if let Some(tabs) = p.extra.as_ref().and_then(|x| x.tabs.as_ref()) {
203        let tabs: Vec<_> = tabs
204            .iter()
205            .filter(|t| t.pos.is_finite())
206            .map(|t| crate::doc::props::TabStop { pos: t.pos.clamp(-1584.0, 1584.0), ..*t })
207            .take(64)
208            .collect();
209        out.tabs = Some(tabs);
210    }
211    out
212}

223    lists: std::collections::HashMap<u32, u32>,

233    /// Tables made at the top of the flow so far (see `crate::doc::kept`).

237fn rev_key(kind: RevisionKind) -> u8 {
238    match kind {
239        RevisionKind::Insert => 0,
240        RevisionKind::Delete => 1,
241        RevisionKind::Format => 2,
242    }

245impl<'a> Builder<'a> {
246    /// A builder for `blocks` into `doc`.
247    fn new(doc: &'a mut Document, blocks: &[FBlock]) -> Builder<'a> {
248        let mut revs = std::collections::HashMap::new();
249        for (i, r) in doc.revisions.iter().enumerate() {
250            if r.move_name.is_none() {
251                let mark = RevMark { author: r.author.clone(), date: r.date.clone(), id: None };
252                revs.entry((rev_key(r.kind), mark)).or_insert(i as u32);
253            }
254        }
255        Builder {
256            doc,
257            revs,
258            run_strings: 0,
259            comments: Default::default(),
260            lists: Default::default(),
261            ranged: super::types::ranged_anchors(blocks),
262            started: Default::default(),
263            ended: Default::default(),
264            boxes: Default::default(),
265            chains: Vec::new(),
266            top_tables: 0,
267        }
268    }

270    /// The document revision for a tracked change, added the first time it is seen.
271    fn revision(&mut self, kind: RevisionKind, m: &RevMark) -> u32 {
272        let key = (rev_key(kind), m.clone());
273        if let Some(i) = self.revs.get(&key) {
274            return *i;
275        }
276        let i = u32::try_from(self.doc.revisions.len()).unwrap_or(u32::MAX);
277        self.doc.revisions.push(Revision::new(kind, m.author.clone(), m.date.clone()));
278        self.revs.insert(key, i);
279        i
280    }

282    /// `cp` as a run may keep it: without its link and font once the runs' copies of them are
283    /// over budget.
284    fn budget(&mut self, mut cp: CharProps) -> CharProps {
285        let n = cp.link.as_ref().map_or(0, String::len).saturating_add(cp.font.as_ref().map_or(0, String::len));
286        match self.run_strings.checked_add(n).filter(|&t| t <= MAX_RUN_STRINGS) {
287            Some(t) => self.run_strings = t,
288            None => {
289                if cp.link.is_some() || cp.font.is_some() {
290                    dropped(Dropped::RunStrings);
291                }
292                (cp.link, cp.font) = (None, None);
293            }
294        }
295        cp
296    }

298    /// Comment `c` (flow id `id`) added to the document with its text, as a reply when its parent
299    /// was added before it; its document id, or `None` when the document has no room for it or
300    /// the flow already used `id`.
301    fn comment(&mut self, id: u32, c: &super::types::FComment) -> Option<u32> {
302        if self.comments.contains_key(&id) {
303            return None;
304        }
305        let blocks = self.blocks(&c.blocks, 1);
306        let part = self.doc.try_add_part(crate::doc::PartKind::Comment, blocks).ok()?;
307        let ids = &self.doc.comments;
308        let Some(doc_id) = crate::doc::free_id(ids.keys().next_back().copied(), ids.keys().copied(), 0) else {
309            self.doc.parts.remove(&part);
310            return None;
311        };
312        let parent = c.parent.and_then(|p| self.comments.get(&p).copied());
313        let comment = crate::doc::Comment {
314            author: clean_text(&c.author),
315            initials: clean_text(&c.initials),
316            date: clean_text(&c.date),
317            parent,
318            resolved: c.resolved,
319            part,
320            ..Default::default()
321        };
322        self.doc.comments.insert(doc_id, comment);
323        self.comments.insert(id, doc_id);
324        Some(doc_id)
325    }

327    /// A note story holding `blocks`, as Word's: its paragraphs in the note text style, the first
328    /// starting with the note's mark and a space. The reference to put in the text, or `None`
329    /// when the document has no room for another story.
330    fn note(&mut self, endnote: bool, custom: &str, blocks: &[FBlock]) -> Option<InlineObject> {
331        use crate::doc::para::NoteKind;
332        let (kind, part, text, mark) = if endnote {
333            (NoteKind::Endnote, crate::doc::PartKind::Endnote, "EndnoteText", "EndnoteReference")
334        } else {
335            (NoteKind::Footnote, crate::doc::PartKind::Footnote, "FootnoteText", "FootnoteReference")
336        };
337        let custom: String = clean_text(custom).chars().filter(|c| !c.is_control()).take(10).collect();
338        let id = self.doc.try_add_part(part, Vec::new()).ok()?;
339        let mut story = self.blocks(blocks, 1);
340        if story.is_empty() {
341            story.push(para_block(Paragraph::new()));
342        }
343        for (i, b) in story.iter_mut().enumerate() {
344            if let Block::Para(p) = Arc::make_mut(b) {
345                if p.props.style.is_none() {
346                    p.props.style = Some(text.into());
347                }
348                if i == 0 {
349                    let starts_blank = p.text.starts_with([' ', '\t']);
350                    let reference = InlineObject::NoteRef { kind, id, custom: custom.clone() };
351                    if p.insert_object(0, reference, &CharProps { style: Some(mark.into()), ..Default::default() }).is_ok() && !starts_blank {
352                        let at = p.text.len().min(crate::doc::para::OBJ.len_utf8());
353                        let _ = p.insert_text(at, " ", &CharProps::default());
354                    }
355                }
356            }
357        }
358        if let Some(part) = self.doc.parts.get_mut(&id) {
359            part.blocks = story;
360        }
361        Some(InlineObject::NoteRef { kind, id, custom })
362    }

364    /// The document object for picture, shape or group `i` (members of groups `depth` deep).
365    fn drawing(&mut self, i: &Inline, depth: usize) -> Option<InlineObject> {
366        // Offsets and distances on the page.
367        let fit_float = |float: &Option<Box<crate::doc::para::Float>>| {
368            float.as_deref().cloned().map(|mut f| {
369                let fit = |v: f32| if v.is_finite() { v.clamp(-1584.0, 1584.0) } else { 0.0 };
370                (f.x, f.y) = (fit(f.x), fit(f.y));
371                (f.dist, f.dist_top, f.dist_bottom) = (fit(f.dist).max(0.0), fit(f.dist_top).max(0.0), fit(f.dist_bottom).max(0.0));
372                f
373            })
374        };
375        match i {
376            Inline::Image(img) => {
377                let key = self.doc.add_media(img.data.to_vec(), &img.ext);
378                let (w, h) = (finite_or(img.w, 72.0).clamp(1.0, 1584.0), finite_or(img.h, 72.0).clamp(1.0, 1584.0));
379                let float = fit_float(&img.float);
380                let ole = img.object.as_deref().and_then(|o| super::odf_object::store(self.doc, o));
381                // A crop of finite fractions that leaves something of the picture.
382                let crop = img.crop.map(|c| if c.is_finite() { c.clamp(0.0, 0.99) } else { 0.0 });
383                let crop = if crop[0] + crop[2] < 0.99 && crop[1] + crop[3] < 0.99 { crop } else { [0.0; 4] };
384                Some(InlineObject::Image {
385                    media: key,
386                    w,
387                    h,
388                    alt: img.alt.clone(),
389                    float: float.unwrap_or_default(),
390                    crop,
391                    fx: Default::default(),
392                    ole,
393                })
394            }
395            // Its drawing is made from the chart by the program that opens the document
396            // (`wordcraft_engine`'s chart drawing), which has the theme's colours.
397            Inline::Chart(c) => {
398                let (w, h) = (finite_or(c.w, 360.0).clamp(1.0, 1584.0), finite_or(c.h, 216.0).clamp(1.0, 1584.0));
399                let mut spec = (*c.spec).clone();
400                spec.sanitize();
401                let source = c.source.as_deref().and_then(|o| super::odf_object::store(self.doc, o));
402                let graphic = crate::doc::graphic::Graphic {
403                    kind: crate::doc::graphic::GraphicKind::Chart,
404                    w,
405                    h,
406                    chart: Some(Arc::new(spec)),
407                    source,
408                    ..Default::default()
409                };
410                Some(InlineObject::Graphic {
411                    w,
412                    h,
413                    alt: clean_text(&c.alt),
414                    float: fit_float(&c.float).unwrap_or_default(),
415                    graphic: Arc::new(graphic),
416                })
417            }
418            Inline::Shape { kind, w, h, float, blocks, look, path } => {
419                // A text box always has its story; a shape only when it holds text.
420                let story = if *kind == crate::doc::para::ShapeKind::TextBox || !blocks.is_empty() {
421                    let story = self.blocks(blocks, 1);
422                    self.doc.try_add_part(crate::doc::PartKind::TextBox, story).ok()
423                } else {
424                    None
425                };
426                // A named box, and the box its text continues in.
427                if let (Some(id), Some(c)) = (story, look.chain.as_deref()) {
428                    if !c.name.is_empty() && self.boxes.len() < MAX_NAMED_BOXES {
429                        self.boxes.entry(c.name.clone()).or_insert(id);
430                    }
431                    if let Some(next) = c.next.as_ref().filter(|n| !n.is_empty() && **n != c.name) {
432                        self.chains.push((id, next.clone()));
433                    }
434                }
435                if let Some(part) = story.and_then(|id| self.doc.parts.get_mut(&id)) {
436                    let emu = |v: f32| if v.is_finite() { (v.clamp(0.0, 1584.0) * 12_700.0).round() as u32 } else { 0 };
437                    part.body.inset = look.inset.map(|i| i.map(emu));
438                    part.body.auto_fit = look.auto_fit;
439                    part.body.vert = look.vert;
440                }
441                let (stroke, stroke_width) = match look.line {
442                    Some((c, w)) if w.is_finite() && w > 0.0 => (Some(c), w.min(12.0)),
443                    _ => (None, 0.0),
444                };
445                // A line may be flat; anything else has some size.
446                let least = if *kind == crate::doc::para::ShapeKind::Line { 0.0 } else { 1.0 };
447                let fit = |v: f32| if v.is_finite() { v.clamp(least, 1584.0) } else { 72.0 };
448                let obj = InlineObject::Shape {
449                    kind: *kind,
450                    w: fit(*w),
451                    h: fit(*h),
452                    fill: look.fill.clone().and_then(|f| f.sanitized()),
453                    stroke,
454                    stroke_width,
455                    float: float.as_deref().cloned().unwrap_or_default(),
456                    story,
457                    freeform: path
458                        .clone()
459                        .filter(|_| *kind == crate::doc::para::ShapeKind::Freeform)
460                        .map(|f| std::sync::Arc::new((*f).clone().sanitized())),
461                    effects: Default::default(),
462                    extra: crate::doc::connector::ShapeExtra {
463                        dash: look.dash.clone(),
464                        arrow_start: look.arrows.0,
465                        arrow_end: look.arrows.1,
466                        // Its drawing id, and what a connector's ends are glued to (sites bounded).
467                        id: look.conn.as_ref().map_or(0, |c| c.id),
468                        start: look.conn.as_ref().and_then(|c| c.start).filter(|e| e.site <= crate::doc::connector::MAX_SITE),
469                        end: look.conn.as_ref().and_then(|c| c.end).filter(|e| e.site <= crate::doc::connector::MAX_SITE),
470                        odf_geometry: look.geometry.clone().filter(|_| {
471                            (*kind == crate::doc::para::ShapeKind::Freeform && path.is_some()) || *kind == crate::doc::para::ShapeKind::Rectangle
472                        }),
473                        ..Default::default()
474                    },
475                };
476                Some(obj)
477            }
478            Inline::Group { w, h, float, members } if depth < MAX_GROUP_DEPTH => {
479                let fit = |v: f32| if v.is_finite() { v.clamp(1.0, 1584.0) } else { 72.0 };
480                let (w, h) = (fit(*w), fit(*h));
481                let off = |v: f32| if v.is_finite() { v.clamp(-1584.0, 1584.0) } else { 0.0 };
482                let children: Vec<GroupChild> = members
483                    .iter()
484                    .take(MAX_GROUP_CHILDREN)
485                    .filter_map(|m| self.drawing(&m.inline, depth + 1).map(|obj| GroupChild { x: off(m.x), y: off(m.y), obj }))
486                    .collect();
487                (!children.is_empty()).then(|| InlineObject::Group {
488                    w,
489                    h,
490                    float: float.as_deref().cloned().unwrap_or_default(),
491                    ch_w: w,
492                    ch_h: h,
493                    children,
494                    canvas: None,
495                })
496            }
497            _ => None,
498        }
499    }

501    /// Keeps a paragraph's markup read from the file (its own, its frame wrap, the index it
502    /// starts) pending until the document is sealed.
503    fn keep_pending(&mut self, p: &Para, rev: u64) {
504        if p.kept.is_none() && p.wrap.is_none() && p.index.is_none() {
505            return;
506        }
507        let group = p.group.as_ref().and_then(|g| g.id).filter(|_| p.kept.is_some());
508        if self.doc.kept.get().is_none() {
509            self.doc.kept = crate::doc::kept::Kept::new(Default::default());
510        }
511        if let Some(k) = self.doc.kept.get_mut() {
512            if let Some(xml) = &p.kept {
513                k.keep_pending(rev, xml.clone());
514            }
515            if let Some(w) = &p.wrap {
516                k.keep_pending_wrap(rev, (**w).clone());
517            }
518            if let Some(ix) = &p.index {
519                k.keep_pending_index(rev, ix.1, ix.0.clone());
520            }
521            if let Some(g) = group {
522                k.keep_pending_group(rev, g);
523            }
524        }
525    }

527    fn para(&mut self, p: &Para, list_num: Option<u32>) -> Paragraph {
528        let mut out = Paragraph::new();
529        out.props = para_props(p, list_num);
530        if let Some(c) = &p.fmt_change {
531            // A list item stays in its list instance (only its level may differ); an old list on a
532            // paragraph that is no list now gets an instance of its own, so Reject can restore it.
533            // The same goes for an item whose list kind changed (numbered to bulleted or back).
534            let old_num = match (c.old.list, p.list, list_num) {
535                (Some(old), Some(now), Some(num)) if old.ordered == now.ordered => Some(num),
536                (Some(old), _, _) => Some(self.new_list(&[old])),
537                _ => None,
538            };
539            let old = para_props(&c.old, old_num);
540            out.props.fmt_change = PropChange::boxed(self.revision(RevisionKind::Format, &c.mark), old);
541        }
542        if p.keep_next {
543            out.props.keep_next = Some(true);
544        }
545        out.section.clone_from(&p.section);
546        // Inlines are appended in order, so build the runs directly: inserting each one through
547        // `Paragraph::insert_text` rescans every earlier run and is quadratic in their number.
548        for i in &p.inlines {
549            match i {
550                Inline::Text(t, f) => {
551                    let t = clean_text(t);
552                    let mut cp = self.budget(char_props(f));
553                    cp.ins = f.ins.as_ref().map(|m| self.revision(RevisionKind::Insert, m));
554                    cp.del = f.del.as_ref().map(|m| self.revision(RevisionKind::Delete, m));
555                    if let Some(c) = &f.fmt_change {
556                        let old = self.budget(char_props(&c.old));
557                        cp.fmt_change = PropChange::boxed(self.revision(RevisionKind::Format, &c.mark), old);
558                    }
559                    if p.kind == Kind::Code {
560                        // Code's font comes from its style, before a change as after it.
561                        if cp.font.as_deref() == Some(MONO_FONT) {
562                            cp.font = None;
563                        }
564                        if let Some(c) = cp.fmt_change.as_deref_mut()
565                            && c.old.font.as_deref() == Some(MONO_FONT)
566                        {
567                            c.old.font = None;
568                        }
569                    }
570                    append(&mut out, &t, &cp, None);
571                }
572                Inline::FieldStart(instr) => {
573                    let obj = InlineObject::FieldStart { instr: clean_text(instr), locked: false, dirty: false };
574                    append(&mut out, "", &CharProps::default(), Some(obj));
575                }
576                Inline::FieldEnd => append(&mut out, "", &CharProps::default(), Some(InlineObject::FieldEnd)),
577                Inline::Image(_) | Inline::Chart(_) | Inline::Shape { .. } | Inline::Group { .. } => {
578                    if let Some(mut obj) = self.drawing(i, 0) {
579                        // In line, raised or lowered: its run's position (as Word keeps it). In
580                        // a paragraph of its own, its run has the paragraph's formatting.
581                        let mut cp = p
582                            .mark
583                            .as_deref()
584                            .filter(|_| p.textless())
585                            .map(|m| self.budget(CharProps { ins: None, del: None, fmt_change: None, ..m.clone() }))
586                            .unwrap_or_default();
587                        if let Some(f) = obj.float_mut()
588                            && f.wrap == crate::doc::para::Wrap::Inline
589                            && f.y != 0.0
590                        {
591                            cp.position = Some(f.y.clamp(-1584.0, 1584.0));
592                            f.y = 0.0;
593                        }
594                        append(&mut out, "", &cp, Some(obj));
595                    }
596                }
597                // A bookmark once (the first of a name): a point, or the start of the text it covers.
598                Inline::Anchor(name) if self.started.insert(name.clone()) => {
599                    append(&mut out, "", &CharProps::default(), Some(InlineObject::BookmarkStart { name: name.clone() }));
600                    if !self.ranged.contains(name) {
601                        append(&mut out, "", &CharProps::default(), Some(InlineObject::BookmarkEnd { name: name.clone() }));
602                    }
603                }
604                Inline::Anchor(_) => {}
605                Inline::AnchorEnd(name) => {
606                    if self.ranged.contains(name) && self.started.contains(name) && self.ended.insert(name.clone()) {
607                        append(&mut out, "", &CharProps::default(), Some(InlineObject::BookmarkEnd { name: name.clone() }));
608                    }
609                }
610                Inline::Figure(_) => {}
611                Inline::Comment { id, comment } => {
612                    // A reply is the comment's, not the text's: it has no range.
613                    if let Some(doc_id) = self.comment(*id, comment)
614                        && comment.parent.is_none()
615                    {
616                        append(&mut out, "", &CharProps::default(), Some(InlineObject::CommentStart { id: doc_id }));
617                    }
618                }
619                Inline::CommentEnd(id) => {
620                    if let Some(doc_id) = self.comments.get(id) {
621                        append(&mut out, "", &CharProps::default(), Some(InlineObject::CommentEnd { id: *doc_id }));
622                    }
623                }
624                Inline::Note { endnote, custom, blocks, fmt } => {
625                    if let Some(obj) = self.note(*endnote, custom, blocks) {
626                        let style = if *endnote { "EndnoteReference" } else { "FootnoteReference" };
627                        let cp = CharProps { style: Some(style.into()), ..self.budget(char_props(fmt)) };
628                        append(&mut out, "", &cp, Some(obj));
629                    }
630                }
631                Inline::Field { instr, result, fmt, locked } => {
632                    let cp = self.budget(char_props(fmt));
633                    let (instr, result) = (clean_text(instr), clean_text(result));
634                    if instr.trim().is_empty() {
635                        append(&mut out, &result, &cp, None);
636                    } else {
637                        let obj = InlineObject::Field { instr, result, locked: *locked, dirty: false, code: false, form: String::new() };
638                        append(&mut out, "", &cp, Some(obj));
639                    }
640                }
641                // A content control around its text.
642                Inline::Control { control, text, fmt } => {
643                    let cp = self.budget(char_props(fmt));
644                    let mut c = control.clone();
645                    c.block = false;
646                    append(&mut out, "", &cp, Some(InlineObject::ControlStart { control: c }));
647                    append(&mut out, &clean_text(text), &cp, None);
648                    append(&mut out, "", &cp, Some(InlineObject::ControlEnd));
649                }
650                Inline::Equation { linear, display } => {
651                    let linear = clean_text(linear);
652                    if !linear.is_empty() {
653                        let obj = InlineObject::Equation { linear, display: *display, math: Default::default() };
654                        append(&mut out, "", &CharProps::default(), Some(obj));
655                    }
656                }
657                Inline::Ruby { base, ruby, fmt } => {
658                    let (base, ruby) = (clean_text(base), clean_text(ruby));
659                    let cp = self.budget(char_props(fmt));
660                    // A source without a base or without a reading (malformed markup) is just its
661                    // text: a guide that annotates nothing is no guide.
662                    if base.is_empty() || ruby.is_empty() {
663                        append(&mut out, if base.is_empty() { &ruby } else { &base }, &cp, None);
664                    } else {
665                        // Ruby attributes a format without them doesn't state (HTML): the
666                        // Phonetic Guide's defaults for a new guide over this run's text.
667                        let obj = ruby_object(base, ruby, cp.size.unwrap_or(RUBY_BASE_SIZE));
668                        append(&mut out, "", &cp, Some(obj));
669                    }
670                }
671            }
672        }
673        if let Some(m) = p.mark.as_deref().filter(|_| p.textless()) {
674            out.mark = self.budget(CharProps { ins: None, del: None, fmt_change: None, ..m.clone() });
675        }
676        out.mark.ins = p.mark_ins.as_deref().map(|m| self.revision(RevisionKind::Insert, m));
677        out.mark.del = p.mark_del.as_deref().map(|m| self.revision(RevisionKind::Delete, m));
678        out.normalize();
679        out.touch();
680        // Its markup kept for the writer, as long as nothing changes the paragraph from here on
681        // (see `crate::doc::kept`: the reader seals it once the document is done).
682        self.keep_pending(p, out.rev);
683        out
684    }

686    /// A new list instance whose levels follow the kinds used in `group` (first use per level).
687    fn new_list(&mut self, group: &[ListInfo]) -> u32 {
688        let num_levels = levels_for(ListKind::Numbered);
689        let bul_levels = levels_for(ListKind::Bullet);
690        let mut kinds: [Option<bool>; 9] = [None; 9];
691        for li in group {
692            if let Some(k) = kinds.get_mut(li.level.min(8) as usize)
693                && k.is_none()
694            {
695                *k = Some(li.ordered);
696            }
697        }
698        let mut starts: [Option<u32>; 9] = [None; 9];
699        let mut labels: [Option<super::types::Label>; 9] = [None; 9];
700        for li in group {
701            if let Some(s) = starts.get_mut(li.level.min(8) as usize)
702                && s.is_none()
703            {
704                *s = Some(li.start);
705            }
706            if let Some(l) = labels.get_mut(li.level.min(8) as usize)
707                && l.is_none()
708            {
709                *l = li.label;
710            }
711        }
712        // Levels nobody used follow the first level's kind.
713        let first = group.first().map(|l| l.ordered).unwrap_or(false);
714        let levels: Vec<Level> = (0..9)
715            .map(|i| {
716                let ordered = kinds.get(i).copied().flatten().unwrap_or(first);
717                let src = if ordered { &num_levels } else { &bul_levels };
718                let mut level = src.get(i).cloned().unwrap_or_default();
719                if ordered && let Some(s) = starts.get(i).copied().flatten() {
720                    level.start = s;
721                }
722                // The source's own label: its format and text, or its bullet.
723                match labels.get(i).copied().flatten() {
724                    Some(l) if ordered && l.format != crate::doc::section::NumFormat::Bullet => {
725                        level.format = l.format;
726                        level.text = l.text(i as u8);
727                    }
728                    Some(l) if !ordered => {
729                        if let Some(c) = l.bullet().filter(|c| !c.is_control() && !level.text.starts_with(*c)) {
730                            let custom = levels_for(ListKind::BulletChar(c));
731                            level.text = c.to_string();
732                            level.chr.font = custom.first().and_then(|l| l.chr.font.clone());
733                        }
734                    }
735                    _ => {}
736                }
737                if let Some(l) = labels.get(i).copied().flatten() {
738                    if let Some((indent, hanging)) = l.position() {
739                        (level.indent, level.hanging) = (indent, hanging);
740                    }
741                    if let Some(a) = l.align {
742                        level.align = a;
743                    }
744                }
745                level
746            })
747            .collect();
748        let numbering = Arc::make_mut(&mut self.doc.numbering);
749        let aid = numbering.abstracts.iter().map(|a| a.id + 1).max().unwrap_or(0);
750        numbering.abstracts.push(AbstractNum { id: aid, name: None, levels, ..Default::default() });
751        let nid = numbering.nums.iter().map(|n| n.id + 1).max().unwrap_or(1).max(1);
752        numbering.nums.push(Num { id: nid, abstract_id: aid, ..Default::default() });
753        nid
754    }

756    fn blocks(&mut self, blocks: &[FBlock], depth: usize) -> Blocks {
757        let mut out = Blocks::new();
758        let mut i = 0;
759        while let Some(b) = blocks.get(i) {
760            match b {
761                FBlock::Para(p) if p.list.is_some() => {
762                    // A run of consecutive list paragraphs is one list, until an item of another
763                    // list or (when the source doesn't tell lists apart) a top-level item of the
764                    // other kind (bulleted or numbered) starts a new one. A list the source
765                    // numbers goes on where it shows up again.
766                    let mut j = i;
767                    let mut group: Vec<ListInfo> = Vec::new();
768                    let mut top: Option<bool> = None;
769                    let id = p.list.map_or(0, |l| l.list);
770                    while let Some(FBlock::Para(q)) = blocks.get(j) {
771                        let Some(li) = q.list else { break };
772                        if li.list != id {
773                            break;
774                        }
775                        if li.level == 0 && id == 0 {
776                            if top.is_some_and(|o| o != li.ordered) {
777                                break;
778                            }
779                            top = Some(li.ordered);
780                        }
781                        group.push(li);
782                        j += 1;
783                    }
784                    let num = match self.lists.get(&id) {
785                        Some(num) if id != 0 => *num,
786                        _ => {
787                            let num = self.new_list(&group);
788                            if id != 0 {
789                                self.lists.insert(id, num);
790                            }
791                            num
792                        }
793                    };
794                    for q in blocks.get(i..j).unwrap_or(&[]) {
795                        if let FBlock::Para(q) = q {
796                            let para = self.para(q, Some(num));
797                            out.push(para_block(para));
798                        }
799                    }
800                    i = j.max(i + 1);
801                    continue;
802                }
803                FBlock::Para(p) => out.push(para_block(self.para(p, None))),
804                FBlock::Table(t) => {
805                    if depth >= MAX_DEPTH {
806                        // Too deep: flatten into paragraphs.
807                        for row in &t.rows {
808                            for c in row {
809                                out.extend(self.blocks(&c.blocks, depth));
810                            }
811                        }
812                    } else if let Some(tb) = self.table(t, depth) {
813                        out.push(Arc::new(Block::Table(tb)));
814                    }
815                }
816            }
817            i += 1;
818        }
819        out
820    }

822    fn table(&mut self, t: &FTable, depth: usize) -> Option<Table> {
823        let rows: Vec<(usize, &Vec<Cell>)> = t.rows.iter().enumerate().filter(|(_, r)| !r.is_empty()).take(crate::doc::table::MAX_ROWS).collect();
824        if rows.is_empty() {
825            return None;
826        }
827        let cols = t.cols().clamp(1, crate::doc::table::MAX_COLS);
828        let width = self.doc.last_section.text_width().max(72.0);
829        // One template row for the grid widths and props; the real rows are built below. Sizing it
830        // to `rows.len()` would allocate a full rows × cols cell grid only to clear it, which a
831        // tall table (many rows, wide from merged cells) turns into a large transient spike.
832        let mut tb = Table::new(1, cols, width);
833        // A table from a format that keeps cell formatting (ODT) has its cells' own borders and no
834        // table style of Word's kind: Table Grid's single spacing would change its text.
835        if t.borderless || t.extra.is_some() {
836            tb.props.style = None;
837        }
838        if t.widths.len() == cols && t.widths.iter().all(|w| w.is_finite() && *w > 1.0) {
839            tb.grid = t.widths.iter().map(|w| w.clamp(6.0, 1584.0)).collect();
840        }
841        tb.rows.clear();
842        if let Some(x) = &t.extra {
843            tb.props.align = x.align;
844            tb.props.width_pct = x.width_pct.filter(|p| p.is_finite() && *p > 0.0).map(|p| p.min(100.0));
845            tb.props.width = x.width.filter(|w| w.is_finite() && *w > 0.0).map(|w| w.min(1584.0));
846            tb.props.fixed |= x.fixed;
847            let space = |v: Option<f32>| v.filter(|v| v.is_finite() && *v > 0.0).map(|v| v.min(1584.0));
848            (tb.props.space_before, tb.props.space_after) = (space(x.space[0]), space(x.space[1]));
849        }
850        for (ri, r) in rows {
851            let mut row = crate::doc::Row { props: RowProps::default(), cells: Vec::new(), controls: Default::default() };
852            row.props.header = !r.is_empty() && r.iter().all(|c| c.header);
853            if let Some(&Some((h, exact))) = t.extra.as_ref().and_then(|x| x.heights.get(ri))
854                && h.is_finite()
855                && h > 0.0
856            {
857                row.props.height = Some(h.min(1584.0));
858                row.props.height_rule = if exact { crate::doc::props::HeightRule::Exact } else { crate::doc::props::HeightRule::AtLeast };
859            }
860            row.props.cant_split = t.extra.as_ref().is_some_and(|x| x.keep_rows.get(ri) == Some(&true));
861            let mut g = 0usize;
862            for c in r.iter().take(crate::doc::table::MAX_COLS) {
863                let span = (c.colspan.clamp(1, 63) as usize).min(cols.saturating_sub(g).max(1));
864                let w: f32 = tb.grid.get(g..(g + span).min(tb.grid.len())).map(|s| s.iter().sum()).unwrap_or(72.0);
865                g += span;
866                let mut blocks = self.blocks(&c.blocks, depth + 1);
867                if blocks.is_empty() || matches!(blocks.last().map(|b| &**b), Some(Block::Table(_))) {
868                    blocks.push(para_block(Paragraph::new()));
869                }
870                let vmerge = if c.covered {
871                    VMerge::Continue
872                } else if c.rowspan > 1 {
873                    VMerge::Restart
874                } else {
875                    VMerge::None
876                };
877                // A background picture goes into the document's media.
878                let picture = c.picture.as_deref().map(|p| {
879                    let media = self.doc.add_media(p.img.data.as_ref().clone(), &p.img.ext);
880                    Box::new(crate::doc::props::CellPicture { media, size: [p.img.w, p.img.h], repeat: p.repeat, align: p.align })
881                });
882                row.cells.push(crate::doc::Cell {
883                    props: CellProps {
884                        width: Some(w),
885                        span: span as u32,
886                        vmerge,
887                        shading: c.shading,
888                        borders: c.borders,
889                        margins: c.margins.filter(|m| m.iter().all(|v| v.is_finite())).map(|m| m.map(|v| v.clamp(0.0, 1584.0))),
890                        valign: c.valign,
891                        text_direction: c.direction,
892                        picture,
893                        ..Default::default()
894                    },
895                    blocks,
896                    controls: Default::default(),
897                });
898            }
899            if row.cells.is_empty() {
900                row.cells.push(crate::doc::Cell::empty());
901            }
902            tb.rows.push(row);
903        }
904        // Its markup kept for the writer, by its place among the top's tables.
905        if depth == 0 {
906            if t.kept.is_some() || t.wrap.is_some() {
907                if self.doc.kept.get().is_none() {
908                    self.doc.kept = crate::doc::kept::Kept::new(Default::default());
909                }
910                if let Some(k) = self.doc.kept.get_mut() {
911                    if let Some(xml) = &t.kept {
912                        k.keep_pending_table(self.top_tables, xml.clone());
913                    }
914                    if let Some(w) = &t.wrap {
915                        k.keep_pending_table_wrap(self.top_tables, (**w).clone());
916                    }
917                }
918            }
919            self.top_tables += 1;
920        }
921        Some(tb)
922    }
923}

934/// [`to_doc`] on a page of the source's own size and margins (`section`), which tables size to.
935/// Most named text boxes linked.
936const MAX_NAMED_BOXES: usize = 10_000;
937
938impl Builder<'_> {
939    /// Each text box whose text continues in another (by name) linked to it, a chain at a time
940    /// (a box is in at most one chain, and a chain doesn't loop back on itself).
941    fn link_boxes(&mut self) {
942        let mut linked_to = std::collections::HashSet::new();
943        for (from, name) in std::mem::take(&mut self.chains) {
944            let Some(&to) = self.boxes.get(&name) else { continue };
945            if to == from || !linked_to.insert(to) {
946                continue;
947            }
948            if let Some(p) = self.doc.parts.get_mut(&from) {
949                p.body.next = Some(to);
950            }
951        }
952        // A link that closes a loop is undone.
953        for id in self.boxes.values().copied().collect::<Vec<_>>() {
954            let mut seen = std::collections::HashSet::new();
955            let mut at = id;
956            while let Some(next) = self.doc.parts.get(&at).and_then(|p| p.body.next) {
957                if !seen.insert(at) || seen.len() > MAX_NAMED_BOXES {
958                    if let Some(p) = self.doc.parts.get_mut(&at) {
959                        p.body.next = None;
960                    }
961                    break;
962                }
963                at = next;
964            }
965        }
966    }
967}

969pub fn to_doc_in(flow: &Flow, section: crate::doc::SectionProps) -> Document {
970    let mut doc = Document::new();
971    doc.last_section = section;
972    doc.numbering = Default::default();
973    let uses_code = uses_kind(&flow.blocks, Kind::Code, 0);
974    if uses_code {
975        Arc::make_mut(&mut doc.styles).upsert(code_style());
976    }
977    let mut b = Builder::new(&mut doc, &flow.blocks);
978    let body = b.blocks(&flow.blocks, 0);
979    b.link_boxes();
980    doc.body = body;
981    let m = &flow.meta;
982    doc.core.title = m.title.clone();
983    doc.core.creator = m.author.clone();
984    doc.core.subject = m.subject.clone();
985    doc.core.keywords = m.keywords.clone();
986    doc.core.description = m.description.clone();
987    doc.core.created = m.created.clone();
988    doc.ensure_nonempty();
989    doc
990}
