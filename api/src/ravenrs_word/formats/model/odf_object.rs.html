1//! ODF embedded objects ([`OdfObject`]) in a document: their files in
2//! [`Document::passthrough`] under `odf-object/<n>/`, linked from the picture showing them by an
3//! [`Embedded`] whose markup only names the object's media type, and taken back out for the ODT
4//! writer. Other formats see the picture alone.
5
6use std::sync::Arc;
7
8use crate::doc::Document;
9use crate::doc::graphic::{Embedded, EmbeddedRef};
10
11use super::types::OdfObject;
12
13/// Relationship kind of an object's file, and of its replacement picture.
14const FILE: &str = "file";
15const REPLACEMENT: &str = "replacement";
16/// Most objects a document keeps, and most files in one.
17pub const MAX_OBJECTS: usize = 1000;
18pub const MAX_FILES: usize = 256;
19
20fn esc(s: &str) -> String {
21    s.replace('&', "&amp;").replace('"', "&quot;").replace('<', "&lt;").replace('>', "&gt;")
22}
23
24fn unesc(s: &str) -> String {
25    s.replace("&quot;", "\"").replace("&lt;", "<").replace("&gt;", ">").replace("&amp;", "&")
26}
27
28/// Keep `obj` in `doc`: its files as passthrough parts, and the link to them. `None` when the
29/// document already holds as many objects as it keeps.
30pub fn store(doc: &mut Document, obj: &OdfObject) -> Option<Arc<Embedded>> {
31    let taken = |n: usize| {
32        let prefix = format!("odf-object/{n}/");
33        doc.passthrough.range(prefix.clone()..).next().is_some_and(|(k, _)| k.starts_with(&prefix))
34    };
35    let n = (0..MAX_OBJECTS).find(|&n| !taken(n))?;
36    let mut refs = Vec::new();
37    for (k, (path, bytes)) in obj.files.iter().take(MAX_FILES).enumerate() {
38        let target = format!("odf-object/{n}/{k}");
39        doc.passthrough.insert(target.clone(), bytes.clone());
40        refs.push(EmbeddedRef { id: path.clone(), kind: FILE.into(), target, external: false });
41    }
42    if let Some(r) = &obj.replacement {
43        let target = format!("odf-object/{n}/replacement");
44        doc.passthrough.insert(target.clone(), r.clone());
45        refs.push(EmbeddedRef { id: String::new(), kind: REPLACEMENT.into(), target, external: false });
46    }
47    let xml = format!("{} mime=\"{}\" ole=\"{}\"/>", Embedded::ODF_OBJECT, esc(&obj.mime), obj.ole);
48    Some(Arc::new(Embedded { xml, refs }))
49}

51/// The object `e` links to in `doc`, when it is an ODF object whose files are all there.
52pub fn load(doc: &Document, e: &Embedded) -> Option<Arc<OdfObject>> {
53    if !e.is_odf_object() {
54        return None;
55    }
56    let attr = |k: &str| {
57        let pat = format!(" {k}=\"");
58        let from = e.xml.find(&pat)? + pat.len();
59        let len = e.xml.get(from..)?.find('"')?;
60        e.xml.get(from..from + len).map(unesc)
61    };
62    let mut obj = OdfObject { mime: attr("mime").unwrap_or_default(), ole: attr("ole").as_deref() == Some("true"), ..Default::default() };
63    for r in e.refs.iter().take(MAX_FILES + 1) {
64        let bytes = doc.passthrough.get(&r.target)?.clone();
65        match r.kind.as_str() {
66            FILE => obj.files.push((r.id.clone(), bytes)),
67            REPLACEMENT => obj.replacement = Some(bytes),
68            _ => {}
69        }
70    }
71    (!obj.files.is_empty()).then(|| Arc::new(obj))
72}
