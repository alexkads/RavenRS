27fn is_code_style(doc: &Document, id: &str) -> bool {
28    let names = |s: &str| {
29        let l = s.to_ascii_lowercase().replace([' ', '-', '_'], "");
30        l == "code" || l.contains("preformatted") || l == "plaintext" || l == "codeblock" || l == "sourcecode"
31    };
32    doc.styles.chain(id).iter().any(|s| names(&s.id) || names(&s.name)) || names(id)
33}

35/// Character formatting from a run's character style chain plus direct formatting (the
36/// paragraph style is left out: it is expressed by the paragraph kind).
37fn effective(doc: &Document, cp: &CharProps) -> CharProps {
38    let mut out = CharProps::default();
39    if let Some(cs) = cp.style.as_deref() {
40        for s in doc.styles.chain(cs) {
41            if s.kind == StyleKind::Character {
42                out.overlay(&s.chr);
43            }
44        }
45    }
46    let mut direct = cp.clone();
47    direct.style = None;
48    out.overlay(&direct);
49    out
50}

52fn fmt_of(doc: &Document, cp: &CharProps, in_code: bool) -> Fmt {
53    let e = effective(doc, cp);
54    let code = !in_code && e.font.as_deref().is_some_and(is_mono);
55    Fmt {
56        bold: e.bold.unwrap_or(false),
57        italic: e.italic.unwrap_or(false),
58        underline: e.underline.is_some_and(|u| u != Underline::None) && e.link.is_none(),
59        strike: e.strike.unwrap_or(false) || e.double_strike.unwrap_or(false),
60        code,
61        sup: e.vert_align == Some(VertAlign::Superscript),
62        sub: e.vert_align == Some(VertAlign::Subscript),
63        link: e.link.as_deref().map(Arc::from),
64        color: match e.color {
65            Some(TextColor::Rgb(c)) => Some(c),
66            _ => None,
67        },
68        size: e.size,
69        font: if code || in_code { None } else { e.shown_font().map(Arc::from) },
70        background: e.highlight.and_then(|h| h.rgb()).or(e.shading),
71        ins: None,
72        del: None,
73        fmt_change: None,
74        extra: None,
75    }
76}

78/// The flow formatting of a run with properties `props`: with `opts.chars`, its extra formatting,
79/// its custom character style by name (only its direct formatting then its own) and a
80/// monospaced font by its name.
81fn run_fmt(doc: &Document, para_style: Option<&str>, props: &CharProps, in_code: bool, opts: FlowOpts) -> Fmt {
82    let mut f = fmt_of(doc, props, in_code);
83    if !opts.chars {
84        return f;
85    }
86    let custom = props.style.as_deref().filter(|id| is_custom_char_style(doc, id));
87    let direct;
88    let props_now = match custom {
89        Some(_) => {
90            direct = CharProps { style: None, ..props.clone() };
91            f = fmt_of(doc, &direct, in_code);
92            &direct
93        }
94        None => props,
95    };
96    let e = effective(doc, props_now);
97    f.extra = extra_of(&e);
98    // Kerning on or off where it differs from the document's default.
99    let kerns = |k: Option<f32>| k.is_some_and(|k| k.is_finite() && k > 0.0);
100    let (kern, default) = (kerns(e.kern.or(doc.styles.default_chr.kern)), kerns(doc.styles.default_chr.kern));
101    if kern != default {
102        f.extra.get_or_insert_with(Default::default).kern = Some(if kern { 1.0 } else { 0.0 });
103    }
104    if let Some(id) = custom {
105        f.extra.get_or_insert_with(Default::default).style = Some(id.to_string());
106    }
107    // A size stated without a complex-script size leaves the complex size to what is above it
108    // (a style's, or this size where none says one): said where it isn't this size, since a
109    // writer that gives every script the one size would otherwise change it.
110    let sized = props.size.is_some() || custom.is_some_and(|id| doc.styles.chain(id).iter().any(|s| s.chr.size.is_some()));
111    if props.size_cs.is_none() && sized && f.extra.as_ref().is_none_or(|x| x.size_cs.is_none()) {
112        let cs = doc.styles.resolve_char(para_style, props).size_cs;
113        if f.size != Some(cs) {
114            f.extra.get_or_insert_with(Default::default).size_cs = Some(cs);
115        }
116    }
117    if f.code {
118        f.code = false;
119        f.font = e.shown_font().map(Arc::from);
120    }
121    f
122}

169/// Whether paragraph style `id` of `doc` is one of its own rather than one the flow model's
170/// paragraph kinds, lists and notes stand for (Normal, the headings, Title, Quote, Code, List
171/// Paragraph, the note texts).
172pub fn is_custom_style(doc: &Document, id: &str) -> bool {
173    // The styles the kinds stand for, and the names formats write fixed styles under.
174    let kinds =
175        ["Normal", "Title", "Quote", super::types::CODE_STYLE, "FootnoteText", "EndnoteText", "Standard", "Footnote", "Endnote", "Quotations"];
176    let heading = id.strip_prefix("Heading").is_some_and(|n| n.len() == 1 && n.chars().all(|c| c.is_ascii_digit()));
177    !heading && !kinds.contains(&id) && doc.styles.get(id).is_some_and(|s| s.kind == crate::doc::styles::StyleKind::Paragraph)
178}

186/// The paragraph formatting only [`Para::extra`] holds that `p` sets itself; `None` when none.
187fn para_extra(doc: &Document, p: &crate::doc::props::ParaProps) -> Option<Box<crate::doc::props::ParaProps>> {
188    let tabs = p.tabs.clone().filter(|t| !t.is_empty()).map(|mut tabs| {
189        // Measured from the paragraph's indent where the document measures them so (ODT).
190        if tabs_from_indent(doc) {
191            let indent = doc.styles.resolve_para(p).indent_left;
192            for t in &mut tabs {
193                t.pos = (t.pos - indent).clamp(-1584.0, 1584.0);
194            }
195        }
196        tabs
197    });
198    let style = p.style.clone().filter(|id| is_custom_style(doc, id));
199    // Its own, else its custom style's: the kind doesn't say (it stops at Heading 6, and is the
200    // heading style's where the style is based on one).
201    let outline_level = p.outline_level.or_else(|| style.as_ref().and_then(|_| doc.styles.resolve_para(p).outline_level));
202    let x = crate::doc::props::ParaProps {
203        tabs,
204        shading: p.shading,
205        style,
206        contextual_spacing: p.contextual_spacing,
207        bidi: p.bidi.filter(|b| *b),
208        drop_cap: p.drop_cap.filter(|l| *l > 1),
209        drop_cap_distance: p.drop_cap_distance.filter(|_| p.drop_cap.is_some_and(|l| l > 1)),
210        keep_lines: p.keep_lines.filter(|k| *k),
211        outline_level,
212        widow_control: p.widow_control.filter(|w| !*w),
213        // Hyphenated (the document hyphenates and the paragraph doesn't opt out): `Some(false)`.
214        suppress_hyphens: (doc.settings.auto_hyphenation && p.suppress_hyphens != Some(true)).then_some(false),
215        ..Default::default()
216    };
217    (x != crate::doc::props::ParaProps::default()).then(|| Box::new(x))
218}

236/// Paragraph kind and list membership of paragraph properties; `blank` = the paragraph shows
237/// nothing (a bottom border then makes it a rule).
238fn props_kind(doc: &Document, props: &ParaProps, blank: bool) -> (Kind, Option<ListInfo>) {
239    let rp = doc.styles.resolve_para(props);
240    let style = props.style.as_deref().unwrap_or("Normal");
241    // A heading's kind is its style's (Heading 2 looks as Heading 2 whatever its outline level,
242    // which the paragraph may set apart); else its outline level's.
243    let style_level =
244        doc.styles.chain(style).iter().find_map(|s| s.id.strip_prefix("Heading").and_then(|n| n.parse::<u8>().ok()).filter(|n| (1..=9).contains(n)));
245    let kind = if let Some(l) = style_level.filter(|_| rp.outline_level.is_some()) {
246        Kind::Heading(l.min(6))
247    } else if let Some(l) = rp.outline_level {
248        Kind::Heading((l + 1).min(6))
249    } else if doc.styles.chain(style).iter().any(|s| s.id == "Title") {
250        Kind::Title
251    } else if doc.styles.chain(style).iter().any(|s| s.id == "Quote" || s.id == "IntenseQuote") {
252        Kind::Quote
253    } else if is_code_style(doc, style) {
254        Kind::Code
255    } else if blank && rp.borders.and_then(|b| b.bottom).is_some_and(|b| b.is_visible()) {
256        Kind::Rule
257    } else {
258        Kind::Normal
259    };
260    let list = rp.numbering.and_then(|n| {
261        let lv = doc.numbering.level(n.num, n.level)?;
262        Some(ListInfo {
263            ordered: !matches!(lv.format, NumFormat::Bullet | NumFormat::None),
264            level: n.level.min(8),
265            start: lv.start,
266            list: n.num,
267            label: Some(super::types::Label::of_level(lv.format, &lv.text).placed(lv.indent, lv.hanging, lv.align)),
268        })
269    });
270    (kind, list)
271}

284/// The flow inlines for a note reference, comment mark or field `o` of `doc`, when `opts` asks
285/// for them; `None` for other objects. `f` is the formatting of the run `o` is in.
286fn rich_inlines(doc: &Document, o: &InlineObject, f: &Fmt, opts: FlowOpts) -> Option<Vec<Inline>> {
287    Some(match o {
288        InlineObject::NoteRef { kind, id, custom } if opts.notes => {
289            // The note's content without its own mark (a note holds no notes).
290            let story = doc.parts.get(id).map(|part| &part.blocks);
291            let mut blocks = story.map(|b| flow_blocks(doc, b, 0, FlowOpts { notes: false, ..opts })).unwrap_or_default();
292            // The space after the mark is the mark's (see `to_doc`'s `note`); a tab or more
293            // spaces there are the note's own.
294            if let Some(FBlock::Para(first)) = blocks.first_mut() {
295                let k = first.inlines.iter().take_while(|i| matches!(i, Inline::Text(t, _) if t.is_empty())).count();
296                first.inlines.drain(..k);
297                if let Some(Inline::Text(t, _)) = first.inlines.first_mut()
298                    && t.starts_with(' ')
299                {
300                    t.remove(0);
301                }
302            }
303            let endnote = *kind == crate::doc::para::NoteKind::Endnote;
304            vec![Inline::Note { endnote, custom: custom.clone(), blocks, fmt: f.clone() }]
305        }
306        // The comment, then the replies in its thread (replies to replies too, each answering the
307        // comment that starts the thread), which have no place of their own.
308        InlineObject::CommentStart { id } if opts.comments => {
309            let Some(c) = doc.comments.get(id) else { return Some(Vec::new()) };
310            let mut out = vec![Inline::Comment { id: *id, comment: Box::new(flow_comment(doc, c, None, opts)) }];
311            let mut thread = std::collections::BTreeSet::from([*id]);
312            let mut k = 0;
313            while let Some(Inline::Comment { id: at, .. }) = out.get(k) {
314                let at = *at;
315                k += 1;
316                for (rid, r) in doc.comments.iter().filter(|(_, r)| r.parent == Some(at)) {
317                    if thread.insert(*rid) {
318                        out.push(Inline::Comment { id: *rid, comment: Box::new(flow_comment(doc, r, Some(*id), opts)) });
319                    }
320                }
321            }
322            out
323        }
324        InlineObject::CommentEnd { id } if opts.comments => {
325            if doc.comments.contains_key(id) {
326                vec![Inline::CommentEnd(*id)]
327            } else {
328                Vec::new()
329            }
330        }
331        // A text box's content, without notes, comments or boxes of its own.
332        // A drawing shape, text box or group (pictures have their own place below).
333        InlineObject::Shape { .. } | InlineObject::Group { .. } if opts.boxes => vec![drawing_inline(doc, o, opts, 0, (1.0, 1.0))?],
334        InlineObject::FieldStart { instr, .. } if opts.fields => vec![Inline::FieldStart(instr.clone())],
335        InlineObject::FieldEnd if opts.fields => vec![Inline::FieldEnd],
336        // Where a bookmark ends, so it covers its text (Word's `_GoBack` is the last edit's spot).
337        InlineObject::BookmarkEnd { name } if opts.fields && name != "_GoBack" => vec![Inline::AnchorEnd(name.clone())],
338        InlineObject::Field { instr, result, locked, .. } if opts.fields => {
339            vec![Inline::Field { instr: instr.clone(), result: result.clone(), fmt: f.clone(), locked: *locked }]
340        }
341        _ => return None,
342    })
343}

345/// Comment `c` (a reply to `parent`) of `doc` as a flow comment: its text holds no comments or
346/// notes of its own.
347/// The flow inline for picture, shape or group `o` of `doc` (a member of groups `depth` deep),
348/// its size scaled by `scale` (a group's members are in its own coordinates). A shape or text
349/// box keeps its text, but not notes, comments or boxes of its own. `None` for what a flow has
350/// no drawing for: a freeform's paths, a picture whose data is missing.
351fn drawing_inline(doc: &Document, o: &InlineObject, opts: FlowOpts, depth: usize, scale: (f32, f32)) -> Option<Inline> {
352    use crate::doc::para::ShapeKind;
353    let (sx, sy) = scale;
354    match o {
355        // A freeform needs its paths.
356        InlineObject::Shape { kind, w, h, float, story, fill, stroke, stroke_width, extra, freeform, .. }
357            if *kind != ShapeKind::Freeform || freeform.is_some() =>
358        {
359            // A connector stays one in formats that glue them (ODT, with boxes); elsewhere it is a
360            // straight line.
361            let kind = match kind {
362                ShapeKind::StraightConnector | ShapeKind::ElbowConnector | ShapeKind::CurvedConnector if !opts.boxes => ShapeKind::Line,
363                k => *k,
364            };
365            let inner = FlowOpts { notes: false, comments: false, boxes: false, ..opts };
366            let part = story.and_then(|id| doc.parts.get(&id));
367            let blocks = part.map(|p| flow_blocks(doc, &p.blocks, 0, inner)).unwrap_or_default();
368            let look = BoxLook {
369                fill: fill.clone().filter(|_| kind != ShapeKind::Line),
370                line: stroke.filter(|_| stroke_width.is_finite() && *stroke_width > 0.0).map(|c| (c, stroke_width.min(12.0))),
371                dash: extra.dash.clone(),
372                arrows: (extra.arrow_start, extra.arrow_end),
373                inset: part.and_then(|p| p.body.inset.is_some().then(|| p.body.insets_pt())),
374                auto_fit: part.is_some_and(|p| p.body.auto_fit),
375                vert: part.map(|p| p.body.vert).unwrap_or_default(),
376                // Kept only while it is still the freeform it was read as.
377                geometry: extra.odf_geometry.clone().filter(|_| (kind == ShapeKind::Freeform && freeform.is_some()) || kind == ShapeKind::Rectangle),
378                conn: (opts.boxes && (extra.id != 0 || extra.start.is_some() || extra.end.is_some()))
379                    .then(|| Box::new(crate::formats::model::Conn { id: extra.id, start: extra.start, end: extra.end })),
380                // A text box is named after its story; a linked one names the box it continues in.
381                chain: story.filter(|_| opts.boxes).map(|id| {
382                    let next = part.and_then(|p| p.body.next).map(|n| format!("Frame{n}"));
383                    Box::new(crate::formats::model::Chain { name: format!("Frame{id}"), next })
384                }),
385            };
386            let path = freeform.clone().filter(|_| kind == ShapeKind::Freeform);
387            Some(Inline::Shape { kind, w: w * sx, h: h * sy, float: kept_float(float), blocks, look, path })
388        }
389        InlineObject::Image { media, w, h, alt, float, ole, crop, .. } => {
390            let data = doc.media.get(media)?;
391            let alt = if o.is_decorative() { "" } else { alt.as_str() };
392            Some(Inline::Image(Img {
393                data: data.clone(),
394                ext: media_ext(media, data),
395                w: w * sx,
396                h: h * sy,
397                alt: alt.to_string(),
398                float: kept_float(float),
399                object: ole.as_deref().filter(|_| opts.boxes).and_then(|e| super::odf_object::load(doc, e)),
400                crop: if opts.paras { *crop } else { [0.0; 4] },
401            }))
402        }
403        InlineObject::Group { w, h, float, ch_w, ch_h, children, .. } if depth < MAX_GROUP_DEPTH => {
404            let ratio = |outer: f32, inner: f32| {
405                let r = outer / inner;
406                if r.is_finite() && r > 0.0 { r } else { 1.0 }
407            };
408            let inner = (sx * ratio(*w, *ch_w), sy * ratio(*h, *ch_h));
409            let members: Vec<Member> = children
410                .iter()
411                .filter_map(|c| {
412                    drawing_inline(doc, &c.obj, opts, depth + 1, inner).map(|inline| Member { x: c.x * inner.0, y: c.y * inner.1, inline })
413                })
414                .collect();
415            (!members.is_empty()).then(|| Inline::Group { w: w * sx, h: h * sy, float: kept_float(float), members })
416        }
417        _ => None,
418    }
419}

421/// The placement a flow keeps for a drawing: when it floats, or turns or mirrors it in line.
422fn kept_float(f: &crate::doc::para::Float) -> Option<Box<crate::doc::para::Float>> {
423    let turned = f.rot.is_finite() && f.rot.rem_euclid(360.0) > 0.001 && f.rot.rem_euclid(360.0) < 359.999;
424    (f.wrap != crate::doc::para::Wrap::Inline || turned || f.flip_h || f.flip_v).then(|| Box::new(f.clone()))
425}

433/// Whether `li`, the list of heading `p`, is the number its outline level has from its heading
434/// style's list, not a list of its own: it states that level, and the heading style of that level
435/// numbers with that list there. Past Heading 6 no such style gives the written outline a number.
436fn numbered_by_outline(doc: &Document, p: &Paragraph, li: &ListInfo) -> bool {
437    use crate::doc::props::NumRef;
438    p.props.outline_level == Some(li.level)
439        && li.level < 6
440        && doc.styles.para_style_props(&format!("Heading{}", li.level + 1)).0.numbering == Some(NumRef { num: li.list, level: li.level })
441}

443/// A paragraph's list membership as the flow keeps it: without the list's identity and label
444/// unless `opts` asks for them, and none for a heading numbered only through its style or
445/// [`numbered_by_outline`], which a format with heading numbering of its own (ODT's outline style)
446/// numbers instead, in its first six levels.
447fn flow_list(doc: &Document, kind: Kind, list: Option<ListInfo>, p: &Paragraph, opts: FlowOpts) -> Option<ListInfo> {
448    let mut li = list?;
449    if !opts.lists {
450        (li.list, li.label) = (0, None);
451    } else if matches!(kind, Kind::Heading(_)) && li.level < 6 && (p.props.numbering.is_none() || numbered_by_outline(doc, p, &li)) {
452        return None;
453    }
454    Some(li)
455}

484/// [`flow_paras`]; `tracked` keeps tracked changes: deleted text stays, and text carries its
485/// insertion and deletion marks.
486fn flow_paras_in(doc: &Document, p: &Paragraph, opts: FlowOpts) -> Vec<Para> {
487    let tracked = opts.tracked;
488    let (kind, list) = para_kind(doc, p);
489    let list = flow_list(doc, kind, list, p, opts);
490    let mut done: Vec<Para> = Vec::new();
491    let borders = if kind == Kind::Rule { None } else { p.props.borders.filter(|b| b.takes_room()) };
492    let mut out = Para {
493        kind,
494        list,
495        align: p.props.align,
496        page_break: p.props.page_break_before.unwrap_or(false),
497        keep_next: p.props.keep_next.unwrap_or(false),
498        inlines: Vec::new(),
499        spacing: direct_spacing(&p.props),
500        borders,
501        section: None,
502        mark_ins: None,
503        mark_del: None,
504        fmt_change: None,
505        extra: None,
506        mark: None,
507        kept: None,
508        wrap: None,
509        index: None,
510        group: None,
511    };
512    if opts.paras {
513        out.extra = para_extra(doc, &p.props);
514    }
515    let in_code = kind == Kind::Code;
516    let mut k = 0usize;
517    let mut control = OpenControl::default();
518    for (range, props) in p.run_ranges() {
519        let Some(text) = p.text.get(range.clone()) else { continue };
520        // Deleted objects and page breaks stay out even when tracking: the flow has no marks for them.
521        let hidden = props.hidden.unwrap_or(false) || props.del.is_some();
522        // Formats that keep hidden text get it, marked hidden.
523        let hidden_text = (props.hidden.unwrap_or(false) && !opts.chars) || (props.del.is_some() && !tracked);
524        let hidden_obj = (props.hidden.unwrap_or(false) && !opts.chars) || props.del.is_some();
525        let mut f = run_fmt(doc, p.props.style.as_deref(), props, in_code, opts);
526        if tracked {
527            (f.ins, f.del) = rev_marks(doc, props);
528            f.fmt_change = props.fmt_change.as_deref().map(|c| Box::new(FmtChange { mark: rev_mark(doc, c.rev), old: fmt_of(doc, &c.old, in_code) }));
529        }
530        let mut buf = String::new();
531        for c in text.chars() {
532            if c == PAGE_BREAK {
533                if !hidden {
534                    out.push_text(&std::mem::take(&mut buf), &f);
535                    let next = Para {
536                        kind,
537                        list: None,
538                        align: p.props.align,
539                        page_break: true,
540                        keep_next: p.props.keep_next.unwrap_or(false),
541                        inlines: Vec::new(),
542                        spacing: direct_spacing(&p.props),
543                        borders,
544                        section: None,
545                        mark_ins: None,
546                        mark_del: None,
547                        fmt_change: None,
548                        extra: None,
549                        mark: None,
550                        kept: None,
551                        wrap: None,
552                        index: None,
553                        group: None,
554                    };
555                    let prev = std::mem::replace(&mut out, next);
556                    if prev.inlines.is_empty() {
557                        out.page_break |= prev.page_break;
558                        out.list = prev.list;
559                    } else {
560                        done.push(prev);
561                    }
562                }
563                continue;
564            }
565            if c != OBJ {
566                buf.push(c);
567                continue;
568            }
569            let obj = p.objects.get(k);
570            k += 1;
571            control.interrupt(obj, &mut out, &f);
572            // Hidden text a format leaves out, a deleted object, or a hidden drawing: not in the
573            // copy (a format keeping hidden text keeps its objects too, as it keeps the text).
574            if hidden_obj || obj.and_then(InlineObject::frame).is_some_and(|(_, _, f)| f.hidden) {
575                continue;
576            }
577            let from = out.inlines.len();
578            if let Some(items) = obj.and_then(|o| rich_inlines(doc, o, &f, opts)) {
579                if !buf.is_empty() {
580                    out.push_text(&std::mem::take(&mut buf), &f);
581                }
582                out.inlines.extend(items);
583                raise_drawings(&mut out.inlines, from, props.position.filter(|_| opts.paras));
584                continue;
585            }
586            match obj {
587                Some(o @ InlineObject::Image { media, w, h, alt, float, ole, crop, .. }) => {
588                    if !buf.is_empty() {
589                        out.push_text(&std::mem::take(&mut buf), &f);
590                    }
591                    // A picture marked decorative keeps its description in the document, but
592                    // exports with none (`alt=""`), so screen readers skip it.
593                    let alt = if o.is_decorative() { "" } else { alt.as_str() };
594                    match doc.media.get(media) {
595                        Some(data) => out.inlines.push(Inline::Image(Img {
596                            data: data.clone(),
597                            ext: media_ext(media, data),
598                            w: *w,
599                            h: *h,
600                            alt: alt.to_string(),
601                            float: kept_float(float).filter(|_| opts.paras),
602                            object: ole.as_deref().filter(|_| opts.boxes).and_then(|e| super::odf_object::load(doc, e)),
603                            crop: if opts.paras { *crop } else { [0.0; 4] },
604                        })),
605                        None if !alt.is_empty() => buf.push_str(alt),
606                        None => {}
607                    }
608                }
609                Some(InlineObject::Graphic { alt, w, h, float, graphic }) => {
610                    if !buf.is_empty() {
611                        out.push_text(&std::mem::take(&mut buf), &f);
612                    }
613                    out.inlines.extend(graphic_inline(doc, (alt, *w, *h), float, graphic, opts));
614                }
615                Some(InlineObject::Equation { linear, display, math }) => {
616                    if !buf.is_empty() {
617                        out.push_text(&std::mem::take(&mut buf), &f);
618                    }
619                    // The linear format is the shared form; an equation known only by its
620                    // structure gets it from there.
621                    let linear = if linear.is_empty() { crate::doc::math::to_linear(&math.nodes) } else { linear.clone() };
622                    out.inlines.push(Inline::Equation { linear, display: *display });
623                }
624                Some(InlineObject::Ruby { base, ruby, .. }) => {
625                    if !buf.is_empty() {
626                        out.push_text(&std::mem::take(&mut buf), &f);
627                    }
628                    // The base text and reading, in the run's formatting; the guide's own
629                    // attributes (alignment, size, raise) stay in the document model.
630                    out.inlines.push(Inline::Ruby { base: base.clone(), ruby: ruby.clone(), fmt: f.clone() });
631                }
632                Some(InlineObject::BookmarkStart { name }) if name != "_GoBack" => {
633                    if !buf.is_empty() {
634                        out.push_text(&std::mem::take(&mut buf), &f);
635                    }
636                    out.inlines.push(Inline::Anchor(name.clone()));
637                }
638                Some(o @ (InlineObject::ControlStart { .. } | InlineObject::ControlEnd)) if control.object(o, opts, &mut buf, &mut out, &f) => {}
639                Some(o) => buf.push_str(shown_text(o)),
640                None => {}
641            }
642            raise_drawings(&mut out.inlines, from, props.position.filter(|_| opts.paras));
643        }
644        if !hidden_text {
645            control.run_end(&buf, &mut out, &f);
646        }
647    }
648    control.finish(&mut out);
649    if tracked {
650        let (ins, del) = rev_marks(doc, &p.mark);
651        (out.mark_ins, out.mark_del) = (ins.map(Box::new), del.map(Box::new));
652        // The formatting change belongs to the paragraph the mark ends.
653        out.fmt_change = p.props.fmt_change.as_deref().map(|c| {
654            let blank = out.inlines.is_empty();
655            let (kind, mut list) = props_kind(doc, &c.old, blank);
656            if !opts.lists
657                && let Some(li) = &mut list
658            {
659                (li.list, li.label) = (0, None);
660            }
661            let old = Para { kind, list, align: c.old.align, page_break: c.old.page_break_before.unwrap_or(false), ..Default::default() };
662            Box::new(FmtChange { mark: rev_mark(doc, c.rev), old })
663        });
664    }
665    // The paragraph that ends a section carries its properties (the last piece of it does).
666    out.section.clone_from(&p.section);
667    out.mark = empty_mark(&out, p, opts);
668    // Still what the file it came from had (a whole paragraph): its markup, for its format. Not
669    // for one starting a page: the break may be its element's or the one before's (a break after
670    // it), and only the model says which is written where.
671    if opts.paras && done.is_empty() && p.props.page_break_before != Some(true) {
672        // With its comments and its text boxes' text, while they're as read.
673        out.kept = doc
674            .kept
675            .of("odt")
676            .and_then(|k| {
677                k.para(p).filter(|xml| (!xml.contains("<office:annotation") || k.comments_as_read(doc, p, xml)) && k.stories_as_read(doc, p))
678            })
679            .cloned();
680        // With the group it is written as read with, if any (see `model::Group`).
681        out.group = out
682            .kept
683            .as_ref()
684            .and_then(|_| doc.kept.of("odt")?.group(p))
685            .map(|id| Box::new(crate::formats::model::Group { links: Vec::new(), id: Some(id) }));
686    }
687    // The sections kept as read that it opens or closes.
688    if opts.paras {
689        out.wrap = doc.kept.of("odt").and_then(|k| k.wrap(p)).map(|w| Box::new(w.clone()));
690    }
691    done.push(out);
692    done
693}

695/// A chart or diagram in a flow: an editable chart for formats that keep charts (ODT), a
696/// diagram's drawing as a group for formats that keep shapes, else its alt text.
697fn graphic_inline(
698    doc: &Document,
699    (alt, w, h): (&str, f32, f32),
700    float: &crate::doc::para::Float,
701    graphic: &crate::doc::graphic::Graphic,
702    opts: FlowOpts,
703) -> Option<Inline> {
704    if let Some(spec) = graphic.chart.as_ref().filter(|_| opts.boxes) {
705        return Some(Inline::Chart(crate::formats::model::Chart {
706            spec: spec.clone(),
707            w,
708            h,
709            alt: alt.to_string(),
710            float: kept_float(float).filter(|_| opts.paras),
711            source: graphic.source.as_deref().and_then(|e| super::odf_object::load(doc, e)),
712        }));
713    }
714    // A diagram (SmartArt) for formats without one: its drawing as a group of shapes, text boxes
715    // and pictures every reader draws.
716    if opts.boxes
717        && let Some(group) = graphic_group(doc, (w, h), float, graphic)
718    {
719        return Some(group);
720    }
721    (!alt.trim().is_empty()).then(|| Inline::Figure(alt.to_string()))
722}

724/// A graphic's drawing items ([`crate::doc::graphic::GraphicItem`]) as a group `w` × `h`
725/// points: its shapes as shapes, its text as borderless text boxes centred on their places, its
726/// pictures as pictures. `None` when nothing in it can be drawn so (paths are left out).
727fn graphic_group(doc: &Document, (w, h): (f32, f32), float: &crate::doc::para::Float, graphic: &crate::doc::graphic::Graphic) -> Option<Inline> {
728    use crate::doc::graphic::{GraphicItem, TextAlign};
729    use crate::doc::para::ShapeKind;
730    let ratio = |outer: f32, inner: f32| {
731        let r = outer / inner;
732        if r.is_finite() && r > 0.0 { r } else { 1.0 }
733    };
734    let (sx, sy) = (ratio(w, graphic.w), ratio(h, graphic.h));
735    let finite = |r: &[f32; 4]| r.iter().all(|v| v.is_finite()) && r[2] >= 0.0 && r[3] >= 0.0;
736    let plain = BoxLook { fill: None, line: None, inset: Some([0.0; 4]), ..BoxLook::default() };
737    let members: Vec<Member> = graphic
738        .items
739        .iter()
740        .take(10_000)
741        .filter_map(|item| match item {
742            GraphicItem::Shape { rect, kind, fill, stroke, stroke_width } if finite(rect) => {
743                let look = BoxLook {
744                    fill: fill.map(crate::doc::fill::Fill::solid).filter(|_| *kind != ShapeKind::Line),
745                    line: stroke.filter(|_| stroke_width.is_finite() && *stroke_width > 0.0).map(|c| (c, stroke_width.min(12.0))),
746                    ..plain.clone()
747                };
748                let inline = Inline::Shape { kind: *kind, w: rect[2] * sx, h: rect[3] * sy, float: None, blocks: Vec::new(), look, path: None };
749                Some(Member { x: rect[0] * sx, y: rect[1] * sy, inline })
750            }
751            GraphicItem::Text { rect, text, size, color, bold, align, font } if finite(rect) && !text.trim().is_empty() => {
752                let size = if size.is_finite() { size.clamp(1.0, 400.0) } else { 11.0 };
753                // One line, centred vertically in its place as the graphic draws it.
754                let line = size * 1.2;
755                let (bw, bh) = (rect[2] * sx, line.max(1.0));
756                let y = rect[1] * sy + (rect[3] * sy - bh) / 2.0;
757                let fmt = Fmt { bold: *bold, color: Some(*color), size: Some(size), font: font.as_deref().map(Arc::from), ..Fmt::default() };
758                let para = Para {
759                    align: Some(match align {
760                        TextAlign::Left => crate::doc::Align::Left,
761                        TextAlign::Right => crate::doc::Align::Right,
762                        _ => crate::doc::Align::Center,
763                    }),
764                    inlines: vec![Inline::Text(text.clone(), fmt)],
765                    spacing: Spacing { before: Some(0.0), after: Some(0.0), ..Spacing::default() },
766                    ..Para::default()
767                };
768                let inline = Inline::Shape {
769                    kind: ShapeKind::Rectangle,
770                    w: bw,
771                    h: bh,
772                    float: None,
773                    blocks: vec![FBlock::Para(para)],
774                    look: plain.clone(),
775                    path: None,
776                };
777                Some(Member { x: rect[0] * sx, y, inline })
778            }
779            GraphicItem::Image { rect, media } if finite(rect) => {
780                let data = doc.media.get(media)?;
781                let img = Img {
782                    data: data.clone(),
783                    ext: media_ext(media, data),
784                    w: rect[2] * sx,
785                    h: rect[3] * sy,
786                    alt: String::new(),
787                    float: None,
788                    object: None,
789                    crop: [0.0; 4],
790                };
791                Some(Member { x: rect[0] * sx, y: rect[1] * sy, inline: Inline::Image(img) })
792            }
793            _ => None,
794        })
795        .collect();
796    if members.is_empty() {
797        return None;
798    }
799    // A group is as big as its members together (ODF 1.3 §10.3.15): an invisible rectangle the
800    // size of the graphic keeps the room around its drawing.
801    let frame = Inline::Shape { kind: ShapeKind::Rectangle, w, h, float: None, blocks: Vec::new(), look: plain, path: None };
802    let all = std::iter::once(Member { x: 0.0, y: 0.0, inline: frame }).chain(members).collect();
803    Some(Inline::Group { w, h, float: kept_float(float), members: all })
804}

814/// The drawings from `from` on (just added) raised by their run's position `raise` (points): a
815/// drawing in line keeps it as its float's `y` (see [`crate::formats::model::Img::float`]).
816fn raise_drawings(inlines: &mut [Inline], from: usize, raise: Option<f32>) {
817    let Some(r) = raise.filter(|r| r.is_finite() && *r != 0.0) else { return };
818    for i in inlines.iter_mut().skip(from) {
819        let float = match i {
820            Inline::Image(img) => &mut img.float,
821            Inline::Chart(c) => &mut c.float,
822            Inline::Shape { float, .. } | Inline::Group { float, .. } => float,
823            _ => continue,
824        };
825        let f = float.get_or_insert_with(Default::default);
826        if f.wrap == crate::doc::para::Wrap::Inline {
827            f.y = r;
828        }
829    }
830}

832/// A content control of plain text being read (formats with form controls): it, its text so
833/// far and that text's formatting.
834#[derive(Default)]
835struct OpenControl(Option<(Box<crate::doc::control::ContentControl>, String, Option<Fmt>)>);
836
837impl OpenControl {
838    /// Object `obj` comes: anything but the control's end makes the control its text after all.
839    fn interrupt(&mut self, obj: Option<&InlineObject>, out: &mut Para, f: &Fmt) {
840        if !matches!(obj, Some(InlineObject::ControlEnd))
841            && let Some((_, text, fmt)) = self.0.take()
842        {
843            out.push_text(&text, &fmt.unwrap_or_else(|| f.clone()));
844        }
845    }
846
847    /// A content control starts or ends (`o`): whether it is one read here (a plain-text
848    /// control, in a flow that keeps them).
849    fn object(&mut self, o: &InlineObject, opts: FlowOpts, buf: &mut String, out: &mut Para, f: &Fmt) -> bool {
850        match o {
851            InlineObject::ControlStart { control: c } if opts.boxes && !c.block && plain_control(&c.kind) => {
852                if !buf.is_empty() {
853                    out.push_text(&std::mem::take(buf), f);
854                }
855                self.0 = Some((c.clone(), String::new(), None));
856                true
857            }
858            InlineObject::ControlEnd if self.0.is_some() => {
859                self.close(buf, out, f);
860                true
861            }
862            _ => false,
863        }
864    }
865
866    /// The control ends: it, with its text (the rest in `buf`), goes into `out`.
867    fn close(&mut self, buf: &mut String, out: &mut Para, f: &Fmt) {
868        if let Some((c, mut text, fmt)) = self.0.take() {
869            text.push_str(&std::mem::take(buf));
870            out.inlines.push(Inline::Control { control: c, text, fmt: fmt.unwrap_or_else(|| f.clone()) });
871        }
872    }
873
874    /// A run ends with `buf`: the control's text when one is open, else the paragraph's.
875    fn run_end(&mut self, buf: &str, out: &mut Para, f: &Fmt) {
876        match &mut self.0 {
877            Some((_, text, fmt)) => {
878                text.push_str(buf);
879                fmt.get_or_insert_with(|| f.clone());
880            }
881            None => out.push_text(buf, f),
882        }
883    }
884
885    /// The paragraph ends: a control it doesn't close is its text.
886    fn finish(&mut self, out: &mut Para) {
887        if let Some((_, text, fmt)) = self.0.take() {
888            out.push_text(&text, &fmt.unwrap_or_default());
889        }
890    }
891}

899/// The index kept as read starting at block `i` of `bl`, while the paragraphs it takes are all
900/// as read: how many blocks, and its markup.
901fn kept_index(doc: &Document, bl: &Blocks, i: usize) -> Option<(usize, std::sync::Arc<str>)> {
902    let k = doc.kept.of("odt")?;
903    let p = bl.get(i)?.as_para()?;
904    let (n, run, xml) = k.index(p)?;
905    let paras: Vec<&Paragraph> = bl.get(i..i + n)?.iter().filter_map(|b| b.as_para()).collect();
906    (paras.len() == *n && crate::doc::kept::run_fingerprint(paras.into_iter()) == *run).then(|| (*n, xml.clone()))
907}

909fn flow_blocks(doc: &Document, bl: &Blocks, depth: usize, tracked: FlowOpts) -> Vec<FBlock> {
910    let mut out = Vec::new();
911    let mut skip = 0usize;
912    for (i, b) in bl.iter().enumerate() {
913        if skip > 0 {
914            skip -= 1;
915            continue;
916        }
917        // An index kept as read, while every paragraph it takes still is what was read: its
918        // flow paragraphs, the first carrying its markup and how many there are.
919        if tracked.paras
920            && depth == 0
921            && let Some(ix) = kept_index(doc, bl, i)
922        {
923            let from = out.len();
924            for b in bl.get(i..i + ix.0).unwrap_or(&[]) {
925                if let Block::Para(p) = &**b {
926                    out.extend(flow_paras_in(doc, p, tracked).into_iter().map(FBlock::Para));
927                }
928            }
929            let n = out.len() - from;
930            if let Some(FBlock::Para(first)) = out.get_mut(from) {
931                first.index = Some(Box::new((ix.1, n)));
932            }
933            skip = ix.0.saturating_sub(1);
934            continue;
935        }
936        match &**b {
937            Block::Para(p) => out.extend(flow_paras_in(doc, p, tracked).into_iter().map(FBlock::Para)),
938            Block::Table(t) if depth < MAX_DEPTH => out.push(FBlock::Table(flow_table(doc, t, depth, tracked))),
939            Block::Table(t) => {
940                for r in &t.rows {
941                    for c in &r.cells {
942                        out.extend(flow_blocks(doc, &c.blocks, depth, tracked));
943                    }
944                }
945            }
946        }
947    }
948    out
949}

954/// What a table's style and borders give one cell, worked out as the layout does, for formats
955/// that keep cell formatting (ODT) but have no table styles of Word's kind: its shading, its four
956/// edges (a side without a line is a "none" border), its vertical alignment, and its text's
957/// character formatting and alignment.
958struct StyledCell {
959    fill: Option<crate::doc::props::Rgb>,
960    borders: crate::doc::props::Borders,
961    v_align: Option<crate::doc::props::VAlign>,
962    chr: CharProps,
963    align: Option<crate::doc::Align>,
964    /// The style's paragraph formatting (spacing, line spacing) under each paragraph's own.
965    para: ParaProps,
966}
967
968fn styled_cell(t: &Table, style: Option<&crate::doc::styles::TableStyleProps>, (ri, ci): (usize, usize), ncols: usize) -> Option<StyledCell> {
969    use crate::doc::props::{Border, BorderStyle, Borders};
970    use crate::doc::styles::{PartSet, TablePart};
971    let row = t.rows.get(ri)?;
972    let cell = row.cells.get(ci)?;
973    let nrows = t.rows.len();
974    let g = t.grid_col(ri, ci);
975    let span = cell.span();
976    let mut out = StyledCell {
977        fill: None,
978        borders: Borders::default(),
979        v_align: None,
980        chr: CharProps::default(),
981        align: None,
982        para: style.map(|s| s.para.clone()).unwrap_or_default(),
983    };
984    let mut part_borders = Vec::new();
985    if let Some(st) = style {
986        let sizes = (st.parts.band_size.unwrap_or(0), st.parts.col_band_size.unwrap_or(0));
987        for part in PartSet::of_cell(&t.props.look, sizes, (ri, nrows), (g, span, ncols)).iter() {
988            let f = st.part(part);
989            out.fill = f.fill.or(out.fill);
990            out.v_align = f.v_align.or(out.v_align);
991            out.chr.overlay(&f.chr);
992            out.align = f.align.or(out.align);
993            if part != TablePart::WholeTable
994                && let Some(b) = f.borders
995            {
996                part_borders.push((part, b));
997            }
998        }
999    }
1000    let mut tb = style.and_then(|s| s.parts.borders).unwrap_or_default();
1001    if let Some(own) = t.props.borders {
1002        tb.overlay(&own);
1003    }
1004    let cb = cell.props.borders.unwrap_or_default();
1005    let rowb = row.props.exceptions.as_ref().and_then(|e| e.borders).unwrap_or_default();
1006    let (first_g, last_g) = (g == row.first_grid(), g + span >= ncols.saturating_sub(row.grid_after()));
1007    let rb = crate::doc::styles::part_borders(&part_borders, (ri, nrows), (first_g, last_g));
1008    let edge = |own: Option<Border>, outer: bool, outer_b: Option<Border>, inner_b: Option<Border>| own.or(if outer { outer_b } else { inner_b });
1009    let none = Border { style: BorderStyle::None, width: 0.0, color: None, space: 0.0 };
1010    let top = edge(cb.top.or(if ri == 0 { rowb.top } else { rowb.between }).or(rb.top), ri == 0, tb.top, tb.between);
1011    let bottom = edge(cb.bottom.or(if ri + 1 == nrows { rowb.bottom } else { rowb.between }).or(rb.bottom), ri + 1 == nrows, tb.bottom, tb.between);
1012    let left = edge(
1013        cb.left.or(if first_g { rowb.left } else { rowb.inside_v }).or(if first_g { rb.left } else { rb.inside_v }),
1014        first_g,
1015        tb.left,
1016        tb.inside_v,
1017    );
1018    let right = edge(
1019        cb.right.or(if last_g { rowb.right } else { rowb.inside_v }).or(if last_g { rb.right } else { rb.inside_v }),
1020        last_g,
1021        tb.right,
1022        tb.inside_v,
1023    );
1024    out.borders = Borders {
1025        top: top.or(Some(none)),
1026        bottom: bottom.or(Some(none)),
1027        left: left.or(Some(none)),
1028        right: right.or(Some(none)),
1029        between: None,
1030        inside_v: None,
1031    };
1032    Some(out)
1033}

1035/// `blocks` with what a table style's character and paragraph formatting changes (see
1036/// [`crate::doc::styles::StyleSheet::table_para_patch`]) stated on each paragraph and run.
1037fn with_cell_text(doc: &Document, blocks: &Blocks, chr: &CharProps, align: Option<crate::doc::Align>, para: &ParaProps) -> Blocks {
1038    if chr.is_empty() && align.is_none() && para.is_empty() {
1039        return blocks.clone();
1040    }
1041    blocks
1042        .iter()
1043        .map(|b| match &**b {
1044            Block::Para(p) => {
1045                let mut p = p.clone();
1046                let pstyle = p.props.style.clone();
1047                if !chr.is_empty() {
1048                    for r in &mut p.runs {
1049                        let mut props = doc.styles.table_char_patch(pstyle.as_deref(), r.props.style.as_deref(), chr);
1050                        props.overlay(&r.props);
1051                        r.props = props;
1052                    }
1053                    let mut mark = doc.styles.table_char_patch(pstyle.as_deref(), p.mark.style.as_deref(), chr);
1054                    mark.overlay(&p.mark);
1055                    p.mark = mark;
1056                }
1057                // A part's alignment is table style paragraph formatting too, under the paragraph's styles.
1058                let table = ParaProps { align: align.or(para.align), ..para.clone() };
1059                if !table.is_empty() {
1060                    let mut props = doc.styles.table_para_patch(&p.props, &table);
1061                    props.overlay(&p.props);
1062                    p.props = props;
1063                }
1064                Arc::new(Block::Para(p))
1065            }
1066            _ => b.clone(),
1067        })
1068        .collect()
1069}

1071fn flow_table(doc: &Document, t: &Table, depth: usize, tracked: FlowOpts) -> FTable {
1072    let mut rows = Vec::new();
1073    // Formats that keep cell formatting get the table style's and the table's borders cell by cell.
1074    let style = t.props.style.as_deref().and_then(|id| doc.styles.table_style(id));
1075    let bake = tracked.paras;
1076    let ncols =
1077        t.rows.iter().map(|r| r.first_grid() + r.cells.iter().map(|c| c.span()).sum::<usize>() + r.grid_after()).max().unwrap_or(0).max(t.grid.len());
1078    for (ri, r) in t.rows.iter().enumerate() {
1079        let header = r.props.header || (ri == 0 && t.props.look.header_row && t.rows.len() > 1);
1080        let mut cells = Vec::new();
1081        for (ci, c) in r.cells.iter().enumerate() {
1082            let styled = if bake { styled_cell(t, style.as_ref(), (ri, ci), ncols) } else { None };
1083            let mut rowspan = 1;
1084            if c.props.vmerge == VMerge::Restart {
1085                let g = t.grid_col(ri, ci);
1086                for (k, below) in t.rows.iter().enumerate().skip(ri + 1) {
1087                    let cont = t.cell_at_grid(k, g).and_then(|i| below.cells.get(i)).is_some_and(|bc| bc.props.vmerge == VMerge::Continue);
1088                    if !cont {
1089                        break;
1090                    }
1091                    rowspan += 1;
1092                }
1093            }
1094            let blocks = match &styled {
1095                Some(st) => flow_blocks(doc, &with_cell_text(doc, &c.blocks, &st.chr, st.align, &st.para), depth + 1, tracked),
1096                None => flow_blocks(doc, &c.blocks, depth + 1, tracked),
1097            };
1098            cells.push(Cell {
1099                blocks,
1100                colspan: c.span() as u32,
1101                rowspan,
1102                covered: c.props.vmerge == VMerge::Continue,
1103                header,
1104                shading: c.props.shading.or(styled.as_ref().and_then(|s| s.fill)),
1105                borders: styled.as_ref().map(|s| s.borders).or(c.props.borders),
1106                valign: c.props.valign.or(styled.as_ref().and_then(|s| s.v_align)),
1107                // Formats that keep cell padding get each cell's, its table's or Word's default.
1108                margins: tracked.paras.then(|| c.props.resolved_margins(t.props.cell_margins.unwrap_or(WORD_CELL_MARGINS))),
1109                direction: c.props.text_direction,
1110                // Its background picture, for formats that keep cell formatting (ODT).
1111                picture: c.props.picture.as_deref().filter(|_| tracked.paras).and_then(|p| {
1112                    let data = doc.media.get(&p.media)?;
1113                    let img = Img {
1114                        data: data.clone(),
1115                        ext: media_ext(&p.media, data),
1116                        w: p.size[0],
1117                        h: p.size[1],
1118                        alt: String::new(),
1119                        float: None,
1120                        object: None,
1121                        crop: [0.0; 4],
1122                    };
1123                    Some(Box::new(crate::formats::model::CellPic { img, repeat: p.repeat, align: p.align }))
1124                }),
1125                width: None,
1126            });
1127        }
1128        rows.push(cells);
1129    }
1130    // Without a table style and without visible table or cell borders the table shows no grid.
1131    let visible = |b: &Option<crate::doc::props::Borders>| b.as_ref().is_some_and(|b| b.any_visible());
1132    let borderless =
1133        t.props.style.is_none() && !visible(&t.props.borders) && t.rows.iter().flat_map(|r| &r.cells).all(|c| !visible(&c.props.borders));
1134    // Formats that keep them get the table's alignment, width and row heights.
1135    let extra = tracked.paras.then(|| {
1136        use crate::doc::props::HeightRule;
1137        let heights = t
1138            .rows
1139            .iter()
1140            .map(|r| match (r.props.height, r.props.height_rule) {
1141                (Some(h), HeightRule::Exact) => Some((h, true)),
1142                (Some(h), HeightRule::AtLeast) => Some((h, false)),
1143                _ => None,
1144            })
1145            .collect::<Vec<_>>();
1146        super::types::TableExtra {
1147            align: t.props.align,
1148            width: t.props.width,
1149            width_pct: t.props.width_pct,
1150            heights,
1151            keep_rows: t.rows.iter().map(|r| r.props.cant_split).collect(),
1152            fixed: t.props.fixed,
1153            space: [t.props.space_before, t.props.space_after],
1154            header_rows: t.rows.iter().take_while(|r| r.props.header).count(),
1155        }
1156    });
1157    // Kept whole for formats that ask for it: its header rows count even without the rest.
1158    let extra = extra.map(Box::new);
1159    let kept = (tracked.paras && depth == 0).then(|| doc.kept.of("odt").and_then(|k| k.table(t)).cloned()).flatten();
1160    let wrap = (tracked.paras && depth == 0).then(|| doc.kept.of("odt").and_then(|k| k.table_wrap(t)).cloned().map(Box::new)).flatten();
1161    FTable { rows, widths: t.grid.clone(), borderless, extra, kept, wrap }
1162}

1202/// List labels in document order for list paragraphs of a flow ("1.", "a.", "•"), computed by a
1203/// simple per-level counter (what text exports show).
1204#[derive(Default)]
1205pub struct ListCounter {
1206    counts: [u32; 9],
1207    ordered: [bool; 9],
1208}
1209
1210impl ListCounter {
1211    pub fn reset(&mut self) {
1212        *self = ListCounter::default();
1213    }
1214    /// Advance for `li` and return its number (1-based; bullets also count).
1215    pub fn next(&mut self, li: ListInfo) -> u32 {
1216        let lv = li.level.min(8) as usize;
1217        if self.ordered.get(lv).copied() != Some(li.ordered)
1218            && let (Some(o), Some(c)) = (self.ordered.get_mut(lv), self.counts.get_mut(lv))
1219        {
1220            *o = li.ordered;
1221            *c = 0;
1222        }
1223        let n = match self.counts.get_mut(lv) {
1224            Some(c) => {
1225                if *c == 0 {
1226                    *c = li.start.saturating_sub(1);
1227                }
1228                *c = c.saturating_add(1);
1229                *c
1230            }
1231            None => 1,
1232        };
1233        for d in self.counts.iter_mut().skip(lv + 1) {
1234            *d = 0;
1235        }
1236        n
1237    }
1238    pub fn label(&mut self, li: ListInfo) -> String {
1239        let n = self.next(li);
1240        if !li.ordered {
1241            return ["•", "◦", "▪"].get(li.level as usize % 3).copied().unwrap_or("•").to_string();
1242        }
1243        match li.level % 3 {
1244            0 => format!("{n}."),
1245            1 => format!("{}.", crate::doc::section::letters(n)),
1246            _ => format!("{}.", crate::doc::section::roman(n)),
1247        }
1248    }
