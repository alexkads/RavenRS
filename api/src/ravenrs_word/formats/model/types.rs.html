23/// Link and font bytes the document's runs may carry between them. A flow's spans share one
24/// copy of a name ([`Fmt`]), but every run of a document owns its own: past this budget, runs
25/// lose their link and font rather than multiply a small file into gigabytes (#3713).
26pub(super) const MAX_RUN_STRINGS: usize = 64 << 20;

28/// `name` as a span's font, unless it is longer than [`MAX_FONT_NAME`] (then reported, see
29/// [`dropped`]).
30pub fn kept_font(name: &str) -> Option<Arc<str>> {
31    let kept = (name.len() <= MAX_FONT_NAME).then(|| name.into());
32    if kept.is_none() {
33        dropped(Dropped::Font);
34    }
35    kept
36}
37
38/// `target` as a span's link, unless it is longer than [`MAX_LINK`] (then reported, see
39/// [`dropped`]).
40pub fn kept_link(target: &str) -> Option<Arc<str>> {
41    let kept = (target.len() <= MAX_LINK).then(|| target.into());
42    if kept.is_none() {
43        dropped(Dropped::Link);
44    }
45    kept
46}

48/// What an importer leaves out to stay within a limit (review of #3873: no limit drops data
49/// without saying so).
50#[derive(Clone, Copy, Debug, PartialEq, Eq)]
51pub(crate) enum Dropped {
52    /// A font name over [`MAX_FONT_NAME`].
53    Font,
54    /// A link target over [`MAX_LINK`].
55    Link,
56    /// Runs' links and fonts past [`MAX_RUN_STRINGS`].
57    RunStrings,
58    /// A style name cut to [`MAX_FONT_NAME`].
59    StyleName,
60    /// ODT `text:s` spaces past the document's budget.
61    Spaces,
62    /// Markdown table rows and columns past the table limits or the document's cell budget.
63    TableCells,
64}
65
66impl Dropped {
67    fn message(self) -> String {
68        match self {
69            Dropped::Font => format!("import: font names longer than {MAX_FONT_NAME} bytes are dropped; that text keeps the surrounding font"),
70            Dropped::Link => format!("import: links longer than {MAX_LINK} bytes are dropped; their text is kept"),
71            Dropped::RunStrings => {
72                format!("import: the document's runs carry more than {MAX_RUN_STRINGS} bytes of links and font names; later runs lose them")
73            }
74            Dropped::StyleName => format!("import: style names longer than {MAX_FONT_NAME} bytes are cut short"),
75            Dropped::Spaces => {
76                "import: the document's `text:s` elements ask for more spaces than it keeps; past that each stands for one".to_string()
77            }
78            Dropped::TableCells => format!(
79                "import: Markdown tables keep {} columns and {} rows at most, and {} cells in all; the rest are left out",
80                crate::doc::table::MAX_COLS,
81                crate::doc::table::MAX_ROWS,
82                crate::formats::markdown::MAX_TABLE_CELLS
83            ),
84        }
85    }
86
87    /// Whether this thread's import has logged this kind ([`dropped`]).
88    #[cfg(test)]
89    pub(crate) fn reported(self) -> bool {
90        REPORTED.with(|r| r.get() & (1u8 << (self as u8)) != 0)
91    }
92}

94/// Log that the importer left something of `kind` out: once per kind per imported document
95/// ([`begin_import`]), not once per item.
96pub(crate) fn dropped(kind: Dropped) {
97    let bit = 1u8 << (kind as u8);
98    if REPORTED.with(|r| r.replace(r.get() | bit)) & bit == 0 {
99        log::warn!("{}", kind.message());
100    }
101}

105    REPORTED.with(|r| r.set(0));
106}

377/// The bookmarks in `blocks` whose end ([`Inline::AnchorEnd`]) follows their start
378/// ([`Inline::Anchor`]): ranges, not points. Tables' cells, notes, comments and shapes' text
379/// are looked through, in document order.
380pub fn ranged_anchors(blocks: &[FBlock]) -> std::collections::HashSet<String> {
381    fn walk(blocks: &[FBlock], depth: usize, started: &mut std::collections::HashSet<String>, ranged: &mut std::collections::HashSet<String>) {
382        if depth > MAX_DEPTH {
383            return;
384        }
385        for b in blocks {
386            match b {
387                FBlock::Para(p) => {
388                    for i in &p.inlines {
389                        match i {
390                            Inline::Anchor(n) => {
391                                started.insert(n.clone());
392                            }
393                            Inline::AnchorEnd(n) if started.contains(n) => {
394                                ranged.insert(n.clone());
395                            }
396                            Inline::Note { blocks, .. } | Inline::Shape { blocks, .. } => walk(blocks, depth + 1, started, ranged),
397                            Inline::Comment { comment, .. } => walk(&comment.blocks, depth + 1, started, ranged),
398                            _ => {}
399                        }
400                    }
401                }
402                FBlock::Table(t) => {
403                    for c in t.rows.iter().flatten() {
404                        walk(&c.blocks, depth + 1, started, ranged);
405                    }
406                }
407            }
408        }
409    }
410    let (mut started, mut ranged) = Default::default();
411    walk(blocks, 0, &mut started, &mut ranged);
412    ranged
413}

415/// `blocks` with each table replaced by the blocks of its cells, row by row (nested tables too),
416/// for places that hold paragraphs only.
417pub fn flatten_tables(blocks: &[FBlock]) -> Vec<FBlock> {
418    let mut out = Vec::new();
419    let mut stack: Vec<std::slice::Iter<'_, FBlock>> = vec![blocks.iter()];
420    while let Some(top) = stack.last_mut() {
421        match top.next() {
422            Some(FBlock::Para(p)) => out.push(FBlock::Para(p.clone())),
423            Some(FBlock::Table(t)) => {
424                // Cells in reverse so they come off the stack in order.
425                for c in t.rows.iter().rev().flat_map(|r| r.iter().rev()) {
426                    stack.push(c.blocks.iter());
427                }
428            }
429            None => {
430                stack.pop();
431            }
432        }
433    }
434    out
435}

481/// Base text size (points) a phonetic guide falls back to when the source states none: the
482/// Phonetic Guide command's own fallback for an unresolved size
483/// (`crates/engine/src/cmd/phonetic.rs`).
484pub const RUBY_BASE_SIZE: f32 = 10.5;
485
486/// Ruby text size for a base of `base_size` points: half the base size, rounded down to a half
487/// point, as the Phonetic Guide command sets it.
488fn ruby_default_size(base_size: f32) -> f32 {
489    (base_size.floor() / 2.0).max(1.0)
490}

492/// The ruby a format without its own ruby markup gets: a [`InlineObject::Ruby`] over `base` with
493/// the Phonetic Guide's defaults (centred, half the base size, raised clear of the base text).
494pub fn ruby_object(base: String, ruby: String, base_size: f32) -> InlineObject {
495    let base_size = if base_size.is_finite() { base_size.clamp(1.0, 1584.0) } else { RUBY_BASE_SIZE };
496    let size = ruby_default_size(base_size);
497    let raise = ruby_raise(0.0, base_size, size).clamp(0.0, 1584.0);
498    InlineObject::Ruby { base, ruby, align: RubyAlign::default(), size, raise, base_size, lang: String::new(), props: Default::default() }
499}

536/// A list label's prefix or suffix (`(`, `.`, `)`, `Chapter `), at most [`Affix::MAX`] bytes
537/// of UTF-8 (kept inline so list paragraphs stay small).
538#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
539pub struct Affix {
540    bytes: [u8; Affix::MAX],
541    len: u8,
542}
543
544impl Affix {
545    pub const MAX: usize = 8;
546
547    /// The characters of `s` that fit in [`Affix::MAX`] bytes, without control characters and
548    /// `%` (which would read as a number placeholder).
549    pub fn new(s: &str) -> Affix {
550        let mut a = Affix::default();
551        for c in s.chars().filter(|c| !c.is_control() && *c != '%') {
552            let mut buf = [0u8; 4];
553            let enc = c.encode_utf8(&mut buf).as_bytes();
554            let at = a.len as usize;
555            let Some(slot) = a.bytes.get_mut(at..at + enc.len()) else { break };
556            slot.copy_from_slice(enc);
557            a.len += enc.len() as u8;
558        }
559        a
560    }
561
562    pub fn as_str(&self) -> &str {
563        self.bytes.get(..self.len as usize).and_then(|b| std::str::from_utf8(b).ok()).unwrap_or("")
564    }
565}

575    pub shown: u8,
576    /// Where the level's text starts and how far its label hangs left of it, in twentieths of a
577    /// point, and how the label sits in that room, when the source says.
578    pub indent: Option<i16>,
579    pub hanging: Option<i16>,
580    pub align: Option<super::Align>,
581}
582
583impl Label {
584    /// A bulleted level's label: `bullet`, or the default bullet.
585    pub fn bulleted(bullet: Option<char>) -> Label {
586        let mut b = [0u8; 4];
587        let prefix = bullet.map(|c| Affix::new(c.encode_utf8(&mut b))).unwrap_or_default();
588        Label { format: crate::doc::section::NumFormat::Bullet, prefix, suffix: Affix::default(), shown: 1, indent: None, hanging: None, align: None }
589    }
590
591    /// Points as twentieths of a point, on the page.
592    pub fn twips(pt: f32) -> Option<i16> {
593        pt.is_finite().then(|| (pt.clamp(-1584.0, 1584.0) * 20.0).round() as i16)
594    }
595
596    /// This label placed `indent` points in with its label `hanging` points left of the text,
597    /// aligned `align`.
598    pub fn placed(mut self, indent: f32, hanging: f32, align: super::Align) -> Label {
599        (self.indent, self.hanging, self.align) = (Label::twips(indent), Label::twips(hanging), Some(align));
600        self
601    }
602
603    /// The indent and hanging, in points, when stated.
604    pub fn position(&self) -> Option<(f32, f32)> {
605        Some((f32::from(self.indent?) / 20.0, f32::from(self.hanging.unwrap_or(0)) / 20.0))
606    }
607
608    /// The bullet of a bulleted level, when it names one.
609    pub fn bullet(&self) -> Option<char> {
610        (self.format == crate::doc::section::NumFormat::Bullet).then(|| self.prefix.as_str().chars().next()).flatten()
611    }
612
613    /// Word's level text for 0-based `level`: `%1.`, `(%1)`, `%1.%2.`, or the bullet.
614    pub fn text(&self, level: u8) -> String {
615        let level = level.min(8);
616        if self.format == crate::doc::section::NumFormat::Bullet {
617            return self.bullet().unwrap_or('•').to_string();
618        }
619        let shown = self.shown.clamp(1, level + 1);
620        let numbers: Vec<String> = (level + 2 - shown..=level + 1).map(|n| format!("%{n}")).collect();
621        format!("{}{}{}", self.prefix.as_str(), numbers.join("."), self.suffix.as_str())
622    }
623
624    /// The label a Word level text (`%1.`, `(%1)`, `%1.%2.`, a bullet) shows.
625    pub fn of_level(format: crate::doc::section::NumFormat, text: &str) -> Label {
626        if format == crate::doc::section::NumFormat::Bullet {
627            return Label::bulleted(text.chars().next());
628        }
629        let first = text.find('%').unwrap_or(text.len());
630        // The suffix follows the last placeholder's digit.
631        let last =
632            text.rfind('%').map(|i| i + text.get(i + 1..).and_then(|r| r.chars().next()).map_or(1, |c| 1 + c.len_utf8())).unwrap_or(text.len());
633        let shown = text.matches('%').count().clamp(1, 9) as u8;
634        Label {
635            format,
636            prefix: Affix::new(text.get(..first).unwrap_or("")),
637            suffix: Affix::new(text.get(last.min(text.len())..).unwrap_or("")),
638            shown,
639            indent: None,
640            hanging: None,
641            align: None,
642        }
643    }
644}

722    /// It holds no text of its own: nothing, or only drawings, bookmarks and field bounds. Its
723    /// [`Para::mark`] formatting is then its own (its height, its drawings' runs).
724    pub fn textless(&self) -> bool {
725        self.inlines.iter().all(|i| {
726            matches!(
727                i,
728                Inline::Image(_)
729                    | Inline::Chart(_)
730                    | Inline::Shape { .. }
731                    | Inline::Group { .. }
732                    | Inline::Figure(_)
733                    | Inline::Anchor(_)
734                    | Inline::AnchorEnd(_)
735                    | Inline::CommentEnd(_)
736                    | Inline::FieldStart(_)
737                    | Inline::FieldEnd
738            )
739        })
740    }
741    /// Append text, merging with the previous span when the formatting matches.
742    pub fn push_text(&mut self, s: &str, f: &Fmt) {
743        if s.is_empty() {
744            return;
745        }
746        if let Some(Inline::Text(t, pf)) = self.inlines.last_mut()
747            && pf == f
748        {
749            t.push_str(s);
750            return;
751        }
752        self.inlines.push(Inline::Text(s.to_string(), f.clone()));
753    }
754    /// The plain text (images and anchors left out; equations as their linear text, ruby as its
755    /// base text).
756    pub fn text(&self) -> String {
757        let mut s = String::new();
758        for i in &self.inlines {
759            match i {
760                Inline::Text(t, _)
761                | Inline::Equation { linear: t, .. }
762                | Inline::Ruby { base: t, .. }
763                | Inline::Field { result: t, .. }
764                | Inline::Control { text: t, .. } => s.push_str(t),
765                Inline::Image(_)
766                | Inline::Anchor(_)
767                | Inline::Figure(_)
768                | Inline::Chart(_)
769                | Inline::Note { .. }
770                | Inline::Comment { .. }
771                | Inline::CommentEnd(_)
772                | Inline::Shape { .. }
773                | Inline::Group { .. }
774                | Inline::FieldStart(_)
775                | Inline::FieldEnd
776                | Inline::AnchorEnd(_) => {}
777            }
778        }
779        s
780    }
781    /// [`Self::text`] as it reads on screen: text set in a symbol font (Symbol, Wingdings…) as
782    /// its Unicode equivalents ([`crate::doc::symbol_fonts`]), for plain-text export.
783    pub fn shown_text(&self) -> String {
784        let mut s = String::new();
785        for i in &self.inlines {
786            match i {
787                Inline::Text(t, f) => match f.font.as_deref() {
788                    Some(font) if crate::doc::symbol_fonts::is_symbol_font(font) => s.push_str(&crate::doc::symbol_fonts::map_text(font, t)),
789                    _ => s.push_str(t),
790                },
791                Inline::Field { instr, .. } if shows_nothing(instr) => {}
792                Inline::Equation { linear: t, .. }
793                | Inline::Ruby { base: t, .. }
794                | Inline::Field { result: t, .. }
795                | Inline::Control { text: t, .. } => s.push_str(t),
796                Inline::Image(_)
797                | Inline::Anchor(_)
798                | Inline::Figure(_)
799                | Inline::Chart(_)
800                | Inline::Note { .. }
801                | Inline::Comment { .. }
802                | Inline::CommentEnd(_)
803                | Inline::Shape { .. }
804                | Inline::Group { .. }
805                | Inline::FieldStart(_)
806                | Inline::FieldEnd
807                | Inline::AnchorEnd(_) => {}
808            }
809        }
810        s
811    }
812    pub fn is_empty(&self) -> bool {
813        self.inlines.iter().all(|i| matches!(i, Inline::Text(t, _) if t.is_empty()))
814    }
815
816    /// Remove whitespace at the very start and end of the paragraph's text.
817    pub fn trim(&mut self) {
818        const WS: &[char] = &[' ', '\t', '\r', '\n'];
819        self.trim_with(WS, WS);
820    }
821
822    /// Remove the characters in `start` from the start of the paragraph's text and those in
823    /// `end` from its end.
824    pub fn trim_with(&mut self, start: &[char], end: &[char]) {
825        // The leading spans that trim to nothing go in one `drain`: removing them one at a time
826        // shifted the rest each time, quadratic in a run of separately formatted line breaks.
827        let k = self.inlines.iter().take_while(|i| matches!(i, Inline::Text(t, _) if t.trim_start_matches(start).is_empty())).count();
828        self.inlines.drain(..k);
829        if let Some(Inline::Text(t, _)) = self.inlines.first_mut() {
830            let cut = t.len() - t.trim_start_matches(start).len();
831            t.replace_range(..cut, "");
832        }
833        while let Some(Inline::Text(t, _)) = self.inlines.last_mut() {
834            let n = t.trim_end_matches(end).len();
835            t.truncate(n);
836            if t.is_empty() {
837                self.inlines.pop();
838            } else {
839                break;
840            }
841        }
842    }
843}

951impl FTable {
952    pub fn cols(&self) -> usize {
953        self.rows.iter().map(|r| r.iter().map(|c| c.colspan.clamp(1, 63) as usize).sum::<usize>()).max().unwrap_or(0)
954    }
955    /// Grid column widths from the cells that give one (after [`Self::insert_covered`], so a
956    /// cell's column counts those spanned from rows above), when every column has one.
957    pub fn widths_from_cells(&mut self) {
958        let mut widths = vec![None; self.cols()];
959        for row in &self.rows {
960            let mut g = 0;
961            for c in row {
962                if let (Some(w), Some(slot)) = (c.width, widths.get_mut(g)) {
963                    slot.get_or_insert(w);
964                }
965                g += c.colspan.clamp(1, 63) as usize;
966            }
967        }
968        self.widths = widths.into_iter().collect::<Option<Vec<f32>>>().unwrap_or_default();
969    }
970
971    /// After an importer that gives `rowspan` but no placeholder cells (HTML): insert covered
972    /// cells into the rows below each spanning cell.
973    pub fn insert_covered(&mut self) {
974        // Each spanning cell touches up to `rowspan - 1` rows below it; a table of many rows each
975        // carrying a large rowspan would otherwise be O(rows × rowspan) covered cells, each of
976        // which becomes a document cell downstream. Cap the total touched so a small file can't
977        // multiply into a huge table, whatever the rowspans claim.
978        const MAX_COVERED_WORK: usize = 1 << 17;
979        let mut work = 0usize;
980        let nrows = self.rows.len();
981        for r in 0..nrows {
982            let mut g = 0usize;
983            let mut ci = 0usize;
984            while let Some(c) = self.rows.get(r).and_then(|row| row.get(ci)) {
985                let (span, rs, covered) = (c.colspan.clamp(1, 63) as usize, c.rowspan.clamp(1, 1000) as usize, c.covered);
986                if rs > 1 && !covered {
987                    for k in 1..rs {
988                        work += 1;
989                        if work > MAX_COVERED_WORK {
990                            return;
991                        }
992                        let Some(row) = self.rows.get_mut(r + k) else { break };
993                        // Find the index in that row at grid column `g`.
994                        let mut x = 0usize;
995                        let mut at = row.len();
996                        for (i, cc) in row.iter().enumerate() {
997                            if x >= g {
998                                at = i;
999                                break;
1000                            }
1001                            x += cc.colspan.clamp(1, 63) as usize;
1002                        }
1003                        if row.len() < 64 {
1004                            row.insert(at, Cell { colspan: span as u32, covered: true, ..Default::default() });
1005                        }
1006                    }
1007                }
1008                g += span;
1009                ci += 1;
1010            }
1011        }
1012    }
1013}
