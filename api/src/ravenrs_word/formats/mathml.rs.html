1//! Equations as MathML (W3C MathML Core / Presentation MathML): written into HTML export and the
2//! HTML a copy puts on the system clipboard, so programs that read MathML can get a formatted
3//! equation instead of its linear text (#816); for a copy of one equation alone, the text of the
4//! platform's MathML clipboard formats, which Word reads first when it pastes math (#1297;
5//! pasting into Word and LibreOffice not yet verified); and read back from pasted or opened HTML.
6//!
7//! The writer walks the equation tree ([`MNode`]) and wraps it in `<semantics>` with the linear
8//! format as an `application/x-unicodemath` annotation, which renderers ignore and WordCraft reads
9//! back exactly. The reader takes that annotation when present, else a TeX annotation, else
10//! builds the tree from the presentation elements (fractions, scripts, radicals, under/over
11//! limits, tables, fences); both bound their nesting and size.
12

29/// An equation as a `<math>` element. `linear` (the linear format) goes in an annotation for an
30/// exact round trip; empty = taken from the tree.
31pub fn to_mathml(nodes: &[MNode], display: bool, linear: &str) -> String {
32    let mut o = String::new();
33    o.push_str(&format!("<math xmlns=\"{NS}\""));
34    if display {
35        o.push_str(" display=\"block\"");
36    }
37    o.push_str("><semantics><mrow>");
38    arg(nodes, &mut o, 0, false);
39    o.push_str("</mrow>");
40    let linear = if linear.is_empty() { to_linear(nodes) } else { linear.to_string() };
41    if !linear.is_empty() {
42        o.push_str(&format!("<annotation encoding=\"{UNICODEMATH}\">{}</annotation>", esc(&linear)));
43    }
44    o.push_str("</semantics></math>");
45    o
46}

53/// An argument as a row of elements. `eq_arr`: a row of an equation array, whose `&` marks are
54/// alignment points, not text.
55fn arg(a: &[MNode], o: &mut String, depth: usize, eq_arr: bool) {
56    if depth > MAX_DEPTH {
57        return;
58    }
59    for n in a {
60        node(n, o, depth + 1, eq_arr);
61    }
62}

75fn run(r: &MRun, o: &mut String, eq_arr: bool) {
76    let color = r.color.map(|c| format!(" mathcolor=\"#{}\"", c.hex())).unwrap_or_default();
77    if r.nor {
78        if !r.text.is_empty() {
79            o.push_str(&format!("<mtext{color}>{}</mtext>", esc(&r.text)));
80        }
81        return;
82    }
83    let bold = matches!(r.sty, Some(MSty::Bold | MSty::BoldItalic));
84    let italic = match r.sty {
85        None => r.scr == MScr::Roman,
86        Some(s) => matches!(s, MSty::Italic | MSty::BoldItalic),
87    };
88    let cs: Vec<char> = r.text.chars().collect();
89    let mut i = 0;
90    while let Some(&c) = cs.get(i) {
91        if c.is_ascii_digit() || (c == '.' && cs.get(i + 1).is_some_and(char::is_ascii_digit)) {
92            let mut num = String::new();
93            while let Some(&d) =
94                cs.get(i).filter(|d| d.is_ascii_digit() || **d == '.' || **d == ',' && cs.get(i + 1).is_some_and(char::is_ascii_digit))
95            {
96                num.push(d);
97                i += 1;
98            }
99            let num: String = if bold { num.chars().map(|d| math_alnum(d, true, false, MScr::Roman)).collect() } else { num };
100            o.push_str(&format!("<mn{color}>{}</mn>", esc(&num)));
101            continue;
102        }
103        i += 1;
104        if c.is_whitespace() || (eq_arr && c == '&') {
105            continue;
106        }
107        if c.is_alphabetic() {
108            if bold || r.scr != MScr::Roman {
109                let m = math_alnum(c, bold, italic, r.scr);
110                let plain = if m == c && !italic { " mathvariant=\"normal\"" } else { "" };
111                o.push_str(&format!("<mi{plain}{color}>{}</mi>", esc(&m.to_string())));
112            } else if italic {
113                o.push_str(&format!("<mi{color}>{}</mi>", esc(&c.to_string())));
114            } else {
115                o.push_str(&format!("<mi mathvariant=\"normal\"{color}>{}</mi>", esc(&c.to_string())));
116            }
117        } else if math_class(c) == MClass::Ord && !c.is_ascii_punctuation() && c != '\u{2061}' {
118            // ∞, ∂, ∅, ℏ…: identifiers, not operators.
119            o.push_str(&format!("<mi{color}>{}</mi>", esc(&c.to_string())));
120        } else {
121            mo(c, &color, o);
122        }
123    }
124}

126/// A function name (`sin`, `lim`): one upright identifier when it's plain text.
127fn func_name(name: &[MNode], o: &mut String, depth: usize) {
128    match name {
129        [MNode::Run(r)] if !r.nor && r.text.chars().all(char::is_alphabetic) && !r.text.is_empty() => {
130            o.push_str(&format!("<mi>{}</mi>", esc(&r.text)));
131        }
132        _ => group(name, o, depth),
133    }
134}

136fn fraction(kind: FracKind, num: &[MNode], den: &[MNode], o: &mut String, depth: usize) {
137    match kind {
138        FracKind::Bar | FracKind::NoBar => {
139            o.push_str(if kind == FracKind::NoBar { "<mfrac linethickness=\"0\">" } else { "<mfrac>" });
140            group(num, o, depth);
141            group(den, o, depth);
142            o.push_str("</mfrac>");
143        }
144        FracKind::Skewed | FracKind::Linear => {
145            o.push_str("<mrow>");
146            group(num, o, depth);
147            mo(if kind == FracKind::Skewed { '⁄' } else { '/' }, "", o);
148            group(den, o, depth);
149            o.push_str("</mrow>");
150        }
151    }
152}

154fn script(kind: ScriptKind, base: &[MNode], sub: &[MNode], sup: &[MNode], o: &mut String, depth: usize) {
155    match kind {
156        ScriptKind::Sup => {
157            o.push_str("<msup>");
158            group(base, o, depth);
159            group(sup, o, depth);
160            o.push_str("</msup>");
161        }
162        ScriptKind::Sub => {
163            o.push_str("<msub>");
164            group(base, o, depth);
165            group(sub, o, depth);
166            o.push_str("</msub>");
167        }
168        ScriptKind::SubSup => {
169            o.push_str("<msubsup>");
170            group(base, o, depth);
171            group(sub, o, depth);
172            group(sup, o, depth);
173            o.push_str("</msubsup>");
174        }
175        ScriptKind::Pre => {
176            o.push_str("<mmultiscripts>");
177            group(base, o, depth);
178            o.push_str("<mprescripts/>");
179            if sub.is_empty() {
180                o.push_str("<none/>");
181            } else {
182                group(sub, o, depth);
183            }
184            if sup.is_empty() {
185                o.push_str("<none/>");
186            } else {
187                group(sup, o, depth);
188            }
189            o.push_str("</mmultiscripts>");
190        }
191    }
192}

194fn write_nary(n: &MNode, o: &mut String, depth: usize) {
195    let MNode::Nary { chr, lim_loc, sub_hide, sup_hide, sub, sup, e, .. } = n else { return };
196    let under_over = match lim_loc {
197        Some(l) => *l == LimLoc::UndOvr,
198        None => !is_integral(*chr),
199    };
200    let has_sub = !*sub_hide && !sub.is_empty();
201    let has_sup = !*sup_hide && !sup.is_empty();
202    let tag = match (under_over, has_sub, has_sup) {
203        (_, false, false) => "",
204        (true, true, true) => "munderover",
205        (true, true, false) => "munder",
206        (true, false, true) => "mover",
207        (false, true, true) => "msubsup",
208        (false, true, false) => "msub",
209        (false, false, true) => "msup",
210    };
211    o.push_str("<mrow>");
212    if !tag.is_empty() {
213        o.push_str(&format!("<{tag}>"));
214    }
215    mo(*chr, " largeop=\"true\" movablelimits=\"false\"", o);
216    if has_sub {
217        group(sub, o, depth);
218    }
219    if has_sup {
220        group(sup, o, depth);
221    }
222    if !tag.is_empty() {
223        o.push_str(&format!("</{tag}>"));
224    }
225    group(e, o, depth);
226    o.push_str("</mrow>");
227}

229fn delimiters(n: &MNode, o: &mut String, depth: usize) {
230    let MNode::Delim { beg, end, sep, grow, elems, .. } = n else { return };
231    let stretchy = if *grow { "" } else { " stretchy=\"false\"" };
232    o.push_str("<mrow>");
233    if let Some(begin) = beg {
234        mo(*begin, &format!(" fence=\"true\" form=\"prefix\"{stretchy}"), o);
235    }
236    for (i, element) in elems.iter().enumerate() {
237        if i > 0 {
238            mo(sep.unwrap_or('|'), &format!(" separator=\"true\"{stretchy}"), o);
239        }
240        group(element, o, depth);
241    }
242    if let Some(close) = end {
243        mo(*close, &format!(" fence=\"true\" form=\"postfix\"{stretchy}"), o);
244    }
245    o.push_str("</mrow>");
246}

248fn node(n: &MNode, o: &mut String, depth: usize, eq_arr: bool) {
249    match n {
250        MNode::Run(r) => run(r, o, eq_arr),
251        MNode::Frac { kind, num, den } => fraction(*kind, num, den, o, depth),
252        MNode::Script { kind, base, sub, sup } => script(*kind, base, sub, sup, o, depth),
253        MNode::Rad { deg, deg_hide, e } => {
254            if *deg_hide || deg.is_empty() {
255                o.push_str("<msqrt>");
256                arg(e, o, depth, false);
257                o.push_str("</msqrt>");
258            } else {
259                o.push_str("<mroot>");
260                group(e, o, depth);
261                group(deg, o, depth);
262                o.push_str("</mroot>");
263            }
264        }
265        MNode::Nary { .. } => write_nary(n, o, depth),
266        MNode::Delim { .. } => delimiters(n, o, depth),
267        MNode::Func { name, e } => {
268            o.push_str("<mrow>");
269            func_name(name, o, depth);
270            mo('\u{2061}', "", o);
271            group(e, o, depth);
272            o.push_str("</mrow>");
273        }
274        MNode::Lim { upper, e, lim } => {
275            let tag = if *upper { "mover" } else { "munder" };
276            o.push_str(&format!("<{tag}>"));
277            func_name(e, o, depth);
278            group(lim, o, depth);
279            o.push_str(&format!("</{tag}>"));
280        }
281        MNode::Acc { chr, e } => {
282            o.push_str("<mover accent=\"true\">");
283            group(e, o, depth);
284            mo(spacing_accent(*chr), "", o);
285            o.push_str("</mover>");
286        }
287        MNode::Bar { top, e } => {
288            if *top {
289                o.push_str("<mover accent=\"true\">");
290                group(e, o, depth);
291                mo('\u{203E}', "", o);
292                o.push_str("</mover>");
293            } else {
294                o.push_str("<munder accentunder=\"true\">");
295                group(e, o, depth);
296                mo('_', "", o);
297                o.push_str("</munder>");
298            }
299        }
300        MNode::BorderBox { hide, strike, e } => {
301            let mut notation: Vec<&str> = Vec::new();
302            for (h, name) in hide.iter().zip(["top", "bottom", "left", "right"]) {
303                if !h {
304                    notation.push(name);
305                }
306            }
307            if notation.len() == 4 {
308                notation = vec!["box"];
309            }
310            for (s, name) in strike.iter().zip(["horizontalstrike", "verticalstrike", "updiagonalstrike", "downdiagonalstrike"]) {
311                if *s {
312                    notation.push(name);
313                }
314            }
315            o.push_str(&format!("<menclose notation=\"{}\">", notation.join(" ")));
316            arg(e, o, depth, false);
317            o.push_str("</menclose>");
318        }
319        MNode::Boxed { e } => group(e, o, depth),
320        MNode::GroupChr { chr, top, e } => {
321            let tag = if *top { "mover" } else { "munder" };
322            o.push_str(&format!("<{tag}>"));
323            group(e, o, depth);
324            mo(*chr, " stretchy=\"true\"", o);
325            o.push_str(&format!("</{tag}>"));
326        }
327        MNode::EqArr { rows } => {
328            // `displaystyle` marks an equation array; a bare one-column table reads as a matrix.
329            o.push_str("<mtable displaystyle=\"true\">");
330            for r in rows {
331                o.push_str("<mtr><mtd>");
332                arg(r, o, depth, true);
333                o.push_str("</mtd></mtr>");
334            }
335            o.push_str("</mtable>");
336        }
337        MNode::Matrix { rows, col_jc } => {
338            o.push_str("<mtable>");
339            for r in rows {
340                o.push_str("<mtr>");
341                for (k, cell) in r.iter().enumerate() {
342                    match col_jc.get(k) {
343                        Some(ColJc::Left) => o.push_str("<mtd columnalign=\"left\">"),
344                        Some(ColJc::Right) => o.push_str("<mtd columnalign=\"right\">"),
345                        _ => o.push_str("<mtd>"),
346                    }
347                    arg(cell, o, depth, false);
348                    o.push_str("</mtd>");
349                }
350                o.push_str("</mtr>");
351            }
352            o.push_str("</mtable>");
353        }
354        MNode::Phant { show, zero_wid, zero_asc, zero_desc, e } => {
355            let mut pad = String::new();
356            for (z, a) in [(zero_wid, "width"), (zero_asc, "height"), (zero_desc, "depth")] {
357                if *z {
358                    pad.push_str(&format!(" {a}=\"0\""));
359                }
360            }
361            if !pad.is_empty() {
362                o.push_str(&format!("<mpadded{pad}>"));
363            }
364            o.push_str(if *show { "<mrow>" } else { "<mphantom>" });
365            arg(e, o, depth, false);
366            o.push_str(if *show { "</mrow>" } else { "</mphantom>" });
367            if !pad.is_empty() {
368                o.push_str("</mpadded>");
369            }
370        }
371    }
372}

374/// Combining accents (how equations keep them) and the spacing characters MathML puts over the
375/// base, both ways.
376const ACCENTS: &[(char, char)] = &[
377    ('\u{0302}', '^'),
378    ('\u{0303}', '~'),
379    ('\u{0304}', '¯'),
380    ('\u{0305}', '\u{203E}'),
381    ('\u{0306}', '˘'),
382    ('\u{0307}', '˙'),
383    ('\u{0308}', '¨'),
384    ('\u{030C}', 'ˇ'),
385    ('\u{0301}', '´'),
386    ('\u{0300}', '`'),
387    ('\u{030A}', '˚'),
388    ('\u{20D7}', '→'),
389    ('\u{20D6}', '←'),
390    ('\u{20E1}', '↔'),
391    ('\u{20D1}', '⇀'),
392    ('\u{20D0}', '↼'),
393];
394
395fn spacing_accent(c: char) -> char {
396    ACCENTS.iter().find(|(comb, _)| *comb == c).map(|(_, s)| *s).unwrap_or(c)
397}
398
399fn combining_accent(c: char) -> Option<char> {
400    let c = match c {
401        'ˆ' => '^',
402        '˜' => '~',
403        '‾' => '\u{203E}',
404        '\u{20D7}' | '\u{0302}' | '\u{0303}' | '\u{0307}' | '\u{0308}' => return Some(c),
405        c => c,
406    };
407    ACCENTS.iter().find(|(_, s)| *s == c).map(|(comb, _)| *comb)
408}

413/// Most elements read from one `<math>` element.
414pub const MAX_NODES: usize = 20_000;

416/// A parsed MathML element (namespace prefixes dropped, names lowercase).
417#[derive(Clone, Debug, Default, PartialEq)]
418pub struct El {
419    pub name: String,
420    pub attrs: Vec<(String, String)>,
421    pub children: Vec<Item>,
422}
423
424#[derive(Clone, Debug, PartialEq)]
425pub enum Item {
426    El(El),
427    Text(String),
428}
429
430impl El {
431    fn attr(&self, name: &str) -> Option<&str> {
432        self.attrs.iter().find(|(k, _)| k == name).map(|(_, v)| v.as_str())
433    }
434    fn els(&self) -> impl Iterator<Item = &El> {
435        self.children.iter().filter_map(|c| match c {
436            Item::El(e) => Some(e),
437            Item::Text(_) => None,
438        })
439    }
440    /// The text inside, all descendants (depth-limited).
441    pub fn text(&self) -> String {
442        let mut s = String::new();
443        text_into(self, &mut s, 0);
444        s
445    }
446}
447
448fn text_into(e: &El, s: &mut String, depth: usize) {
449    if depth > MAX_DEPTH {
450        return;
451    }
452    for c in &e.children {
453        match c {
454            Item::Text(t) => s.push_str(t),
455            Item::El(c) => text_into(c, s, depth + 1),
456        }
457    }
458}

460/// The element name without a namespace prefix (`m:math`, `mml:mfrac`).
461pub fn local(name: &str) -> &str {
462    name.rsplit(':').next().unwrap_or(name)
463}
464
465/// The linear format of a `<math>` element, and whether it's a display equation. `None` when it
466/// holds nothing.
467pub fn read(math: &El) -> Option<(String, bool)> {
468    let display =
469        math.attr("display").is_some_and(|d| d.eq_ignore_ascii_case("block")) || math.attr("mode").is_some_and(|d| d.eq_ignore_ascii_case("display"));
470    let linear = linear_of(math);
471    let linear: String = linear.chars().filter(|c| !c.is_control()).collect();
472    let linear = linear.trim().to_string();
473    (!linear.is_empty()).then_some((linear, display))
474}

476fn linear_of(math: &El) -> String {
477    // Our own copy carries the linear format exactly.
478    if let Some(sem) = math.els().find(|e| local(&e.name) == "semantics") {
479        let ann = |enc: &dyn Fn(&str) -> bool| {
480            sem.els()
481                .filter(|e| local(&e.name) == "annotation")
482                .find(|e| e.attr("encoding").is_some_and(|v| enc(&v.to_ascii_lowercase())))
483                .map(El::text)
484                .filter(|t| !t.trim().is_empty())
485        };
486        if let Some(t) = ann(&|v| v == UNICODEMATH) {
487            return t;
488        }
489        let tree = to_tree(math);
490        if !tree.is_empty() {
491            return to_linear(&tree);
492        }
493        if let Some(t) = ann(&|v| v.contains("tex")) {
494            return to_linear(&crate::doc::math_latex::parse_latex(&t));
495        }
496        return String::new();
497    }
498    to_linear(&to_tree(math))
499}

501/// A MathML element's children as an equation tree.
502pub fn to_tree(math: &El) -> Arg {
503    let mut budget = MAX_NODES;
504    crate::doc::math::merge_runs(children(math, 0, &mut budget))
505}

507fn children(e: &El, depth: usize, budget: &mut usize) -> Arg {
508    let els: Vec<&El> = e.els().collect();
509    seq(&els, depth, budget)
510}

512/// A sequence of elements; an n-ary operator without an operand takes the next element.
513fn seq(els: &[&El], depth: usize, budget: &mut usize) -> Arg {
514    let mut out: Arg = Vec::new();
515    let mut i = 0;
516    while let Some(el) = els.get(i) {
517        i += 1;
518        // A function name (`<mi>sin</mi>`) applies to the next element even without U+2061,
519        // so it doesn't merge with it into `sinx`. Before an operator (`(`, U+2061) the runs
520        // merge and the linear format finds the function itself.
521        if local(&el.name) == "mi"
522            && let Some(next) = els.get(i).filter(|n| local(&n.name) != "mo")
523        {
524            let t = el.text();
525            let t = t.trim();
526            if FUNCTION_NAMES.contains(&t) && *budget > 0 {
527                *budget -= 1;
528                i += 1;
529                out.push(MNode::Func { name: vec![MNode::Run(MRun::plain(t))], e: elem(next, depth + 1, budget) });
530                continue;
531            }
532        }
533        let mut a = elem(el, depth, budget);
534        if let [MNode::Nary { e, .. }] = a.as_mut_slice()
535            && e.is_empty()
536            && let Some(next) = els.get(i)
537        {
538            i += 1;
539            *e = elem(next, depth, budget);
540        }
541        out.extend(a);
542    }
543    out
544}

546/// The `k`th child element as an argument.
547fn nth(e: &El, k: usize, depth: usize, budget: &mut usize) -> Arg {
548    e.els().nth(k).map(|c| elem(c, depth, budget)).unwrap_or_default()

551/// A lone operator character (`<mo>∑</mo>`).
552fn lone_char(a: &[MNode]) -> Option<char> {
553    match a {
554        [MNode::Run(r)] => {
555            let mut cs = r.text.trim().chars();
556            let c = cs.next()?;
557            cs.next().is_none().then_some(c)
558        }
559        _ => None,
560    }
561}

563/// A plain word (`lim`, `max`).
564fn word(a: &[MNode]) -> bool {
565    matches!(a, [MNode::Run(r)] if r.text.chars().count() > 1 && r.text.chars().all(char::is_alphabetic))
566}

568fn elem(e: &El, depth: usize, budget: &mut usize) -> Arg {
569    if depth > MAX_DEPTH || *budget == 0 {
570        return Vec::new();
571    }
572    *budget -= 1;
573    let d = depth + 1;
574    match local(&e.name) {
575        "mi" | "mn" | "mo" | "ms" => {
576            let t = e.text();
577            let t = t.trim();
578            if t.is_empty() {
579                return Vec::new();
580            }
581            let mut r = MRun::new(t);
582            if local(&e.name) == "mi" && e.attr("mathvariant") == Some("normal") && t.chars().count() == 1 {
583                r.sty = Some(MSty::Plain);
584            }
585            vec![MNode::Run(r)]
586        }
587        "mtext" => {
588            let t = e.text();
589            if t.trim().is_empty() {
590                return Vec::new();
591            }
592            vec![MNode::Run(MRun { text: t, nor: true, ..Default::default() })]
593        }
594        "mfrac" => {
595            let thin = e.attr("linethickness").is_some_and(|t| {
596                let s = t.trim();
597                s.eq_ignore_ascii_case("none") || s.trim_end_matches(|c: char| c.is_alphabetic() || c == '%').trim() == "0"
598            });
599            let kind = if e.attr("bevelled").is_some_and(|b| b.eq_ignore_ascii_case("true")) {
600                FracKind::Skewed
601            } else if thin {
602                FracKind::NoBar
603            } else {
604                FracKind::Bar
605            };
606            vec![MNode::Frac { kind, num: nth(e, 0, d, budget), den: nth(e, 1, d, budget) }]
607        }
608        "msqrt" => vec![MNode::Rad { deg: Vec::new(), deg_hide: true, e: children(e, d, budget) }],
609        "mroot" => vec![MNode::Rad { deg: nth(e, 1, d, budget), deg_hide: false, e: nth(e, 0, d, budget) }],
610        "msup" | "msub" | "msubsup" => {
611            let name = local(&e.name);
612            let base = nth(e, 0, d, budget);
613            let (sub, sup) = match name {
614                "msup" => (Vec::new(), nth(e, 1, d, budget)),
615                "msub" => (nth(e, 1, d, budget), Vec::new()),
616                _ => (nth(e, 1, d, budget), nth(e, 2, d, budget)),
617            };
618            if let Some(c) = lone_char(&base).filter(|c| is_nary_char(*c)) {
619                return vec![nary(c, LimLoc::SubSup, sub, sup)];
620            }
621            let kind = match name {
622                "msup" => ScriptKind::Sup,
623                "msub" => ScriptKind::Sub,
624                _ => ScriptKind::SubSup,
625            };
626            vec![MNode::Script { kind, base, sub, sup }]
627        }
628        "munder" | "mover" | "munderover" => {
629            let name = local(&e.name);
630            let base = nth(e, 0, d, budget);
631            let (under, over) = match name {
632                "munder" => (nth(e, 1, d, budget), Vec::new()),
633                "mover" => (Vec::new(), nth(e, 1, d, budget)),
634                _ => (nth(e, 1, d, budget), nth(e, 2, d, budget)),
635            };
636            if let Some(c) = lone_char(&base).filter(|c| is_nary_char(*c)) {
637                return vec![nary(c, LimLoc::UndOvr, under, over)];
638            }
639            if name == "munderover" {
640                let inner = MNode::Lim { upper: false, e: base, lim: under };
641                return vec![MNode::Lim { upper: true, e: vec![inner], lim: over }];
642            }
643            let upper = name == "mover";
644            let mark = if upper { lone_char(&over) } else { lone_char(&under) };
645            match mark {
646                Some('\u{203E}' | '¯' | '‾') if upper => vec![MNode::Bar { top: true, e: base }],
647                Some('_' | '\u{203E}' | '¯') if !upper => vec![MNode::Bar { top: false, e: base }],
648                Some(c @ ('⏞' | '⏟' | '⏜' | '⏝' | '⎴' | '⎵' | '←' | '→' | '↔' | '⇐' | '⇒' | '⇔')) if !word(&base) => {
649                    vec![MNode::GroupChr { chr: c, top: upper, e: base }]
650                }
651                Some(c) if upper && combining_accent(c).is_some() => vec![MNode::Acc { chr: combining_accent(c).unwrap_or(c), e: base }],
652                _ => vec![MNode::Lim { upper, e: base, lim: if upper { over } else { under } }],
653            }
654        }
655        "mmultiscripts" => {
656            let els: Vec<&El> = e.els().collect();
657            let base = els.first().map(|b| elem(b, d, budget)).unwrap_or_default();
658            let pre_at = els.iter().position(|c| local(&c.name) == "mprescripts");
659            let script =
660                |k: usize, budget: &mut usize| els.get(k).filter(|c| local(&c.name) != "none").map(|c| elem(c, d, budget)).unwrap_or_default();
661            match pre_at {
662                Some(p) => {
663                    let sub = script(p + 1, budget);
664                    let sup = script(p + 2, budget);
665                    let pre_node = MNode::Script { kind: ScriptKind::Pre, base, sub, sup };
666                    if p > 1 {
667                        let post_sub = script(1, budget);
668                        let post_sup = script(2, budget);
669                        let kind = match (post_sub.is_empty(), post_sup.is_empty()) {
670                            (true, true) => return vec![pre_node],
671                            (false, true) => ScriptKind::Sub,
672                            (true, false) => ScriptKind::Sup,
673                            (false, false) => ScriptKind::SubSup,
674                        };
675                        vec![MNode::Script { kind, base: vec![pre_node], sub: post_sub, sup: post_sup }]
676                    } else {
677                        vec![pre_node]
678                    }
679                }
680                None => {
681                    let sub = script(1, budget);
682                    let sup = script(2, budget);
683                    let kind = match (sub.is_empty(), sup.is_empty()) {
684                        (true, true) => return base,
685                        (false, true) => ScriptKind::Sub,
686                        (true, false) => ScriptKind::Sup,
687                        (false, false) => ScriptKind::SubSup,
688                    };
689                    vec![MNode::Script { kind, base, sub, sup }]
690                }
691            }
692        }
693        "mfenced" => {
694            let first = |a: &str, dflt: char| e.attr(a).map(|v| v.trim().chars().next()).unwrap_or(Some(dflt));
695            let beg = first("open", '(');
696            let end = first("close", ')');
697            let sep = first("separators", ',');
698            let elems: Vec<Arg> = e.els().map(|c| elem(c, d, budget)).collect();
699            vec![MNode::Delim { beg, end, sep, grow: true, shp_match: false, elems }]
700        }
701        "mtable" => {
702            let mtr_list: Vec<&El> = e.els().filter(|r| matches!(local(&r.name), "mtr" | "mlabeledtr")).take(256).collect();
703            let rows: Vec<Vec<Arg>> =
704                mtr_list.iter().map(|r| r.els().filter(|c| local(&c.name) == "mtd").take(256).map(|c| children(c, d, budget)).collect()).collect();
705            if rows.is_empty() {
706                return Vec::new();
707            }
708            // One column is an equation array only when marked as one (`displaystyle`, alignment
709            // marks, labelled rows, as TeX's align and our own writer give); else a column vector.
710            let marked =
711                e.attr("displaystyle").is_some_and(|v| v.trim() == "true") || e.els().any(|r| local(&r.name) == "mlabeledtr") || has_align_mark(e, 0);
712            if marked && rows.iter().all(|r| r.len() <= 1) {
713                return vec![MNode::EqArr { rows: rows.into_iter().map(|mut r| r.pop().unwrap_or_default()).collect() }];
714            }
715            let mut col_jc: Vec<ColJc> = e
716                .attr("columnalign")
717                .map(|ca| {
718                    ca.split_whitespace()
719                        .map(|a| match a {
720                            "left" => ColJc::Left,
721                            "right" => ColJc::Right,
722                            _ => ColJc::Center,
723                        })
724                        .collect()
725                })
726                .unwrap_or_default();
727            if col_jc.is_empty()
728                && let Some(first_row) = mtr_list.first()
729            {
730                let cell_aligns: Vec<ColJc> = first_row
731                    .els()
732                    .filter(|c| local(&c.name) == "mtd")
733                    .map(|c| match c.attr("columnalign") {
734                        Some("left") => ColJc::Left,
735                        Some("right") => ColJc::Right,
736                        _ => ColJc::Center,
737                    })
738                    .collect();
739                if cell_aligns.iter().any(|a| *a != ColJc::Center) {
740                    col_jc = cell_aligns;
741                }
742            }
743            vec![MNode::Matrix { rows, col_jc }]
744        }
745        "mpadded" => {
746            let zero_wid = e.attr("width").is_some_and(|w| w.trim() == "0");
747            let zero_asc = e.attr("height").is_some_and(|h| h.trim() == "0");
748            let zero_desc = e.attr("depth").is_some_and(|d| d.trim() == "0");
749            let ch = children(e, d, budget);
750            if zero_wid || zero_asc || zero_desc {
751                if let [MNode::Phant { show, e, .. }] = ch.as_slice() {
752                    vec![MNode::Phant { show: *show, zero_wid, zero_asc, zero_desc, e: e.clone() }]
753                } else {
754                    vec![MNode::Phant { show: true, zero_wid, zero_asc, zero_desc, e: ch }]
755                }
756            } else {
757                ch
758            }
759        }
760        "mphantom" => vec![MNode::Phant { show: false, zero_wid: false, zero_asc: false, zero_desc: false, e: children(e, d, budget) }],
761        "menclose" => {
762            let n = e.attr("notation").unwrap_or("longdiv");
763            let has = |k: &str| n.split_whitespace().any(|w| w.eq_ignore_ascii_case(k));
764            if has("radical") {
765                return vec![MNode::Rad { deg: Vec::new(), deg_hide: true, e: children(e, d, budget) }];
766            }
767            // An actuarial symbol is the top and right edges.
768            let actuarial = has("actuarial");
769            let is_box = has("box") || has("roundedbox") || has("circle");
770            let (top, right) = (has("top") || actuarial, has("right") || actuarial);
771            let has_border = is_box || top || has("bottom") || has("left") || right;
772            let strike =
773                [has("horizontalstrike"), has("verticalstrike"), has("updiagonalstrike") || has("cross"), has("downdiagonalstrike") || has("cross")];
774            if has_border || strike.iter().any(|&s| s) {
775                let hide = if is_box {
776                    [false; 4]
777                } else if has_border {
778                    [!top, !has("bottom"), !has("left"), !right]
779                } else {
780                    [true; 4]
781                };
782                return vec![MNode::BorderBox { hide, strike, e: children(e, d, budget) }];
783            }
784            children(e, d, budget)
785        }
786        // The first child of `semantics` is the presentation; annotations aren't content.
787        "semantics" => e.els().next().map(|c| elem(c, d, budget)).unwrap_or_default(),
788        "annotation" | "annotation-xml" | "mspace" | "none" | "mprescripts" | "maligngroup" | "malignmark" => Vec::new(),
789        _ => children(e, d, budget),
790    }
791}

793/// Whether `e` holds a `maligngroup` or `malignmark` near the top (in a cell or a row inside it).
794fn has_align_mark(e: &El, depth: usize) -> bool {
795    depth <= 4 && e.els().any(|c| matches!(local(&c.name), "maligngroup" | "malignmark") || has_align_mark(c, depth + 1))
796}

799    let lim_loc = Some(lim_loc).filter(|l| *l != if is_integral(chr) { LimLoc::SubSup } else { LimLoc::UndOvr });

801}
