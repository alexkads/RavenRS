1//! MHT / MHTML (Single File Web Page): a web page and its pictures in one MIME message (RFC 2557
2//! `multipart/related`, RFC 2045 transfer encodings). The page is the part the message's `start`
3//! parameter names, else its first HTML part; it is read by [`crate::formats::html`], with `img` sources
4//! found among the other parts by `Content-Location` (also relative to the page's) or `cid:`
5//! `Content-ID`, and the page opens as Word opens one ([`crate::formats::html::import_page`]). Word keeps
6//! HTML `w:altChunk`s in this form too.
7
8use std::collections::HashMap;
9use std::sync::Arc;
10
11use crate::doc::Document;
12
13/// Most parts read from one message (nested multiparts included).
14const MAX_PARTS: usize = 10_000;
15/// Deepest `multipart` nesting followed.
16const MAX_NESTING: usize = 8;
17/// Longest header block looked at for one part.
18const MAX_HEADER: usize = 64 * 1024;
19/// Longest boundary (RFC 2046 §5.1.1).
20const MAX_BOUNDARY: usize = 70;
21/// Picture bytes one page may load, counting every `img` that names a part.
22const PICTURE_BUDGET: usize = 1 << 30;
23
24/// One leaf part of the message: its headers (names lower-cased) and decoded body.
25struct Part {
26    content_type: String,
27    location: Option<String>,
28    id: Option<String>,
29    /// Shared by every `img` that names it.
30    body: Arc<Vec<u8>>,
31}
32
33impl Part {
34    fn mime(&self) -> String {
35        self.content_type.split(';').next().unwrap_or("").trim().to_ascii_lowercase()
36    }
37}

39/// Does `bytes` look like a MIME message (an `.mht` file)?
40pub fn is_mht(bytes: &[u8]) -> bool {
41    let head = String::from_utf8_lossy(bytes.get(..bytes.len().min(4096)).unwrap_or(bytes)).to_ascii_lowercase();
42    head.starts_with("mime-version:") || head.starts_with("content-type: multipart/") || head.contains("\ncontent-type: multipart/related")
43}

45/// Import an MHT file. A message without an HTML part reads as its text (an empty document
46/// when there's none).
47pub fn import(bytes: &[u8]) -> Document {
48    import_on(bytes, &crate::doc::SectionProps::default())
49}

51/// [`import`] onto `page`: like HTML, an MHT file gives no page size (#1381).
52pub fn import_on(bytes: &[u8], page: &crate::doc::SectionProps) -> Document {
53    let mut parts = Vec::new();
54    let (headers, body) = split_headers(bytes);
55    let start = param(header(&headers, "content-type").unwrap_or(""), "start").map(|s| s.trim_matches(['<', '>']).to_string());
56    collect(&headers, body, &mut parts, 0);
57    let root = start
58        .as_deref()
59        .and_then(|s| parts.iter().position(|p| p.id.as_deref() == Some(s)))
60        .or_else(|| parts.iter().position(|p| matches!(p.mime().as_str(), "text/html" | "application/xhtml+xml")));
61    let Some(root) = root.and_then(|i| parts.get(i)) else {
62        let text = parts.iter().find(|p| p.mime() == "text/plain").map(|p| p.body.as_slice()).unwrap_or(&[]);
63        return crate::formats::txt::import_on(text, page);
64    };
65    let html = crate::formats::html::decode_as(&root.body, param(&root.content_type, "charset"));
66    let base = root.location.as_deref().unwrap_or("");
67    // Parts by `Content-Location` (lower case) and `Content-ID`; the first of each wins.
68    let mut by_location = HashMap::new();
69    let mut by_id = HashMap::new();
70    for p in &parts {
71        if let Some(l) = &p.location {
72            by_location.entry(l.to_ascii_lowercase()).or_insert(p);
73        }
74        if let Some(i) = &p.id {
75            by_id.entry(i.as_str()).or_insert(p);
76        }
77    }
78    let left = std::cell::Cell::new(PICTURE_BUDGET);
79    let loader = |src: &str| -> Option<Arc<Vec<u8>>> {
80        let src = src.trim();
81        let part = match src.get(..4).filter(|s| s.eq_ignore_ascii_case("cid:")).and(src.get(4..)) {
82            Some(cid) => by_id.get(cid),
83            // By its location as written, or resolved against the page's.
84            None => by_location.get(&src.to_ascii_lowercase()).or_else(|| by_location.get(&resolve(base, src).to_ascii_lowercase())),
85        }?;
86        left.set(left.get().checked_sub(part.body.len())?);
87        Some(part.body.clone())
88    };
89    crate::formats::html::import_on(&html, &loader, page)
90}

92/// `reference` resolved against the page's location `base`, as RFC 2557 §5 asks (RFC 3986
93/// §5.2.2): a reference with a scheme stays as it is, `//host/…` takes the base's scheme, `/…`
94/// its scheme and host, an empty path its path and query, anything else is merged with its path
95/// (§5.2.3); `.` and `..` segments go (§5.2.4) and fragments are dropped. Nothing is fetched.
96fn resolve(base: &str, reference: &str) -> String {
97    let (b, r) = (UriParts::of(base), UriParts::of(reference));
98    let t = if r.scheme.is_some() {
99        UriParts { path: remove_dots(&r.path), ..r }
100    } else if r.authority.is_some() {
101        UriParts { scheme: b.scheme, path: remove_dots(&r.path), ..r }
102    } else if r.path.is_empty() {
103        UriParts { query: r.query.or(b.query), ..b }
104    } else if r.path.starts_with('/') {
105        UriParts { path: remove_dots(&r.path), query: r.query, ..b }
106    } else {
107        let merged = if b.authority.is_some() && b.path.is_empty() {
108            format!("/{}", r.path)
109        } else {
110            format!("{}{}", b.path.rfind('/').and_then(|i| b.path.get(..=i)).unwrap_or(""), r.path)
111        };
112        UriParts { path: remove_dots(&merged), query: r.query, ..b }
113    };
114    t.to_string()
115}

117/// A URI's parts (RFC 3986 §3): `scheme:`, `//authority`, path and `?query`; the fragment left out.
118struct UriParts<'a> {
119    scheme: Option<&'a str>,
120    authority: Option<&'a str>,
121    path: String,
122    query: Option<&'a str>,
123}
124
125impl<'a> UriParts<'a> {
126    fn of(uri: &'a str) -> Self {
127        let uri = uri.split('#').next().unwrap_or("");
128        // A one-letter scheme is a Windows drive (`C:/…`), not a scheme.
129        let scheme = uri.split_once(':').map(|(k, _)| k).filter(|k| {
130            k.len() > 1 && k.starts_with(|c: char| c.is_ascii_alphabetic()) && k.chars().all(|c| c.is_ascii_alphanumeric() || "+-.".contains(c))
131        });
132        let rest = scheme.and_then(|k| uri.get(k.len() + 1..)).unwrap_or(uri);
133        let (authority, rest) = match rest.strip_prefix("//") {
134            Some(r) => {
135                let end = r.find(['/', '?']).unwrap_or(r.len());
136                (r.get(..end), r.get(end..).unwrap_or(""))
137            }
138            None => (None, rest),
139        };
140        let (path, query) = match rest.split_once('?') {
141            Some((p, q)) => (p, Some(q)),
142            None => (rest, None),
143        };
144        UriParts { scheme, authority, path: path.to_string(), query }
145    }
146}
147
148impl std::fmt::Display for UriParts<'_> {
149    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
150        if let Some(s) = self.scheme {
151            write!(f, "{s}:")?;
152        }
153        if let Some(a) = self.authority {
154            write!(f, "//{a}")?;
155        }
156        f.write_str(&self.path)?;
157        if let Some(q) = self.query {
158            write!(f, "?{q}")?;
159        }
160        Ok(())
161    }
162}

164/// `path` without its `.` and `..` segments (RFC 3986 §5.2.4); `..` stops at the root.
165fn remove_dots(path: &str) -> String {
166    let mut out: Vec<&str> = Vec::new();
167    let segments: Vec<&str> = path.split('/').collect();
168    for (i, seg) in segments.iter().enumerate() {
169        match *seg {
170            "." => {}
171            ".." => {
172                if out.len() > 1 || out.first().is_some_and(|f| !f.is_empty()) {
173                    out.pop();
174                }
175            }
176            s => out.push(s),
177        }
178        if i + 1 == segments.len() && matches!(*seg, "." | "..") {
179            out.push("");
180        }
181    }
182    out.join("/")
183}

185/// Split a part into its unfolded header lines and its body (after the first empty line).
186fn split_headers(b: &[u8]) -> (Vec<(String, String)>, &[u8]) {
187    // Whichever blank line comes first ends the headers.
188    let blank = [find_bytes(b, b"\r\n\r\n").map(|i| (i, 4)), find_bytes(b, b"\n\n").map(|i| (i, 2))].into_iter().flatten().min();
189    let (head, body) = match blank {
190        Some((i, n)) => (b.get(..i).unwrap_or(&[]), b.get(i + n..).unwrap_or(&[])),
191        // Headers only, or a body without them.
192        None if b.first().is_some_and(|c| c.is_ascii_alphabetic()) && find_bytes(b, b":").is_some() => (b, &[][..]),
193        None => (&[][..], b),
194    };
195    let head = String::from_utf8_lossy(head.get(..head.len().min(MAX_HEADER)).unwrap_or(head));
196    let mut headers: Vec<(String, String)> = Vec::new();
197    for line in head.lines() {
198        if line.starts_with([' ', '\t']) {
199            if let Some((_, v)) = headers.last_mut() {
200                v.push(' ');
201                v.push_str(line.trim());
202            }
203        } else if let Some((k, v)) = line.split_once(':') {
204            headers.push((k.trim().to_ascii_lowercase(), v.trim().to_string()));
205        }
206    }
207    (headers, body)
208}

210fn header<'a>(h: &'a [(String, String)], name: &str) -> Option<&'a str> {
211    h.iter().find(|(k, _)| k == name).map(|(_, v)| v.as_str())
212}

214/// A `; name=value` parameter of a header value (quotes removed), name case-insensitive.
215fn param<'a>(value: &'a str, name: &str) -> Option<&'a str> {
216    // Split outside quoted strings and comments (RFC 2045 §5.1, RFC 822 §3.3).
217    let mut quoted = false;
218    let mut escaped = false;
219    let mut comment_depth = 0usize;
220    value
221        .split(|c| {
222            if escaped {
223                escaped = false;
224            } else if (quoted || comment_depth > 0) && c == '\\' {
225                escaped = true;
226            } else if !quoted && c == '(' {
227                comment_depth = comment_depth.saturating_add(1);
228            } else if comment_depth > 0 {
229                if c == ')' {
230                    comment_depth -= 1;
231                }
232            } else if c == '"' {
233                quoted = !quoted;
234            }
235            c == ';' && !quoted && comment_depth == 0
236        })
237        .skip(1)
238        .find_map(|p| {
239            let (k, v) = p.split_once('=')?;
240            k.trim().eq_ignore_ascii_case(name).then(|| v.trim().trim_matches('"'))
241        })
242}

244/// Gather the leaf parts of a (possibly multipart) entity.
245fn collect(headers: &[(String, String)], body: &[u8], out: &mut Vec<Part>, depth: usize) {
246    if out.len() >= MAX_PARTS {
247        return;
248    }
249    let ct = header(headers, "content-type").unwrap_or("text/plain").to_string();
250    if ct.trim_start().to_ascii_lowercase().starts_with("multipart/") {
251        let Some(boundary) = param(&ct, "boundary").filter(|b| !b.is_empty() && b.len() <= MAX_BOUNDARY) else { return };
252        if depth >= MAX_NESTING {
253            return;
254        }
255        for chunk in multipart_bodies(body, boundary.as_bytes()) {
256            let (h, b) = split_headers(chunk);
257            collect(&h, b, out, depth + 1);
258            if out.len() >= MAX_PARTS {
259                return;
260            }
261        }
262        return;
263    }
264    let enc = header(headers, "content-transfer-encoding").unwrap_or("").trim().to_ascii_lowercase();
265    let body = match enc.as_str() {
266        "quoted-printable" => quoted_printable(body),
267        "base64" => crate::formats::model::base64_decode(&String::from_utf8_lossy(body)).unwrap_or_default(),
268        _ => body.to_vec(),
269    };
270    let clean = |s: &str| s.trim().trim_matches(['<', '>']).to_string();
271    out.push(Part {
272        content_type: ct,
273        location: header(headers, "content-location").map(clean),
274        id: header(headers, "content-id").map(clean),
275        body: Arc::new(body),
276    });
277}

279/// The bodies between `--boundary` lines, up to the closing `--boundary--`.
280fn multipart_bodies<'a>(body: &'a [u8], boundary: &[u8]) -> Vec<&'a [u8]> {
281    let mut delim = b"--".to_vec();
282    delim.extend_from_slice(boundary);
283    let mut out = Vec::new();
284    // Delimiters sit at line starts.
285    let mut starts = Vec::new();
286    let mut i = 0;
287    while let Some(k) = body.get(i..).and_then(|rest| find_bytes(rest, &delim)) {
288        let at = i + k;
289        if at == 0 || body.get(at - 1) == Some(&b'\n') {
290            starts.push(at);
291            if starts.len() > MAX_PARTS {
292                break;
293            }
294        }
295        i = at + delim.len();
296    }
297    for w in starts.windows(2) {
298        let (Some(&a), Some(&b)) = (w.first(), w.get(1)) else { continue };
299        let Some(seg) = body.get(a + delim.len()..b) else { continue };
300        if seg.starts_with(b"--") {
301            break;
302        }
303        // The rest of the delimiter line, then the part; the line break before the next
304        // delimiter belongs to it.
305        let seg = match find_bytes(seg, b"\n") {
306            Some(n) => seg.get(n + 1..).unwrap_or(&[]),
307            None => &[],
308        };
309        let seg = seg.strip_suffix(b"\r\n").or_else(|| seg.strip_suffix(b"\n")).unwrap_or(seg);
310        out.push(seg);
311        if out.len() >= MAX_PARTS {
312            break;
313        }
314    }
315    out
316}

318fn find_bytes(hay: &[u8], needle: &[u8]) -> Option<usize> {
319    if needle.is_empty() {
320        return None;
321    }
322    hay.windows(needle.len()).position(|w| w == needle)
323}

325/// Decode quoted-printable (RFC 2045 §6.7): `=XX` bytes and `=` soft line breaks; anything
326/// malformed is kept as it stands.
327fn quoted_printable(b: &[u8]) -> Vec<u8> {
328    let hex = |c: u8| (c as char).to_digit(16).and_then(|d| u8::try_from(d).ok());
329    let mut out = Vec::with_capacity(b.len());
330    let mut i = 0;
331    while let Some(&c) = b.get(i) {
332        if c != b'=' {
333            out.push(c);
334            i += 1;
335            continue;
336        }
337        match (b.get(i + 1).copied(), b.get(i + 2).copied()) {
338            (Some(b'\r'), Some(b'\n')) => i += 3,
339            (Some(b'\n'), _) => i += 2,
340            (Some(h), Some(l)) if let (Some(h), Some(l)) = (hex(h), hex(l)) => {
341                out.push((h << 4) | l);
342                i += 3;
343            }
344            _ => {
345                out.push(c);
346                i += 1;
347            }
348        }
349    }
350    out
351}

465        assert_eq!(d.plain_text(crate::doc::StoryRef::Body).trim(), "Just text");
466        assert!(is_mht(m.as_bytes()) && !is_mht(b"<html><p>x</p>"));
467    }
