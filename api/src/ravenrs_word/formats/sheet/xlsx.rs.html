15/// The relationships of a part in `dir` (targets resolved to package paths).
16pub(super) fn relationships(xml: &[u8], dir: &str) -> Vec<Rel> {
17    let mut v = Vec::new();
18    walk(xml, |ev| {
19        if let Ev::Start(e) = ev
20            && e.local_name().as_ref() == b"Relationship"
21            && v.len() < MAX_SHEETS * 2
22        {
23            if attr(e, b"TargetMode").is_some_and(|m| m.eq_ignore_ascii_case("External")) {
24                return;
25            }
26            let target = attr(e, b"Target").unwrap_or_default();
27            v.push(Rel { id: attr(e, b"Id").unwrap_or_default(), kind: attr(e, b"Type").unwrap_or_default(), target: join(dir, &target) });
28        }
29    });
30    v
31}

33/// A relative part name resolved against `dir` (`/` = package root).
34fn join(dir: &str, target: &str) -> String {
35    let mut parts: Vec<&str> = if target.starts_with('/') { Vec::new() } else { dir.split('/').filter(|p| !p.is_empty()).collect() };
36    for seg in target.split('/') {
37        match seg {
38            "" | "." => {}
39            ".." => {
40                parts.pop();
41            }
42            s => parts.push(s),
43        }
44    }
45    parts.join("/")
46}

48fn dir_of(path: &str) -> &str {
49    path.rsplit_once('/').map_or("", |(d, _)| d)
50}

52fn rels_of(path: &str) -> String {
53    match path.rsplit_once('/') {
54        Some((d, f)) => format!("{d}/_rels/{f}.rels"),
55        None => format!("_rels/{path}.rels"),
56    }

71fn workbook(pkg: &mut Package<'_>) -> Result<Workbook, String> {
72    let path = pkg.workbook_path().ok_or("the workbook part is missing")?;
73    let xml = pkg.read(&path)?;
74    let rels = pkg.read(&rels_of(&path)).map(|r| relationships(&r, dir_of(&path))).unwrap_or_default();
75    let mut listed: Vec<(String, String)> = Vec::new();
76    let mut date1904 = false;
77    walk(&xml, |ev| {
78        if let Ev::Start(e) = ev {
79            match e.local_name().as_ref() {
80                b"sheet" if listed.len() < MAX_SHEETS => listed.push((attr(e, b"name").unwrap_or_default(), attr(e, b"id").unwrap_or_default())),
81                b"workbookPr" => date1904 = attr(e, b"date1904").is_some_and(|v| v == "1" || v.eq_ignore_ascii_case("true")),
82                _ => {}
83            }
84        }
85    });
86    // Worksheets only: chart and dialog sheets hold no cells. Indexed by id once (the first
87    // relationship with an id wins), not searched for every sheet.
88    let mut worksheets: HashMap<&str, &str> = HashMap::new();
89    for r in rels.iter().filter(|r| r.kind.ends_with("/worksheet")) {
90        worksheets.entry(r.id.as_str()).or_insert(r.target.as_str());
91    }
92    let sheets = listed.into_iter().filter_map(|(name, id)| Some(SheetRef { name, path: worksheets.get(id.as_str())?.to_string() })).collect();
93    let part = |suffix: &str| rels.iter().find(|r| r.kind.ends_with(suffix)).map(|r| r.target.clone());
94    Ok(Workbook { sheets, date1904, shared: part("/sharedStrings"), styles: part("/styles") })
95}

101/// The shared string table (§18.4): each `si`'s text, rich-text runs joined, phonetic runs
102/// (`rPh`) left out.
103fn shared_strings(xml: &[u8]) -> Vec<String> {
104    let mut out = Vec::new();
105    let mut cur: Option<(String, usize)> = None;
106    let (mut in_t, mut phonetic) = (false, 0usize);
107    walk(xml, |ev| match ev {
108        Ev::Start(e) => match e.local_name().as_ref() {
109            b"si" => cur = Some((String::new(), 0)),
110            b"t" => in_t = true,
111            b"rPh" => phonetic += 1,
112            _ => {}
113        },
114        Ev::End(n) => match n.as_slice() {
115            b"si" => {
116                if let Some((s, _)) = cur.take()
117                    && out.len() < MAX_SHARED
118                {
119                    out.push(s);
120                }
121            }
122            b"t" => in_t = false,
123            b"rPh" => phonetic = phonetic.saturating_sub(1),
124            _ => {}
125        },
126        Ev::Text(t) => {
127            if in_t
128                && phonetic == 0
129                && let Some((s, n)) = cur.as_mut()
130            {
131                push_capped(s, n, &t);
132            }
133        }
134    });
135    out
136}

138/// Each cell format's (`cellXfs/xf`, by `s` index) date kind.
139fn cell_formats(xml: &[u8]) -> Vec<Option<DateKind>> {
140    let mut custom: HashMap<u32, Option<DateKind>> = HashMap::new();
141    let mut xfs: Vec<u32> = Vec::new();
142    let mut in_xfs = false;
143    walk(xml, |ev| match ev {
144        Ev::Start(e) => match e.local_name().as_ref() {
145            b"numFmt" if custom.len() < MAX_XFS => {
146                if let Some(id) = attr(e, b"numFmtId").and_then(|v| v.trim().parse().ok()) {
147                    custom.insert(id, custom_date(&attr(e, b"formatCode").unwrap_or_default()));
148                }
149            }
150            b"cellXfs" => in_xfs = true,
151            b"xf" if in_xfs && xfs.len() < MAX_XFS => xfs.push(attr(e, b"numFmtId").and_then(|v| v.trim().parse().ok()).unwrap_or(0)),
152            _ => {}
153        },
154        Ev::End(n) if n.as_slice() == b"cellXfs" => in_xfs = false,
155        _ => {}
156    });
157    xfs.into_iter().map(|id| custom.get(&id).copied().unwrap_or_else(|| builtin_date(id))).collect()
158}

160/// A cell reference's 0-based column (`C5` → 2); `None` when it isn't one.
161fn ref_col(r: &str) -> Option<usize> {
162    let mut col = 0usize;
163    let mut letters = 0;
164    for c in r.chars() {
165        if c.is_ascii_alphabetic() {
166            letters += 1;
167            if letters > 3 {
168                return None;
169            }
170            col = col * 26 + (c.to_ascii_uppercase() as usize - 'A' as usize + 1);
171        } else {
172            break;
173        }
174    }
175    col.checked_sub(1)
176}

190pub(super) fn xlsx_rows(pkg: &mut Package<'_>, path: &str) -> Result<Vec<Vec<String>>, String> {
191    let wb = workbook(pkg)?;
192    let shared = match wb.shared.as_deref().or(pkg.has("xl/sharedStrings.xml").then_some("xl/sharedStrings.xml")).map(str::to_string) {
193        Some(p) => shared_strings(&pkg.read(&p)?),
194        None => Vec::new(),
195    };
196    let formats = match wb.styles.as_deref().or(pkg.has("xl/styles.xml").then_some("xl/styles.xml")).map(str::to_string) {
197        Some(p) => cell_formats(&pkg.read(&p)?),
198        None => Vec::new(),
199    };
200    let xml = pkg.read(path)?;
201    // Rows by number, so out-of-order rows still come out in order.
202    let mut rows: BTreeMap<u64, Vec<String>> = BTreeMap::new();
203    let mut cells = 0usize;
204    let mut budget = TextBudget::default();
205    let (mut in_data, mut row_no, mut next_col) = (false, 0u64, 0usize);
206    let mut cell: Option<XCell> = None;
207    let (mut in_v, mut in_is_t, mut phonetic) = (false, false, 0usize);
208    walk(&xml, |ev| match ev {
209        Ev::Start(e) => match e.local_name().as_ref() {
210            b"sheetData" => in_data = true,
211            b"row" if in_data => {
212                row_no = attr(e, b"r").and_then(|v| v.trim().parse().ok()).unwrap_or(row_no.saturating_add(1));
213                next_col = 0;
214            }
215            b"c" if in_data => {
216                // A reference that isn't one (or is past the last column) drops the cell.
217                let col = match attr(e, b"r") {
218                    Some(r) => ref_col(r.trim()).unwrap_or(usize::MAX),
219                    None => next_col,
220                };
221                next_col = col.saturating_add(1);
222                let style = attr(e, b"s").and_then(|v| v.trim().parse().ok()).unwrap_or(0);
223                cell = Some(XCell { col, kind: attr(e, b"t").unwrap_or_default(), style, ..Default::default() });
224            }
225            b"v" => in_v = cell.is_some(),
226            b"t" => in_is_t = cell.is_some(),
227            b"rPh" => phonetic += 1,
228            _ => {}
229        },
230        Ev::Text(t) => {
231            if let Some(c) = cell.as_mut() {
232                if in_v {
233                    push_capped(&mut c.value, &mut c.value_chars, &t);
234                } else if in_is_t && phonetic == 0 {
235                    push_capped(&mut c.inline, &mut c.inline_chars, &t);
236                }
237            }
238        }
239        Ev::End(n) => match n.as_slice() {
240            b"sheetData" => in_data = false,
241            b"v" => in_v = false,
242            b"t" => in_is_t = false,
243            b"rPh" => phonetic = phonetic.saturating_sub(1),
244            b"c" => {
245                let Some(c) = cell.take() else { return };
246                let text = xlsx_value(&c, &shared, &formats, wb.date1904);
247                if text.is_empty() || c.col >= MAX_COLS {
248                    return;
249                }
250                if !rows.contains_key(&row_no) && rows.len() >= MAX_ROWS {
251                    return;
252                }
253                if !budget.take(text.len()) {
254                    return;
255                }
256                let row = rows.entry(row_no).or_default();
257                if row.len() <= c.col {
258                    let grow = c.col + 1 - row.len();
259                    if cells.saturating_add(grow) > MAX_CELLS {
260                        return;
261                    }
262                    cells += grow;
263                    row.resize(c.col + 1, String::new());
264                }
265                if let Some(slot) = row.get_mut(c.col) {
266                    *slot = text;
267                }
268            }
269            _ => {}
270        },
271    });
272    let mut grid = Grid::default();
273    for (_, r) in rows {
274        grid.push(r, 1);
275    }
276    Ok(grid.rows)
277}

279fn xlsx_value(c: &XCell, shared: &[String], formats: &[Option<DateKind>], date1904: bool) -> String {
280    let v = c.value.trim();
281    let kind = c.kind.trim();
282    if kind.eq_ignore_ascii_case("s") {
283        v.parse::<usize>().ok().and_then(|i| shared.get(i)).cloned().unwrap_or_default()
284    } else if kind.eq_ignore_ascii_case("inlineStr") {
285        if c.inline.is_empty() { c.value.clone() } else { c.inline.clone() }
286    } else if kind.eq_ignore_ascii_case("str") || kind.eq_ignore_ascii_case("e") {
287        c.value.clone()
288    } else if kind.eq_ignore_ascii_case("b") {
289        if v == "1" || v.eq_ignore_ascii_case("true") { "TRUE" } else { "FALSE" }.into()
290    } else if kind.eq_ignore_ascii_case("d") {
291        iso_datetime(v).unwrap_or_else(|| v.to_string())
292    } else if v.is_empty() {
293        c.inline.clone()
294    } else {
295        let Ok(x) = v.parse::<f64>() else { return v.to_string() };
296        match formats.get(c.style).copied().flatten() {
297            Some(kind) => serial_date(x, date1904, kind).unwrap_or_else(|| number(x)),
298            None => number(x),
299        }
300    }
301}

303/// A number as a spreadsheet shows it in General format: up to 15 significant digits, no
304/// trailing zeros.
305pub(super) fn number(x: f64) -> String {
306    if !x.is_finite() {
307        return x.to_string();
308    }
309    if x.fract() == 0.0 && x.abs() < 1e15 {
310        // `+ 0.0` turns negative zero into positive zero, which would print as "-0".
311        return format!("{:.0}", x + 0.0);
312    }
313    let rounded: f64 = format!("{x:.14e}").parse().unwrap_or(x);
314    if rounded.abs() >= 1e15 || rounded.abs() < 1e-9 { format!("{rounded:E}") } else { rounded.to_string() }
