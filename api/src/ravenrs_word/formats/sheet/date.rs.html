1/// Whether a number format shows a date, a time or both.
2#[derive(Clone, Copy, Debug, PartialEq, Eq)]
3pub(super) enum DateKind {
4    Date,
5    Time,
6    DateTime,
7}

9/// The built-in number formats (§18.8.30) that show dates or times.
10pub(super) fn builtin_date(id: u32) -> Option<DateKind> {
11    match id {
12        14..=17 | 27..=36 | 50..=58 => Some(DateKind::Date),
13        18..=21 | 45..=47 => Some(DateKind::Time),
14        22 => Some(DateKind::DateTime),
15        _ => None,
16    }
17}

19/// Whether a custom format code (§18.8.31) shows a date or time: its first section, without
20/// quoted text, escaped characters, `[…]` (colours, locales, conditions) and `_x` / `*x`
21/// padding, has `d` or `y` (date), `h` or `s` (time), or `m` alone (a month).
22pub(super) fn custom_date(code: &str) -> Option<DateKind> {
23    let mut plain = String::new();
24    let mut chars = code.chars();
25    while let Some(c) = chars.next() {
26        match c {
27            ';' => break,
28            '"' => {
29                for q in chars.by_ref() {
30                    if q == '"' {
31                        break;
32                    }
33                }
34            }
35            '[' => {
36                for q in chars.by_ref() {
37                    if q == ']' {
38                        break;
39                    }
40                }
41            }
42            '\\' | '_' | '*' => {
43                chars.next();
44            }
45            c => plain.extend(c.to_lowercase()),
46        }
47    }
48    let has = |c: char| plain.contains(c);
49    let date = has('d') || has('y');
50    let time = has('h') || has('s');
51    match (date, time) {
52        (true, true) => Some(DateKind::DateTime),
53        (true, false) => Some(DateKind::Date),
54        (false, true) => Some(DateKind::Time),
55        (false, false) if has('m') && !has('0') && !has('#') => Some(DateKind::Date),
56        _ => None,
57    }
58}

60/// A serial date-time (days since the workbook's epoch) in ISO 8601 form.
61pub(super) fn serial_date(serial: f64, date1904: bool, kind: DateKind) -> Option<String> {
62    // Day 2958465 is 9999-12-31.
63    if !serial.is_finite() || !(0.0..2_958_466.0).contains(&serial) {
64        return None;
65    }
66    let mut days = serial.floor() as i64;
67    let mut secs = ((serial - serial.floor()) * 86_400.0).round() as i64;
68    if secs >= 86_400 {
69        days += 1;
70        secs -= 86_400;
71    }
72    // Days from 1970-01-01. The 1900 system counts a 29 February 1900 that never was, so days
73    // before it start a day later.
74    let epoch = if date1904 {
75        -24_107
76    } else if days < 60 {
77        -25_568
78    } else {
79        -25_569
80    };
81    let (y, m, d) = civil(days + epoch);
82    let (h, mi, s) = (secs / 3600, secs / 60 % 60, secs % 60);
83    let time = if s == 0 { format!("{h:02}:{mi:02}") } else { format!("{h:02}:{mi:02}:{s:02}") };
84    Some(match kind {
85        DateKind::Date => format!("{y:04}-{m:02}-{d:02}"),
86        DateKind::Time => time,
87        DateKind::DateTime => format!("{y:04}-{m:02}-{d:02} {time}"),
88    })
89}

91/// Days since 1970-01-01 to a proleptic Gregorian (year, month, day).
92///
93/// The caller's serial guard keeps `z` inside chrono's range, but the Gregorian calendar repeats
94/// every 400 years (146_097 days), so this stays exact for any `z` without hand-rolled calendar
95/// math: chrono converts the day within its cycle and whole cycles go back into the year (an
96/// i128 so the extremes can't overflow). 1970-01-01 is `num_days_from_ce` 719_163.
97pub(crate) fn civil(z: i64) -> (i64, i64, i64) {
98    use chrono::Datelike;
99    let (cycles, rest) = (z.div_euclid(146_097), z.rem_euclid(146_097));
100    match i32::try_from(rest + 719_163).ok().and_then(chrono::NaiveDate::from_num_days_from_ce_opt) {
101        Some(d) => {
102            let y = i128::from(d.year()) + i128::from(cycles) * 400;
103            (i64::try_from(y).unwrap_or(if y < 0 { i64::MIN } else { i64::MAX }), i64::from(d.month()), i64::from(d.day()))
104        }
105        // Unreachable: `rest + 719_163` is always a valid CE day.
106        None => (1970, 1, 1),
107    }
108}

110/// An ISO 8601 date or date-time (`2024-03-05`, `2024-03-05T13:45:00.000`) as `yyyy-mm-dd`,
111/// with ` hh:mm[:ss]` when the time isn't midnight.
112pub(super) fn iso_datetime(v: &str) -> Option<String> {
113    let (date, time) = v.split_once('T').unwrap_or((v, ""));
114    let mut it = date.splitn(3, '-');
115    let y: i64 = it.next()?.parse().ok()?;
116    let m: u32 = it.next()?.parse().ok()?;
117    let d_part = it.next()?;
118    let d_digits: String = d_part.chars().take_while(char::is_ascii_digit).collect();
119    let d: u32 = d_digits.parse().ok()?;
120    if !(1..=12).contains(&m) || !(1..=31).contains(&d) {
121        return None;
122    }
123    let mut out = format!("{y:04}-{m:02}-{d:02}");
124    let time_clean = time.split(['Z', '+', '-', '.']).next().unwrap_or("");
125    let hms: Vec<u32> = time_clean.split(':').filter_map(|p| p.parse().ok()).collect();
126    if let [h, mi, rest @ ..] = hms.as_slice() {
127        let s = rest.first().copied().unwrap_or(0);
128        if (*h, *mi, s) != (0, 0, 0) {
129            out.push_str(&if s == 0 { format!(" {h:02}:{mi:02}") } else { format!(" {h:02}:{mi:02}:{s:02}") });
130        }
131    }
132    Some(out)
133}

135/// An ISO 8601 duration time (`PT13H45M00S`) or clock time (`13:45[:00]`) as `hh:mm[:ss]`.
136pub(super) fn iso_time(v: &str) -> Option<String> {
137    let t = v.trim();
138    if let Some(t) = t.strip_prefix("PT") {
139        let (mut h, mut m, mut s) = (0u64, 0u64, 0f64);
140        let mut num = String::new();
141        for c in t.chars() {
142            match c {
143                'H' => h = std::mem::take(&mut num).parse().ok()?,
144                'M' => m = std::mem::take(&mut num).parse().ok()?,
145                'S' => s = std::mem::take(&mut num).parse().ok()?,
146                c if num.len() < 20 => num.push(c),
147                _ => return None,
148            }
149        }
150        let mut s = s.round() as u64;
151        m = m.saturating_add(s / 60);
152        s %= 60;
153        h = h.saturating_add(m / 60);
154        m %= 60;
155        return Some(if s == 0 { format!("{h:02}:{m:02}") } else { format!("{h:02}:{m:02}:{s:02}") });
156    }
157    // xs:time: hh:mm[:ss[.sss]][Z|+...]
158    let hms: Vec<u32> =
159        t.split(['Z', '+', '-']).next().unwrap_or("").split(':').filter_map(|p| p.split('.').next().and_then(|x| x.parse().ok())).collect();
160    if let [h, mi, rest @ ..] = hms.as_slice() {
161        let s = rest.first().copied().unwrap_or(0);
162        return Some(if s == 0 { format!("{h:02}:{mi:02}") } else { format!("{h:02}:{mi:02}:{s:02}") });
163    }
164    None
165}
