7/// Plain text of tokens (arguments that are names, URLs, lengths).
8pub(super) fn raw_text(toks: &[Tok]) -> String {
9    let mut s = String::new();
10    for t in toks {
11        match t {
12            Tok::Text(x) => s.push_str(x),
13            Tok::Tie => s.push('~'),
14            Tok::Cmd(c) => match text_symbol(c) {
15                Some(sym) => s.push_str(sym),
16                None if c == "\\" => s.push(' '),
17                None => {
18                    s.push('\\');
19                    s.push_str(c);
20                }
21            },
22            Tok::Verb(v) | Tok::Math(v, _) => s.push_str(v),
23            Tok::Amp => s.push('&'),
24            _ => {}
25        }
26    }
27    s.trim().to_string()
28}

30/// Tokens back as LaTeX source (comments and layout whitespace normalised).
31pub(super) fn source_text(toks: &[Tok]) -> String {
32    let mut s = String::new();
33    for t in toks {
34        match t {
35            Tok::Text(x) => s.push_str(x),
36            Tok::Open => s.push('{'),
37            Tok::Close => s.push('}'),
38            Tok::Tie => s.push('~'),
39            Tok::Amp => s.push('&'),
40            Tok::Par => s.push(' '),
41            Tok::Cmd(c) => {
42                s.push('\\');
43                s.push_str(c);
44                if c.chars().all(|c| c.is_ascii_alphabetic()) {
45                    s.push(' ');
46                }
47            }
48            Tok::Math(m, _) => {
49                s.push('$');
50                s.push_str(m);
51                s.push('$');
52            }
53            Tok::Verb(v) | Tok::Verbatim(v) => s.push_str(v),
54            Tok::Begin(n) => s.push_str(&format!("\\begin{{{n}}}")),
55            Tok::End(n) => s.push_str(&format!("\\end{{{n}}}")),
56        }
57    }
58    s.trim().to_string()
59}

61/// A `tabular` body as a table.
62pub(super) fn table(toks: &[Tok], spec: &str, depth: usize) -> FTable {
63    let (spec_cols, widths) = spec_widths(spec);
64    let mut has_rules = spec.contains('|');
65    // Split rows at `\\` and cells at `&`, outside braces and nested environments.
66    let mut rows: Vec<Vec<&[Tok]>> = Vec::new();
67    let mut row: Vec<&[Tok]> = Vec::new();
68    let mut start = 0usize;
69    let mut brace = 0usize;
70    let mut env = 0usize;
71    for (i, t) in toks.iter().enumerate() {
72        match t {
73            Tok::Open => brace += 1,
74            Tok::Close => brace = brace.saturating_sub(1),
75            Tok::Begin(_) => env += 1,
76            Tok::End(_) => env = env.saturating_sub(1),
77            Tok::Cmd(c) if brace == 0 && env == 0 && matches!(c.as_str(), "hline" | "cline" | "toprule" | "midrule" | "bottomrule") => {
78                has_rules = true;
79            }
80            Tok::Amp if brace == 0 && env == 0 => {
81                row.push(toks.get(start..i).unwrap_or_default());
82                start = i + 1;
83            }
84            Tok::Cmd(c) if brace == 0 && env == 0 && (c == "\\" || c == "tabularnewline") => {
85                row.push(toks.get(start..i).unwrap_or_default());
86                rows.push(std::mem::take(&mut row));
87                start = i + 1;
88            }
89            _ => {}
90        }
91        if rows.len() >= crate::doc::table::MAX_ROWS {
92            break;
93        }
94    }
95    if start < toks.len() {
96        row.push(toks.get(start..).unwrap_or_default());
97    }
98    if !row.is_empty() {
99        rows.push(row);
100    }
101    let mut out = FTable { rows: Vec::new(), widths: Vec::new(), borderless: !has_rules, extra: None, kept: None, wrap: None };
102    // Rows still covered by a `\multirow` above, per grid column.
103    let mut covering: Vec<u32> = Vec::new();
104    for r in rows {
105        let mut cells = Vec::new();
106        let mut g = 0usize;
107        // A document row keeps MAX_COLS cells; the rest would only widen `covering`, which is
108        // walked once per row.
109        for cell_toks in r.into_iter().take(crate::doc::table::MAX_COLS) {
110            let mut cell = Cell::default();
111            let mut content = trim_cell(cell_toks);
112            // `\multicolumn{n}{spec}{content}`, `\multirow{n}{width}{content}`. Each wrapper
113            // rescans its content, so only MAX_NEST of them are unwrapped (a cell nests one or
114            // two); further ones stay in the content.
115            let mut sub = Parser::new(content, depth + 1);
116            let mut wrappers = 0usize;
117            loop {
118                sub.skip_spaces();
119                match sub.t.get(sub.pos) {
120                    Some(Tok::Cmd(c)) if c == "multicolumn" && wrappers < MAX_NEST => {
121                        wrappers += 1;
122                        sub.pos += 1;
123                        cell.colspan = sub.arg_raw().trim().parse::<u32>().unwrap_or(1).clamp(1, 63);
124                        let _ = sub.arg_raw();
125                        content = sub.arg_tokens();
126                        sub = Parser::new(content, depth + 1);
127                    }
128                    Some(Tok::Cmd(c)) if (c == "multirow" || c == "multirow*") && wrappers < MAX_NEST => {
129                        wrappers += 1;
130                        sub.pos += 1;
131                        let _ = sub.opt_arg();
132                        cell.rowspan = sub.arg_raw().trim().parse::<i64>().map(|n| n.unsigned_abs().clamp(1, 1000) as u32).unwrap_or(1);
133                        let _ = sub.opt_arg();
134                        let _ = sub.arg_raw();
135                        let _ = sub.opt_arg();
136                        content = sub.arg_tokens();
137                        sub = Parser::new(content, depth + 1);
138                    }
139                    Some(Tok::Cmd(c)) if c == "cellcolor" => {
140                        sub.pos += 1;
141                        let model = sub.opt_arg();
142                        let spec = sub.arg_raw();
143                        cell.shading = parse_color(model.as_deref(), &spec);
144                    }
145                    _ => break,
146                }
147            }
148            if covering.get(g).copied().unwrap_or(0) > 0 {
149                cell.covered = true;
150            }
151            if !cell.covered && depth < MAX_NEST {
152                sub.run(None);
153                let (blocks, _) = sub.finish();
154                cell.blocks = blocks;
155            }
156            let span = cell.colspan as usize;
157            if covering.len() < g + span {
158                covering.resize(g + span, 0);
159            }
160            for k in g..g + span {
161                if let Some(c) = covering.get_mut(k)
162                    && cell.rowspan > 1
163                    && !cell.covered
164                {
165                    *c = cell.rowspan;
166                }
167            }
168            g += span;
169            cells.push(cell);
170        }
171        for c in covering.iter_mut() {
172            *c = c.saturating_sub(1);
173        }
174        // `\hline` alone after the last `\\` is not a row.
175        let empty = cells.iter().all(|c| !c.covered && c.blocks.is_empty());
176        if empty && cells.len() <= 1 {
177            continue;
178        }
179        out.rows.push(cells);
180    }
181    let cols = out.cols();
182    if !widths.is_empty() && widths.len() == cols && spec_cols == cols {
183        out.widths = widths;
184    }
185    out
186}

188/// A cell's tokens without the rules (`\hline`…) and spaces around them.
189fn trim_cell(toks: &[Tok]) -> &[Tok] {
190    let skip = |t: &Tok| match t {
191        Tok::Cmd(c) => {
192            matches!(c.as_str(), "hline" | "toprule" | "midrule" | "bottomrule" | "endhead" | "endfirsthead" | "endfoot" | "endlastfoot" | "noalign")
193        }
194        Tok::Text(s) => s.trim().is_empty(),
195        _ => false,
196    };
197    let mut a = 0;
198    while let Some(t) = toks.get(a) {
199        if skip(t) {
200            a += 1;
201        } else if matches!(t, Tok::Cmd(c) if c == "cline" || c == "cmidrule" || c == "addlinespace") {
202            // `\cline{2-3}`, `\cmidrule(lr){1-2}`: the command and its argument.
203            a += 1;
204            while let Some(Tok::Text(s)) = toks.get(a) {
205                if s.trim().is_empty() || s.trim_start().starts_with('(') || s.trim_start().starts_with('[') {
206                    a += 1;
207                } else {
208                    break;
209                }
210            }
211            if toks.get(a) == Some(&Tok::Open) {
212                while let Some(t) = toks.get(a) {
213                    a += 1;
214                    if *t == Tok::Close {
215                        break;
216                    }
217                }
218            }
219        } else {
220            break;
221        }
222    }
223    let mut b = toks.len();
224    while b > a && toks.get(b - 1).is_some_and(skip) {
225        b -= 1;
226    }
227    toks.get(a..b).unwrap_or_default()
228}
