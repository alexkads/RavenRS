188/// May `\\name` from an equation's text be written to the exported file as a command? Only the
189/// names known here. An equation's text comes from the document (a file someone else may have
190/// made), and `\\input`, `\\write` or `\\def` in it would act on the machine of whoever compiles
191/// the export.
192fn known_math_command(name: &str) -> bool {
193    math_symbol(name).is_some() || MATH_FUNCTIONS.contains(&name) || MATH_COMMANDS.contains(&name)
194}
195
196fn math_symbol(name: &str) -> Option<char> {
197    MATH_SYMBOLS.iter().find(|(_, n)| *n == name).map(|(c, _)| *c).or_else(|| MATH_ALIASES.iter().find(|(n, _)| *n == name).map(|(_, c)| *c))
198}

200/// For every `l` in `chars`, the index of the `r` that closes it, found in one stack pass.
201/// Looking groups up here keeps scanning linear: rescanning to the end of the input for every
202/// unmatched `(` made long equations quadratic.
203pub(super) fn closers(chars: &[char], l: char, r: char) -> Vec<Option<usize>> {
204    let mut out = vec![None; chars.len()];
205    let mut open = Vec::new();
206    for (i, c) in chars.iter().enumerate() {
207        if *c == l {
208            open.push(i);
209        } else if *c == r
210            && let Some(o) = open.pop()
211            && let Some(slot) = out.get_mut(o)
212        {
213            *slot = Some(i);
214        }
215    }
216    out
217}

219/// The balanced group opening at `open` (see [`closers`]): its content and the index after it.
220pub(super) fn balanced(chars: &[char], closers: &[Option<usize>], open: usize) -> Option<(String, usize)> {
221    let close = (*closers.get(open)?)?;
222    Some((chars.get(open + 1..close).unwrap_or_default().iter().collect(), close + 1))
223}

230fn linear_to_latex_at(linear: &str, depth: usize) -> String {
231    let chars: Vec<char> = linear.chars().collect();
232    let parens = closers(&chars, '(', ')');
233    let mut o = String::new();
234    let mut i = 0;
235    let sub = |s: &str| if depth < MAX_NEST { linear_to_latex_at(s, depth + 1) } else { escape_math_text(s) };
236    while let Some(&c) = chars.get(i) {
237        // (a)/(b) → \frac{a}{b}
238        // (The `/` is looked for before the group is copied out: copying every group of a deeply
239        // nested equation only to find no `/` after it was quadratic.)
240        if c == '('
241            && let Some(Some(close)) = parens.get(i)
242            && chars.get(close + 1) == Some(&'/')
243            && let Some((num, after)) = balanced(&chars, &parens, i)
244            && let Some((den, end)) = balanced(&chars, &parens, after + 1)
245        {
246            o.push_str(&format!("\\frac{{{}}}{{{}}}", sub(&num), sub(&den)));
247            i = end;
248            continue;
249        }
250        match c {
251            '^' | '_' => {
252                // Never `^^`: TeX reads `^^5c` as a backslash, which would start a command.
253                if c == '^' && o.ends_with('^') {
254                    o.push_str("{}");
255                }
256                o.push(c);
257                match balanced(&chars, &parens, i + 1) {
258                    Some((inner, end)) => {
259                        o.push_str(&format!("{{{}}}", sub(&inner)));
260                        i = end;
261                    }
262                    None => i += 1,
263                }
264                continue;
265            }
266            '√' => {
267                match balanced(&chars, &parens, i + 1) {
268                    Some((inner, end)) => {
269                        match inner.split_once('&') {
270                            Some((n, x)) => o.push_str(&format!("\\sqrt[{}]{{{}}}", sub(n), sub(x))),
271                            None => o.push_str(&format!("\\sqrt{{{}}}", sub(&inner))),
272                        }
273                        i = end;
274                    }
275                    None => {
276                        o.push_str("\\surd ");
277                        i += 1;
278                    }
279                }
280                continue;
281            }
282            '\\' => {
283                // Linear format control words (`\alpha`) are LaTeX's too.
284                let name: String = chars.iter().skip(i + 1).take_while(|c| c.is_ascii_alphabetic()).collect();
285                if name.is_empty() {
286                    o.push_str("\\backslash ");
287                    i += 1;
288                } else if !known_math_command(&name) {
289                    // Not a command we know: written as the text it is.
290                    o.push_str("\\backslash ");
291                    o.push_str(&name);
292                    i += 1 + name.len();
293                } else {
294                    o.push('\\');
295                    o.push_str(&name);
296                    i += 1 + name.len();
297                    if chars.get(i).is_some_and(char::is_ascii_alphabetic) {
298                        o.push(' ');
299                    }
300                }
301                continue;
302            }
303            '#'..='&' => {
304                o.push('\\');
305                o.push(c);
306            }
307            '{' => o.push_str("\\{"),
308            '}' => o.push_str("\\}"),
309            '~' => o.push_str("\\sim "),
310            c => match MATH_SYMBOLS.iter().find(|(s, _)| *s == c) {
311                Some((_, name)) => {
312                    o.push('\\');
313                    o.push_str(name);
314                    // A control word needs a space only before a letter.
315                    if chars.get(i + 1).is_some_and(char::is_ascii_alphabetic) {
316                        o.push(' ');
317                    }
318                }
319                None if c.is_control() => {}
320                None => o.push(c),
321            },
322        }
323        i += 1;
324    }
325    o.trim().to_string()
326}

328fn escape_math_text(s: &str) -> String {
329    let mut o = String::new();
330    for c in s.chars().filter(|c| !matches!(c, '{' | '}' | '\\' | '$' | '%' | '#' | '&') && !c.is_control()) {
331        // Never `^^` (see `linear_to_latex_at`).
332        if c == '^' && o.ends_with('^') {
333            continue;
334        }
335        o.push(c);
336    }
337    o
338}

340/// LaTeX math as a linear-format equation.
341pub fn latex_to_linear(src: &str) -> String {
342    let toks = math_tokens(src);
343    let mut pos = 0;
344    let s = math_seq(&toks, &mut pos, 0, false);
345    s.split_whitespace().collect::<Vec<_>>().join(" ")
346}

348#[derive(Clone, Debug, PartialEq)]
349enum MTok {
350    Cmd(String),
351    Open,
352    Close,
353    OptOpen,
354    OptClose,
355    Char(char),
356}
357
358fn math_tokens(src: &str) -> Vec<MTok> {
359    let chars: Vec<char> = src.chars().collect();
360    let mut out = Vec::new();
361    let mut i = 0;
362    while let Some(&c) = chars.get(i) {
363        match c {
364            '\\' => {
365                let name: String = chars.iter().skip(i + 1).take_while(|c| c.is_ascii_alphabetic()).collect();
366                if name.is_empty() {
367                    if let Some(&n) = chars.get(i + 1) {
368                        out.push(MTok::Cmd(n.to_string()));
369                    }
370                    i += 2;
371                } else {
372                    i += 1 + name.len();
373                    out.push(MTok::Cmd(name));
374                }
375            }
376            '{' => {
377                out.push(MTok::Open);
378                i += 1;
379            }
380            '}' => {
381                out.push(MTok::Close);
382                i += 1;
383            }
384            '[' => {
385                out.push(MTok::OptOpen);
386                i += 1;
387            }
388            ']' => {
389                out.push(MTok::OptClose);
390                i += 1;
391            }
392            '%' => {
393                // A comment, to the end of the line.
394                while chars.get(i).is_some_and(|c| *c != '\n') {
395                    i += 1;
396                }
397            }
398            c => {
399                out.push(MTok::Char(if c.is_whitespace() { ' ' } else { c }));
400                i += 1;
401            }
402        }
403    }
404    out
405}

407/// One argument: a `{…}` group or a single token.
408fn math_arg(t: &[MTok], pos: &mut usize, depth: usize) -> String {
409    while t.get(*pos) == Some(&MTok::Char(' ')) {
410        *pos += 1;
411    }
412    match t.get(*pos) {
413        Some(MTok::Open) => {
414            *pos += 1;
415            let s = math_seq(t, pos, depth + 1, false);
416            if t.get(*pos) == Some(&MTok::Close) {
417                *pos += 1;
418            }
419            s
420        }
421        Some(MTok::Close) | None => String::new(),
422        Some(_) => {
423            let end = *pos + 1;
424            let mut p = *pos;
425            let s = math_seq(t.get(..end).unwrap_or_default(), &mut p, depth + 1, false);
426            *pos = end;
427            s
428        }
429    }
430}

432/// An optional `[…]` argument.
433fn math_opt(t: &[MTok], pos: &mut usize, depth: usize) -> Option<String> {
434    if t.get(*pos) != Some(&MTok::OptOpen) {
435        return None;
436    }
437    *pos += 1;
438    let s = math_seq(t, pos, depth + 1, true);
439    if t.get(*pos) == Some(&MTok::OptClose) {
440        *pos += 1;
441    }
442    Some(s)
443}

445/// Parenthesise a fraction part or script unless it is a single symbol.
446fn wrap_part(s: &str) -> String {
447    let s = s.trim();
448    if s.chars().count() == 1 || (!s.is_empty() && s.chars().all(|c| c.is_ascii_digit())) { s.to_string() } else { format!("({s})") }
449}
450
451fn math_seq(t: &[MTok], pos: &mut usize, depth: usize, in_opt: bool) -> String {
452    let mut o = String::new();
453    while let Some(tok) = t.get(*pos) {
454        match tok {
455            MTok::Close => break,
456            MTok::OptClose if in_opt => break,
457            MTok::Open => {
458                *pos += 1;
459                if depth < MAX_NEST {
460                    o.push_str(&math_seq(t, pos, depth + 1, false));
461                }
462                if t.get(*pos) == Some(&MTok::Close) {
463                    *pos += 1;
464                }
465                continue;
466            }
467            MTok::OptOpen => o.push('['),
468            MTok::OptClose => o.push(']'),
469            MTok::Char('^') | MTok::Char('_') => {
470                let c = if *tok == MTok::Char('^') { '^' } else { '_' };
471                *pos += 1;
472                let a = if depth < MAX_NEST { math_arg(t, pos, depth) } else { String::new() };
473                o.push(c);
474                o.push_str(&wrap_part(&a));
475                continue;
476            }
477            MTok::Char('&') => o.push(' '),
478            MTok::Char('~') => o.push(' '),
479            MTok::Char(c) => o.push(*c),
480            MTok::Cmd(name) => {
481                *pos += 1;
482                let name = name.as_str();
483                if depth >= MAX_NEST {
484                    continue;
485                }
486                match name {
487                    "frac" | "dfrac" | "tfrac" | "cfrac" => {
488                        let a = math_arg(t, pos, depth);
489                        let b = math_arg(t, pos, depth);
490                        o.push_str(&format!("{}/{}", wrap_part(&a), wrap_part(&b)));
491                    }
492                    "binom" | "dbinom" | "tbinom" => {
493                        let a = math_arg(t, pos, depth);
494                        let b = math_arg(t, pos, depth);
495                        o.push_str(&format!("({a}¦{b})"));
496                    }
497                    "sqrt" => {
498                        let n = math_opt(t, pos, depth);
499                        let a = math_arg(t, pos, depth);
500                        match n {
501                            Some(n) => o.push_str(&format!("√({n}&{a})")),
502                            None => o.push_str(&format!("√({a})")),
503                        }
504                    }
505                    "left" | "right" | "big" | "Big" | "bigg" | "Bigg" | "bigl" | "bigr" | "Bigl" | "Bigr" | "middle" => {
506                        // Keep the delimiter, drop the sizing; `\left.` is no delimiter.
507                        if t.get(*pos) == Some(&MTok::Char('.')) {
508                            *pos += 1;
509                        }
510                    }
511                    "text" | "textrm" | "textit" | "textbf" | "mathrm" | "mathit" | "mathbf" | "mathsf" | "mathtt" | "mathcal" | "mathbb"
512                    | "mathfrak" | "boldsymbol" | "bm" | "pmb" | "operatorname" | "mbox" | "hbox" | "displaystyle" | "textstyle" | "scriptstyle" => {
513                        if t.get(*pos) == Some(&MTok::Open) || !matches!(name, "displaystyle" | "textstyle" | "scriptstyle") {
514                            let a = math_arg(t, pos, depth);
515                            o.push_str(&a);
516                        }
517                    }
518                    "overline" | "bar" => o.push_str(&format!("{}¯", wrap_part(&math_arg(t, pos, depth)))),
519                    "hat" | "widehat" => o.push_str(&format!("{}̂", math_arg(t, pos, depth))),
520                    "tilde" | "widetilde" => o.push_str(&format!("{}̃", math_arg(t, pos, depth))),
521                    "vec" => o.push_str(&format!("{}⃗", math_arg(t, pos, depth))),
522                    "dot" => o.push_str(&format!("{}̇", math_arg(t, pos, depth))),
523                    "ddot" => o.push_str(&format!("{}̈", math_arg(t, pos, depth))),
524                    "acute" => o.push_str(&format!("{}́", math_arg(t, pos, depth))),
525                    "grave" => o.push_str(&format!("{}̀", math_arg(t, pos, depth))),
526                    "breve" => o.push_str(&format!("{}̆", math_arg(t, pos, depth))),
527                    "check" => o.push_str(&format!("{}̌", math_arg(t, pos, depth))),
528                    "pmod" => {
529                        let a = math_arg(t, pos, depth);
530                        o.push_str(&format!(" (mod {a})"));
531                    }
532                    "bmod" => o.push_str(" mod "),
533                    "begin" | "end" => {
534                        // Inner environments (`matrix`, `cases`, `aligned`…): keep the content.
535                        let _ = math_arg(t, pos, depth);
536                        if name == "begin" {
537                            let _ = math_opt(t, pos, depth);
538                        }
539                    }
540                    "," | ";" | ":" | "quad" | "qquad" | " " | "enspace" | "thinspace" => o.push(' '),
541                    "!" | "limits" | "nolimits" | "nonumber" | "notag" => {}
542                    "\\" | "cr" => o.push(' '),
543                    "{" | "}" | "%" | "#" | "&" | "$" | "_" => o.push_str(name),
544                    "label" | "tag" => {
545                        let _ = math_arg(t, pos, depth);
546                    }
547                    n if MATH_FUNCTIONS.contains(&n) => {
548                        o.push_str(n);
549                        o.push(' ');
550                    }
551                    n => match math_symbol(n) {
552                        Some(c) => o.push(c),
553                        // Linear format knows LaTeX-style control words too.
554                        None => {
555                            o.push('\\');
556                            o.push_str(n);
557                            o.push(' ');
558                        }
559                    },
560                }
561                continue;
562            }
563        }
564        *pos += 1;
565    }
566    o
567}
