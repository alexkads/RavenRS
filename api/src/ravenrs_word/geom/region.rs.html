1//! The conventions of the user's region that new documents and the measurement fields follow
2//! (#838): the paper a new blank document starts on, the unit lengths are shown in, and the
3//! decimal separator. Word takes them from the system region, not from the interface language
4//! (an English interface in Germany still gets A4 and centimetres).
5//!
6//! The territory lists come from Unicode CLDR's supplemental `measurementData` (Unicode License):
7//! `paperSize` US-Letter and `measurementSystem` US. The decimal separators follow CLDR's number
8//! symbols for the common locales. The front end detects the region (it knows the platform); the
9//! engine only sees this plain value.
10

15/// Territories whose default paper is US Letter (CLDR `paperSize type="US-Letter"`).
16const LETTER: &[&str] = &["BZ", "CA", "CL", "CO", "CR", "GT", "MX", "NI", "PA", "PH", "PR", "SV", "US", "VE"];
17
18/// Territories that measure in inches (CLDR `measurementSystem type="US"`).
19const INCHES: &[&str] = &["LR", "MM", "US"];
20
21/// Languages whose numbers use a decimal comma in most of their territories.
22const COMMA_LANGUAGES: &[&str] = &[
23    "af", "az", "be", "bg", "bs", "ca", "cs", "da", "de", "el", "es", "et", "eu", "fi", "fo", "fr", "gl", "hr", "hu", "hy", "id", "is", "it", "ka",
24    "kk", "ky", "lb", "lt", "lv", "mk", "nb", "nl", "nn", "no", "pl", "pt", "ro", "ru", "sk", "sl", "sq", "sr", "sv", "tr", "uk", "uz", "vi",
25];
26
27/// Territories where those languages use a decimal point after all.
28const POINT_EXCEPTIONS: &[(&str, &[&str])] =
29    &[("de", &["CH", "LI"]), ("it", &["CH"]), ("es", &["CU", "DO", "GT", "HN", "MX", "NI", "PA", "PE", "PR", "SV", "US", "419"])];
30
31/// Territories where English (and other decimal-point languages) is written with a decimal
32/// comma: most of continental Europe, South Africa and much of South America.
33const COMMA_TERRITORIES: &[&str] = &[
34    "AR", "AT", "BE", "BG", "BO", "BR", "CL", "CO", "CZ", "DE", "DK", "EC", "EE", "ES", "FI", "FR", "GR", "HR", "HU", "ID", "IS", "IT", "LT", "LU",
35    "LV", "NL", "NO", "PL", "PT", "PY", "RO", "RS", "RU", "SE", "SI", "SK", "TR", "UA", "UY", "VE", "VN", "ZA",
36];

38/// A region: a language and a territory, read from a locale tag like `pl-PL`, `de_AT.UTF-8` or
39/// `en-US`. The default is `en-US` (Letter, inches, decimal point).
40#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
41pub struct Region {
42    /// Lower-case language subtag (`pl`); empty when the tag has none.
43    pub language: String,
44    /// Upper-case territory (`PL`, or a UN M.49 number like `419`); empty when the tag has none.
45    pub territory: String,
46    /// US Letter or not, when a separate paper locale says so (Unix `LC_PAPER`); otherwise the
47    /// territory decides.
48    #[serde(default, skip_serializing_if = "Option::is_none")]
49    pub letter: Option<bool>,
50    /// Inches or not, when a separate measurement locale says so (Unix `LC_MEASUREMENT`);
51    /// otherwise the territory decides.
52    #[serde(default, skip_serializing_if = "Option::is_none")]
53    pub inches: Option<bool>,
54}
55
56impl Default for Region {
57    fn default() -> Self {
58        Region { language: "en".into(), territory: "US".into(), letter: None, inches: None }
59    }
60}
61
62impl Region {
63    /// The region a BCP 47 or POSIX locale tag names (`pl-PL`, `pl_PL.UTF-8@euro`,
64    /// `zh-Hans-CN`, `en-US-u-ms-metric`). An empty or unreadable tag, `C` and `POSIX` are
65    /// `en-US`; a bare `en` is the United States, any other bare language has no territory.
66    pub fn from_tag(tag: &str) -> Region {
67        // Locale tags are short; anything long is junk and only its start is read.
68        let tag: String = tag.chars().take(64).collect();
69        let tag = tag.split(['.', '@']).next().unwrap_or_default().trim();
70        if tag.is_empty() || tag.eq_ignore_ascii_case("c") || tag.eq_ignore_ascii_case("posix") {
71            return Region::default();
72        }
73        let mut parts = tag.split(['-', '_']);
74        let language = parts.next().unwrap_or_default();
75        if !(2..=3).contains(&language.len()) || !language.chars().all(|c| c.is_ascii_alphabetic()) {
76            return Region::default();
77        }
78        let language = language.to_ascii_lowercase();
79        let mut territory = String::new();
80        for p in parts {
81            // A singleton starts the extensions (`-u-…`, `-x-…`): no territory after it.
82            if p.len() == 1 {
83                break;
84            }
85            let alpha = p.len() == 2 && p.chars().all(|c| c.is_ascii_alphabetic());
86            let digits = p.len() == 3 && p.chars().all(|c| c.is_ascii_digit());
87            if alpha || digits {
88                territory = p.to_ascii_uppercase();
89                break;
90            }
91        }
92        if territory.is_empty() && language == "en" {
93            territory = "US".into();
94        }
95        Region { language, territory, letter: None, inches: None }
96    }
97
98    /// This region with another territory (from a platform setting that names only the
99    /// territory, like Windows' home location). A value that isn't a territory code is ignored.
100    pub fn with_territory(mut self, territory: &str) -> Region {
101        let t = territory.trim();
102        let ok = (t.len() == 2 && t.chars().all(|c| c.is_ascii_alphabetic())) || (t.len() == 3 && t.chars().all(|c| c.is_ascii_digit()));
103        if ok {
104            self.territory = t.to_ascii_uppercase();
105        }
106        self
107    }
108
109    /// This region with the paper and the measurement system taken from their own locale tags,
110    /// when the platform sets them apart (Unix `LC_PAPER`, `LC_MEASUREMENT`): `LANG=pl_PL.UTF-8`
111    /// with `LC_PAPER=en_US.UTF-8` writes `2,54 cm` on Letter.
112    pub fn with_paper_and_measurement(mut self, paper: Option<&str>, measurement: Option<&str>) -> Region {
113        if let Some(t) = paper {
114            self.letter = Some(Region::from_tag(t).uses_letter());
115        }
116        if let Some(t) = measurement {
117            self.inches = Some(Region::from_tag(t).units() == Unit::Inches);
118        }
119        self
120    }
121
122    /// The system region from the locale variables `var` reads (the process environment) and,
123    /// when none is set, the platform's first preferred locale `platform` gives. One tag for
124    /// everything first (`WORDCRAFT_REGION`, then `LC_ALL`); otherwise Unix sets each convention
125    /// on its own: numbers (`LC_NUMERIC`, else `LANG`), paper (`LC_PAPER`) and measurement
126    /// (`LC_MEASUREMENT`). Empty, `C` and `POSIX` values count as unset.
127    pub fn from_locale_vars(var: impl Fn(&str) -> Option<String>, platform: impl FnOnce() -> Option<String>) -> Region {
128        let var = |name: &str| var(name).filter(|v| usable_locale(v));
129        if let Some(tag) = var("WORDCRAFT_REGION").or_else(|| var("LC_ALL")) {
130            return Region::from_tag(&tag);
131        }
132        let base = var("LC_NUMERIC").or_else(|| var("LANG")).unwrap_or_else(|| platform().unwrap_or_default());
133        Region::from_tag(&base).with_paper_and_measurement(var("LC_PAPER").as_deref(), var("LC_MEASUREMENT").as_deref())
134    }
135
136    /// Does a new document start on US Letter here (otherwise A4)?
137    pub fn uses_letter(&self) -> bool {
138        self.letter.unwrap_or_else(|| LETTER.contains(&self.territory.as_str()))
139    }
140
141    /// The default paper's name and portrait size in points: Letter (612 × 792) or A4.
142    pub fn paper(&self) -> (&'static str, f32, f32) {
143        if self.uses_letter() { ("Letter", 612.0, 792.0) } else { ("A4", 595.28, 841.89) }
144    }
145
146    /// The unit lengths are shown in by default: inches in the US, centimetres elsewhere.
147    pub fn units(&self) -> Unit {
148        if self.inches.unwrap_or_else(|| INCHES.contains(&self.territory.as_str())) { Unit::Inches } else { Unit::Centimeters }
149    }
150
151    /// The decimal separator numbers are written with here (`,` or `.`).
152    pub fn decimal(&self) -> char {
153        let lang = self.language.as_str();
154        let t = self.territory.as_str();
155        if COMMA_LANGUAGES.contains(&lang) {
156            let point = POINT_EXCEPTIONS.iter().any(|(l, ts)| *l == lang && ts.contains(&t));
157            return if point { '.' } else { ',' };
158        }
159        if lang == "en" && COMMA_TERRITORIES.contains(&t) { ',' } else { '.' }
160    }
161
162    /// `pt` in `unit` written the way this region writes numbers (`2,54 cm` in Poland).
163    pub fn format(&self, unit: Unit, pt: f32) -> String {
164        unit.format_with(pt, self.decimal())
165    }
166}
167
168/// A locale variable that names a locale (not empty, `C` or `POSIX`).
169fn usable_locale(v: &str) -> bool {
170    let v = v.trim();
171    !v.is_empty() && !v.eq_ignore_ascii_case("c") && !v.eq_ignore_ascii_case("posix") && !v.starts_with("C.")
172}

178    #[test]
179    fn reads_locale_tags() {
180        assert_eq!(Region::from_tag("pl-PL"), Region { language: "pl".into(), territory: "PL".into(), ..Region::default() });
181        assert_eq!(Region::from_tag("de_AT.UTF-8@euro").territory, "AT");
182        assert_eq!(Region::from_tag("zh-Hans-CN").territory, "CN");
183        assert_eq!(Region::from_tag("es-419").territory, "419");
184        assert_eq!(Region::from_tag("en-u-ms-metric").territory, "US", "bare English is the US");
185        assert_eq!(Region::from_tag("fr").territory, "");
186        for junk in ["", "C", "POSIX", "1234", "x", "--", "ünïcode-ÄÄ", &"a".repeat(10_000)] {
187            assert_eq!(Region::from_tag(junk), Region::default(), "{junk:?}");
188        }
189        assert_eq!(Region::from_tag("en-US").with_territory("de").territory, "DE");
190        assert_eq!(Region::from_tag("en-US").with_territory("Germany").territory, "US");
191    }

193    /// Paper and units follow the territory, not the language (#838).
194    #[test]
195    fn paper_and_units_follow_the_territory() {
196        let pl = Region::from_tag("pl-PL");
197        assert_eq!((pl.paper().0, pl.units(), pl.decimal()), ("A4", Unit::Centimeters, ','));
198        let us = Region::from_tag("en-US");
199        assert_eq!((us.paper().0, us.units(), us.decimal()), ("Letter", Unit::Inches, '.'));
200        let ca = Region::from_tag("fr-CA");
201        assert_eq!((ca.paper().0, ca.units(), ca.decimal()), ("Letter", Unit::Centimeters, ','));
202        let en_de = Region::from_tag("en-DE");
203        assert_eq!((en_de.paper().0, en_de.units(), en_de.decimal()), ("A4", Unit::Centimeters, ','));
204        let gb = Region::from_tag("en-GB");
205        assert_eq!((gb.paper().0, gb.units(), gb.decimal()), ("A4", Unit::Centimeters, '.'));
206        assert_eq!(Region::from_tag("de-CH").decimal(), '.');
207        assert_eq!(Region::from_tag("es-MX").paper().0, "Letter");
208        assert_eq!(Region::from_tag("ja-JP").paper().0, "A4");
209        assert_eq!(pl.format(Unit::Centimeters, 72.0), "2,54 cm");
210        // Unix sets paper and measurement apart from the number locale.
211        let mixed = Region::from_tag("pl_PL.UTF-8").with_paper_and_measurement(Some("en_US.UTF-8"), None);
212        assert_eq!((mixed.paper().0, mixed.units(), mixed.decimal()), ("Letter", Unit::Centimeters, ','));
213        let mixed = Region::from_tag("en_US.UTF-8").with_paper_and_measurement(Some("de_DE.UTF-8"), Some("de_DE.UTF-8"));
214        assert_eq!((mixed.paper().0, mixed.units(), mixed.decimal()), ("A4", Unit::Centimeters, '.'));
215        assert_eq!(Region::from_tag("pl-PL").with_paper_and_measurement(Some(""), Some("junk")).paper().0, "Letter", "junk is en-US");
216        assert_eq!(us.format(Unit::Inches, 90.0), "1.25\"");
217    }

219    #[test]
220    fn region_from_locale_variables() {
221        // #1381: the CLI and the app read the system region the same way.
222        let env = |pairs: &'static [(&'static str, &'static str)]| move |n: &str| pairs.iter().find(|(k, _)| *k == n).map(|(_, v)| v.to_string());
223        let none = || None;
224        assert_eq!(Region::from_locale_vars(env(&[("LANG", "it_IT.UTF-8")]), none).paper().0, "A4");
225        assert_eq!(Region::from_locale_vars(env(&[("LANG", "en_US.UTF-8")]), none).paper().0, "Letter");
226        assert_eq!(Region::from_locale_vars(env(&[("LANG", "it_IT.UTF-8"), ("LC_ALL", "en_US")]), none).paper().0, "Letter");
227        assert_eq!(Region::from_locale_vars(env(&[("LANG", "en_US"), ("WORDCRAFT_REGION", "de-DE")]), none).territory, "DE");
228        assert_eq!(Region::from_locale_vars(env(&[("LANG", "en_US"), ("LC_PAPER", "it_IT")]), none).paper().0, "A4");
229        assert_eq!(Region::from_locale_vars(env(&[("LANG", "C.UTF-8")]), || Some("it-IT".into())).territory, "IT");
230        assert_eq!(Region::from_locale_vars(env(&[]), none), Region::default());
231    }
