1//! Pixel-budget arithmetic shared by every crate that decodes or allocates rasters: sizes are
2//! checked before anything is allocated, so a hostile header or request cannot exhaust memory.
3
4/// Bytes a decoded picture may take as RGBA8 (what the `image` crate's default decode budget is).
5pub const MAX_DECODED_BYTES: u64 = 512 * 1024 * 1024;
6
7/// Most pixels one page raster may hold (400 MB of RGBA8); larger requests are scaled down.
8pub const MAX_RASTER_PIXELS: u64 = 100_000_000;
9
10/// Bytes of a `w`×`h` RGBA8 raster, None on overflow.
11pub fn rgba_bytes(w: u32, h: u32) -> Option<u64> {
12    (w as u64).checked_mul(h as u64)?.checked_mul(4)
13}
14
15/// Whether a `w`×`h` RGBA8 raster fits `budget` bytes.
16pub fn rgba_fits(w: u32, h: u32, budget: u64) -> bool {
17    w > 0 && h > 0 && rgba_bytes(w, h).is_some_and(|b| b <= budget)
18}
19
20/// Whether decoding a `w`×`h` picture fits `budget` bytes at its peak: the decoder's buffer and
21/// the RGBA conversion (or its premultiplied copy) are alive together, so two RGBA buffers. For a
22/// decoder whose buffer isn't four bytes a pixel, see [`decode_peak_fits`].
23pub fn decode_fits(w: u32, h: u32, budget: u64) -> bool {
24    decode_peak_fits(w, h, 4, budget)
25}
26
27/// [`decode_fits`] for a decoder that hands back `native` bytes a pixel (8 for 16-bit RGBA, 16 for
28/// float RGBA): its buffer lives alongside the RGBA8 conversion, then the conversion alongside its
29/// premultiplied copy, so the peak is the larger of `native + 4` and `8` bytes a pixel.
30pub fn decode_peak_fits(w: u32, h: u32, native: u64, budget: u64) -> bool {
31    let per_pixel = native.saturating_add(4).max(8);
32    w > 0 && h > 0 && (w as u64).checked_mul(h as u64).and_then(|p| p.checked_mul(per_pixel)).is_some_and(|b| b <= budget)
33}
34
35/// `w`×`h` shrunk (aspect kept, each side at least 1) until it holds at most `max_pixels`.
36pub fn fit_pixels(w: u32, h: u32, max_pixels: u64) -> (u32, u32) {
37    let (w, h) = (w.max(1), h.max(1));
38    let max_pixels = max_pixels.max(1);
39    let area = w as u64 * h as u64;
40    if area <= max_pixels {
41        return (w, h);
42    }
43    let f = (max_pixels as f64 / area as f64).sqrt();
44    let nh = ((h as f64 * f) as u64).clamp(1, h as u64);
45    let nw = ((w as f64 * f) as u64).clamp(1, w as u64).min(max_pixels / nh).max(1);
46    let nh = nh.min(max_pixels / nw).max(1);
47    (nw as u32, nh as u32)
48}

50#[cfg(test)]
51mod tests {
52    use super::*;
53
54    #[test]
55    fn rgba_budget_boundary() {
56        assert_eq!(rgba_bytes(16, 8), Some(512));
57        assert!(rgba_fits(16, 8, 512));
58        assert!(!rgba_fits(16, 8, 511));
59        assert!(!rgba_fits(0, 8, 512), "empty is not a picture");
60        assert!(decode_fits(16, 8, 1024));
61        assert!(!decode_fits(16, 8, 1023), "decode + conversion hold two buffers");
62        assert!(!decode_fits(0, 8, u64::MAX));
63    }
64
65    /// A 16-bit RGBA decode (8 bytes a pixel) lives alongside its 4-byte RGBA8 conversion: 12 bytes
66    /// a pixel at the peak, not the 8 two RGBA8 buffers take.
67    #[test]
68    fn wide_decoder_buffers_are_charged_at_their_size() {
69        assert!(decode_peak_fits(16, 8, 4, 1024), "RGBA8: two RGBA8 buffers");
70        assert!(decode_peak_fits(16, 8, 1, 1024), "grey: the premultiplied copy is the peak");
71        assert!(!decode_peak_fits(16, 8, 8, 1024), "RGBA16 needs 1536 bytes");
72        assert!(decode_peak_fits(16, 8, 8, 1536));
73        assert!(!decode_peak_fits(16, 8, 16, 2559), "RGBA32F: 20 bytes a pixel");
74        assert!(!decode_peak_fits(u32::MAX, u32::MAX, u64::MAX, u64::MAX), "overflow is refused");
75    }
76
77    #[test]
78    fn fit_pixels_keeps_normal_sizes_and_shrinks_huge_ones() {
79        assert_eq!(fit_pixels(612, 792, MAX_RASTER_PIXELS), (612, 792));
80        let (w, h) = fit_pixels(16_000, 16_000, 1_000_000);
81        assert!(w as u64 * h as u64 <= 1_000_000 && w >= 900, "{w}x{h}");
82        let (w, h) = fit_pixels(16_000, 1, 100);
83        assert!(w as u64 * h as u64 <= 100 && h == 1, "{w}x{h}");
84        assert_eq!(fit_pixels(0, 5, 10).0, 1);
85    }
86}
