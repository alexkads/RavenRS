103    pub total_borders: Option<Borders>,
104    /// The odd column bands (`band1Vert` region).
105    pub col_band_fill: Option<Rgb>,
106    pub col_band_chr: CharProps,
107    pub col_band_borders: Option<Borders>,
108    /// Per-part formatting the named fields above don't carry: every field of the parts without
109    /// one (even bands, corner cells), and the cell text alignment of every part. Read and write
110    /// parts through [`TableStyleParts::part`] / [`TableStyleParts::set_part`].
111    #[serde(skip_serializing_if = "BTreeMap::is_empty")]
112    pub regions: BTreeMap<TablePart, PartFormat>,

115/// A conditional formatting part of a table style (`w:tblStylePr/@w:type`, ECMA-376 §17.7.6).
116#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize, Deserialize)]
117#[serde(rename_all = "camelCase")]
118pub enum TablePart {
119    WholeTable,
120    FirstRow,
121    LastRow,
122    FirstCol,
123    LastCol,
124    Band1Vert,
125    Band2Vert,
126    Band1Horz,
127    Band2Horz,
128    NeCell,
129    NwCell,
130    SeCell,
131    SwCell,
132}
133
134impl TablePart {
135    /// Every part, lowest precedence first (ECMA-376 §17.7.6: whole table, bands, first/last
136    /// column, first/last row, corner cells).
137    pub const ALL: [TablePart; 13] = [
138        TablePart::WholeTable,
139        TablePart::Band1Vert,
140        TablePart::Band2Vert,
141        TablePart::Band1Horz,
142        TablePart::Band2Horz,
143        TablePart::FirstCol,
144        TablePart::LastCol,
145        TablePart::FirstRow,
146        TablePart::LastRow,
147        TablePart::NeCell,
148        TablePart::NwCell,
149        TablePart::SeCell,
150        TablePart::SwCell,
151    ];
152    /// The `w:type` value.
153    pub fn ooxml(self) -> &'static str {
154        match self {
155            TablePart::WholeTable => "wholeTable",
156            TablePart::FirstRow => "firstRow",
157            TablePart::LastRow => "lastRow",
158            TablePart::FirstCol => "firstCol",
159            TablePart::LastCol => "lastCol",
160            TablePart::Band1Vert => "band1Vert",
161            TablePart::Band2Vert => "band2Vert",
162            TablePart::Band1Horz => "band1Horz",
163            TablePart::Band2Horz => "band2Horz",
164            TablePart::NeCell => "neCell",
165            TablePart::NwCell => "nwCell",
166            TablePart::SeCell => "seCell",
167            TablePart::SwCell => "swCell",
168        }
169    }
170    pub fn from_ooxml(s: &str) -> Option<TablePart> {
171        TablePart::ALL.into_iter().find(|p| p.ooxml() == s)
172    }
173    /// A part spanning whole rows (its left/right edges are the table's sides, `inside_v` the
174    /// edges between its cells).
175    pub fn is_row(self) -> bool {
176        matches!(self, TablePart::FirstRow | TablePart::LastRow | TablePart::Band1Horz | TablePart::Band2Horz)
177    }
178    /// A part spanning whole columns (its top/bottom edges are the table's, `between` the edges
179    /// between its cells).
180    pub fn is_column(self) -> bool {
181        matches!(self, TablePart::FirstCol | TablePart::LastCol | TablePart::Band1Vert | TablePart::Band2Vert)
182    }
183    fn bit(self) -> u16 {
184        1 << (TablePart::ALL.iter().position(|p| *p == self).unwrap_or(0) as u16)
185    }
186}

188/// The table style parts one cell is in (always the whole table).
189#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
190pub struct PartSet(u16);
191
192impl PartSet {
193    /// The parts of the cell at row `ri` of `nrows`, grid column `g` spanning `span` of `ncols`,
194    /// in a table with options `look`, row bands of `row_size` rows and column bands of
195    /// `col_size` columns (a size of zero turns those bands off, as in Word).
196    pub fn of_cell(
197        look: &TableLook,
198        (row_size, col_size): (u32, u32),
199        (ri, nrows): (usize, usize),
200        (g, span, ncols): (usize, usize, usize),
201    ) -> PartSet {
202        let header = look.header_row && ri == 0;
203        let total = look.total_row && ri + 1 == nrows && nrows > 1;
204        let first_col = look.first_column && g == 0;
205        let last_col = look.last_column && g.saturating_add(span) >= ncols && ncols > 1;
206        let rows = row_size.min(1000) as usize;
207        let cols = col_size.min(1000) as usize;
208        let row_band = look.banded_rows && !header && rows > 0;
209        let odd_row = (ri.saturating_sub(usize::from(look.header_row)) / rows.max(1)).is_multiple_of(2);
210        let col_band = look.banded_columns && !first_col && cols > 0;
211        let odd_col = (g.saturating_sub(usize::from(look.first_column)) / cols.max(1)).is_multiple_of(2);
212        let mut bits = TablePart::WholeTable.bit();
213        for (on, p) in [
214            (col_band && odd_col, TablePart::Band1Vert),
215            (col_band && !odd_col, TablePart::Band2Vert),
216            (row_band && odd_row, TablePart::Band1Horz),
217            (row_band && !odd_row, TablePart::Band2Horz),
218            (first_col, TablePart::FirstCol),
219            (last_col, TablePart::LastCol),
220            (header, TablePart::FirstRow),
221            (total, TablePart::LastRow),
222            (header && last_col, TablePart::NeCell),
223            (header && first_col, TablePart::NwCell),
224            (total && last_col, TablePart::SeCell),
225            (total && first_col, TablePart::SwCell),
226        ] {
227            if on {
228                bits |= p.bit();
229            }
230        }
231        PartSet(bits)
232    }
233    pub fn contains(self, p: TablePart) -> bool {
234        self.0 & p.bit() != 0
235    }
236    /// The parts, lowest precedence first.
237    pub fn iter(self) -> impl Iterator<Item = TablePart> {
238        TablePart::ALL.into_iter().filter(move |p| self.contains(*p))
239    }
240}

242/// A cell's borders from the table style's parts it is in, later parts winning. Row parts give
243/// their left/right edges at the table's sides and `inside_v` between cells; column parts their
244/// top/bottom edges at the table's top and bottom and `between` between rows; corner cells their
245/// four edges.
246pub fn part_borders(parts: &[(TablePart, Borders)], (ri, nrows): (usize, usize), (first_g, last_g): (bool, bool)) -> Borders {
247    let mut out = Borders::default();
248    for (part, b) in parts {
249        let b = *b;
250        let mapped = if part.is_row() {
251            Borders {
252                top: b.top,
253                bottom: b.bottom,
254                left: if first_g { b.left } else { b.inside_v },
255                right: if last_g { b.right } else { b.inside_v },
256                between: None,
257                inside_v: None,
258            }
259        } else if part.is_column() {
260            Borders {
261                top: if ri == 0 { b.top } else { b.between },
262                bottom: if ri + 1 >= nrows { b.bottom } else { b.between },
263                left: b.left,
264                right: b.right,
265                between: None,
266                inside_v: None,
267            }
268        } else {
269            Borders { between: None, inside_v: None, ..b }
270        };
271        out.overlay(&mapped);
272    }
273    out
274}

276/// What one table style part gives its cells.
277#[derive(Clone, Debug, PartialEq, Serialize, Deserialize, Default)]
278#[serde(default, rename_all = "camelCase")]
279pub struct PartFormat {
280    pub chr: CharProps,
281    /// Cell shading.
282    pub fill: Option<Rgb>,
283    /// Cell borders (the whole table's are the table borders).
284    pub borders: Option<Borders>,
285    /// Horizontal alignment of the cells' paragraphs (`w:pPr/w:jc`).
286    pub align: Option<Align>,
287    /// Vertical alignment in the cells (`w:tcPr/w:vAlign`).
288    pub v_align: Option<VAlign>,
289}
290
291impl PartFormat {
292    pub fn is_empty(&self) -> bool {
293        *self == PartFormat::default()
294    }
295    /// What `patch` sets wins; borders merge edge by edge.
296    pub fn overlay(&mut self, patch: &PartFormat) {
297        self.chr.overlay(&patch.chr);
298        self.fill = patch.fill.or(self.fill);
299        self.borders = match (self.borders, patch.borders) {
300            (Some(mut b), Some(p)) => {
301                b.overlay(&p);
302                Some(b)
303            }
304            (b, p) => p.or(b),
305        };
306        self.align = patch.align.or(self.align);
307        self.v_align = patch.v_align.or(self.v_align);
308    }
309}

311impl TableStyleParts {
312    /// Apply `patch` (a style further down a based-on chain) over these parts: what it sets wins,
313    /// the rest is inherited; borders merge edge by edge.
314    pub fn overlay(&mut self, patch: &TableStyleParts) {
315        let merge = |b: Option<Borders>, p: Option<Borders>| match (b, p) {
316            (Some(mut b), Some(p)) => {
317                b.overlay(&p);
318                Some(b)
319            }
320            (b, p) => p.or(b),
321        };
322        self.borders = merge(self.borders, patch.borders);
323        self.header_borders = merge(self.header_borders, patch.header_borders);
324        self.band_borders = merge(self.band_borders, patch.band_borders);
325        self.band_size = patch.band_size.or(self.band_size);
326        self.col_band_size = patch.col_band_size.or(self.col_band_size);
327        self.band_chr.overlay(&patch.band_chr);
328        self.fill = patch.fill.or(self.fill);
329        self.cell_margins = patch.cell_margins.or(self.cell_margins);
330        self.header_fill = patch.header_fill.or(self.header_fill);
331        self.header_chr.overlay(&patch.header_chr);
332        self.band_fill = patch.band_fill.or(self.band_fill);
333        self.first_col_chr.overlay(&patch.first_col_chr);
334        self.total_chr.overlay(&patch.total_chr);
335        self.total_border_top = patch.total_border_top.or(self.total_border_top);
336        self.first_col_fill = patch.first_col_fill.or(self.first_col_fill);
337        self.first_col_borders = merge(self.first_col_borders, patch.first_col_borders);
338        self.last_col_fill = patch.last_col_fill.or(self.last_col_fill);
339        self.last_col_chr.overlay(&patch.last_col_chr);
340        self.last_col_borders = merge(self.last_col_borders, patch.last_col_borders);
341        self.total_fill = patch.total_fill.or(self.total_fill);
342        self.total_borders = merge(self.total_borders, patch.total_borders);
343        self.col_band_fill = patch.col_band_fill.or(self.col_band_fill);
344        self.col_band_chr.overlay(&patch.col_band_chr);
345        self.col_band_borders = merge(self.col_band_borders, patch.col_band_borders);
346        for (k, f) in &patch.regions {
347            self.regions.entry(*k).or_default().overlay(f);
348        }
349    }

351    /// Part `p`'s formatting. The whole table's text formatting is the style's own
352    /// ([`Style::table_part`] adds it); its borders are the table borders.
353    pub fn part(&self, p: TablePart) -> PartFormat {
354        let extra = self.regions.get(&p).cloned().unwrap_or_default();
355        let named = |chr: &CharProps, fill: Option<Rgb>, borders: Option<Borders>| PartFormat {
356            chr: chr.clone(),
357            fill,
358            borders,
359            align: extra.align,
360            v_align: extra.v_align,
361        };
362        match p {
363            TablePart::WholeTable => {
364                PartFormat { chr: CharProps::default(), fill: self.fill, borders: self.borders, align: None, v_align: extra.v_align }
365            }
366            TablePart::FirstRow => named(&self.header_chr, self.header_fill, self.header_borders),
367            TablePart::LastRow => {
368                // The built-in top rule is the region's top edge unless its borders set one.
369                let borders = match (self.total_borders, self.total_border_top) {
370                    (Some(mut b), top) => {
371                        b.top = b.top.or(top);
372                        Some(b)
373                    }
374                    (None, top) => top.map(|t| Borders { top: Some(t), ..Borders::default() }),
375                };
376                named(&self.total_chr, self.total_fill, borders)
377            }
378            TablePart::FirstCol => named(&self.first_col_chr, self.first_col_fill, self.first_col_borders),
379            TablePart::LastCol => named(&self.last_col_chr, self.last_col_fill, self.last_col_borders),
380            TablePart::Band1Vert => named(&self.col_band_chr, self.col_band_fill, self.col_band_borders),
381            TablePart::Band1Horz => named(&self.band_chr, self.band_fill, self.band_borders),
382            _ => extra,
383        }
384    }
385
386    /// Replace part `p`'s formatting (for the whole table only its fill, borders and vertical
387    /// alignment: [`Style::set_table_part`] stores the rest).
388    pub fn set_part(&mut self, p: TablePart, f: PartFormat) {
389        let align_only = PartFormat { align: f.align, v_align: f.v_align, ..PartFormat::default() };
390        let PartFormat { chr, fill, borders, .. } = f.clone();
391        let rest = match p {
392            TablePart::WholeTable => {
393                self.fill = fill;
394                self.borders = borders;
395                PartFormat { v_align: f.v_align, ..PartFormat::default() }
396            }
397            TablePart::FirstRow => {
398                (self.header_chr, self.header_fill, self.header_borders) = (chr, fill, borders);
399                align_only
400            }
401            TablePart::LastRow => {
402                (self.total_chr, self.total_fill) = (chr, fill);
403                self.total_border_top = borders.and_then(|b| b.top);
404                self.total_borders = borders.map(|b| Borders { top: None, ..b }).filter(|b| *b != Borders::default());
405                align_only
406            }
407            TablePart::FirstCol => {
408                (self.first_col_chr, self.first_col_fill, self.first_col_borders) = (chr, fill, borders);
409                align_only
410            }
411            TablePart::LastCol => {
412                (self.last_col_chr, self.last_col_fill, self.last_col_borders) = (chr, fill, borders);
413                align_only
414            }
415            TablePart::Band1Vert => {
416                (self.col_band_chr, self.col_band_fill, self.col_band_borders) = (chr, fill, borders);
417                align_only
418            }
419            TablePart::Band1Horz => {
420                (self.band_chr, self.band_fill, self.band_borders) = (chr, fill, borders);
421                align_only
422            }
423            _ => f,
424        };
425        if rest.is_empty() {
426            self.regions.remove(&p);
427        } else {
428            self.regions.insert(p, rest);
429        }
430    }
431}

433impl Style {
434    /// Table style part `p` as this style alone sets it (the whole table's text formatting and
435    /// paragraph alignment are the style's own).
436    pub fn table_part(&self, p: TablePart) -> PartFormat {
437        let mut f = self.table.as_ref().map(|t| t.part(p)).unwrap_or_default();
438        if p == TablePart::WholeTable {
439            f.chr = self.chr.clone();
440            f.align = self.para.align;
441        }
442        f
443    }
444    /// Replace part `p` of this table style.
445    pub fn set_table_part(&mut self, p: TablePart, f: PartFormat) {
446        if p == TablePart::WholeTable {
447            self.chr = f.chr.clone();
448            self.para.align = f.align;
449        }
450        let mut t = self.table.take().unwrap_or_default();
451        t.set_part(p, f);
452        self.table = (t != TableStyleParts::default()).then_some(t);
453    }
454}

528        let mut s = StyleSheet {
529            default_chr: CharProps { font: Some(BODY_FONT.into()), size: Some(12.0), kern: Some(1.0), lang: Some("en-US".into()), ..chr() },
530            default_para: ParaProps {

534                widow_control: Some(true),
535                ..para()
536            },
537            default_para_present: true,
538            styles: Vec::new(),
539            latent_styles: None,
540        };
541        let mut add = |st: Style| s.styles.push(st);
542        add(Style { id: "Normal".into(), name: "Normal".into(), priority: Some(0), quick: true, ..p_style() });
543        add(Style {
544            id: "DefaultParagraphFont".into(),
545            name: "Default Paragraph Font".into(),
546            kind: StyleKind::Character,
547            priority: Some(1),
548            hidden: true,
549            builtin: true,
550            ..Default::default()

780    /// Find by display name (case-insensitive) or id.
781    pub fn find(&self, name_or_id: &str) -> Option<&Style> {
782        self.get(name_or_id).or_else(|| self.styles.iter().find(|s| s.name.eq_ignore_ascii_case(name_or_id)))
783    }
784    /// The Quick Styles gallery, in priority order.
785    pub fn gallery(&self) -> Vec<&Style> {
786        let mut v: Vec<&Style> = self.styles.iter().filter(|s| s.quick && !s.hidden && s.kind != StyleKind::Table).collect();
787        v.sort_by_key(|s| (s.priority.unwrap_or(99), s.name.clone()));
788        v
789    }
790    /// The `based_on` chain from the root down to `id` (bounded, cycle-safe).
791    pub fn chain(&self, id: &str) -> Vec<&Style> {
792        let mut out = Vec::new();
793        let mut cur = self.get(id);
794        while let Some(s) = cur {
795            if out.len() >= 32 || out.iter().any(|o: &&Style| o.id == s.id) {
796                break;
797            }
798            out.push(s);
799            cur = s.based_on.as_deref().and_then(|b| self.get(b));
800        }
801        out.reverse();
802        out
803    }
804    /// Which indents (left, first line) of a paragraph that gets its list from style `id` the style
805    /// chain sets over the list level's own. Word applies the level's indents on top of the most
806    /// derived style that gives a list, so that style and the ones based on it win over the level,
807    /// and the styles it is based on don't (Heading 3 with its own left indent on Heading 2's
808    /// list). `(false, false)` when no style in the chain gives a list.
809    pub fn indents_over_list(&self, id: &str) -> (bool, bool) {
810        let chain = self.chain(id);
811        let Some(k) = chain.iter().rposition(|s| s.para.numbering.is_some()) else { return (false, false) };
812        let set = |v: Option<f32>, chars: Option<f32>| v.is_some() || chars.is_some_and(|c| c != 0.0);
813        chain.get(k..).unwrap_or_default().iter().fold((false, false), |(l, f), s| {
814            (l || set(s.para.indent_left, s.para.indent_left_chars), f || set(s.para.indent_first, s.para.indent_first_chars))
815        })
816    }
817    /// Table style `id` with its based-on chain merged, base styles first (`None` when there is no
818    /// such table style). Styles of other kinds in the chain are skipped.
819    pub fn table_style(&self, id: &str) -> Option<TableStyleProps> {
820        let chain: Vec<&Style> = self.chain(id).into_iter().filter(|s| s.kind == StyleKind::Table).collect();
821        if chain.last().is_none_or(|s| s.id != id) {
822            return None;
823        }
824        let mut out = TableStyleProps::default();
825        for s in chain {
826            out.para.overlay(&s.para);
827            out.chr.overlay(&s.chr);
828            if let Some(t) = &s.table {
829                out.parts.overlay(t);
830            }
831        }
832        out.para.style = None;
833        out.chr.style = None;
834        Some(out)
835    }
836    /// Add or replace a style.
837    pub fn upsert(&mut self, st: Style) {
838        match self.styles.iter_mut().find(|s| s.id == st.id) {
839            Some(s) => *s = st,
840            None => self.styles.push(st),
841        }
842    }
843    /// For a file from a producer that never writes `w:qFormat` (LibreOffice): give it Word's
844    /// usual Quick Styles gallery. Each built-in style whose id or name matches a quick style of
845    /// [`StyleSheet::builtin`] becomes quick (taking the built-in's priority when it has none), and
846    /// the quick built-ins the file lacks are added with their linked styles. Existing ids and
847    /// names are never overwritten. Callers use it only when no style in the file is quick.
848    pub fn add_default_gallery(&mut self) {
849        let builtin = StyleSheet::builtin();
850        for q in builtin.styles.iter().filter(|s| s.quick && s.kind != StyleKind::Table) {
851            let found =
852                self.styles.iter().position(|s| s.id == q.id).or_else(|| self.styles.iter().position(|s| s.name.eq_ignore_ascii_case(&q.name)));
853            if let Some(st) = found.and_then(|i| self.styles.get_mut(i)) {
854                if st.builtin && st.kind == q.kind {
855                    st.quick = true;
856                    st.priority = st.priority.or(q.priority);
857                }
858                continue;
859            }
860            for sid in std::iter::once(q.id.as_str()).chain(q.linked.as_deref()) {
861                if let Some(b) = builtin.get(sid)
862                    && self.get(sid).is_none()
863                    && self.find(&b.name).is_none()
864                {
865                    self.styles.push(b.clone());
866                }
867            }
868        }
869    }
870    /// Remove style `id` and return it. Styles based on it are rebased onto its own base, and
871    /// `next` / `linked` references to it are dropped. Text that uses it is the document's to
872    /// retarget ([`crate::doc::Document::restyle`]).
873    pub fn remove(&mut self, id: &str) -> Option<Style> {
874        let i = self.styles.iter().position(|s| s.id == id)?;
875        let gone = self.styles.remove(i);
876        for s in &mut self.styles {
877            if s.based_on.as_deref() == Some(id) {
878                s.based_on = gone.based_on.clone().filter(|b| *b != s.id);
879            }
880            if s.next.as_deref() == Some(id) {
881                s.next = None;
882            }
883            if s.linked.as_deref() == Some(id) {
884                s.linked = None;
885            }
886        }
887        Some(gone)
888    }
889    /// An id not used yet, derived from a display name.
890    pub fn new_id(&self, name: &str) -> String {
891        let base: String = name.chars().filter(|c| c.is_alphanumeric()).collect();
892        let base = if base.is_empty() { "Style".to_string() } else { base };
893        if self.get(&base).is_none() {
894            return base;
895        }
896        (1..10_000).map(|i| format!("{base}{i}")).find(|id| self.get(id).is_none()).unwrap_or(base)
897    }
