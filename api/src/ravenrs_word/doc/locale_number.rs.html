1//! Numbers written the way a language writes them: `1 250,50` in Polish, `1.250,50` in German,
2//! `1,250.50` in English. Used by table sort and table formulas.
3
4use crate::doc::collate::language;
5
6/// The decimal separator of language `lang` (a BCP 47 tag; `None` means English).
7pub fn decimal_separator(lang: Option<&str>) -> char {
8    let tag = lang.unwrap_or("").trim().to_ascii_lowercase().replace('_', "-");
9    // Regions that write a decimal point where the language usually writes a comma.
10    const POINT_REGIONS: &[&str] =
11        &["de-ch", "de-li", "it-ch", "es-mx", "es-us", "es-419", "es-pr", "es-do", "es-gt", "es-hn", "es-ni", "es-pa", "es-sv", "es-cu", "es-pe"];
12    if POINT_REGIONS.iter().any(|r| tag == *r || tag.starts_with(&format!("{r}-"))) {
13        return '.';
14    }
15    const COMMA: &[&str] = &[
16        "af", "az", "be", "bg", "bs", "ca", "cs", "da", "de", "el", "es", "et", "eu", "fi", "fo", "fr", "gl", "hr", "hu", "hy", "id", "is", "it",
17        "ka", "kk", "ky", "lb", "lt", "lv", "mk", "mn", "nb", "nl", "nn", "no", "pl", "pt", "ro", "ru", "sk", "sl", "sq", "sr", "sv", "tr", "uk",
18        "uz", "vi",
19    ];
20    if COMMA.contains(&language(lang).as_str()) { ',' } else { '.' }
21}

23const SPACES: &[char] = &[' ', '\u{a0}', '\u{202f}', '\u{2009}'];
24const CURRENCY: &[char] = &['$', '€', '£', '¥', '¢', '₽', '₴', '₺', '₹', '₩', '₪', '₫', '฿', '₱', '₸', '₼', '₾'];
25/// Currency signs spelled with more than one character: the Panamanian balboa (`B/.`).
26const CURRENCY_SYMBOLS: &[&str] = &["B/."];
27
28/// Trim spaces and single-character currency signs from either end.
29fn trim_currency(s: &str) -> &str {
30    s.trim_matches(|c: char| c.is_whitespace() || CURRENCY.contains(&c))
31}
32
33/// Strip a currency symbol or a short currency word (`zł`, `kr`, `PLN`, `B/.`) from either end.
34/// As Word does, a multi-character sign is a word of its own (`B/. 1,250.50`, `1,250.50 B/.`);
35/// `B/.1,250.50` and `B/ 1,250.50` are not numbers.
36fn strip_currency(s: &str) -> &str {
37    let mut s = trim_currency(s);
38    if let Some(rest) =
39        CURRENCY_SYMBOLS.iter().find_map(|sym| s.strip_prefix(sym).filter(|rest| rest.chars().next().is_some_and(char::is_whitespace)))
40    {
41        s = trim_currency(rest);
42    }
43    if let Some(rest) =
44        CURRENCY_SYMBOLS.iter().find_map(|sym| s.strip_suffix(sym).filter(|rest| rest.chars().next_back().is_some_and(char::is_whitespace)))
45    {
46        s = trim_currency(rest);
47    }
48    if let Some((head, word)) = s.rsplit_once(SPACES)
49        && (1..=3).contains(&word.chars().count())
50        && word.chars().all(|c| c.is_alphabetic() || c == '.')
51    {
52        s = head;
53    }
54    if let Some((word, tail)) = s.split_once(SPACES)
55        && word.chars().count() == 3
56        && word.chars().all(|c| c.is_ascii_uppercase())
57    {
58        s = tail;
59    }
60    s.trim_matches(|c: char| c.is_whitespace() || CURRENCY.contains(&c))
61}

63/// Digits of an integer part with group separators `groups`: the first group has 1–3 digits,
64/// every later one exactly 3.
65fn grouped_digits(int: &str, groups: &[char]) -> Option<String> {
66    if int.chars().all(|c| c.is_ascii_digit()) {
67        return Some(int.to_string());
68    }
69    let parts: Vec<&str> = int.split(groups).collect();
70    let (first, rest) = parts.split_first()?;
71    // A first group of `0` is never grouping (`0.125` is a fraction, not 125).
72    let ok = (1..=3).contains(&first.len())
73        && !first.starts_with('0')
74        && first.chars().all(|c| c.is_ascii_digit())
75        && rest.iter().all(|g| g.len() == 3 && g.chars().all(|c| c.is_ascii_digit()));
76    ok.then(|| parts.concat())
77}

79/// Parse `s` as a number written in language `lang`, or `None` when it isn't one. The decimal
80/// separator is the language's; group separators are spaces (and `.` or `'` where the decimal
81/// separator is a comma, `,` or `'` where it is a point) between groups of three digits. A
82/// currency sign or short currency word is ignored, and so is a trailing `%` (`12 %` is 12, as
83/// table sort reads it). In comma languages a lone `.` that cannot be a group separator (`1.5`)
84/// is read as a decimal point. A trailing separator with no digits after it (`1.`, an ordinal
85/// in a numbered column) is the whole number. A minus may be `-`, `−` or an en dash.
86pub fn parse_number(s: &str, lang: Option<&str>) -> Option<f64> {
87    let dec = decimal_separator(lang);
88    let s = s.trim();
89    let s = s.strip_suffix('%').map_or(s, str::trim_end);
90    let mut body = strip_currency(s);
91    let mut neg = false;
92    if let Some(rest) = body.strip_prefix(['-', '\u{2212}', '\u{2013}']) {
93        neg = true;
94        body = strip_currency(rest);
95    } else if let Some(rest) = body.strip_prefix('+') {
96        body = strip_currency(rest);
97    }
98    if body.is_empty() || body.len() > 400 {
99        return None;
100    }
101    let mut groups: Vec<char> = SPACES.to_vec();
102    groups.extend(['\'', '\u{2019}']);
103    groups.push(if dec == ',' { '.' } else { ',' });
104    let (int, frac) = match body.split_once(dec) {
105        Some((i, f)) => (i, Some(f)),
106        None => (body, None),
107    };
108    let int_digits = match grouped_digits(int, &groups) {
109        Some(d) => d,
110        None if dec == ',' && frac.is_none() && int.matches('.').count() == 1 => {
111            return parse_number(&format!("{}{}", if neg { "-" } else { "" }, int.replace('.', ",")), lang);
112        }
113        None => return None,
114    };
115    let frac_digits = match frac {
116        Some(f) if f.chars().all(|c| c.is_ascii_digit()) => f,
117        Some(_) => return None,
118        None => "",
119    };
120    if int_digits.is_empty() && frac_digits.is_empty() {
121        return None;
122    }
123    let v: f64 = format!("0{int_digits}.{frac_digits}0").parse().ok()?;
124    if !v.is_finite() {
125        return None;
126    }
127    Some(if neg { -v } else { v })
128}

130/// Write `v` the way language `lang` writes a formula result: whole numbers without decimals,
131/// others with two decimals and the language's decimal separator (`5,75` in Polish).
132pub fn format_number(v: f64, lang: Option<&str>) -> String {
133    if !v.is_finite() {
134        return "0".into();
135    }
136    if v.fract() == 0.0 && v.abs() < 9.2e18 {
137        return format!("{}", v as i64);
138    }
139    let s = format!("{v:.2}");
140    let s = if s == "-0.00" { "0.00".to_string() } else { s };
141    if decimal_separator(lang) == ',' { s.replace('.', ",") } else { s }
142}

148    #[test]
149    fn polish_numbers() {
150        let pl = Some("pl-PL");
151        assert_eq!(parse_number("1,5", pl), Some(1.5));
152        assert_eq!(parse_number("3,50", pl), Some(3.5));
153        assert_eq!(parse_number("1 250,00", pl), Some(1250.0));
154        assert_eq!(parse_number("1\u{a0}250,00 zł", pl), Some(1250.0));
155        assert_eq!(parse_number("1.250,00", pl), Some(1250.0));
156        assert_eq!(parse_number("1.5", pl), Some(1.5));
157        assert_eq!(parse_number("-2,25", pl), Some(-2.25));
158        assert_eq!(parse_number("PLN 12", pl), Some(12.0));
159        assert_eq!(parse_number("-B/. 1 250,50", pl), Some(-1250.5));
160        assert_eq!(parse_number("abc", pl), None);
161        assert_eq!(parse_number("1,2,3", pl), None);
162        // From review: ordinals, a leading zero, percentages and an en dash minus.
163        assert_eq!(parse_number("1.", pl), Some(1.0), "a numbered column: 1.");
164        assert_eq!(parse_number("10.", pl), Some(10.0));
165        assert_eq!(parse_number("0.125", pl), Some(0.125), "a leading 0 is never a group");
166        assert_eq!(parse_number("0,125", pl), Some(0.125));
167        assert_eq!(parse_number("12 %", pl), Some(12.0));
168        assert_eq!(parse_number("12%", pl), Some(12.0));
169        assert_eq!(parse_number("\u{2013}2,5", pl), Some(-2.5));
170        assert_eq!(parse_number("%", pl), None);
171        assert_eq!(parse_number(".", pl), None);
172        assert_eq!(format_number(5.75, pl), "5,75");
173        assert_eq!(format_number(1350.0, pl), "1350");
174        assert_eq!(format_number(-0.001, pl), "0,00", "a rounded zero has no sign");
175    }

180        assert_eq!(parse_number(&"9".repeat(400), Some("pl-PL")), None);
181        assert_eq!(parse_number(&format!("-{}", "1".repeat(350)), None), None);
182        let big = parse_number(&"1".repeat(300), None);
183        assert!(big.is_some_and(f64::is_finite), "{big:?}");
184    }

197    #[test]
198    fn balboa_numbers() {
199        let en = Some("en-US");
200        // Word 16.0 reads `B/.` as a word of its own, before or after the number.
201        assert_eq!(parse_number("B/. 1,234.50", en), Some(1234.5));
202        assert_eq!(parse_number("1,234.50 B/.", en), Some(1234.5));
203        assert_eq!(parse_number("B/.\u{a0}1,234.50", en), Some(1234.5), "a no-break space after the sign");
204        assert_eq!(parse_number("B/. 1,234.50 B/.", en), Some(1234.5), "the sign at both ends");
205        assert_eq!(parse_number("-B/. 1,234.50", en), Some(-1234.5), "a minus before the sign");
206        assert_eq!(parse_number("B/. 1,234.50", Some("es-PA")), Some(1234.5), "the balboa's own region");
207        // The sign has no space of its own in Word: touching the number is text.
208        assert_eq!(parse_number("B/.1,234.50", en), None);
209        assert_eq!(parse_number("1,234.50B/.", en), None);
210        assert_eq!(parse_number("B/ 1,234.50", en), None, "the balboa is B/. in Word");
211        assert_eq!(parse_number("B/.", en), None);
212        // The slash and the full stop make `B/.` ambiguous with a fraction: the sign is matched at
213        // the ends of the text only, so a fraction is never read as a balboa amount and a sign in
214        // the middle is not a sign.
215        assert_eq!(parse_number("1/.5", en), None, "a fraction is not a price");
216        assert_eq!(parse_number("1/.5", Some("es-PA")), None);
217        assert_eq!(parse_number("1/.5 B/.", en), None, "the sign at the end does not rescue a fraction");
218        assert_eq!(parse_number("B/. 1/.5", en), None);
219        assert_eq!(parse_number("2 B/. 1,234.50", en), None, "the sign sits before the number");
220        assert_eq!(parse_number("1,234.50 B/. 1,234.50", en), None, "the sign sits between two numbers");
221        assert_eq!(parse_number("1,234.50/2", en), None, "a division is not a balboa amount");
222        assert_eq!(parse_number("B/. 1,234.50,50", en), None, "a second decimal separator");
223    }
