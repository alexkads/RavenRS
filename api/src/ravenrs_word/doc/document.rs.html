12/// [`Document::media`] by content, so adding media is linear in the bytes added rather than a
13/// scan of every earlier image (CWE-407). Rebuilt whenever `media` changed size behind its back;
14/// a hit is always checked against `media`, so a stale entry can at worst miss a duplicate.
15#[derive(Clone, Default)]
16pub(crate) struct MediaIndex {
17    by_bytes: std::collections::HashMap<Arc<Vec<u8>>, String>,
18    /// `media.len()` when the index last matched it.
19    len: usize,
20    /// The number after the last generated `image{n}` key, so taken names are probed once.
21    next: usize,
22}
23
24impl MediaIndex {
25    fn rebuild(&mut self, media: &BTreeMap<String, Arc<Vec<u8>>>) {
26        self.by_bytes.clear();
27        // The first key (in key order) of equal bytes wins, as a scan would find it.
28        for (k, v) in media.iter().rev() {
29            self.by_bytes.insert(v.clone(), k.clone());
30        }
31        self.len = media.len();
32        self.next = 0;
33    }
34
35    fn find(&self, media: &BTreeMap<String, Arc<Vec<u8>>>, bytes: &Vec<u8>) -> Option<String> {
36        self.by_bytes.get(bytes).filter(|k| media.get(*k).is_some_and(|v| **v == *bytes)).cloned()
37    }
38}

98    fn fix_story_blocks(bl: &mut Blocks) {
99        if !bl.iter().any(|b| matches!(**b, Block::Para(_))) || matches!(bl.last().map(|b| &**b), Some(Block::Table(_))) {
100            bl.push(para_block(Paragraph::new()));
101        }
102        for b in bl.iter_mut() {
103            if let Block::Table(t) = &**b {
104                let needs = t
105                    .rows
106                    .iter()
107                    .any(|r| r.cells.iter().any(|c| c.blocks.is_empty() || matches!(c.blocks.last().map(|b| &**b), Some(Block::Table(_)))));
108                if needs && let Block::Table(t) = Arc::make_mut(b) {
109                    for r in &mut t.rows {
110                        for c in &mut r.cells {
111                            Self::fix_cell_blocks(&mut c.blocks);
112                        }
113                    }
114                }
115            }
116        }
117    }

166            comments: other_comments,

212    /// The block list that contains the block at `path`.
213    pub fn container(&self, s: StoryRef, path: &Path) -> Option<&Blocks> {
214        let mut bl = self.story(s)?;
215        let parent = path.parent();
216        if !parent.len().is_multiple_of(3) {
217            return None;
218        }
219        crate::doc::work::steps(1);
220        for ch in parent.chunks(3) {
221            crate::doc::work::steps(1);
222            let [b, r, c] = ch else { return None };
223            let Block::Table(t) = &**bl.get(*b as usize)? else { return None };
224            bl = &t.rows.get(*r as usize)?.cells.get(*c as usize)?.blocks;
225        }
226        Some(bl)
227    }

229    pub fn container_mut(&mut self, s: StoryRef, path: &Path) -> Result<&mut Blocks> {
230        let bad = || DocError::BadPath(path.to_string());
231        let parent: Vec<u32> = path.parent().to_vec();
232        if !parent.len().is_multiple_of(3) {
233            return Err(bad());
234        }
235        let mut bl = self.story_mut(s)?;
236        for ch in parent.chunks(3) {
237            let [b, r, c] = ch else { return Err(bad()) };
238            let blk = bl.get_mut(*b as usize).ok_or_else(bad)?;
239            let Block::Table(t) = Arc::make_mut(blk) else { return Err(bad()) };
240            bl = &mut t.rows.get_mut(*r as usize).ok_or_else(bad)?.cells.get_mut(*c as usize).ok_or_else(bad)?.blocks;
241        }
242        Ok(bl)
243    }

297    /// The position just after the shape that owns text box story `part` (body first, then the
298    /// other parts), for leaving the text box.
299    pub fn text_box_anchor(&self, part: u32) -> Option<Pos> {
300        let stories = std::iter::once(StoryRef::Body).chain(self.parts.keys().filter(|k| **k != part).map(|k| StoryRef::Part(*k)));
301        for story in stories {
302            for path in self.para_paths(story) {
303                let Some(p) = self.para(story, &path) else { continue };
304                for (off, o) in p.objects_with_offsets() {
305                    if o.text_boxes().contains(&part) {
306                        return Some(Pos { story, path, off: off + para::OBJ.len_utf8() });
307                    }
308                }
309            }
310        }
311        None
312    }

316        let all = self.para_paths(s);

321        let all = self.para_paths(s);

323        i.checked_sub(1).and_then(|j| all.get(j).cloned())

326    /// Clamp a position into the document (valid path, char boundary).
327    pub fn clamp(&self, p: &Pos) -> Pos {
328        if let Some(para) = self.para_at(p) {
329            return Pos { story: p.story, path: p.path.clone(), off: para.clamp(p.off) };
330        }
331        let story = if self.story(p.story).is_some() { p.story } else { StoryRef::Body };
332        let all = self.para_paths(story);
333        let path = all.iter().rev().find(|q| **q <= p.path).or(all.first()).cloned().unwrap_or_else(|| Path::top(0));
334        let off = self.para(story, &path).map(|q| q.clamp(p.off)).unwrap_or(0);
335        Pos { story, path, off }
336    }

380    /// The footnote (`endnote` false) or endnote options in effect in section `sect`: its own,
381    /// else the document's, else Word's defaults (footnotes at the page bottom in 1, 2, 3;
382    /// endnotes at the end of the document in i, ii, iii; numbered on through the document).
383    /// Placements that don't apply to the kind fall back to its default.
384    pub fn note_options(&self, sect: &SectionProps, endnote: bool) -> section::NoteOptions {
385        use section::NotePos;
386        let (own, doc, fmt) = if endnote {
387            (&sect.endnote_pr, &self.settings.endnote_pr, self.settings.endnote_format)
388        } else {
389            (&sect.footnote_pr, &self.settings.footnote_pr, self.settings.footnote_format)
390        };
391        let pos = own.pos.or(doc.pos);
392        let pos = if endnote {
393            pos.filter(|p| matches!(p, NotePos::SectEnd | NotePos::DocEnd)).unwrap_or(NotePos::DocEnd)
394        } else {
395            pos.filter(|p| matches!(p, NotePos::PageBottom | NotePos::BeneathText)).unwrap_or(NotePos::PageBottom)
396        };
397        section::NoteOptions {
398            pos,
399            num_fmt: own.num_fmt.unwrap_or(fmt),
400            num_start: own.num_start.or(doc.num_start).unwrap_or(1).clamp(1, section::MAX_NOTE_START),
401            num_restart: own.num_restart.or(doc.num_restart).unwrap_or_default(),
402        }
403    }

424    /// Remove the section break on the mark of top-level body paragraph `block`: its section
425    /// joins the following one ([`SectionProps::absorb`]). False when there is no break there.
426    pub fn remove_section_break(&mut self, block: usize) -> bool {
427        if !matches!(self.body.get(block).map(|b| &**b), Some(Block::Para(p)) if p.section.is_some()) {
428            return false;
429        }
430        let Some(Block::Para(p)) = self.body.get_mut(block).map(Arc::make_mut) else { return false };
431        let Some(before) = p.section.take() else { return false };
432        self.section_mut(block).absorb(&before);
433        true

442    /// [`Document::add_part`], or an error, adding nothing, when every part id is taken.
443    pub fn try_add_part(&mut self, kind: PartKind, blocks: Blocks) -> Result<u32> {
444        let id = free_id(self.parts.keys().next_back().copied(), self.parts.keys().copied(), 1)
445            .ok_or_else(|| DocError::Invalid("no free part id".into()))?;
446        let mut part = Part { kind, blocks, body: TextBody::default() };
447        if part.blocks.is_empty() {
448            part.blocks.push(para_block(Paragraph::new()));
449        }
450        self.parts.insert(id, part);
451        Ok(id)

477    /// The stories a reader sees, in reading order: the body, then the headers and footers in
478    /// use, each followed by the notes and text boxes it shows. Hidden stories (a removed header,
479    /// a first-page or even-page header whose option is off, a linked text box's own text) and
480    /// comment text aren't included.
481    pub fn shown_stories(&self) -> Vec<StoryRef> {
482        let followers: std::collections::BTreeSet<u32> = self.text_box_chains().into_iter().flat_map(|c| c.into_iter().skip(1)).collect();
483        let follow = |o: &InlineObject| {
484            let mut v = notes_and_boxes(o);
485            v.retain(|(id, _)| !followers.contains(id));
486            v
487        };
488        let mut out = self.reachable(vec![StoryRef::Body], &follow);
489        let mut seen: std::collections::BTreeSet<StoryRef> = out.iter().copied().collect();
490        let mut roots = Vec::new();
491        let mut root_seen = std::collections::BTreeSet::new();
492        let (mut headers, mut footers) = (section::HeaderSet::default(), section::HeaderSet::default());
493        for (_, sec) in self.sections() {
494            // Unset versions are inherited from the previous section.
495            for (set, own) in [(&mut headers, &sec.headers), (&mut footers, &sec.footers)] {
496                set.default = own.default.or(set.default);
497                set.first = own.first.or(set.first);
498                set.even = own.even.or(set.even);
499                let first = set.first.filter(|_| sec.title_page);
500                let even = set.even.filter(|_| self.settings.even_odd_headers);
501                // By the first page each one shows on: page 1, then page 2 (even), then page 3.
502                let order = if sec.title_page { [first, even, set.default] } else { [set.default, even, None] };
503                for id in order.into_iter().flatten() {
504                    if self.parts.get(&id).is_some_and(|p| matches!(p.kind, PartKind::Header | PartKind::Footer)) && root_seen.insert(id) {
505                        roots.push(StoryRef::Part(id));
506                    }
507                }
508            }
509        }
510        for st in self.reachable(roots, &follow) {
511            if seen.insert(st) {
512                out.push(st);
513            }
514        }
515        out
516    }

518    /// Drop the text box stories no box shows any more (a deleted or cut box's text), nested
519    /// ones too. Undo snapshots and the clipboard keep their own copies. Free for a document
520    /// without text boxes; otherwise one pass over its paragraphs. Returns how many went.
521    pub fn prune_text_boxes(&mut self) -> usize {
522        if !self.parts.values().any(|p| p.kind == PartKind::TextBox) {
523            return 0;
524        }
525        let roots = std::iter::once(StoryRef::Body)
526            .chain(self.parts.iter().filter(|(_, p)| p.kind != PartKind::TextBox).map(|(id, _)| StoryRef::Part(*id)))
527            .collect();
528        let live = self.reachable(roots, &|o| o.text_boxes().into_iter().map(|id| (id, &[PartKind::TextBox][..])).collect());
529        let before = self.parts.len();
530        self.parts.retain(|id, p| p.kind != PartKind::TextBox || live.contains(&StoryRef::Part(*id)));
531        let gone = before - self.parts.len();
532        if gone > 0 {
533            // A link to a box that went ends the chain there (its id may be reused).
534            let ids: std::collections::BTreeSet<u32> = self.parts.keys().copied().collect();
535            for p in self.parts.values_mut() {
536                if p.body.next.is_some_and(|n| !ids.contains(&n)) {
537                    p.body.next = None;
538                }
539            }
540        }
541        gone
542    }

544    /// The chains of linked text boxes (Create Link), each in order from the box holding the
545    /// text: every chain has two or more boxes. A link to a part that isn't a text box, a box
546    /// already linked from another, or links running in a circle are ignored (those boxes show
547    /// their own text); a chain stops after [`MAX_LINKED_BOXES`].
548    pub fn text_box_chains(&self) -> Vec<Vec<u32>> {
549        let is_box = |id: u32| self.parts.get(&id).is_some_and(|p| p.kind == PartKind::TextBox);
550        // Who links to whom: the first box (by id) to link to a box wins.
551        let mut prev: BTreeMap<u32, u32> = BTreeMap::new();
552        for (id, p) in &self.parts {
553            if p.kind != PartKind::TextBox {
554                continue;
555            }
556            if let Some(n) = p.body.next.filter(|n| *n != *id && is_box(*n)) {
557                prev.entry(n).or_insert(*id);
558            }
559        }
560        let next = |id: u32| self.parts.get(&id).and_then(|p| p.body.next).filter(|n| prev.get(n) == Some(&id));
561        let mut out = Vec::new();
562        for (id, p) in &self.parts {
563            if p.kind != PartKind::TextBox || prev.contains_key(id) || next(*id).is_none() {
564                continue;
565            }
566            let mut chain = vec![*id];
567            let mut at = *id;
568            while let Some(n) = next(at) {
569                if chain.len() >= MAX_LINKED_BOXES || chain.contains(&n) {
570                    break;
571                }
572                chain.push(n);
573                at = n;
574            }
575            out.push(chain);
576        }
577        out
578    }

585    /// The size (width, height) of the shapes showing text boxes `ids`, in any story (a group
586    /// member's as the group stretches it).
587    pub fn text_box_sizes(&self, ids: &std::collections::BTreeSet<u32>) -> BTreeMap<u32, (f32, f32)> {
588        let mut out = BTreeMap::new();
589        let stories = std::iter::once(&self.body).chain(self.parts.values().map(|p| &p.blocks));
590        for blocks in stories {
591            for b in blocks {
592                edit::each_para(b, 0, &mut |p| {
593                    for o in &p.objects {
594                        match o {
595                            InlineObject::Shape { story: Some(id), w, h, .. } if ids.contains(id) => {
596                                out.entry(*id).or_insert((*w, *h));
597                            }
598                            InlineObject::Group { w, h, .. } => {
599                                for (r, c) in o.group_rects(0.0, 0.0, *w, *h) {
600                                    if let Some(id) = c.text_box().filter(|id| ids.contains(id)) {
601                                        out.entry(id).or_insert((r[2], r[3]));
602                                    }
603                                }
604                            }
605                            _ => {}
606                        }
607                    }
608                });
609            }
610        }
611        out
612    }

640    /// The footnotes and endnotes referenced inside text box story `part` (nested boxes too), in
641    /// reading order.
642    pub fn notes_in_text_box(&self, part: u32) -> Vec<u32> {
643        let mut out = Vec::new();
644        let mut budget = BoxBudget::default();
645        if let Some(p) = self.parts.get(&part).filter(|p| p.kind == PartKind::TextBox)
646            && budget.enter(part)
647        {
648            self.walk_objects(&p.blocks, &mut budget, &mut |p, k| {
649                if let Some(InlineObject::NoteRef { id, .. }) = p.objects.get(k) {
650                    out.push(*id);
651                }
652                true
653            });
654            budget.leave();
655        }
656        out
657    }

659    fn walk_objects(&self, blocks: &[Arc<Block>], budget: &mut BoxBudget, f: &mut dyn FnMut(&Paragraph, usize) -> bool) {
660        for b in blocks {
661            edit::each_para(b, 0, &mut |p| {
662                for (k, o) in p.objects.iter().enumerate() {
663                    if !f(p, k) {
664                        continue;
665                    }
666                    for id in o.text_boxes() {
667                        if let Some(part) = self.parts.get(&id).filter(|p| p.kind == PartKind::TextBox)
668                            && budget.enter(id)
669                        {
670                            self.walk_objects(&part.blocks, budget, f);
671                            budget.leave();
672                        }
673                    }
674                }
675            });
676        }
677    }

679    /// `roots`, then every story their objects lead to (`follow`: an object's story id and the
680    /// part kinds that count), transitively, in the order found.
681    fn reachable(&self, roots: Vec<StoryRef>, follow: &dyn Fn(&InlineObject) -> Vec<(u32, &'static [PartKind])>) -> Vec<StoryRef> {
682        let mut out = roots;
683        let mut seen: std::collections::BTreeSet<StoryRef> = out.iter().copied().collect();
684        let mut i = 0;
685        while let Some(s) = out.get(i).copied() {
686            i += 1;
687            let mut found = Vec::new();
688            for b in self.story(s).into_iter().flatten() {
689                edit::each_para(b, 0, &mut |p| {
690                    found.extend(p.objects.iter().flat_map(follow));
691                });
692            }
693            for (id, kinds) in found {
694                if self.parts.get(&id).is_some_and(|p| kinds.contains(&p.kind)) && seen.insert(StoryRef::Part(id)) {
695                    out.push(StoryRef::Part(id));
696                }
697            }
698        }
699        out
700    }

702    /// A drawing id no shape uses yet (for gluing a connector to a shape that has none).
703    pub fn next_shape_id(&self) -> u32 {
704        let mut max = 0u32;
705        let mut see = |o: &InlineObject| match o {
706            InlineObject::Shape { extra, .. } => max = max.max(extra.id),
707            InlineObject::Group { children, .. } => {
708                for c in children {
709                    if let InlineObject::Shape { extra, .. } = &c.obj {
710                        max = max.max(extra.id);
711                    }
712                }
713            }
714            _ => {}
715        };
716        for blocks in std::iter::once(&self.body).chain(self.parts.values().map(|p| &p.blocks)) {
717            for b in blocks {
718                edit::each_para(b, 0, &mut |p| p.objects.iter().for_each(&mut see));
719            }
720        }
721        max.saturating_add(1).max(1)
722    }

724    /// Change every inline object in every story (paragraphs inside tables too), copying only
725    /// the blocks whose paragraphs `wants` says have objects to change.
726    pub fn objects_mut(&mut self, wants: &dyn Fn(&InlineObject) -> bool, f: &mut dyn FnMut(&mut InlineObject)) {
727        let mut stories: Vec<&mut Blocks> = vec![&mut self.body];
728        stories.extend(self.parts.values_mut().map(|p| &mut p.blocks));
729        for blocks in stories {
730            for b in blocks.iter_mut() {
731                let mut any = false;
732                edit::each_para(b, 0, &mut |p| any |= p.objects.iter().any(wants));
733                if any {
734                    edit::each_para_mut(Arc::make_mut(b), 0, &mut |p| {
735                        if p.objects.iter().any(wants) {
736                            p.objects.iter_mut().filter(|o| wants(o)).for_each(&mut *f);
737                            p.touch();
738                        }
739                    });
740                }
741            }
742        }
743    }

745    /// [`Document::objects_mut`] for filling in what an object shows from what it is (a chart's
746    /// drawing from its data), which doesn't change the document: the markup kept as read for
747    /// the paragraphs and tables holding them (see [`crate::doc::kept`]) follows them.
748    pub fn fill_objects(&mut self, wants: &dyn Fn(&InlineObject) -> bool, f: &mut dyn FnMut(&mut InlineObject)) {
749        // The stories (text boxes', notes') as read before, when filling changes one: they stay
750        // as read after it, so the paragraphs showing them keep their markup.
751        let holds = |b: &Arc<Block>| {
752            let mut any = false;
753            edit::each_para(b, 0, &mut |p| any |= p.objects.iter().any(wants));
754            any
755        };
756        let as_read: Vec<u32> = match self.kept.get() {
757            Some(k) if !k.story_fps.is_empty() && self.parts.values().any(|p| p.blocks.iter().any(holds)) => {
758                k.story_fps.iter().filter(|(id, fp)| **fp == crate::doc::kept::story_fingerprint(self, **id)).map(|(id, _)| *id).collect()
759            }
760            _ => Vec::new(),
761        };
762        let mut moves: Vec<crate::doc::kept::Moved> = Vec::new();
763        let mut stories: Vec<&mut Blocks> = vec![&mut self.body];
764        stories.extend(self.parts.values_mut().map(|p| &mut p.blocks));
765        for blocks in stories {
766            for b in blocks.iter_mut() {
767                let mut any = false;
768                edit::each_para(b, 0, &mut |p| any |= p.objects.iter().any(wants));
769                if !any {
770                    continue;
771                }
772                let before = crate::doc::kept::identities(b);
773                edit::each_para_mut(Arc::make_mut(b), 0, &mut |p| {
774                    if p.objects.iter().any(wants) {
775                        p.objects.iter_mut().filter(|o| wants(o)).for_each(&mut *f);
776                        p.touch();
777                    }
778                });
779                moves.push((before, crate::doc::kept::identities(b)));
780            }
781        }
782        let refreshed: Vec<(u32, u64)> = as_read.into_iter().map(|id| (id, crate::doc::kept::story_fingerprint(self, id))).collect();
783        if let Some(k) = self.kept.get_mut() {
784            for (before, after) in moves {
785                k.follow(&before, &after);
786            }
787            k.story_fps.extend(refreshed);
788        }
789    }

808    /// Add a media item; returns its key (deduplicated by content).
809    pub fn add_media(&mut self, bytes: Vec<u8>, ext: &str) -> String {
810        let ix = &mut self.media_index;
811        if ix.len != self.media.len() {
812            ix.rebuild(&self.media);
813        }
814        if let Some(k) = ix.find(&self.media, &bytes) {
815            return k;
816        }
817        if ix.by_bytes.contains_key(&bytes) {
818            // Stale (`media` was edited directly without changing its size): look again.
819            ix.rebuild(&self.media);
820            if let Some(k) = ix.find(&self.media, &bytes) {
821                return k;
822            }
823        }
824        let n = self.media.len().saturating_add(1);
825        let mut i = n.max(ix.next);
826        let mut key = format!("image{i}.{ext}");
827        while self.media.contains_key(&key) {
828            i = i.saturating_add(1);
829            key = format!("image{i}.{ext}");
830        }
831        let bytes = Arc::new(bytes);
832        self.media.insert(key.clone(), bytes.clone());
833        ix.by_bytes.entry(bytes).or_insert_with(|| key.clone());
834        ix.len = self.media.len();
835        ix.next = i.saturating_add(1);
836        key
837    }

864    /// Ranges that stay editable while the document is protected (Restrict Editing exceptions),
865    /// in document order across every story: the [`InlineObject::PermStart`]'s id, group and
866    /// editor, its start (just after the marker) and its end (at the matching
867    /// [`InlineObject::PermEnd`], or the end of the story when it has none).
868    pub fn perm_ranges(&self) -> Vec<PermRange> {
869        let mut out: Vec<PermRange> = Vec::new();
870        let stories = std::iter::once(StoryRef::Body).chain(self.parts.keys().map(|k| StoryRef::Part(*k)));
871        for story in stories {
872            // Open ranges by id, latest last (an end closes the latest open range of its id).
873            let mut open: BTreeMap<u32, Vec<usize>> = BTreeMap::new();
874            for path in self.para_paths(story) {
875                let Some(p) = self.para(story, &path) else { continue };
876                for (off, o) in p.objects_with_offsets() {
877                    match o {
878                        InlineObject::PermStart { id, group, editor } => {
879                            let start = Pos { story, path: path.clone(), off: off + para::OBJ.len_utf8() };
880                            open.entry(*id).or_default().push(out.len());
881                            out.push(PermRange { id: *id, group: group.clone(), editor: editor.clone(), start: start.clone(), end: start });
882                        }
883                        InlineObject::PermEnd { id } => {
884                            if let Some(i) = open.get_mut(id).and_then(Vec::pop)
885                                && let Some(r) = out.get_mut(i)
886                            {
887                                r.end = Pos { story, path: path.clone(), off };
888                            }
889                        }
890                        _ => {}
891                    }
892                }
893            }
894            // A range left open runs to the end of its story.
895            let end = self.end_of(story);
896            for i in open.into_values().flatten() {
897                if let Some(r) = out.get_mut(i) {
898                    r.end = end.clone();
899                }
900            }
901        }
902        out
903    }
