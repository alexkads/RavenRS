1//! Character, paragraph, table and cell properties.
2//!
3//! Every field is an `Option`: `None` means "inherit" (from the style chain, then document
4//! defaults). [`CharProps::overlay`] and friends apply a patch on top of a base.
5
6use serde::{Deserialize, Serialize};
7
8macro_rules! overlay_fields {
9    ($dst:ident, $src:ident; $($f:ident),*) => {
10        $( if $src.$f.is_some() { $dst.$f = $src.$f.clone(); } )*
11    };
12}
13
14macro_rules! drop_same_fields {
15    ($dst:ident, $base:ident; $($f:ident),*) => {
16        $( if $dst.$f == $base.$f { $dst.$f = None; } )*
17    };
18}

71    fn borders_overlay_per_side() {
72        let thin = Border::single(0.5);
73        let thick = Border::single(2.0);
74        let nil = Border { style: BorderStyle::None, width: 0.0, color: None, space: 0.0 };
75        let mut b = Borders::all(thin);
76        b.overlay(&Borders { top: Some(thick), ..Default::default() });
77        assert_eq!(b.top, Some(thick));
78        assert_eq!(b.bottom, Some(thin));
79        b.overlay(&Borders::box_(nil));
80        assert_eq!(b.left, Some(nil)); // explicit nil overrides that side
81        assert_eq!(b.between, Some(thin)); // untouched sides survive
82        b.overlay(&Borders::default());
83        assert_eq!(b.left, Some(nil)); // empty patch changes nothing
84    }

144    /// #865: a colour reads as `format.color` takes it (`RRGGBB`, `#RRGGBB`, `auto`), while what
145    /// documents and settings already hold reads and writes exactly as before.
146    #[test]
147    fn text_color_reads_hex_and_keeps_its_serialized_forms() {
148        let read = |j: &str| serde_json::from_str::<TextColor>(j);
149        let ink = TextColor::Rgb(Rgb(0x1B, 0x16, 0x11));
150        assert_eq!(read(r#""1B1611""#).unwrap(), ink);
151        assert_eq!(read(r##""#1b1611""##).unwrap(), ink);
152        assert_eq!(read(r#""auto""#).unwrap(), TextColor::Auto);
153        assert_eq!(read(r#""Automatic""#).unwrap(), TextColor::Auto);
154        for (c, j) in [(TextColor::Auto, r#""Auto""#), (ink, r#"{"Rgb":[27,22,17]}"#)] {
155            assert_eq!(serde_json::to_string(&c).unwrap(), j, "written as before");
156            assert_eq!(read(j).unwrap(), c);
157        }
158        for bad in [r#""1B161""#, r#""zzzzzz""#, r#""""#, r#"{"Rgb":"1B16"}"#, r#"{"Rgb":[1,2]}"#, r#"{"Hex":[1,2,3]}"#, "7", "[27,22,17]"] {
159            assert!(read(bad).is_err(), "{bad}");
160        }
161        assert_eq!(read(r#"{"Rgb":"1B1611"}"#).unwrap(), ink);
162        let p: CharProps = serde_json::from_value(serde_json::json!({"color": "#1B1611", "bold": true})).unwrap();
163        assert_eq!((p.color, p.bold), (Some(ink), Some(true)));
164    }

189    #[test]
190    fn symbol_font_is_optional_in_json_and_survives_font_changes() {
191        // Older documents (no `symFont`) still load.
192        let old: CharProps = serde_json::from_str(r#"{"font":"Arial"}"#).unwrap();
193        assert_eq!(old.sym_font, None);
194        let p = CharProps { font: Some("Arial".into()), sym_font: Some("Wingdings".into()), ..Default::default() };
195        let s = serde_json::to_string(&p).unwrap();
196        assert!(s.contains(r#""symFont":"Wingdings""#), "{s}");
197        assert_eq!(serde_json::from_str::<CharProps>(&s).unwrap(), p);
198        // A font change keeps the symbol's own font; Clear Formatting too.
199        let changed = p.clone().overlaid(&CharProps { font: Some("Times New Roman".into()), ..Default::default() });
200        assert_eq!(changed.sym_font.as_deref(), Some("Wingdings"));
201        assert_eq!(p.cleared().sym_font.as_deref(), Some("Wingdings"));
202    }
