25#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash, Serialize, Deserialize, Default)]
26#[serde(rename_all = "camelCase")]
27pub enum RevisionKind {
28    #[default]
29    Insert,
30    Delete,
31    Format,
32}
33
34#[derive(Clone, Debug, PartialEq, Serialize, Deserialize, Default)]
35#[serde(default)]
36pub struct Revision {
37    pub kind: RevisionKind,
38    pub author: String,
39    pub date: String,
40    /// A move (ECMA-376 §17.13.5.21–28): the name linking the source (an `Delete` revision,
41    /// `w:moveFrom`) with the destination (an `Insert` revision, `w:moveTo`). `None` = a plain
42    /// insertion or deletion.
43    #[serde(skip_serializing_if = "Option::is_none")]
44    pub move_name: Option<String>,
45}
46
47impl Revision {
48    pub fn new(kind: RevisionKind, author: impl Into<String>, date: impl Into<String>) -> Revision {
49        Revision { kind, author: author.into(), date: date.into(), move_name: None }
50    }
51}
52
53impl Document {
54    /// The move name of revision `idx`, when it is one end of a move.
55    pub fn move_name(&self, idx: Option<u32>) -> Option<&str> {
56        idx.and_then(|i| self.revisions.get(i as usize)).and_then(|r| r.move_name.as_deref())
57    }
58}

60/// A building block (Insert › Quick Parts, AutoText…) kept in the document's glossary
61/// (ECMA-376 §17.12: `w:docPart`).
62#[derive(Clone, Debug, PartialEq, Serialize, Deserialize, Default)]
63#[serde(default, rename_all = "camelCase")]
64pub struct BuildingBlock {
65    /// The entry's name (`w:docPartPr/w:name`).
66    pub name: String,
67    /// Its gallery (`w:category/w:gallery`: `docParts` for Quick Parts, `autoTxt` for AutoText…).
68    pub gallery: String,
69    /// Its category within the gallery (`w:category/w:name`, e.g. "General").
70    pub category: String,
71    #[serde(skip_serializing_if = "String::is_empty")]
72    pub description: String,
73    /// The entry's identity (`w:guid`), kept as read.
74    #[serde(skip_serializing_if = "String::is_empty")]
75    pub guid: String,
76    /// The paragraph style it's associated with (`w:style`), kept as read.
77    #[serde(skip_serializing_if = "String::is_empty")]
78    pub style: String,
79    /// Its types (`w:types/w:type`: `normal`, `bbPlcHdr` for a content control's placeholder…).
80    #[serde(skip_serializing_if = "Vec::is_empty")]
81    pub types: Vec<String>,
82    /// How it's inserted (`w:behaviors/w:behavior`: `content`, `p`, `pg`).
83    #[serde(skip_serializing_if = "Vec::is_empty")]
84    pub behaviors: Vec<String>,
85    /// The content (`w:docPartBody`).
86    pub blocks: Blocks,
87}
