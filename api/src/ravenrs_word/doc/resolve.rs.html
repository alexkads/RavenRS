79impl ResolvedChar {
80    fn from(c: &CharProps) -> ResolvedChar {
81        ResolvedChar {
82            // A symbol character is shown in its own font (`w:sym`), whatever the run's font.
83            font: c.shown_font().map(str::to_string).unwrap_or_else(|| crate::doc::styles::BODY_FONT.to_string()),
84            size: c.size.unwrap_or(crate::doc::props::DEFAULT_FONT_SIZE).clamp(1.0, 1638.0),
85            bold: c.bold.unwrap_or(false),
86            italic: c.italic.unwrap_or(false),
87            underline: c.underline.unwrap_or_default(),
88            underline_color: c.underline_color,
89            strike: c.strike.unwrap_or(false),
90            double_strike: c.double_strike.unwrap_or(false),
91            color: c.color.unwrap_or_default(),
92            highlight: c.highlight.and_then(Highlight::rgb),
93            shading: c.shading,
94            border: c.border.filter(Border::is_visible),
95            vert_align: c.vert_align.unwrap_or_default(),
96            caps: c.caps.unwrap_or(false),
97            small_caps: c.small_caps.unwrap_or(false),
98            hidden: c.hidden.unwrap_or(false),
99            spacing: c.spacing.unwrap_or(0.0).clamp(-100.0, 100.0),
100            scale: c.scale.unwrap_or(100.0).clamp(1.0, 600.0),
101            position: c.position.unwrap_or(0.0).clamp(-800.0, 800.0),
102            kern: c.kern.filter(|k| k.is_finite()).unwrap_or(0.0).clamp(0.0, 1638.0),
103            emphasis: c.emphasis.unwrap_or_default(),
104            ligatures: c.ligatures,
105            outline: c.outline.unwrap_or(false),
106            shadow: c.shadow.unwrap_or(false),
107            emboss: c.emboss.unwrap_or(false),
108            engrave: c.engrave.unwrap_or(false),
109            text_effects: c.text_effects.as_deref().map(crate::doc::wordart::TextEffects::sanitized).filter(|t| !t.is_empty()).map(Box::new),
110            link: c.link.clone(),
111            ins: c.ins,
112            del: c.del,
113            fmt: c.fmt_change.as_ref().map(|f| f.rev),
114            no_proof: c.no_proof.unwrap_or(false),
115            lang: c.lang.clone(),
116            rtl: c.rtl.unwrap_or(false),
117            cs: c.cs.unwrap_or(false),
118            font_cs: c.font_cs.clone().or_else(|| c.font.clone()).unwrap_or_else(|| crate::doc::styles::BODY_FONT.to_string()),
119            font_ea: c.font_ea.clone(),
120            size_cs: c.size_cs.or(c.size).unwrap_or(crate::doc::props::DEFAULT_FONT_SIZE).clamp(1.0, 1638.0),
121            bold_cs: c.bold_cs.or(c.bold).unwrap_or(false),
122            italic_cs: c.italic_cs.or(c.italic).unwrap_or(false),
123        }
124    }
125    /// The formatting complex-script text in this run is drawn with: the complex-script font,
126    /// size, bold and italic in place of the others (Word formats Arabic, Persian and Hebrew
127    /// characters, and every character of an `rtl`/`cs` run, this way).
128    pub fn complex(&self) -> ResolvedChar {
129        ResolvedChar { font: self.font_cs.clone(), size: self.size_cs, bold: self.bold_cs, italic: self.italic_cs, ..self.clone() }
130    }
131    /// Whether the text is kerned: Word kerns only a run with Kerning for fonts set (`w:kern`) at
132    /// a size of at least that many points, never by default (measured in Word for Windows).
133    pub fn kerned(&self) -> bool {
134        self.kern > 0.0 && self.size >= self.kern
135    }
136    /// Does a character of this run use the complex-script properties?
137    pub fn uses_complex(&self, c: char) -> bool {
138        self.rtl || self.cs || crate::doc::bidi::is_complex_script(c)
139    }
140    /// The size glyphs are drawn and measured at. Word sets superscript and subscript text at
141    /// 0.65 of the run's size rounded to the half point, a font size the format can store
142    /// (measured in Word for Mac from digit advances: 8 pt → 5, 9 → 6, 11 → 7, 12 → 8,
143    /// 16 → 10.5, 24 → 15.5, 36 → 23.5).
144    pub fn draw_size(&self) -> f32 {
145        if self.vert_align == VertAlign::Baseline { self.size } else { (self.size * 0.65 * 2.0).round() / 2.0 }
146    }
147    /// Baseline shift, points (positive = up).
148    pub fn baseline_shift(&self) -> f32 {
149        self.position
150            + match self.vert_align {
151                VertAlign::Baseline => 0.0,
152                VertAlign::Superscript => self.size * 0.33,
153                VertAlign::Subscript => -self.size * 0.14,
154            }
155    }
156}

231const MAX_INDENT: f32 = 1584.0; // 22"
232
233impl ResolvedPara {
234    /// Visual (left, right) indents. Indents, like alignment, are logical: `indent_left` is the
235    /// start (leading) edge, which is the right edge of a right-to-left paragraph (ISO 29500
236    /// `w:ind/@w:start`, read from `w:left` too).
237    pub fn visual_indents(&self) -> (f32, f32) {
238        if self.bidi { (self.indent_right, self.indent_left) } else { (self.indent_left, self.indent_right) }
239    }
240
241    /// Turn the indents given in characters into points, one character being `size` points: the
242    /// size of the paragraph mark's font (as Word measures `w:firstLineChars` and friends).
243    pub fn set_char_unit(&mut self, size: f32) {
244        let size = if size.is_finite() { size.clamp(1.0, 1638.0) } else { 12.0 };
245        let [l, r, f] = self.indent_chars;
246        for (chars, pts) in [(l, &mut self.indent_left), (r, &mut self.indent_right), (f, &mut self.indent_first)] {
247            if chars != 0.0 {
248                *pts = (chars * size).clamp(-MAX_INDENT, MAX_INDENT);
249            }
250        }
251    }
252}

254impl ResolvedPara {
255    fn from(p: &ParaProps, style: &str) -> ResolvedPara {
256        let clampi = |v: Option<f32>| v.unwrap_or(0.0).clamp(-MAX_INDENT, MAX_INDENT);
257        let mut tabs: Vec<TabStop> =
258            p.tabs.clone().unwrap_or_default().into_iter().filter(|t| t.align != TabAlign::Clear && t.pos.is_finite()).collect();
259        tabs.sort_by(|a, b| a.pos.total_cmp(&b.pos));
260        ResolvedPara {
261            style: style.to_string(),
262            align: p.align.unwrap_or_default(),
263            indent_left: clampi(p.indent_left),
264            indent_right: clampi(p.indent_right),
265            indent_first: clampi(p.indent_first),
266            space_before: if p.space_before_auto == Some(true) { AUTO_SPACING } else { p.space_before.unwrap_or(0.0).clamp(0.0, 1584.0) },
267            space_after: if p.space_after_auto == Some(true) { AUTO_SPACING } else { p.space_after.unwrap_or(0.0).clamp(0.0, 1584.0) },
268            before_auto: p.space_before_auto == Some(true),
269            after_auto: p.space_after_auto == Some(true),
270            line_spacing: match p.line_spacing.unwrap_or_default() {
271                LineSpacing::Multiple(m) => LineSpacing::Multiple(if m.is_finite() { m.clamp(0.06, 132.0) } else { 1.0 }),
272                LineSpacing::AtLeast(v) => LineSpacing::AtLeast(if v.is_finite() { v.clamp(0.0, 1584.0) } else { 0.0 }),
273                LineSpacing::Exactly(v) => LineSpacing::Exactly(if v.is_finite() { v.clamp(0.7, 1584.0) } else { 12.0 }),
274            },
275            contextual_spacing: p.contextual_spacing.unwrap_or(false),
276            mirror_indents: p.mirror_indents.unwrap_or(false),
277            adjust_right_indent: p.adjust_right_indent.unwrap_or(true),
278            keep_next: p.keep_next.unwrap_or(false),
279            keep_lines: p.keep_lines.unwrap_or(false),
280            page_break_before: p.page_break_before.unwrap_or(false),
281            widow_control: p.widow_control.unwrap_or(true),
282            outline_level: p.outline_level.filter(|l| *l < 9),
283            numbering: p.numbering.filter(|n| n.num != 0),
284            tabs,
285            shading: p.shading,
286            borders: p.borders.filter(Borders::any_visible),
287            padding: p.borders.map_or([0.0; 4], |b| {
288                [b.top, b.left, b.bottom, b.right]
289                    .map(|s| s.filter(|l| !l.is_visible() && l.space.is_finite() && l.space > 0.0).map_or(0.0, |l| l.space.min(1584.0)))
290            }),
291            bidi: p.bidi.unwrap_or(false),
292            drop_cap: p.drop_cap.unwrap_or(0).min(10),
293            drop_cap_margin: p.drop_cap_margin.unwrap_or(false),
294            drop_cap_distance: p.drop_cap_distance.filter(|d| d.is_finite()).unwrap_or(0.0).clamp(0.0, MAX_DROP_CAP_DISTANCE),
295            drop_cap_font: p.drop_cap_font.clone().filter(|f| !f.trim().is_empty()),
296            // Explicit zero dimensions cancel a frame; omitted dimensions still fit its text.
297            frame: p.frame.clone().filter(|f| p.drop_cap.unwrap_or(0) == 0 && (f.w != Some(0.0) || f.h != Some(0.0))).map(|mut f| {
298                f.w = f.w.filter(|v| *v != 0.0);
299                f.h = f.h.filter(|v| *v != 0.0);
300                f
301            }),
302            suppress_hyphens: p.suppress_hyphens.unwrap_or(false),
303            suppress_line_numbers: p.suppress_line_numbers.unwrap_or(false),
304            kinsoku: p.kinsoku.unwrap_or(true),
305            word_wrap: p.word_wrap.unwrap_or(true),
306            overflow_punct: p.overflow_punct.unwrap_or(true),
307            top_line_punct: p.top_line_punct.unwrap_or(false),
308            auto_space_de: p.auto_space_de.unwrap_or(true),
309            auto_space_dn: p.auto_space_dn.unwrap_or(true),
310            snap_to_grid: p.snap_to_grid.unwrap_or(true),
311            indent_chars: [p.indent_left_chars, p.indent_right_chars, p.indent_first_chars]
312                .map(|c| c.filter(|c| c.is_finite()).unwrap_or(0.0).clamp(-MAX_CHARS, MAX_CHARS)),
313        }
314    }
315}

320impl StyleSheet {
321    /// Paragraph properties after defaults and the style chain (without direct formatting).
322    pub fn para_style_props(&self, style: &str) -> (ParaProps, CharProps) {
323        self.para_style_props_over(style, self.default_para.clone(), self.default_chr.clone())
324    }
325
326    /// The paragraph style chain of `style` applied over `pp` / `cp`. A style the document doesn't
327    /// define is the default paragraph style, Normal, as in Word: a footer naming a `Footer` style
328    /// that `styles.xml` leaves out is formatted as Normal, not with the document defaults alone
329    /// (#5077).
330    fn para_style_props_over(&self, style: &str, mut pp: ParaProps, mut cp: CharProps) -> (ParaProps, CharProps) {
331        let style = if self.get(style).is_some() { style } else { "Normal" };
332        for s in self.chain(style) {
333            if s.kind == StyleKind::Paragraph {
334                let mut sp = s.para.clone();
335                // Tabs accumulate along the chain (with Clear removing inherited stops).
336                if let (Some(base), Some(add)) = (pp.tabs.clone(), sp.tabs.clone()) {
337                    let mut merged: Vec<TabStop> = base.into_iter().filter(|t| !add.iter().any(|a| (a.pos - t.pos).abs() < 0.5)).collect();
338                    merged.extend(add);
339                    sp.tabs = Some(merged);
340                }
341                pp.overlay(&sp);
342                cp.overlay(&s.chr);
343            }
344        }
345        pp.style = None;
346        (pp, cp)
347    }
348
349    /// What a table style's paragraph formatting `table` changes for a paragraph with `p`, once
350    /// the paragraph style chain is applied over it (ECMA-376 §17.7.2): each property it gives
351    /// that the chain doesn't set. Formats without table styles state these directly.
352    pub fn table_para_patch(&self, p: &ParaProps, table: &ParaProps) -> ParaProps {
353        let style = p.style.clone().unwrap_or_else(|| "Normal".into());
354        let mut with = self.default_para.clone();
355        with.overlay(&ParaProps { style: None, ..table.clone() });
356        let (with, _) = self.para_style_props_over(&style, with, CharProps::default());
357        let (without, _) = self.para_style_props_over(&style, self.default_para.clone(), CharProps::default());
358        diff_patch(&with, &without)
359    }
360
361    /// What a table style's run formatting `table` changes for a run of character style
362    /// `run_style` in a paragraph of style `para_style`, once the paragraph and character style
363    /// chains are applied over it (ECMA-376 §17.7.2).
364    pub fn table_char_patch(&self, para_style: Option<&str>, run_style: Option<&str>, table: &CharProps) -> CharProps {
365        let chain = |defaults: CharProps| {
366            let (_, mut cp) = self.para_style_props_over(para_style.unwrap_or("Normal"), ParaProps::default(), defaults);
367            if let Some(cs) = run_style {
368                for s in self.chain(cs) {
369                    if s.kind == StyleKind::Character || s.kind == StyleKind::Paragraph {
370                        cp.overlay(&s.chr);
371                    }
372                }
373            }
374            cp
375        };
376        let mut with = self.default_chr.clone();
377        with.overlay(&CharProps { style: None, ..table.clone() });
378        diff_patch(&chain(with), &chain(self.default_chr.clone()))
379    }
380
381    /// Resolve a paragraph's properties.
382    pub fn resolve_para(&self, p: &ParaProps) -> ResolvedPara {
383        self.resolve_para_in(p, None)
384    }
385
386    /// Resolve a paragraph's properties; `table` is the paragraph formatting of the table style of
387    /// the table the paragraph is in. The order is that of ECMA-376 §17.7.2: document defaults, the
388    /// table style, the paragraph style chain, direct formatting. So the table style replaces
389    /// document defaults (Table Grid's single spacing beats the defaults' 8 pt after), but any
390    /// property the paragraph style chain sets, Normal included, beats the table style.
391    pub fn resolve_para_in(&self, p: &ParaProps, table: Option<&ParaProps>) -> ResolvedPara {
392        let style = p.style.clone().unwrap_or_else(|| "Normal".into());
393        let mut defaults = self.default_para.clone();
394        if let Some(t) = table {
395            defaults.overlay(&ParaProps { style: None, ..t.clone() });
396        }
397        let (mut base, _) = self.para_style_props_over(&style, defaults, CharProps::default());
398        let mut direct = p.clone();
399        if let (Some(b), Some(d)) = (base.tabs.clone(), direct.tabs.clone()) {
400            let mut merged: Vec<TabStop> = b.into_iter().filter(|t| !d.iter().any(|a| (a.pos - t.pos).abs() < 0.5)).collect();
401            merged.extend(d);
402            direct.tabs = Some(merged);
403        }
404        base.overlay(&direct);
405        let mut rp = ResolvedPara::from(&base, &style);
406        // Indents in characters: measured in the style's font size here; layout measures them in
407        // the paragraph mark's (`set_char_unit`), which direct formatting may change.
408        if rp.indent_chars.iter().any(|c| *c != 0.0) {
409            rp.set_char_unit(self.resolve_char(Some(&style), &CharProps::default()).size);
410        }
411        rp
412    }
413
414    /// Resolve character formatting for a run in a paragraph of style `para_style`.
415    pub fn resolve_char(&self, para_style: Option<&str>, c: &CharProps) -> ResolvedChar {
416        self.resolve_char_in(para_style, None, c)
417    }
418
419    /// Resolve character formatting for a run in a paragraph of style `para_style`; `table` is the
420    /// run formatting of the table style (with the cell's conditional formatting) of the table the
421    /// paragraph is in. Order: document defaults, the table style, the paragraph style chain, the
422    /// character style chain, direct formatting (ECMA-376 §17.7.2).
423    pub fn resolve_char_in(&self, para_style: Option<&str>, table: Option<&CharProps>, c: &CharProps) -> ResolvedChar {
424        let mut defaults = self.default_chr.clone();
425        if let Some(t) = table {
426            defaults.overlay(&CharProps { style: None, ..t.clone() });
427        }
428        let (_, mut cp) = self.para_style_props_over(para_style.unwrap_or("Normal"), ParaProps::default(), defaults);
429        if let Some(cs) = c.style.as_deref() {
430            for s in self.chain(cs) {
431                if s.kind == StyleKind::Character || s.kind == StyleKind::Paragraph {
432                    cp.overlay(&s.chr);
433                }
434            }
435        }
436        // Only a character itself has a symbol font of its own (`w:sym`); a style never gives one.
437        cp.sym_font = None;
438        let mut direct = c.clone();
439        direct.style = None;
440        cp.overlay(&direct);
441        ResolvedChar::from(&cp)
442    }
443}

445/// The properties of `with` that differ from `without` (both fully resolved property sets of one
446/// kind), as a set of only those; empty when serialisation fails.
447fn diff_patch<T: Serialize + serde::de::DeserializeOwned + Default>(with: &T, without: &T) -> T {
448    let (Ok(serde_json::Value::Object(w)), Ok(serde_json::Value::Object(o))) = (serde_json::to_value(with), serde_json::to_value(without)) else {
449        return T::default();
450    };
451    let patch: serde_json::Map<String, serde_json::Value> = w.into_iter().filter(|(k, v)| o.get(k) != Some(v)).collect();
452    serde_json::from_value(serde_json::Value::Object(patch)).unwrap_or_default()
453}

561    /// ECMA-376 §17.7.2 for text in a table: document defaults, then the table style, then the
562    /// paragraph style chain (Normal included), then direct formatting.
563    #[test]
564    fn table_style_sits_between_defaults_and_paragraph_style() {
565        let mut s = StyleSheet::builtin();
566        s.styles.push(crate::doc::styles::Style {
567            id: "Spaced".into(),
568            based_on: Some("Normal".into()),
569            para: ParaProps { space_after: Some(12.0), ..Default::default() },
570            chr: CharProps { color: Some(TextColor::Rgb(Rgb(0, 0, 0xFF))), ..Default::default() },
571            ..Default::default()
572        });
573        let table_p = ParaProps { space_after: Some(0.0), line_spacing: Some(LineSpacing::Multiple(1.0)), ..Default::default() };
574        let red = TextColor::Rgb(Rgb(0xC0, 0, 0));
575        let table_c = CharProps { bold: Some(true), color: Some(red), size: Some(9.0), ..Default::default() };
576        // Document defaults alone (outside a table).
577        let out = s.resolve_para(&ParaProps::default());
578        assert_eq!((out.space_after, out.line_spacing), (8.0, LineSpacing::Multiple(1.15)));
579        // The table style replaces the defaults.
580        let p = s.resolve_para_in(&ParaProps::default(), Some(&table_p));
581        assert_eq!((p.space_after, p.line_spacing), (0.0, LineSpacing::Multiple(1.0)));
582        let c = s.resolve_char_in(None, Some(&table_c), &CharProps::default());
583        assert!(c.bold);
584        assert_eq!((c.color, c.size), (red, 9.0));
585        // A paragraph style beats it, for what the style sets only.
586        let spaced = ParaProps { style: Some("Spaced".into()), ..Default::default() };
587        let p = s.resolve_para_in(&spaced, Some(&table_p));
588        assert_eq!((p.space_after, p.line_spacing), (12.0, LineSpacing::Multiple(1.0)));
589        let c = s.resolve_char_in(Some("Spaced"), Some(&table_c), &CharProps::default());
590        assert!(c.bold);
591        assert_eq!((c.color, c.size), (TextColor::Rgb(Rgb(0, 0, 0xFF)), 9.0));
592        // A character style and direct formatting beat both.
593        let c =
594            s.resolve_char_in(Some("Spaced"), Some(&table_c), &CharProps { style: Some("Emphasis".into()), bold: Some(false), ..Default::default() });
595        assert!(!c.bold && c.italic);
596        let p = s.resolve_para_in(&ParaProps { space_after: Some(3.0), ..spaced }, Some(&table_p));
597        assert_eq!(p.space_after, 3.0);
598        // Normal setting a property itself beats the table style too (Word draws such rows tall).
599        if let Some(n) = s.get_mut("Normal") {
600            n.para.space_after = Some(10.0);
601            n.chr.size = Some(11.0);
602        }
603        assert_eq!(s.resolve_para_in(&ParaProps::default(), Some(&table_p)).space_after, 10.0);
604        assert_eq!(s.resolve_char_in(None, Some(&table_c), &CharProps::default()).size, 11.0);
605    }
