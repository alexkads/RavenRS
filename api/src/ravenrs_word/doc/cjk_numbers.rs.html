1//! East Asian list and page number formats (ECMA-376 §17.18.59 ST_NumberFormat): Chinese,
2//! Japanese and Korean counting systems (with units 十/百/千 and myriads 万/億), legal (financial)
3//! numerals, digit-by-digit ideographs, the heavenly stems and earthly branches, kana and hangul
4//! sequences and enclosed numbers. Written from the number systems' rules; a value outside a
5//! sequence's range falls back to plain decimal.
6
7/// Whether a counting system writes the "one" before a unit.
8#[derive(Clone, Copy, PartialEq, Eq)]
9enum OmitOne {
10    /// Always written: legal numerals (壹拾, 壱百).
11    Never,
12    /// Dropped only before a leading ten: 十, 十一, 十万 but 一百一十 (Chinese).
13    LeadingTen,
14    /// Dropped before 十, 百 and 千 inside every group of four: 十, 百, 千 but 一万 (Japanese,
15    /// Korean).
16    Units,
17}
18
19/// A myriad-based counting system (groups of four digits, like 1|2345|6789).
20struct Counting {
21    /// The digits 1 to 9 (index 0 unused).
22    digits: [char; 10],
23    /// The units ten, hundred, thousand.
24    units: [char; 3],
25    /// The myriads: ten thousand, hundred million.
26    myriads: [char; 2],
27    /// The character written once for a run of zeros between two digits (Chinese 零); `None`
28    /// skips zeros (Japanese, Korean).
29    zero: Option<char>,
30    omit_one: OmitOne,
31}
32
33impl Counting {
34    fn format(&self, n: u32) -> String {
35        if n == 0 {
36            return self.zero.unwrap_or('〇').to_string();
37        }
38        let groups = [n / 100_000_000, (n / 10_000) % 10_000, n % 10_000];
39        let mut out = String::new();
40        let mut zero_pending = false;
41        for (gi, &g) in groups.iter().enumerate() {
42            if g == 0 {
43                zero_pending |= !out.is_empty();
44                continue;
45            }
46            let ds = [g / 1000, (g / 100) % 10, (g / 10) % 10, g % 10];
47            for (pos, &d) in ds.iter().enumerate() {
48                if d == 0 {
49                    zero_pending |= !out.is_empty();
50                    continue;
51                }
52                if zero_pending && let Some(z) = self.zero {
53                    out.push(z);
54                }
55                zero_pending = false;
56                // pos 0 = thousands, 1 = hundreds, 2 = tens, 3 = ones.
57                let unit = match pos {
58                    0 => self.units.get(2),
59                    1 => self.units.get(1),
60                    2 => self.units.first(),
61                    _ => None,
62                };
63                let omit = d == 1
64                    && unit.is_some()
65                    && match self.omit_one {
66                        OmitOne::Never => false,
67                        OmitOne::LeadingTen => pos == 2 && out.is_empty(),
68                        OmitOne::Units => true,
69                    };
70                if !omit {
71                    out.push(self.digits.get(d as usize).copied().unwrap_or('?'));
72                }
73                if let Some(&u) = unit {
74                    out.push(u);
75                }
76            }
77            // Trailing zeros of a group are not a gap: the next group starts a new place.
78            zero_pending = false;
79            match gi {
80                0 => out.extend(self.myriads.get(1)),
81                1 => out.extend(self.myriads.first()),
82                _ => {}
83            }
84        }
85        out
86    }
87}

166/// Japanese digits with 万/億 marking each group of four: 一二, 一万二三四五.
167pub(crate) fn japanese_digital_ten_thousand(n: u32) -> String {
168    let groups = [(n / 100_000_000, Some('億')), ((n / 10_000) % 10_000, Some('万')), (n % 10_000, None)];
169    let mut out = String::new();
170    for (g, mark) in groups {
171        let s = digits(g, &CN_DIGITS);
172        if out.is_empty() {
173            if g == 0 && mark.is_some() {
174                continue;
175            }
176        } else {
177            // Inner groups keep their four places: 一万〇〇〇一.
178            out.extend(std::iter::repeat_n('〇', 4usize.saturating_sub(s.chars().count())));
179        }
180        out.push_str(&s);
181        out.extend(mark);
182    }
183    out
184}

199/// Native Korean numbers: 하나, 둘, 셋 … 열, 열하나 … 아흔아홉 (decimal from 100 on).
200pub(crate) fn korean_legal(n: u32) -> String {
201    if n == 0 || n > 99 {
202        return n.to_string();
203    }
204    let (t, o) = ((n / 10) as usize, (n % 10) as usize);
205    format!("{}{}", KO_TENS.get(t).copied().unwrap_or(""), KO_ONES.get(o).copied().unwrap_or(""))
206}

213fn digits(n: u32, set: &[char; 10]) -> String {
214    n.to_string().bytes().map(|b| set.get(b.saturating_sub(b'0') as usize).copied().unwrap_or('?')).collect()
215}

230/// The sixty-year cycle: 甲子, 乙丑, 丙寅 … 癸亥.
231pub(crate) fn ideograph_zodiac_traditional(n: u32) -> String {
232    if !(1..=60).contains(&n) {
233        return n.to_string();
234    }
235    let i = (n - 1) as usize;
236    [STEMS.get(i % 10), BRANCHES.get(i % 12)].into_iter().flatten().collect()
237}

239/// The `n`th (1-based) item of a sequence, or decimal past its end.
240fn nth(set: &[char], n: u32) -> String {
241    n.checked_sub(1).and_then(|i| set.get(i as usize)).map(ToString::to_string).unwrap_or_else(|| n.to_string())
242}
243
244/// Code points `first..` enclosing 1 to `count`, decimal beyond.
245fn enclosed(n: u32, first: u32, count: u32) -> String {
246    if (1..=count).contains(&n)
247        && let Some(c) = char::from_u32(first + n - 1)
248    {
249        return c.to_string();
250    }
251    n.to_string()
252}

330    fn chinese_no_zero_after_group_trailing_zeros() {
331        assert_eq!(chinese_counting_thousand(201_000), "二十万一千");
332        assert_eq!(chinese_counting_thousand(120_000), "十二万");
333        assert_eq!(chinese_counting_thousand(100_100), "十万零一百");
334        assert_eq!(taiwanese_counting_thousand(201_000), "二十萬一千");
335        assert_eq!(chinese_counting_thousand(100_000_001), "一亿零一");
336        assert_eq!(chinese_counting_thousand(100_010_000), "一亿零一万");
337        assert_eq!(chinese_counting_thousand(120_000_001), "一亿二千万零一");

340    #[test]
341    fn chinese_counting_systems() {
342        let c = |n| chinese_counting(n);
343        assert_eq!([c(1), c(2), c(3), c(10), c(11), c(20), c(21), c(99)], ["一", "二", "三", "十", "十一", "二十", "二十一", "九十九"]);
344        assert_eq!([c(100), c(101), c(1000)], ["一〇〇", "一〇一", "一〇〇〇"]);
345        let t = |n| chinese_counting_thousand(n);
346        assert_eq!(
347            [t(10), t(11), t(20), t(101), t(110), t(111), t(1000), t(1010)],
348            ["十", "十一", "二十", "一百零一", "一百一十", "一百一十一", "一千", "一千零一十"]
349        );
350        assert_eq!([t(10_000), t(10_001), t(100_100), t(1_000_000), t(100_000_001)], ["一万", "一万零一", "十万零一百", "一百万", "一亿零一"]);
351        assert_eq!(t(u32::MAX), "四十二亿九千四百九十六万七千二百九十五");
352        let l = |n| chinese_legal_simplified(n);
353        assert_eq!([l(1), l(3), l(10), l(11), l(101), l(1000)], ["壹", "叁", "壹拾", "壹拾壹", "壹佰零壹", "壹仟"]);
354        assert_eq!([taiwanese_counting_thousand(10_001), ideograph_legal_traditional(20)], ["一萬零一", "貳拾"]);
355        assert_eq!([ideograph_digital(10), ideograph_digital(101)], ["一〇", "一〇一"]);
356    }
