5/// An inline object anchored at a U+FFFC in the paragraph text.
6#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
7#[serde(tag = "type", rename_all = "camelCase")]
8pub enum InlineObject {
9    Image {
10        /// Key in `Document::media`.
11        media: String,
12        /// Display size, points.
13        w: f32,
14        h: f32,
15        #[serde(default)]
16        alt: String,
17        #[serde(default)]
18        float: Float,

22        /// Brightness, contrast and transparency read from the file (see [`crate::doc::picfx`]).
23        #[serde(default, skip_serializing_if = "crate::doc::picfx::PicFx::is_identity")]
24        fx: crate::doc::picfx::PicFx,
25        /// The OLE object (embedded or linked file) or Word signature line this picture shows, as
26        /// read from a file: saving writes the object back, not only its picture. See
27        /// [`crate::doc::graphic::Embedded`].
28        #[serde(default, skip_serializing_if = "Option::is_none")]
29        ole: Option<std::sync::Arc<crate::doc::graphic::Embedded>>,

31    /// A chart or SmartArt diagram (see [`crate::doc::graphic`]).
32    Graphic {
33        w: f32,
34        h: f32,
35        #[serde(default)]
36        alt: String,
37        #[serde(default)]

41    Shape {
42        kind: ShapeKind,
43        w: f32,
44        h: f32,
45        fill: Option<crate::doc::fill::Fill>,
46        stroke: Option<crate::doc::props::Rgb>,
47        #[serde(default)]
48        stroke_width: f32,

50        float: Float,
51        /// Text box content: `Document::parts` id.
52        #[serde(default)]
53        story: Option<u32>,
54        /// A freeform's geometry (`ShapeKind::Freeform`): its paths, or an ink stroke.
55        #[serde(default, skip_serializing_if = "Option::is_none")]
56        freeform: Option<std::sync::Arc<crate::doc::freeform::Freeform>>,
57        /// Shadow, glow and soft edges.

59        effects: crate::doc::effects::ShapeEffects,
60        /// Drawing id, text warp, connector ends and arrowheads.
61        #[serde(default, skip_serializing_if = "crate::doc::connector::ShapeExtra::is_empty")]
62        extra: crate::doc::connector::ShapeExtra,
63    },
64    /// Pictures, shapes and text boxes grouped into one object (Layout › Arrange › Group): it
65    /// moves, wraps and resizes as one. Its members are laid out in the group's own coordinate
66    /// space, `ch_w` × `ch_h`, which is stretched over the group's `w` × `h` (DrawingML's
67    /// `a:chExt` and `a:ext`), so resizing the group scales them.
68    Group {
69        w: f32,
70        h: f32,
71        #[serde(default)]
72        float: Float,
73        /// Size of the members' coordinate space (their offsets and sizes are in it).
74        ch_w: f32,
75        ch_h: f32,
76        children: Vec<GroupChild>,
77        /// A Drawing Canvas (Insert › Shapes › New Drawing Canvas, DrawingML's `wpc:wpc`) rather
78        /// than a group: it has its own background and outline, and its frame is independent of
79        /// its members: resizing it doesn't scale them (`ch_w` × `ch_h` follow `w` × `h`), and
80        /// what lies outside the frame is clipped.
81        #[serde(default, skip_serializing_if = "Option::is_none")]
82        canvas: Option<CanvasStyle>,
83    },
84    /// A field: `instr` is the field code (`PAGE`, `NUMPAGES`, `DATE \@ "M/d/yyyy"`, `TOC \o "1-3"`…);
85    /// `result` the cached display text. A `locked` field keeps its result when fields update.
86    Field {
87        instr: String,
88        #[serde(default)]
89        result: String,
90        #[serde(default)]
91        locked: bool,
92        /// `w:dirty` (ECMA-376 §17.16.5.35): the cached result is stale — another program that
93        /// couldn't compute it set the flag so Word refreshes the field on open. Kept through
94        /// saves; cleared when WordCraft computes a result, like Word's F9.
95        #[serde(default, skip_serializing_if = "std::ops::Not::not")]
96        dirty: bool,
97        /// Shows its code (`{ PAGE }`) instead of its result, the other way round from the
98        /// document-wide View › Field Codes setting (`fields.toggleCode`, Shift+F9). A view
99        /// toggle: never written to files.
100        #[serde(default, skip_serializing_if = "std::ops::Not::not")]
101        code: bool,
102        /// A legacy form field's definition (`FORMCHECKBOX`, `FORMTEXT`, `FORMDROPDOWN`): the
103        /// source's `w:ffData`, raw OOXML kept for round trip. Empty for other fields.
104        #[serde(default, skip_serializing_if = "String::is_empty")]
105        form: String,
106    },
107    /// The start of a field whose result is ordinary content — formatted, possibly spanning
108    /// paragraphs — up to the matching [`InlineObject::FieldEnd`]. Imported TOCs and citation
109    /// managers' `ADDIN` fields are kept this way; `instr` is the field code. See
110    /// [`crate::doc::fields`].
111    FieldStart {
112        instr: String,
113        #[serde(default)]
114        locked: bool,
115        /// `w:dirty` on the begin marker, as [`InlineObject::Field::dirty`].
116        #[serde(default, skip_serializing_if = "std::ops::Not::not")]
117        dirty: bool,
118    },
119    /// The end of the innermost open [`InlineObject::FieldStart`].
120    FieldEnd,
121    /// The start of a content control whose content is what follows, up to the matching
122    /// [`InlineObject::ControlEnd`]. See [`crate::doc::control`].
123    ControlStart {
124        control: Box<crate::doc::control::ContentControl>,
125    },
126    /// The end of the innermost open [`InlineObject::ControlStart`].
127    ControlEnd,
128    NoteRef {
129        kind: NoteKind,
130        /// `Document::parts` id of the note's story.
131        id: u32,
132        /// Custom mark (empty = automatic number).
133        #[serde(default)]
134        custom: String,
135    },
136    BookmarkStart {
137        name: String,
138    },
139    BookmarkEnd {
140        name: String,
141    },
142    CommentStart {
143        id: u32,
144    },
145    CommentEnd {
146        id: u32,
147    },
148    /// The start of a range that stays editable while the document is protected (Restrict
149    /// Editing exceptions, `w:permStart`): for everyone (`group` = `everyone`), another editing
150    /// group, or one editor (`editor`, a user name or address).
151    PermStart {
152        id: u32,
153        #[serde(default, skip_serializing_if = "String::is_empty")]
154        group: String,
155        #[serde(default, skip_serializing_if = "String::is_empty")]
156        editor: String,
157    },
158    /// The end of the editable range [`InlineObject::PermStart`] with the same `id` opened.
159    PermEnd {
160        id: u32,
161    },
162    /// An equation: its linear format (`x=(-b±√(b^2-4ac))/2a`, for plain text) and structure.
163    /// A display equation sits on a line of its own.
164    Equation {
165        linear: String,
166        #[serde(default)]
167        display: bool,
168        /// The structure; empty = parse `linear`.
169        #[serde(default, skip_serializing_if = "crate::doc::math::Math::is_empty")]
170        math: crate::doc::math::Math,
171    },
172    /// Ruby (Phonetic Guide): small `ruby` text (furigana, pinyin…) set over `base` text. The base
173    /// text takes the formatting of the object's run; it reads as the paragraph's text.
174    Ruby {
175        base: String,
176        ruby: String,
177        #[serde(default)]
178        align: RubyAlign,
179        /// Ruby text size, points (`w:hps`).
180        size: f32,
181        /// From the base text's baseline up to the ruby text's baseline, points (`w:hpsRaise`).
182        raise: f32,
183        /// The base text size the ruby was set for, points (`w:hpsBaseText`).
184        #[serde(default)]
185        base_size: f32,
186        /// Language of the ruby (`w:lid`, e.g. `ja-JP`); empty = unspecified.
187        #[serde(default, skip_serializing_if = "String::is_empty")]
188        lang: String,
189        /// Formatting of the ruby text over the run's (font, colour…); its size is `size`.
190        #[serde(default, skip_serializing_if = "CharProps::is_empty")]
191        props: Box<CharProps>,
192    },
193    /// A positional tab (Word's alignment tab, `w:ptab`): it goes to the left edge, centre or
194    /// right edge of the margins or of the indents, whatever the paragraph's tab stops, and the

196    PositionalTab {
197        #[serde(default)]

199        #[serde(default)]

207        xml: String,

236impl InlineObject {
237    /// A picture, chart or diagram, shape, text box or group: something drawn with a frame, sized and wrapped.
238    pub fn is_drawing(&self) -> bool {
239        matches!(self, InlineObject::Image { .. } | InlineObject::Graphic { .. } | InlineObject::Shape { .. } | InlineObject::Group { .. })
240    }
241    /// A drawing's size and placement.
242    pub fn frame(&self) -> Option<(f32, f32, &Float)> {
243        match self {
244            InlineObject::Image { w, h, float, .. }
245            | InlineObject::Graphic { w, h, float, .. }
246            | InlineObject::Shape { w, h, float, .. }
247            | InlineObject::Group { w, h, float, .. } => Some((*w, *h, float)),
248            _ => None,
249        }
250    }
251    /// A drawing's placement, to change.
252    pub fn float_mut(&mut self) -> Option<&mut Float> {
253        match self {
254            InlineObject::Image { float, .. }
255            | InlineObject::Graphic { float, .. }
256            | InlineObject::Shape { float, .. }
257            | InlineObject::Group { float, .. } => Some(float),
258            _ => None,
259        }
260    }
261    /// Resize a drawing (a group's members scale with it; a canvas's keep their size and place).
262    pub fn set_size(&mut self, nw: f32, nh: f32) {
263        if let InlineObject::Image { w, h, .. }
264        | InlineObject::Graphic { w, h, .. }
265        | InlineObject::Shape { w, h, .. }
266        | InlineObject::Group { w, h, .. } = self
267        {
268            *w = nw;
269            *h = nh;
270        }
271        if let InlineObject::Group { ch_w, ch_h, canvas: Some(_), .. } = self {
272            *ch_w = nw;
273            *ch_h = nh;
274        }
275    }
276    /// A Drawing Canvas's background and outline (`None` for anything else).
277    pub fn canvas_style(&self) -> Option<&CanvasStyle> {
278        match self {
279            InlineObject::Group { canvas, .. } => canvas.as_ref(),
280            _ => None,
281        }
282    }
283    /// Whether this is a Drawing Canvas.
284    pub fn is_canvas(&self) -> bool {
285        self.canvas_style().is_some()
286    }
287    /// The text box story this object shows, if it is a text box.
288    pub fn text_box(&self) -> Option<u32> {
289        match self {
290            InlineObject::Shape { story, .. } => *story,
291            _ => None,
292        }
293    }
294    /// The text box stories this object shows: a text box's, or its group members'.
295    pub fn text_boxes(&self) -> Vec<u32> {
296        match self {
297            InlineObject::Shape { story: Some(id), .. } => vec![*id],
298            InlineObject::Group { children, .. } => children.iter().filter_map(|c| c.obj.text_box()).collect(),
299            _ => Vec::new(),
300        }
301    }
302    /// The text box story slots of this object and its group members, to repoint.
303    pub fn text_box_slots(&mut self) -> Vec<&mut Option<u32>> {
304        match self {
305            InlineObject::Shape { story, .. } => vec![story],
306            InlineObject::Group { children, .. } => children
307                .iter_mut()
308                .filter_map(|c| match &mut c.obj {
309                    InlineObject::Shape { story, .. } => Some(story),
310                    _ => None,
311                })
312                .collect(),
313            _ => Vec::new(),
314        }
315    }
316    /// A group's members placed in a `w` × `h` box at (`x`, `y`): each member's rectangle
317    /// (x, y, w, h) and the member. Empty for anything else.
318    pub fn group_rects(&self, x: f32, y: f32, w: f32, h: f32) -> Vec<([f32; 4], &InlineObject)> {
319        let InlineObject::Group { children, .. } = self else { return Vec::new() };
320        self.member_rects(x, y, w, h).into_iter().filter_map(|(i, r)| Some((r, &children.get(i)?.obj))).collect()
321    }
322    /// [`InlineObject::group_rects`] with each member's index in `children` instead of the member.
323    pub fn member_rects(&self, x: f32, y: f32, w: f32, h: f32) -> Vec<(usize, [f32; 4])> {
324        let InlineObject::Group { children, .. } = self else { return Vec::new() };
325        let (sx, sy) = self.group_scale(w, h);
326        let fin = |v: f32| if v.is_finite() { v } else { 0.0 };
327        children
328            .iter()
329            .take(MAX_GROUP_CHILDREN)
330            .enumerate()
331            .filter_map(|(i, c)| {
332                let (cw, chh, _) = c.obj.frame()?;
333                let r = [x + fin(c.x) * sx, y + fin(c.y) * sy, (fin(cw) * sx).clamp(0.0, 4000.0), (fin(chh) * sy).clamp(0.0, 4000.0)];
334                (!matches!(c.obj, InlineObject::Group { .. })).then_some((i, r))
335            })
336            .collect()
337    }
338    /// How much a group drawn `w` × `h` scales its members' space (x, y): 1 for a canvas, whose
339    /// members keep their size.
340    pub fn group_scale(&self, w: f32, h: f32) -> (f32, f32) {
341        let InlineObject::Group { ch_w, ch_h, .. } = self else { return (1.0, 1.0) };
342        let fin = |v: f32| if v.is_finite() { v } else { 0.0 };
343        let sx = if fin(*ch_w) > 0.0 { fin(w) / ch_w } else { 1.0 };
344        let sy = if fin(*ch_h) > 0.0 { fin(h) / ch_h } else { 1.0 };
345        (sx, sy)
346    }
347    /// A legacy form field's definition (`FORMCHECKBOX`, `FORMTEXT`, `FORMDROPDOWN`), read from
348    /// the `w:ffData` it keeps; `None` for anything else.
349    pub fn legacy_form(&self) -> Option<crate::doc::legacy_form::LegacyForm> {
350        match self {
351            InlineObject::Field { form, .. } if !form.is_empty() => crate::doc::legacy_form::LegacyForm::parse(form),
352            _ => None,
353        }
354    }
355    /// Zero-width markers don't take part in layout.
356    pub fn is_marker(&self) -> bool {
357        matches!(
358            self,
359            InlineObject::BookmarkStart { .. }
360                | InlineObject::BookmarkEnd { .. }
361                | InlineObject::CommentStart { .. }
362                | InlineObject::CommentEnd { .. }
363                | InlineObject::PermStart { .. }
364                | InlineObject::PermEnd { .. }
365                | InlineObject::FieldStart { .. }
366                | InlineObject::FieldEnd
367                | InlineObject::ControlStart { .. }
368                | InlineObject::ControlEnd
369        )
370    }
371    /// A drawing's description (alternative text), wherever it is kept.
372    pub fn alt_text(&self) -> &str {
373        match self {
374            InlineObject::Image { alt, .. } | InlineObject::Graphic { alt, .. } => alt,
375            InlineObject::Shape { float, .. } | InlineObject::Group { float, .. } => &float.alt.text,
376            _ => "",
377        }
378    }
379    /// Set a drawing's description (capped at [`MAX_ALT_TEXT`](super::MAX_ALT_TEXT) characters); `false` if this is no drawing.
380    pub fn set_alt_text(&mut self, text: &str) -> bool {
381        let text = cap_alt(text);
382        match self {
383            InlineObject::Image { alt, .. } | InlineObject::Graphic { alt, .. } => *alt = text,
384            InlineObject::Shape { float, .. } | InlineObject::Group { float, .. } => float.alt.text = text,
385            _ => return false,
386        }
387        true
388    }
389    /// A drawing marked decorative.
390    pub fn is_decorative(&self) -> bool {
391        self.frame().is_some_and(|(_, _, f)| f.alt.decorative)
392    }
393    pub fn is_floating(&self) -> bool {
394        match self {
395            InlineObject::Image { float, .. }
396            | InlineObject::Graphic { float, .. }
397            | InlineObject::Shape { float, .. }
398            | InlineObject::Group { float, .. } => float.wrap != Wrap::Inline,
399            _ => false,
400        }
401    }
402    /// The geometry of an ink stroke (a freeform shape drawn with a pen), if this is one.
403    pub fn ink(&self) -> Option<&crate::doc::freeform::Freeform> {
404        match self {
405            InlineObject::Shape { freeform: Some(f), .. } if f.is_ink() => Some(f),
406            _ => None,
407        }
408    }
409    /// The line a footnote or endnote separator draws (ECMA-376 §17.11: the `w:separator` and
410    /// `w:continuationSeparator` run elements): a short line, or with `continuation` one across
411    /// the column. Kept as the run element itself.
412    pub fn note_separator(continuation: bool) -> InlineObject {
413        let el = if continuation { "w:continuationSeparator" } else { "w:separator" };
414        InlineObject::Opaque { format: "docx".into(), xml: format!("<w:r><{el}/></w:r>"), text: String::new() }
415    }
416    /// For a [`InlineObject::note_separator`]: whether it is the continuation separator.
417    pub fn as_note_separator(&self) -> Option<bool> {
418        match self {
419            InlineObject::Opaque { format, xml, .. } if format == "docx" => match xml.as_str() {
420                "<w:r><w:separator/></w:r>" => Some(false),
421                "<w:r><w:continuationSeparator/></w:r>" => Some(true),
422                _ => None,
423            },
424            _ => None,
425        }
426    }
427    /// The text this object contributes to plain-text extraction.
428    pub fn plain_text(&self) -> &str {
429        match self {
430            InlineObject::Field { result, .. } => result,
431            InlineObject::Equation { linear, .. } => linear,
432            InlineObject::Ruby { base, .. } => base,
433            InlineObject::PositionalTab { .. } => "\t",
434            InlineObject::Opaque { text, .. } => text,
435            _ => "",
436        }
437    }
