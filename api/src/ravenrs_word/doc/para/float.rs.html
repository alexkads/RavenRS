3/// How a floating object is placed.
4#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash, Serialize, Deserialize, Default)]
5#[serde(rename_all = "camelCase")]
6pub enum Wrap {
7    /// In line with text (not floating).
8    #[default]
9    Inline,
10    Square,
11    Tight,
12    Through,
13    TopAndBottom,
14    BehindText,
15    InFrontOfText,
16}

18/// Which sides of a Square, Tight or Through object text may flow on (`@wrapText`, ECMA-376
19/// Part 1, §20.4.3.7 `ST_WrapText`).
20#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash, Serialize, Deserialize, Default)]
21#[serde(rename_all = "camelCase")]
22pub enum WrapText {
23    /// Both sides of the object.
24    #[default]
25    BothSides,
26    /// Only the left side: the band to the object's right stays empty.
27    Left,
28    /// Only the right side.
29    Right,
30    /// Only whichever side has more room, decided per line.
31    Largest,
32}
33
34impl WrapText {
35    /// From the OOXML value (`bothSides`, `left`, `right`, `largest`).
36    pub fn from_ooxml(v: &str) -> Option<WrapText> {
37        let v = v.trim();
38        if v.eq_ignore_ascii_case("bothSides") || v.eq_ignore_ascii_case("both") {
39            Some(WrapText::BothSides)
40        } else if v.eq_ignore_ascii_case("left") {
41            Some(WrapText::Left)
42        } else if v.eq_ignore_ascii_case("right") {
43            Some(WrapText::Right)
44        } else if v.eq_ignore_ascii_case("largest") {
45            Some(WrapText::Largest)
46        } else {
47            None
48        }
49    }
50
51    /// The OOXML value.
52    pub fn ooxml(self) -> &'static str {
53        match self {
54            WrapText::BothSides => "bothSides",
55            WrapText::Left => "left",
56            WrapText::Right => "right",
57            WrapText::Largest => "largest",
58        }
59    }
60
61    pub fn is_both(&self) -> bool {
62        *self == WrapText::BothSides
63    }
64}

93/// Alignment of a floating object within its [`Anchor`] area, instead of an offset.
94#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash, Serialize, Deserialize)]
95#[serde(rename_all = "camelCase")]
96pub enum FloatAlign {
97    /// Left or top.
98    Start,
99    Center,
100    /// Right or bottom.
101    End,
102    Inside,
103    Outside,
104}
105
106impl FloatAlign {
107    /// From an OOXML alignment (`wp:align`, `w:tblpXSpec`, `w:tblpYSpec`).
108    pub fn from_ooxml(v: &str) -> Option<FloatAlign> {
109        match v {
110            "left" | "top" => Some(FloatAlign::Start),
111            "center" => Some(FloatAlign::Center),
112            "right" | "bottom" => Some(FloatAlign::End),
113            "inside" => Some(FloatAlign::Inside),
114            "outside" => Some(FloatAlign::Outside),
115            _ => None,
116        }
117    }
118
119    /// The OOXML name on the horizontal (`left`, `right`) or vertical (`top`, `bottom`) axis.
120    pub fn ooxml(self, horizontal: bool) -> &'static str {
121        match (self, horizontal) {
122            (FloatAlign::Start, true) => "left",
123            (FloatAlign::Start, false) => "top",
124            (FloatAlign::Center, _) => "center",
125            (FloatAlign::End, true) => "right",
126            (FloatAlign::End, false) => "bottom",
127            (FloatAlign::Inside, _) => "inside",
128            (FloatAlign::Outside, _) => "outside",
129        }
130    }
131}

133/// Floating placement (ignored when `wrap` is `Inline`).
134#[derive(Clone, Debug, PartialEq, Serialize, Deserialize, Default)]
135#[serde(default, rename_all = "camelCase")]
136pub struct Float {
137    pub wrap: Wrap,
138    /// The sides text wraps on (Square, Tight and Through).
139    #[serde(skip_serializing_if = "WrapText::is_both")]
140    pub wrap_text: WrapText,
141    pub h_rel: Anchor,
142    pub v_rel: Anchor,
143    /// Offsets, points (used when the matching alignment is `None`).
144    pub x: f32,
145    pub y: f32,
146    /// Alignments within the anchor area; they take precedence over the offsets.
147    #[serde(skip_serializing_if = "Option::is_none")]
148    pub h_align: Option<FloatAlign>,
149    #[serde(skip_serializing_if = "Option::is_none")]
150    pub v_align: Option<FloatAlign>,
151    /// Distance from surrounding text at the left and right (points).
152    pub dist: f32,
153    /// Distance from surrounding text above and below (points).
154    pub dist_top: f32,
155    pub dist_bottom: f32,
156    /// Room around the object for effects such as shadows (left, top, right, bottom, points),
157    /// inline or floating: lines and surrounding text keep clear of it. Rotation's own overhang
158    /// is not in here: it follows from [`Float::rot`] (see [`Float::spin_pad`]).
159    pub effect: [f32; 4],
160    /// Rotation clockwise about the object's centre, degrees (DrawingML `a:xfrm/@rot`). Offsets
161    /// and the size stay those of the unrotated frame, as in Word; see [`Float::spin`].
162    #[serde(skip_serializing_if = "is_zero")]
163    pub rot: f32,
164    /// Mirrored left to right / top to bottom (`a:xfrm/@flipH`, `@flipV`).
165    #[serde(skip_serializing_if = "is_false")]
166    pub flip_h: bool,
167    #[serde(skip_serializing_if = "is_false")]
168    pub flip_v: bool,
169    /// The outline Tight and Through wrapping follow (`wp:wrapPolygon`), in the object's own
170    /// (unrotated) frame; `None` = derived from the object (see [`crate::doc::wrap`]).
171    #[serde(skip_serializing_if = "Option::is_none")]
172    pub wrap_polygon: Option<std::sync::Arc<crate::doc::wrap::WrapPolygon>>,
173    /// What the drawing is, for screen readers and exports (`wp:docPr` `@descr`, `@title` and the
174    /// decorative mark). See [`InlineObject::alt_text`](super::InlineObject::alt_text).
175    #[serde(skip_serializing_if = "AltText::is_empty")]
176    pub alt: AltText,
177    /// Hidden (`wp:docPr/@hidden`, VML `visibility:hidden`): kept and saved, but neither drawn nor
178    /// laid out, as Word for Windows does (Word for Mac draws hidden VML shapes).
179    #[serde(skip_serializing_if = "is_false")]
180    pub hidden: bool,
181    /// Anchored in a table cell but placed on the page, not in the cell (`wp:anchor/@layoutInCell`
182    /// off, any VML shape in a header or footer): a picture watermark in a header that starts with a table.
183    #[serde(skip_serializing_if = "is_false")]
184    pub outside_cell: bool,
185    /// Stacking order among floating objects (`wp:anchor/@relativeHeight`): higher is drawn on
186    /// top. 0 = unknown, drawn in document order (objects inserted here, or from other formats).
187    #[serde(skip_serializing_if = "is_zero_u32")]
188    pub z: u32,
189    /// The page a drawing anchored to a page is on, from 1 (ODF's `text:anchor-page-number`,
190    /// §19.872); 0 = anchored to its paragraph or not said. Kept for saving: the layout places a
191    /// drawing on its anchor paragraph's page.
192    #[serde(skip_serializing_if = "is_zero_u32")]
193    pub page: u32,
194    /// Turned, but the file keeps no room for the turn (`wp:effectExtent` missing or zero, as
195    /// files from other tools have): Word then keeps an inline one's line to its frame and lets
196    /// the turned object overhang it (see [`Float::line_pad`]). Cleared by [`Float::set_spin`].
197    #[serde(skip_serializing_if = "is_false")]
198    pub bare_turn: bool,
199}

204/// A drawing's alternative text (Picture/Shape Format › Accessibility › Alt Text).
205#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize, Default)]
206#[serde(default, rename_all = "camelCase")]
207pub struct AltText {
208    /// The description of a shape, text box or group. Pictures and charts keep theirs in their
209    /// own `alt`; read it with [`InlineObject::alt_text`](super::InlineObject::alt_text), which knows where it is.
210    #[serde(skip_serializing_if = "String::is_empty")]
211    pub text: String,
212    /// The short title (`wp:docPr/@title`).
213    #[serde(skip_serializing_if = "String::is_empty")]
214    pub title: String,
215    /// Decorative: it adds nothing to the content, so screen readers skip it and the
216    /// accessibility checker doesn't ask for a description.
217    #[serde(skip_serializing_if = "is_false")]
218    pub decorative: bool,
219}
220
221impl AltText {
222    pub fn is_empty(&self) -> bool {
223        self.text.is_empty() && self.title.is_empty() && !self.decorative
224    }
225}

244impl Float {
245    /// The rotation and flips, the angle normalised to `0..360` (a NaN angle is no rotation).
246    pub fn spin(&self) -> crate::geom::Spin {
247        crate::geom::Spin::new(self.rot, self.flip_h, self.flip_v)
248    }
249    /// Set the rotation and flips (the angle normalised).
250    pub fn set_spin(&mut self, s: crate::geom::Spin) {
251        self.rot = crate::geom::normalize_degrees(s.deg);
252        self.flip_h = s.flip_h;
253        self.flip_v = s.flip_v;
254        self.bare_turn = false;
255    }
256    /// How far a `w` × `h` frame's rotated bounds reach past it on each side (x, y): the room a
257    /// rotated object takes beyond its frame (zero unrotated; negative where the turned object
258    /// is narrower than its frame, as a wide picture turned 90° is).
259    pub fn spin_pad(&self, w: f32, h: f32) -> (f32, f32) {
260        let (w, h) = (crate::geom::finite(w).clamp(0.0, 4000.0), crate::geom::finite(h).clamp(0.0, 4000.0));
261        let (bw, bh) = self.spin().extent(w, h);
262        ((bw - w) / 2.0, (bh - h) / 2.0)
263    }
264    /// A [`Float::bare_turn`] still inline and without effects room: made floating, or given room
265    /// for a shadow or glow since, it saves, and so lays out, with the turn's room again.
266    pub fn is_bare_turn(&self) -> bool {
267        self.bare_turn && self.wrap == Wrap::Inline && self.effect_extent() == [0.0; 4]
268    }
269    /// The room an inline `w` × `h` object's line keeps past its frame on each side (x, y) for its
270    /// turn: its rotated bounds ([`Float::spin_pad`]) — or, for a [`Float::is_bare_turn`], none, except
271    /// that Word 2013+ layout (`modern`) exchanges the frame's width and height from 45° up to 135°
272    /// and from 225° up to 315°.
273    pub fn line_pad(&self, w: f32, h: f32, modern: bool) -> (f32, f32) {
274        if !self.is_bare_turn() {
275            return self.spin_pad(w, h);
276        }
277        let (w, h) = (crate::geom::finite(w).clamp(0.0, 4000.0), crate::geom::finite(h).clamp(0.0, 4000.0));
278        let deg = crate::geom::normalize_degrees(self.rot);
279        if modern && ((45.0..135.0).contains(&deg) || (225.0..315.0).contains(&deg)) { ((h - w) / 2.0, (w - h) / 2.0) } else { (0.0, 0.0) }
280    }
281    /// The effect extents, finite and clamped (left, top, right, bottom).
282    pub fn effect_extent(&self) -> [f32; 4] {
283        self.effect.map(|v| if v.is_finite() { v.clamp(0.0, 1584.0) } else { 0.0 })
284    }
