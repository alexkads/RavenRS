71/// Hard limits that keep hostile input from exhausting memory.
72pub const MAX_ROWS: usize = 32_767;
73pub const MAX_COLS: usize = 63;
74/// Most tables nested one inside another. Pasting and inserting stop there, so no edit builds a
75/// document deep enough to overflow the stack when it is saved, copied or dropped (#1596), or
76/// deeper than a `.json` save reads back: serde_json reads 127 nested containers, a table level
77/// takes six, and the deepest cell of a header, note or text box needs up to 15 more for its
78/// paragraph and the richest content measured (#3618). The DOCX reader and writer keep 24.
79pub const MAX_TABLE_DEPTH: usize = 18;

81impl Table {
82    /// Grid columns `lo..hi` that [`Table::slice`] copies for `rows`/`cols` (widened over merged cells),
83    /// and whether any cell touches them (Cut clears the same cells it copies).
84    pub fn slice_bounds(&self, rows: (usize, usize), cols: (usize, usize)) -> (usize, usize, bool) {
85        let ((r0, r1), (g0, g1)) = (rows, cols);
86        let kept = || self.rows.iter().take(r1.saturating_add(1)).skip(r0);
87        // Grid columns `lo..hi` the cells touching them cover: a cell merged across that is partly
88        // in them widens them, for every row, so the copy isn't ragged. Each round only widens
89        // (at most once per grid column), so a bound of the grid's width plus one round is enough.
90        let (mut lo, mut hi) = (g0, g1.saturating_add(1));
91        let mut any = false;
92        for _ in 0..=MAX_COLS.saturating_add(1) {
93            let (mut nlo, mut nhi, mut found) = (lo, hi, false);
94            for row in kept() {
95                let mut g = 0usize;
96                for cell in &row.cells {
97                    let end = g.saturating_add(cell.span());
98                    if g < hi && end > lo {
99                        (nlo, nhi, found) = (nlo.min(g), nhi.max(end), true);
100                    }
101                    g = end;
102                }
103            }
104            any = found;
105            if (nlo, nhi) == (lo, hi) {
106                break;
107            }
108            (lo, hi) = (nlo, nhi);
109        }
110        (lo, hi, any)
111    }

113    /// A copy of rows `rows` of this table holding, of each row, the cells that reach into grid
114    /// columns `cols` (both inclusive): what Copy takes from selected rows or cells. Content
115    /// controls around rows and the table aren't copied (they'd lose their other end); a cell
116    /// that continues a vertical merge from a row left out starts the merge instead.
117    pub fn slice(&self, rows: (usize, usize), cols: (usize, usize)) -> Table {
118        let (r0, r1) = rows;
119        let kept = || self.rows.iter().take(r1.saturating_add(1)).skip(r0);
120        let (lo, hi, any) = self.slice_bounds(rows, cols);
121        let mut out_rows = Vec::new();
122        for row in kept() {
123            let mut g = 0usize;
124            let mut cells = Vec::new();
125            for cell in &row.cells {
126                let end = g.saturating_add(cell.span());
127                if any && g < hi && end > lo {
128                    cells.push(cell.clone());
129                }
130                g = end;
131            }
132            if !cells.is_empty() {
133                out_rows.push(Row { props: row.props.clone(), cells, controls: ControlWrap::default() });
134            }
135        }
136        if let Some(first) = out_rows.first_mut() {
137            for c in &mut first.cells {
138                if c.props.vmerge == VMerge::Continue {
139                    c.props.vmerge = VMerge::Restart;
140                }
141            }
142        }
143        let whole = lo == 0 && hi >= self.grid.len();
144        let grid = if whole { self.grid.clone() } else { self.grid.get(lo..hi.min(self.grid.len())).map(<[f32]>::to_vec).unwrap_or_default() };
145        let mut props = self.props.clone();
146        if !whole {
147            // Part of the columns: as wide as they are, not the whole table's width.
148            if props.width.is_some() || props.width_pct.is_some() {
149                props.width = Some(grid.iter().sum());
150            }
151            props.width_pct = None;
152        }
153        Table { props, grid, rows: out_rows, controls: ControlWrap::default() }
154    }

156    /// A `rows × cols` table spanning `width` points, "Table Grid" style.
157    pub fn new(rows: usize, cols: usize, width: f32) -> Table {
158        let rows = rows.clamp(1, MAX_ROWS);
159        let cols = cols.clamp(1, MAX_COLS);
160        let w = (width / cols as f32).max(6.0);
161        Table {
162            props: TableProps { style: Some("TableGrid".into()), ..Default::default() },
163            grid: vec![w; cols],
164            rows: (0..rows)
165                .map(|_| Row {
166                    props: RowProps::default(),
167                    cells: (0..cols)
168                        .map(|_| {
169                            let mut c = Cell::empty();
170                            c.props.width = Some(w);
171                            c
172                        })
173                        .collect(),
174                    controls: ControlWrap::default(),
175                })
176                .collect(),
177            controls: ControlWrap::default(),
178        }
179    }

181    pub fn cols(&self) -> usize {
182        let row_cols =
183            |r: &Row| r.first_grid().saturating_add(r.cells.iter().map(Cell::span).fold(0, usize::saturating_add)).saturating_add(r.grid_after());
184        self.grid.len().max(self.rows.iter().map(row_cols).max().unwrap_or(0))
185    }

188    pub fn grid_col(&self, r: usize, c: usize) -> usize {
189        self.rows.get(r).map(|row| row.first_grid() + row.cells.iter().take(c).map(Cell::span).sum::<usize>()).unwrap_or(0)

192    /// Cell index in row `r` that covers grid column `g`.
193    pub fn cell_at_grid(&self, r: usize, g: usize) -> Option<usize> {
194        let row = self.rows.get(r)?;
195        let mut x = row.first_grid();
196        if g < x {
197            return None;
198        }
199        for (i, c) in row.cells.iter().enumerate() {
200            if g < x + c.span() {
201                return Some(i);
202            }
203            x += c.span();
204        }
205        None
206    }

208    /// Insert a row copied (formatting, empty text) from row `like` at index `at`.
209    pub fn insert_row(&mut self, at: usize, like: usize) {
210        let Some(src) = self.rows.get(like.min(self.rows.len().saturating_sub(1))) else { return };
211        if self.rows.len() >= MAX_ROWS {
212            return;
213        }
214        let mut row = Row { props: src.props.clone(), cells: Vec::with_capacity(src.cells.len()), controls: ControlWrap::default() };
215        row.props.header = false;
216        for c in &src.cells {
217            let mut nc = Cell::empty();
218            nc.props = c.props.clone();
219            if nc.props.vmerge == VMerge::Restart {
220                nc.props.vmerge = VMerge::None;
221            }
222            row.cells.push(nc);
223        }
224        self.rows.insert(at.min(self.rows.len()), row);
225    }

227    pub fn delete_row(&mut self, r: usize) {
228        if r >= self.rows.len() {
229            return;
230        }
231        // A vertical merge that starts in this row now starts in the row below.
232        let starts: Vec<usize> = (0..self.rows.get(r).map_or(0, |row| row.cells.len()))
233            .filter(|&c| self.rows.get(r).and_then(|row| row.cells.get(c)).is_some_and(|x| x.props.vmerge == VMerge::Restart))
234            .map(|c| self.grid_col(r, c))
235            .collect();
236        for g in starts {
237            let continues = |t: &Self, row: usize| {
238                t.cell_at_grid(row, g).and_then(|i| t.rows.get(row)?.cells.get(i)).is_some_and(|x| x.props.vmerge == VMerge::Continue)
239            };
240            if !continues(self, r + 1) {
241                continue;
242            }
243            let more = continues(self, r + 2);
244            if let Some(c) = self.cell_at_grid(r + 1, g).and_then(|i| self.rows.get_mut(r + 1)?.cells.get_mut(i)) {
245                c.props.vmerge = if more { VMerge::Restart } else { VMerge::None };
246            }
247        }
248        self.rows.remove(r);
249    }

251    /// Split the table before row `at`: the rows from `at` on move to the returned table, which
252    /// has this table's properties and grid. A vertical merge that the split cuts ends in this
253    /// table's last row and starts again in the new table's first row. A row content control
254    /// around rows on both sides closes after this table's last row and opens again (without its
255    /// `w:id`) before the new table's first row. Content controls that close after this table
256    /// close after the new one.
257    pub fn split_off(&mut self, at: usize) -> Table {
258        let at = at.min(self.rows.len());
259        let mut open = Vec::new();
260        for row in self.rows.iter().take(at) {
261            open.extend(row.controls.open.iter().cloned());
262            let close = usize::try_from(row.controls.close).unwrap_or(usize::MAX).min(open.len());
263            open.truncate(open.len() - close);
264        }
265        let mut tail = Table {
266            props: self.props.clone(),
267            grid: self.grid.clone(),
268            rows: self.rows.split_off(at),
269            controls: ControlWrap { open: Vec::new(), close: std::mem::take(&mut self.controls.close) },
270        };
271        if let Some(last) = self.rows.last_mut() {
272            last.controls.close = last.controls.close.saturating_add(u32::try_from(open.len()).unwrap_or(u32::MAX));
273            for c in &mut last.cells {
274                if c.props.vmerge == VMerge::Restart {
275                    c.props.vmerge = VMerge::None;
276                }
277            }
278        }
279        for c in &mut open {
280            c.id = None;
281        }
282        if let Some(first) = tail.rows.first_mut() {
283            let own = std::mem::replace(&mut first.controls.open, open);
284            first.controls.open.extend(own);
285        }
286        let starts: Vec<(usize, usize)> = (0..tail.rows.first().map_or(0, |row| row.cells.len()))
287            .filter(|&c| tail.rows.first().and_then(|row| row.cells.get(c)).is_some_and(|x| x.props.vmerge == VMerge::Continue))
288            .map(|c| (c, tail.grid_col(0, c)))
289            .collect();
290        for (c, g) in starts {
291            let more = tail.cell_at_grid(1, g).and_then(|i| tail.rows.get(1)?.cells.get(i)).is_some_and(|x| x.props.vmerge == VMerge::Continue);
292            if let Some(x) = tail.rows.first_mut().and_then(|r| r.cells.get_mut(c)) {
293                x.props.vmerge = if more { VMerge::Restart } else { VMerge::None };
294            }
295        }
296        tail
297    }

299    /// Insert a grid column at grid index `g` (cells spanning it widen instead).
300    pub fn insert_col(&mut self, g: usize) {
301        if self.grid.len() >= MAX_COLS {
302            return;
303        }
304        let g = g.min(self.grid.len());
305        let w = self.grid.get(g.min(self.grid.len().saturating_sub(1))).copied().unwrap_or(72.0);
306        self.grid.insert(g, w);
307        for r in 0..self.rows.len() {
308            let mut x = 0;
309            let mut idx = None;
310            if let Some(row) = self.rows.get(r) {
311                for (i, c) in row.cells.iter().enumerate() {
312                    if g <= x {
313                        idx = Some((i, false));
314                        break;
315                    }
316                    if g < x + c.span() {
317                        idx = Some((i, true));
318                        break;
319                    }
320                    x += c.span();
321                }
322            }
323            if let Some(row) = self.rows.get_mut(r) {
324                match idx {
325                    Some((i, true)) => {
326                        if let Some(c) = row.cells.get_mut(i) {
327                            // From the clamped span: a `.json` file can carry any `u32` (#3637).
328                            c.props.span = (c.span() + 1).min(MAX_COLS) as u32;
329                        }
330                    }
331                    Some((i, false)) => {
332                        let mut c = Cell::empty();
333                        c.props.width = Some(w);
334                        row.cells.insert(i, c);
335                    }
336                    None => {
337                        let mut c = Cell::empty();
338                        c.props.width = Some(w);
339                        row.cells.push(c);
340                    }
341                }
342            }
343        }
344    }

346    /// Delete grid column `g` (spanning cells shrink).
347    pub fn delete_col(&mut self, g: usize) {
348        if g >= self.grid.len() || self.grid.len() <= 1 {
349            return;
350        }
351        self.grid.remove(g);
352        for row in &mut self.rows {
353            let mut x = 0;
354            let mut rm = None;
355            for (i, c) in row.cells.iter_mut().enumerate() {
356                if g < x + c.span() {
357                    if c.span() > 1 {
358                        c.props.span = (c.span() - 1) as u32;
359                    } else {
360                        rm = Some(i);
361                    }
362                    break;
363                }
364                x += c.span();
365            }
366            if let Some(i) = rm {
367                row.cells.remove(i);
368            }
369        }
370        self.rows.retain(|r| !r.cells.is_empty());
371    }

373    /// Merge cells `c0..=c1` of row `r` horizontally (content is concatenated).
374    pub fn merge_right(&mut self, r: usize, c0: usize, c1: usize) {
375        let Some(row) = self.rows.get_mut(r) else { return };
376        if c1 <= c0 || c1 >= row.cells.len() {
377            return;
378        }
379        let removed: Vec<Cell> = row.cells.drain(c0 + 1..=c1).collect();
380        if let Some(first) = row.cells.get_mut(c0) {
381            for c in removed {
382                first.props.span = (first.span() + c.span()).min(MAX_COLS) as u32;
383                let nonempty: Vec<_> = c.blocks.into_iter().filter(|b| b.as_para().is_none_or(|p| !p.is_empty())).collect();
384                first.blocks.extend(nonempty);
385            }
386            if first.blocks.len() > 1 && first.blocks.first().and_then(|b| b.as_para()).is_some_and(Paragraph::is_empty) {
387                first.blocks.remove(0);
388            }
389        }
390    }

392    /// Merge a rectangle of cells (rows r0..=r1, grid columns g0..=g1).
393    pub fn merge(&mut self, r0: usize, r1: usize, g0: usize, g1: usize) {
394        for r in r0..=r1.min(self.rows.len().saturating_sub(1)) {
395            let (Some(a), Some(b)) = (self.cell_at_grid(r, g0), self.cell_at_grid(r, g1)) else { continue };
396            self.merge_right(r, a, b);
397            if let Some(c) = self.rows.get_mut(r).and_then(|row| row.cells.get_mut(a)) {
398                c.props.vmerge = if r1 == r0 {
399                    VMerge::None
400                } else if r == r0 {
401                    VMerge::Restart
402                } else {
403                    VMerge::Continue
404                };
405            }
406        }
407        // Move content of continued cells into the restart cell.
408        if r1 > r0
409            && let Some(a) = self.cell_at_grid(r0, g0)
410        {
411            let mut moved = Vec::new();
412            for r in r0 + 1..=r1.min(self.rows.len().saturating_sub(1)) {
413                if let Some(c) = self.cell_at_grid(r, g0).and_then(|ci| self.rows.get_mut(r).and_then(|row| row.cells.get_mut(ci))) {
414                    let blocks = std::mem::replace(&mut c.blocks, vec![para_block(Paragraph::new())]);
415                    moved.extend(blocks.into_iter().filter(|b| b.as_para().is_none_or(|p| !p.is_empty())));
416                }
417            }
418            if let Some(c) = self.rows.get_mut(r0).and_then(|row| row.cells.get_mut(a)) {
419                c.blocks.extend(moved);
420            }
421        }
422    }

424    /// Split cell `c` of row `r` into `n` cells horizontally. Where the cell's grid lines don't
425    /// fall on the equal parts, the grid is refined inside the cell only: the new borders divide it
426    /// into equal parts, and the cells of other rows widen over the new grid columns, so their
427    /// extent and the table's width stay as they were. Returns the parts' spans, left to right.
428    pub fn split_cell(&mut self, r: usize, c: usize, n: usize) -> Vec<usize> {
429        let n = n.clamp(1, MAX_COLS);
430        let Some(span) = self.rows.get(r).and_then(|row| row.cells.get(c)).map(Cell::span) else { return Vec::new() };
431        if n <= 1 {
432            return vec![span];
433        }
434        let g = self.grid_col(r, c);
435        let mut end = g.saturating_add(span);
436        // A grid narrower than the row (a damaged file) gets the columns the cell needs.
437        while self.grid.len() < end && self.grid.len() < MAX_COLS {
438            self.grid.push(72.0);
439        }
440        if self.grid.len() < end {
441            return vec![span];
442        }
443        for w in self.grid.get_mut(g..end).into_iter().flatten() {
444            if !(w.is_finite() && *w > 0.0) {
445                *w = 72.0;
446            }
447        }
448        const EPS: f32 = 0.01;
449        let total: f32 = self.grid.get(g..end).map(|x| x.iter().sum()).unwrap_or(0.0);
450        // Grid lines where the cell's parts start (after the first, at `g`).
451        let mut lines = Vec::with_capacity(n);
452        for k in 1..n {
453            let x = total * k as f32 / n as f32;
454            let mut acc = 0.0;
455            let mut hit = None;
456            for j in g..end {
457                let w = self.grid.get(j).copied().unwrap_or(0.0);
458                if j > g && (x - acc).abs() <= EPS {
459                    hit = Some((j, None));
460                    break;
461                }
462                if x < acc + w - EPS {
463                    if x > acc + EPS {
464                        hit = Some((j, Some((x - acc, w - (x - acc)))));
465                    }
466                    break;
467                }
468                acc += w;
469            }
470            match hit {
471                // An existing grid line is where the border goes.
472                Some((j, None)) => lines.push(j),
473                // A new grid line inside column `j`: every cell over `j` widens over both halves.
474                Some((j, Some((left, right)))) if self.grid.len() < MAX_COLS => {
475                    if let Some(w) = self.grid.get_mut(j) {
476                        *w = left;
477                    }
478                    self.grid.insert(j + 1, right);
479                    for rr in 0..self.rows.len() {
480                        if let Some(ci) = self.cell_at_grid(rr, j)
481                            && let Some(cell) = self.rows.get_mut(rr).and_then(|row| row.cells.get_mut(ci))
482                        {
483                            cell.props.span = cell.props.span.max(1).saturating_add(1);
484                        }
485                    }
486                    end += 1;
487                    lines.push(j + 1);
488                }
489                _ => {}
490            }
491        }
492        lines.dedup();
493        // The cell now spans `g..end`: cut it at the new lines.
494        let grid = self.grid.clone();
495        let width = |a: usize, b: usize| grid.get(a..b).map(|x| x.iter().sum::<f32>());
496        let Some(row) = self.rows.get_mut(r) else { return Vec::new() };
497        let mut a = g;
498        let mut parts = Vec::with_capacity(lines.len() + 1);
499        for &b in lines.iter().chain(std::iter::once(&end)) {
500            if b > a {
501                parts.push((a, b));
502                a = b;
503            }
504        }
505        for (k, &(a, b)) in parts.iter().enumerate() {
506            if k == 0 {
507                if let Some(cell) = row.cells.get_mut(c) {
508                    cell.props.span = (b - a) as u32;
509                    cell.props.width = width(a, b);
510                }
511            } else {
512                let mut nc = Cell::empty();
513                nc.props.span = (b - a) as u32;
514                nc.props.width = width(a, b);
515                row.cells.insert((c + k).min(row.cells.len()), nc);
516            }
517        }
518        parts.iter().map(|&(a, b)| b - a).collect()
519    }

521    /// The rows cell `c` of row `r` covers: (first row, count). A vertically merged cell runs from
522    /// its `Restart` cell down through the `Continue` cells that start at the same grid column.
523    pub fn vertical_extent(&self, r: usize, c: usize) -> Option<(usize, usize)> {
524        let g = self.grid_col(r, c);
525        let at = |rr: usize| -> Option<VMerge> {
526            let ci = self.cell_at_grid(rr, g)?;
527            if self.grid_col(rr, ci) != g {
528                return None;
529            }
530            Some(self.rows.get(rr)?.cells.get(ci)?.props.vmerge)
531        };
532        let mut top = r;
533        if self.rows.get(r)?.cells.get(c)?.props.vmerge == VMerge::Continue {
534            while let Some(up) = top.checked_sub(1).and_then(at) {
535                if up == VMerge::None {
536                    break;
537                }
538                top -= 1;
539                if up == VMerge::Restart {
540                    break;
541                }
542            }
543        }
544        let mut k = 1;
545        if at(top) != Some(VMerge::None) {
546            while at(top + k) == Some(VMerge::Continue) {
547                k += 1;
548            }
549        }
550        Some((top, k))
551    }

553    /// Split Cells: cut cell `c` of row `r` into `cols` × `rows` cells. A cell one row high gets
554    /// `rows - 1` new rows under it, over which the row's other cells merge down; a vertically
555    /// merged cell is regrouped instead, so `rows` must divide its height. The content stays in
556    /// the first part. Returns false (with the table unchanged) when the split can't be made.
557    pub fn split_cell_grid(&mut self, r: usize, c: usize, cols: usize, rows: usize) -> bool {
558        let (cols, rows) = (cols.clamp(1, MAX_COLS), rows.clamp(1, MAX_ROWS));
559        let Some((top, k)) = self.vertical_extent(r, c) else { return false };
560        let g = self.grid_col(r, c);
561        let Some(ci) = self.cell_at_grid(top, g) else { return false };
562        if (k > 1 && k % rows != 0) || (k == 1 && self.rows.len().saturating_add(rows - 1) > MAX_ROWS) {
563            return false;
564        }
565        let height = if k > 1 { k } else { rows };
566        if k > 1 && rows > 1 {
567            let q = k / rows;
568            for i in 0..k {
569                if let Some(cj) = self.cell_at_grid(top + i, g)
570                    && let Some(cell) = self.rows.get_mut(top + i).and_then(|row| row.cells.get_mut(cj))
571                {
572                    cell.props.vmerge = if q == 1 {
573                        VMerge::None
574                    } else if i % q == 0 {
575                        VMerge::Restart
576                    } else {
577                        VMerge::Continue
578                    };
579                }
580            }
581        } else if k == 1 && rows > 1 {
582            let Some(src) = self.rows.get_mut(top) else { return false };
583            let mut like = Row { props: src.props.clone(), cells: Vec::with_capacity(src.cells.len()), controls: ControlWrap::default() };
584            like.props.header = false;
585            for (i, cell) in src.cells.iter_mut().enumerate() {
586                let mut nc = Cell::empty();
587                nc.props = cell.props.clone();
588                if i == ci {
589                    cell.props.vmerge = VMerge::None;
590                    nc.props.vmerge = VMerge::None;
591                } else {
592                    // The row's other cells merge down over the new rows.
593                    if cell.props.vmerge == VMerge::None {
594                        cell.props.vmerge = VMerge::Restart;
595                    }
596                    nc.props.vmerge = VMerge::Continue;
597                }
598                like.cells.push(nc);
599            }
600            for _ in 1..rows {
601                self.rows.insert(top + 1, like.clone());
602            }
603        }
604        if cols > 1 {
605            let spans = self.split_cell(top, ci, cols);
606            let total: usize = spans.iter().sum();
607            let vmerge = self.rows.get(top).and_then(|row| row.cells.get(ci)).map(|cell| cell.props.vmerge).unwrap_or_default();
608            if let Some(row) = self.rows.get_mut(top) {
609                for cell in row.cells.iter_mut().skip(ci + 1).take(spans.len().saturating_sub(1)) {
610                    cell.props.vmerge = vmerge;
611                }
612            }
613            // The cells under it, which the new grid lines widened, are cut the same way.
614            let grid = self.grid.clone();
615            let width = |a: usize, b: usize| grid.get(a..b).map(|x| x.iter().sum::<f32>());
616            for rr in top + 1..top + height {
617                let Some(cj) = self.cell_at_grid(rr, g).filter(|&cj| self.grid_col(rr, cj) == g) else { continue };
618                let Some(row) = self.rows.get_mut(rr) else { continue };
619                let Some(cell) = row.cells.get_mut(cj).filter(|cell| cell.span() == total) else { continue };
620                let mut a = g;
621                let mut parts = Vec::with_capacity(spans.len());
622                for (n, &span) in spans.iter().enumerate() {
623                    let mut part = if n == 0 { cell.clone() } else { Cell::empty() };
624                    part.props.vmerge = cell.props.vmerge;
625                    part.props.span = span as u32;
626                    part.props.width = width(a, a + span);
627                    a += span;
628                    parts.push(part);
629                }
630                row.cells.splice(cj..=cj, parts);
631            }
632        }
633        true
634    }
635
636    pub fn total_width(&self) -> f32 {
637        self.grid.iter().sum()
638    }
