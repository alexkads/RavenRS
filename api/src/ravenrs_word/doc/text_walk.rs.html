6/// Words in `s`, as Word counts them: tokens separated by whitespace that contain at least one
7/// alphanumeric char, plus each Han ideograph and kana as a word of its own (Chinese and Japanese
8/// are written without spaces, #4547; Korean is spaced, so Hangul counts by tokens).
9pub fn count_words(s: &str) -> usize {
10    // The half-width voiced sound marks belong to the kana before them, not a word of their own.
11    let word_char = |c: char| c.is_alphanumeric() && !matches!(c, '\u{FF9E}' | '\u{FF9F}');
12    let tokens = s.split(|c: char| c.is_whitespace() || c == para::OBJ || is_cjk_word(c)).filter(|w| w.chars().any(word_char)).count();
13    tokens.saturating_add(s.chars().filter(|c| is_cjk_word(*c)).count())
14}

16/// Words in a generated list label: whitespace-separated tokens holding a letter or digit, so a
17/// CJK label such as 第十二章 stays one word (how Word counts those is unverified).
18fn count_label_words(s: &str) -> usize {
19    s.split(char::is_whitespace).filter(|w| w.chars().any(char::is_alphanumeric)).count()
20}

22/// A Han ideograph or kana letter (not CJK punctuation), which Word counts as one word.
23fn is_cjk_word(c: char) -> bool {
24    matches!(c,
25        '\u{3005}'..='\u{3007}' // 々 〆 〇
26        | '\u{3031}'..='\u{3035}' | '\u{303B}' // vertical kana repeat marks, 〻
27        | '\u{3040}'..='\u{30FF}' // hiragana, katakana
28        | '\u{31F0}'..='\u{31FF}' // katakana phonetic extensions
29        | '\u{3400}'..='\u{4DBF}' // extension A
30        | '\u{4E00}'..='\u{9FFF}' // unified ideographs
31        | '\u{F900}'..='\u{FAFF}' // compatibility ideographs
32        | '\u{FF66}'..='\u{FF9D}' // half-width katakana
33        | '\u{1B000}'..='\u{1B16F}' // kana supplement and extended
34        | '\u{20000}'..='\u{3FFFF}' // extensions B and later
35    ) && c.is_alphanumeric()
36}

38/// Characters in `s`, as the Word Count dialog counts them ("Characters (with spaces)"):
39/// paragraph breaks aren't characters (#2797).
40pub fn count_chars(s: &str) -> usize {
41    s.chars().filter(|c| *c != '\n' && *c != '\r').count()
42}
43
44/// Characters in `s` ignoring every whitespace character, as the Word Count dialog counts them
45/// ("Characters (no spaces)"; #2797).
46pub fn count_chars_no_spaces(s: &str) -> usize {
47    s.chars().filter(|c| !c.is_whitespace()).count()
48}

61pub(crate) fn walk(bl: &Blocks, prefix: &mut Vec<u32>, out: &mut Vec<Path>, depth: usize) {
62    if depth > PARA_WALK_DEPTH {
63        return;
64    }
65    for (i, b) in bl.iter().enumerate() {
66        crate::doc::work::steps(1);
67        prefix.push(i as u32);
68        match &**b {
69            Block::Para(_) => out.push(Path(prefix.clone())),
70            Block::Table(t) => {
71                for (r, row) in t.rows.iter().enumerate() {
72                    for (c, cell) in row.cells.iter().enumerate() {
73                        prefix.push(r as u32);
74                        prefix.push(c as u32);
75                        walk(&cell.blocks, prefix, out, depth + 1);
76                        prefix.pop();
77                        prefix.pop();
78                    }
79                }
80            }
81        }
82        prefix.pop();
83    }
84}

86/// [`walk`] for the paragraphs from path `a` to `b` inclusive, skipping the blocks outside.
87pub(crate) fn walk_between(bl: &Blocks, prefix: &mut Vec<u32>, out: &mut Vec<Path>, depth: usize, (a, b): (&[u32], &[u32])) {
88    use std::cmp::Ordering::{Greater, Less};
89    if depth > PARA_WALK_DEPTH {
90        return;
91    }
92    // Where the paths starting with `q` lie against `x`: all before, all after, or either.
93    let side = |q: &[u32], x: &[u32]| {
94        let m = q.len().min(x.len());
95        q[..m].cmp(&x[..m])
96    };
97    for (i, blk) in bl.iter().enumerate() {
98        prefix.push(i as u32);
99        if side(prefix, b) == Greater {
100            prefix.pop();
101            break;
102        }
103        if side(prefix, a) != Less {
104            match &**blk {
105                Block::Para(_) => {
106                    if prefix.as_slice() >= a && prefix.as_slice() <= b {
107                        out.push(Path(prefix.clone()));
108                    }
109                }
110                Block::Table(t) => {
111                    for (r, row) in t.rows.iter().enumerate() {
112                        for (c, cell) in row.cells.iter().enumerate() {
113                            prefix.push(r as u32);
114                            prefix.push(c as u32);
115                            if side(prefix, a) != Less && side(prefix, b) != Greater {
116                                walk_between(&cell.blocks, prefix, out, depth + 1, (a, b));
117                            }
118                            prefix.pop();
119                            prefix.pop();
120                        }
121                    }
122                }
123            }
124        }
125        prefix.pop();
126    }
127}

129/// Which text of a paragraph [`text_of`] collects.
130#[derive(Clone, Copy)]
131pub(crate) enum TextMode {
132    /// Every run, tracked deletions included ([`Paragraph::plain_text`]).
133    Plain,
134    /// As shown on screen ([`Paragraph::shown_text`]).
135    Shown,
136    /// Without tracked deletions ([`Paragraph::final_text`]).
137    Final,
138}
139
140pub(crate) fn text_of(bl: &Blocks, out: &mut String, depth: usize, mode: TextMode) {
141    if depth > 16 {
142        return;
143    }
144    for b in bl {
145        match &**b {
146            Block::Para(p) => {
147                out.push_str(&match mode {
148                    TextMode::Plain => p.plain_text(),
149                    TextMode::Shown => p.shown_text(),
150                    TextMode::Final => p.final_text(),
151                });
152                out.push('\n');
153            }
154            Block::Table(t) => {
155                for row in &t.rows {
156                    let mut cells = Vec::new();
157                    for c in &row.cells {
158                        let mut s = String::new();
159                        text_of(&c.blocks, &mut s, depth + 1, mode);
160                        cells.push(s.trim_end_matches('\n').replace('\n', " "));
161                    }
162                    out.push_str(&cells.join("\t"));
163                    out.push('\n');
164                }
165            }
166        }
167    }
168}

170pub(crate) fn story_word_count_of(
171    bl: &Blocks,
172    numbering: &Numbering,
173    styles: &styles::StyleSheet,
174    counters: &mut numbering::Counters,
175    depth: usize,
176) -> usize {
177    if depth > 16 {
178        return 0;
179    }
180    let mut total = 0usize;
181    for b in bl {
182        match &**b {
183            Block::Para(p) => {
184                let n = p.props.numbering.or_else(|| styles.resolve_para(&p.props).numbering).filter(|n| n.num != 0);
185                if let Some(n) = n
186                    && let Some((label, def)) = counters.next_label(numbering, n.num, n.level)
187                    && !matches!(def.format, section::NumFormat::Bullet | section::NumFormat::None)
188                {
189                    total = total.saturating_add(count_label_words(&label));
190                }
191                total = total.saturating_add(count_words(&p.plain_text()));
192            }
193            Block::Table(t) => {
194                for row in &t.rows {
195                    for c in &row.cells {
196                        total = total.saturating_add(story_word_count_of(&c.blocks, numbering, styles, counters, depth + 1));
197                    }
198                }
199            }
200        }
201    }
202    total
