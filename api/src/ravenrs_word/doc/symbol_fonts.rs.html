1//! Word's symbol fonts — Symbol, Wingdings, Wingdings 2, Wingdings 3 and Webdings — mapped to
2//! standard Unicode characters, so text set in them shows the intended symbols when the font
3//! isn't installed (they are proprietary and never bundled) and exports as real characters.
4//!
5//! Word stores a symbol-font character either as the private-use code U+F000 + the font's own
6//! 8-bit code (U+F020–U+F0FF, what Insert › Symbol and `w:sym` write) or as the plain 8-bit code
7//! (U+0020–U+00FF) in a run whose font is the symbol font. Both map the same way. The stored text
8//! never changes: the mapping is for display, layout and plain-text output only.
9//!
10//! Sources (Unicode's own published data; no font file was read):
11//! - Symbol: the Adobe Symbol encoding table published by Unicode,
12//!   <https://www.unicode.org/Public/MAPPINGS/VENDORS/ADOBE/symbol.txt> (codes it maps to the
13//!   private use area are left out; 0x6D maps to GREEK SMALL LETTER MU).
14//! - Wingdings, Wingdings 2, Wingdings 3, Webdings: the correspondence tables in "Status of
15//!   encoding of Wingdings and Webdings Symbols", Unicode L2/12-368 (WG2 N4384, M. Suignard,
16//!   2012), the frozen repertoire of the Unicode 7.0 encoding (earlier drafts L2/11-052,
17//!   L2/11-344), checked against the Unicode Character Database names. The one character not
18//!   encoded (Wingdings 0xFF) has no entry.
19//!
20//! Tables are indexed by code − 0x20; 0 = no Unicode equivalent.
21
22/// The symbol table for a single family name, by lowercase family name.
23fn single_table(family: &str) -> Option<&'static [u32; 224]> {
24    let f = family.trim();
25    if f.eq_ignore_ascii_case("symbol") || f.eq_ignore_ascii_case("symbol (pcl6)") {
26        Some(&SYMBOL)
27    } else if f.eq_ignore_ascii_case("wingdings") {
28        Some(&WINGDINGS)
29    } else if f.eq_ignore_ascii_case("wingdings 2") {
30        Some(&WINGDINGS_2)
31    } else if f.eq_ignore_ascii_case("wingdings 3") {
32        Some(&WINGDINGS_3)
33    } else if f.eq_ignore_ascii_case("webdings") {
34        Some(&WEBDINGS)
35    } else {
36        None
37    }
38}

40/// The symbol fonts this module knows, by lowercase family name.
41fn table(family: &str) -> Option<&'static [u32; 224]> {
42    if family.contains(';') { family.split(';').map(str::trim).filter(|s| !s.is_empty()).find_map(single_table) } else { single_table(family) }
43}

45/// Whether `family` is one of Word's symbol fonts (Symbol, Wingdings, Wingdings 2, Wingdings 3,
46/// Webdings), whose characters [`symbol_char`] maps.
47pub fn is_symbol_font(family: &str) -> bool {
48    table(family).is_some()
49}
50
51/// The standard Unicode character for `c` in a run set in the symbol font `family`: `c` is the
52/// font's code stored as U+F020–U+F0FF or as U+0020–U+00FF (except the soft hyphen U+00AD).
53/// `None` for other fonts, other characters and codes with no Unicode equivalent.
54pub fn symbol_char(family: &str, c: char) -> Option<char> {
55    let t = table(family)?;
56    let u = u32::from(c);
57    let code = match u {
58        0xF020..=0xF0FF => u - 0xF000,
59        // A soft hyphen stays a soft hyphen (Word stores `w:softHyphen` as U+00AD).
60        0xAD => return None,
61        0x20..=0xFF => u,
62        _ => return None,
63    };
64    let i = usize::try_from(code.checked_sub(0x20)?).ok()?;
65    t.get(i).copied().filter(|v| *v != 0).and_then(char::from_u32)
66}
67
68/// `text` (in a run set in `family`) with every symbol-font character replaced by its Unicode
69/// equivalent; unchanged for other fonts.
70pub fn map_text(family: &str, text: &str) -> String {
71    if !is_symbol_font(family) {
72        return text.to_string();
73    }
74    text.chars().map(|c| symbol_char(family, c).unwrap_or(c)).collect()
75}

236    #[test]
237    fn both_storage_forms_map_alike() {
238        for f in ["Symbol", "Wingdings", "Wingdings 2", "Wingdings 3", "Webdings"] {
239            assert!(is_symbol_font(f));
240            for code in 0x20u32..=0xFF {
241                let (Some(pua), Some(plain)) = (char::from_u32(0xF000 + code), char::from_u32(code)) else { continue };
242                if code == 0xAD {
243                    assert_eq!(symbol_char(f, plain), None);
244                    continue;
245                }
246                assert_eq!(symbol_char(f, pua), symbol_char(f, plain), "{f} {code:#X}");
247            }
248        }
249    }
250
251    #[test]
252    fn pcl6_symbol_encoding() {
253        for family in ["Symbol (PCL6)", " symbol (pcl6) "] {
254            assert!(is_symbol_font(family));
255            for code in 0x20..=0xFF {
256                let c = char::from_u32(code).unwrap();
257                assert_eq!(symbol_char(family, c), symbol_char("Symbol", c));
258                let pua = char::from_u32(0xF000 + code).unwrap();
259                assert_eq!(symbol_char(family, pua), symbol_char("Symbol", pua));
260            }
261            assert_eq!(symbol_char(family, '\u{F0B3}'), Some('≥'));
262        }
263        assert!(!is_symbol_font("Symbol (unknown)"));
264    }

266    #[test]
267    fn symbol_uses_the_adobe_symbol_encoding() {
268        assert_eq!(symbol_char("Symbol", '\u{F061}'), Some('α'));
269        assert_eq!(symbol_char("symbol", 'a'), Some('α'));
270        assert_eq!(symbol_char("Symbol", 'm'), Some('μ'));
271        assert_eq!(symbol_char("Symbol", 'W'), Some('Ω'));
272        assert_eq!(symbol_char("Symbol", '\u{F0B7}'), Some('•'));
273        assert_eq!(symbol_char("Symbol", '\u{F0A5}'), Some('∞'));
274        assert_eq!(symbol_char("Symbol", '\u{F0D6}'), Some('√'));
275        // The radical extender is private use in the Adobe table: no equivalent.
276        assert_eq!(symbol_char("Symbol", '\u{F060}'), None);
277    }
