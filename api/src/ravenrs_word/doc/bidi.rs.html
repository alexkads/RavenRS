1//! Script classification for bidirectional and complex-script text (Persian, Arabic, Hebrew…).
2//!
3//! The full Unicode Bidirectional Algorithm (UAX #9) runs in layout; this module only answers
4//! the cheap questions the model and the engine need: does a character belong to a
5//! right-to-left script, and is it formatted with the complex-script properties?
6
7/// A character of a right-to-left script: Hebrew, Arabic (Persian, Urdu…), Syriac, Thaana, N'Ko,
8/// Samaritan, Mandaic, and their presentation forms. Includes the Arabic-Indic and Persian
9/// (Extended Arabic-Indic) digits and the Arabic combining marks.
10pub fn is_rtl_script(c: char) -> bool {
11    matches!(u32::from(c),
12        0x0590..=0x08FF
13        | 0xFB1D..=0xFDFF
14        | 0xFE70..=0xFEFF
15        | 0x10800..=0x10FFF
16        | 0x1E800..=0x1EFFF)
17}
18
19/// A character Word formats with the complex-script properties (`w:cs` font, `w:szCs`, `w:bCs`,
20/// `w:iCs`): right-to-left scripts, the Indic scripts, Thai and Lao.
21pub fn is_complex_script(c: char) -> bool {
22    is_rtl_script(c) || matches!(u32::from(c), 0x0900..=0x0EFF)
23}
24
25/// A character that belongs to a word for word selection and Ctrl+arrow movement: letters,
26/// digits, `_` and `'`, plus what sits inside Persian and Arabic words: the zero-width non-joiner
27/// (نیم‌فاصله, as in «می‌خواهم»), the zero-width joiner, tatweel and the Arabic combining marks.
28pub fn is_word_char(c: char) -> bool {
29    c.is_alphanumeric()
30        || c == '_'
31        || c == '\''
32        || matches!(u32::from(c), 0x200C | 0x200D | 0x0640 | 0x0610..=0x061A | 0x064B..=0x065F | 0x0670 | 0x06D6..=0x06DC | 0x06DF..=0x06E8 | 0x06EA..=0x06ED | 0x08D3..=0x08FF | 0x0300..=0x036F)
33}
34
35/// Does `text` contain a right-to-left character?
36pub fn has_rtl(text: &str) -> bool {
37    text.chars().any(is_rtl_script)
38}
39
40/// Does the first strong directional character in `text` belong to a right-to-left script?
41///
42/// Follows UAX #9 rules P2 and P3: characters with strong bidirectional class AL or R
43/// (Arabic, Hebrew, Syriac, Thaana…) give `true`; strong L characters (Latin, Greek,
44/// Cyrillic, CJK…) give `false`. Neutral, weak and whitespace characters are skipped.
45pub fn first_strong_is_rtl(text: &str) -> bool {
46    for c in text.chars() {
47        if c == '\u{200F}' {
48            return true;
49        }
50        if c == '\u{200E}' {
51            return false;
52        }
53        if c.is_alphabetic() {
54            return is_rtl_script(c);
55        }
56    }
57    false
58}

60#[cfg(test)]
61mod tests {
62    use super::*;
63
64    #[test]
65    fn classifies_persian_and_latin() {
66        // Persian letters, the Persian-only letters (پ چ ژ گ ک ی), ZWNJ-free word, digits.
67        assert!("سلام".chars().all(is_rtl_script));
68        assert!("پچژگکی".chars().all(is_rtl_script));
69        assert!("۰۱۲۳۴۵۶۷۸۹".chars().all(is_rtl_script));
70        assert!(is_rtl_script('\u{064E}'), "fatha (a combining mark) is Arabic script");
71        assert!("שלום".chars().all(is_rtl_script));
72        assert!(!"Hello 123".chars().any(is_rtl_script));
73        assert!(!is_rtl_script('\u{200C}'), "ZWNJ is a format character, not a letter");
74        assert!(is_complex_script('क') && !is_rtl_script('क'));
75        assert!(has_rtl("Word ورد 2024") && !has_rtl("Word 2024"));
76        assert!("می‌خواهم".chars().all(is_word_char), "ZWNJ is inside the word");
77        assert!("کِتاب".chars().all(is_word_char), "so are harakat");
78        assert!(!is_word_char(' ') && !is_word_char('،'));
79        assert!(first_strong_is_rtl("سلام"));
80        assert!(first_strong_is_rtl("من ناوم هێڤاره"));
81        assert!(first_strong_is_rtl("  123. سلام"));
82        assert!(!first_strong_is_rtl("Hello سلام"));
83        assert!(!first_strong_is_rtl("123 Hello"));
84        assert!(!first_strong_is_rtl(""));
85    }
86}
