1//! Equations: the Office Math object model (OMML, ECMA-376 Part 1 §22.1) as a tree, its linear
2//! format (UnicodeMath, the plain text Word shows for an equation) and a parser for that format.
3//!
4//! An equation is a list of [`MNode`]s (an *argument*); structures hold arguments of their own
5//! (a fraction's numerator and denominator, a radical's degree and base…). Layout reads the tree;
6//! DOCX import builds it from OMML and keeps the source XML so a saved file carries the equation
7//! exactly as Word wrote it.
8
9use serde::{Deserialize, Serialize};

11use crate::doc::props::Rgb;

13/// A sequence of math nodes (OMML's `m:e`, `m:num`, `m:sub`…).

25/// An equation's structure, plus what's needed to write it back.
26#[derive(Clone, Debug, PartialEq, Default, Serialize, Deserialize)]
27pub struct Math {
28    #[serde(default, skip_serializing_if = "Vec::is_empty")]
29    pub nodes: Arg,
30    /// The source `m:oMath` element (DOCX import), written back unchanged on save. Empty for
31    /// equations made here.

39    pub linear: bool,
40}

46impl Math {
47    pub fn is_empty(&self) -> bool {
48        self.nodes.is_empty()
49    }
50
51    /// Rough heap bytes a clone copies: every node, its argument lists and its run text.
52    /// Iterative, so it costs at most the size of the structure.
53    pub(crate) fn heap_bytes(&self) -> usize {
54        let mut n = self.omml.len();
55        let mut todo: Vec<&MNode> = self.nodes.iter().collect();
56        while let Some(node) = todo.pop() {
57            n = n.saturating_add(std::mem::size_of::<MNode>());
58            if let MNode::Run(r) = node {
59                n = n.saturating_add(r.text.len()).saturating_add(r.font.as_ref().map_or(0, String::len));
60            }
61            each_arg(node, &mut |a| {
62                n = n.saturating_add(std::mem::size_of::<Arg>());
63                todo.extend(a);
64            });
65        }
66        n
67    }
68}

380/// Mathematical alphanumeric for `c` in a style and alphabet (Unicode block U+1D400).
381pub fn math_alnum(c: char, bold: bool, italic: bool, scr: MScr) -> char {
382    let off = |base: u32, k: u32| char::from_u32(base + k).unwrap_or(c);
383    let upper = c.is_ascii_uppercase();
384    let lower = c.is_ascii_lowercase();
385    if upper || lower {
386        let k = if upper { c as u32 - 'A' as u32 } else { c as u32 - 'a' as u32 };
387        // Letters that live in the Letterlike Symbols block.
388        let hole = |table: &[(char, char)]| table.iter().find(|(f, _)| *f == c).map(|(_, t)| *t);
389        let (base_u, base_l) = match (scr, bold, italic) {
390            (MScr::Roman, false, false) => return c,
391            (MScr::Roman, true, false) => (0x1D400, 0x1D41A),
392            (MScr::Roman, false, true) => {
393                if c == 'h' {
394                    return '\u{210E}';
395                }
396                (0x1D434, 0x1D44E)
397            }
398            (MScr::Roman, true, true) => (0x1D468, 0x1D482),
399            (MScr::Script, false, _) => {
400                if let Some(t) = hole(&[
401                    ('B', 'ℬ'),
402                    ('E', 'ℰ'),
403                    ('F', 'ℱ'),
404                    ('H', 'ℋ'),
405                    ('I', 'ℐ'),
406                    ('L', 'ℒ'),
407                    ('M', 'ℳ'),
408                    ('R', 'ℛ'),
409                    ('e', 'ℯ'),
410                    ('g', 'ℊ'),
411                    ('o', 'ℴ'),
412                ]) {
413                    return t;
414                }
415                (0x1D49C, 0x1D4B6)
416            }
417            (MScr::Script, true, _) => (0x1D4D0, 0x1D4EA),
418            (MScr::Fraktur, false, _) => {
419                if let Some(t) = hole(&[('C', 'ℭ'), ('H', 'ℌ'), ('I', 'ℑ'), ('R', 'ℜ'), ('Z', 'ℨ')]) {
420                    return t;
421                }
422                (0x1D504, 0x1D51E)
423            }
424            (MScr::Fraktur, true, _) => (0x1D56C, 0x1D586),
425            (MScr::DoubleStruck, _, _) => {
426                if let Some(t) = hole(&[('C', 'ℂ'), ('H', 'ℍ'), ('N', 'ℕ'), ('P', 'ℙ'), ('Q', 'ℚ'), ('R', 'ℝ'), ('Z', 'ℤ')]) {
427                    return t;
428                }
429                (0x1D538, 0x1D552)
430            }
431            (MScr::SansSerif, false, false) => (0x1D5A0, 0x1D5BA),
432            (MScr::SansSerif, true, false) => (0x1D5D4, 0x1D5EE),
433            (MScr::SansSerif, false, true) => (0x1D608, 0x1D622),
434            (MScr::SansSerif, true, true) => (0x1D63C, 0x1D656),
435            (MScr::Monospace, _, _) => (0x1D670, 0x1D68A),
436        };
437        return off(if upper { base_u } else { base_l }, k);
438    }
439    if c.is_ascii_digit() {
440        let k = c as u32 - '0' as u32;
441        return match (scr, bold) {
442            (MScr::Roman, true) => off(0x1D7CE, k),
443            (MScr::DoubleStruck, _) => off(0x1D7D8, k),
444            (MScr::SansSerif, false) => off(0x1D7E2, k),
445            (MScr::SansSerif, true) => off(0x1D7EC, k),
446            (MScr::Monospace, _) => off(0x1D7F6, k),
447            _ => c,
448        };
449    }
450    // Greek (roman alphabet only).
451    if scr == MScr::Roman && (bold || italic) {
452        let (cap, small) = match (bold, italic) {
453            (true, false) => (0x1D6A8, 0x1D6C2),
454            (false, true) => (0x1D6E2, 0x1D6FC),
455            _ => (0x1D71C, 0x1D736),
456        };
457        let u = c as u32;
458        if (0x391..=0x3A9).contains(&u) && u != 0x3A2 {
459            return off(cap, u - 0x391);
460        }
461        if (0x3B1..=0x3C9).contains(&u) {
462            return off(small, u - 0x3B1);
463        }
464        let extra = match c {
465            '∂' => Some(25),
466            'ϵ' => Some(26),
467            'ϑ' => Some(27),
468            'ϰ' => Some(28),
469            'ϕ' => Some(29),
470            'ϱ' => Some(30),
471            'ϖ' => Some(31),
472            _ => None,
473        };
474        if let Some(e) = extra
475            && italic
476        {
477            return off(small, e);
478        }
479    }
480    c
481}

486/// The equation as linear-format text (`x=(-b±√(b^2-4ac))/2a`).
487pub fn to_linear(nodes: &[MNode]) -> String {
488    let mut s = String::new();
489    lin_arg(nodes, &mut s, 0);
490    s
491}
492
493fn lin_arg(a: &[MNode], s: &mut String, depth: usize) {
494    if depth > MAX_DEPTH {
495        return;
496    }
497    for (i, n) in a.iter().enumerate() {
498        lin_node(n, s, depth + 1);
499        // A bare script or limit ends at a space: a letter, digit or period after it is its own item.
500        if matches!(n, MNode::Script { kind, .. } if *kind != ScriptKind::Pre) || matches!(n, MNode::Lim { .. }) {
501            let ends_bare = s.chars().next_back().is_some_and(|c| c.is_alphanumeric() || c == '.');
502            if ends_bare && a.get(i + 1).and_then(first_lin_char).is_some_and(|c| c.is_alphanumeric() || c == '.') {
503                s.push(' ');
504            }
505        }
506    }
507}

509/// The first character `n` is written with, when that is cheap to tell (letters and digits).
510fn first_lin_char(n: &MNode) -> Option<char> {
511    match n {
512        MNode::Run(r) => r.text.chars().next(),
513        MNode::Func { name, .. } => name.first().and_then(first_lin_char),
514        MNode::Script { kind, base, .. } if *kind != ScriptKind::Pre => base.first().and_then(first_lin_char),
515        MNode::Lim { e, .. } => e.first().and_then(first_lin_char),
516        MNode::Frac { num, .. } => num.first().and_then(first_lin_char),
517        _ => None,
518    }
519}

521/// An argument as an operand: bare when it's a single simple atom, parenthesised otherwise.
522fn lin_operand(a: &[MNode], s: &mut String, depth: usize) {
523    let mut inner = String::new();
524    lin_arg(a, &mut inner, depth);
525    let simple = match a {
526        [MNode::Run(r)] => {
527            let mut cs = r.text.chars();
528            match (cs.next(), cs.next()) {
529                (Some(c), None) => !"+-−=/^_ ".contains(c),
530                _ => r.text.chars().all(|c| c.is_alphanumeric() || c == '.'),
531            }
532        }
533        [MNode::Delim { .. }] => true,
534        _ => false,
535    };
536    if simple && !inner.is_empty() {
537        s.push_str(&inner);
538    } else {
539        s.push('(');
540        s.push_str(&inner);
541        s.push(')');
542    }
543}

545/// A script or limit operand. The parser reads a run of letters and digits up to the next space or
546/// operator, so `V_dd` is written bare as Word's Linear format shows it (`V_(dd)` reads back the same).
547fn lin_script(a: &[MNode], s: &mut String, depth: usize) {
548    // Bare only when the parser reads the whole run back, in the same style.
549    if let [MNode::Run(r)] = a
550        && r.text.chars().nth(1).is_some()
551    {
552        let upright_name = r.sty == Some(MSty::Plain) && crate::doc::math_linear::FUNCTION_NAMES.contains(&r.text.as_str());
553        let number = r.text.starts_with(|c: char| c.is_ascii_digit()) && r.text.chars().all(|c| c.is_alphanumeric() || c == '.');
554        let word = r.text.starts_with(char::is_alphabetic) && r.text.chars().all(char::is_alphanumeric);
555        let upright_other = r.sty == Some(MSty::Plain) && !upright_name;
556        if upright_other || (!number && !word) {
557            s.push('(');
558            s.push_str(&r.text);
559            s.push(')');
560            return;
561        }
562    }
563    lin_operand(a, s, depth);
564}

566fn lin_node(n: &MNode, s: &mut String, depth: usize) {
567    match n {
568        MNode::Run(r) => s.push_str(&r.text),
569        MNode::Frac { kind, num, den } => {
570            lin_operand(num, s, depth);
571            s.push(match kind {
572                FracKind::Skewed => '⁄',
573                FracKind::Linear => '∕',
574                FracKind::NoBar => '¦',
575                FracKind::Bar => '/',
576            });
577            lin_operand(den, s, depth);
578        }
579        MNode::Script { kind, base, sub, sup } => {
580            if *kind == ScriptKind::Pre {
581                s.push('_');
582                lin_script(sub, s, depth);
583                s.push('^');
584                lin_script(sup, s, depth);
585                s.push('▒');
586                lin_operand(base, s, depth);
587                return;
588            }
589            lin_operand(base, s, depth);
590            if matches!(kind, ScriptKind::Sub | ScriptKind::SubSup) {
591                s.push('_');
592                lin_script(sub, s, depth);
593            }
594            if matches!(kind, ScriptKind::Sup | ScriptKind::SubSup) {
595                s.push('^');
596                lin_script(sup, s, depth);
597            }
598        }
599        MNode::Rad { deg, deg_hide, e } => {
600            if *deg_hide || deg.is_empty() {
601                s.push('√');
602                lin_operand(e, s, depth);
603            } else {
604                s.push_str("√(");
605                lin_arg(deg, s, depth);
606                s.push('&');
607                lin_arg(e, s, depth);
608                s.push(')');
609            }
610        }
611        MNode::Nary { chr, sub_hide, sup_hide, sub, sup, e, .. } => {
612            s.push(*chr);
613            if !sub_hide && !sub.is_empty() {
614                s.push('_');
615                lin_script(sub, s, depth);
616            }
617            if !sup_hide && !sup.is_empty() {
618                s.push('^');
619                lin_script(sup, s, depth);
620            }
621            s.push('▒');
622            lin_operand(e, s, depth);
623        }
624        MNode::Delim { beg, end, elems, .. } => {
625            // A standard pair writes as itself; anything else uses explicit `├X … Y┤`.
626            let pair = matches!((beg, end), (Some(b), Some(e)) if crate::doc::math_linear::closer(*b) == Some(*e));
627            let invisible = beg.is_none() && end.is_none();
628            if pair {
629                s.extend(*beg);
630            } else if invisible {
631                s.push('〖');
632            } else {
633                s.push('├');
634                s.extend(*beg);
635            }
636            for (i, e) in elems.iter().enumerate() {
637                if i > 0 {
638                    s.push('│');
639                }
640                lin_arg(e, s, depth);
641            }
642            if pair {
643                s.extend(*end);
644            } else if invisible {
645                s.push('〗');
646            } else {
647                s.extend(*end);
648                s.push('┤');
649            }
650        }
651        MNode::Func { name, e } => {
652            lin_arg(name, s, depth);
653            s.push('\u{2061}');
654            lin_operand(e, s, depth);
655        }
656        MNode::Lim { upper, e, lim } => {
657            lin_operand(e, s, depth);
658            s.push(if *upper { '┴' } else { '┬' });
659            lin_script(lim, s, depth);
660        }
661        MNode::Acc { chr, e } => {
662            lin_operand(e, s, depth);
663            s.push(*chr);
664        }
665        MNode::Bar { top, e } => {
666            s.push(if *top { '¯' } else { '▁' });
667            lin_operand(e, s, depth);
668        }
669        MNode::BorderBox { e, .. } => {
670            s.push('▭');
671            lin_operand(e, s, depth);
672        }
673        MNode::Boxed { e } => {
674            s.push('□');
675            lin_operand(e, s, depth);
676        }
677        MNode::GroupChr { chr, e, .. } => {
678            s.push(*chr);
679            lin_operand(e, s, depth);
680        }
681        // A numbered equation (`E=mc^2#(1)`) is written as its one row.
682        MNode::EqArr { rows } if rows.len() == 1 && rows.first().is_some_and(|r| has_number_mark(r)) => {
683            if let Some(r) = rows.first() {
684                lin_arg(r, s, depth);
685            }
686        }
687        MNode::EqArr { rows } => {
688            s.push_str("█(");
689            for (i, r) in rows.iter().enumerate() {
690                if i > 0 {
691                    s.push('@');
692                }
693                lin_arg(r, s, depth);
694            }
695            s.push(')');
696        }
697        MNode::Matrix { rows, .. } => {
698            s.push_str("■(");
699            for (i, r) in rows.iter().enumerate() {
700                if i > 0 {
701                    s.push('@');
702                }
703                for (j, c) in r.iter().enumerate() {
704                    if j > 0 {
705                        s.push('&');
706                    }
707                    lin_arg(c, s, depth);
708                }
709            }
710            s.push(')');
711        }
712        MNode::Phant { e, .. } => {
713            s.push('⟡');
714            lin_operand(e, s, depth);
715        }
716    }
717}

768/// Merge adjacent runs with the same formatting; recurse into every structure.
769pub fn merge_runs(a: Arg) -> Arg {
770    let mut out: Arg = Vec::with_capacity(a.len());
771    for n in a {
772        let n = match n {
773            MNode::Run(r) => {
774                if let Some(MNode::Run(prev)) = out.last_mut()
775                    && prev.sty == r.sty
776                    && prev.scr == r.scr
777                    && prev.nor == r.nor
778                    && prev.lit == r.lit
779                    && prev.size == r.size
780                    && prev.color == r.color
781                    && prev.font == r.font
782                    && r.brk.is_none()
783                {
784                    prev.text.push_str(&r.text);
785                    continue;
786                }
787                MNode::Run(r)
788            }
789            other => crate::doc::math_linear::map_args(other, &merge_runs),
790        };
791        out.push(n);
792    }
793    out
794}

796/// Deepest node (a leaf counts 1) the parsers wrap further. Half of [`MAX_DEPTH`]: serde_json
797/// stops reading at 128 nested containers and a node costs two (object and array).
798pub(crate) const WRAP_DEPTH: usize = MAX_DEPTH / 2;
799
800/// Calls `f` with every argument of `n`.
801fn each_arg<'a>(n: &'a MNode, f: &mut impl FnMut(&'a [MNode])) {
802    match n {
803        MNode::Run(_) => {}
804        MNode::Frac { num, den, .. } => {
805            f(num);
806            f(den);
807        }
808        MNode::Script { base, sub, sup, .. } => {
809            f(base);
810            f(sub);
811            f(sup);
812        }
813        MNode::Rad { deg, e, .. } => {
814            f(deg);
815            f(e);
816        }
817        MNode::Nary { sub, sup, e, .. } => {
818            f(sub);
819            f(sup);
820            f(e);
821        }
822        MNode::Delim { elems, .. } => elems.iter().for_each(|a| f(a)),
823        MNode::Func { name, e } => {
824            f(name);
825            f(e);
826        }
827        MNode::Lim { e, lim, .. } => {
828            f(e);
829            f(lim);
830        }
831        MNode::Acc { e, .. }
832        | MNode::Bar { e, .. }
833        | MNode::BorderBox { e, .. }
834        | MNode::Boxed { e }
835        | MNode::GroupChr { e, .. }
836        | MNode::Phant { e, .. } => f(e),
837        MNode::EqArr { rows } => rows.iter().for_each(|a| f(a)),
838        MNode::Matrix { rows, .. } => rows.iter().flatten().for_each(|a| f(a)),
839    }
840}
841
842/// `n` is already nested too deeply to be wrapped in another structure. The parsers use this so
843/// `x^1^1^1…` stays shallow (a left-nested tree 10 000 deep overflows the stack in `Clone`,
844/// `Drop`, `PartialEq` and serde). Iterative, stops at the first too-deep path; costs at most the
845/// size of `n`, which the parsers' input caps bound.
846pub(crate) fn too_deep(n: &MNode) -> bool {
847    let mut stack: Vec<(&MNode, usize)> = vec![(n, 1)];
848    while let Some((n, d)) = stack.pop() {
849        if d >= WRAP_DEPTH {
850            return true;
851        }
852        each_arg(n, &mut |a| stack.extend(a.iter().map(|c| (c, d + 1))));
853    }
854    false
855}
856
857/// Any of `a` is too deep to be wrapped ([`too_deep`]).
858pub(crate) fn any_too_deep(a: &[MNode]) -> bool {
859    a.iter().any(too_deep)
860}
