345/// The character for a name (`alpha` → α), including `doubleR`, `scriptL`, `frakturg`.
346pub fn symbol(name: &str) -> Option<char> {
347    if let Some((_, c)) = SYMBOLS.iter().find(|(n, _)| *n == name) {
348        return Some(*c);
349    }
350    for (prefix, scr) in
351        [("double", crate::doc::math::MScr::DoubleStruck), ("script", crate::doc::math::MScr::Script), ("fraktur", crate::doc::math::MScr::Fraktur)]
352    {
353        if let Some(rest) = name.strip_prefix(prefix) {
354            let mut cs = rest.chars();
355            if let (Some(c), None) = (cs.next(), cs.next())
356                && c.is_ascii_alphanumeric()
357            {
358                let m = crate::doc::math::math_alnum(c, false, false, scr);
359                if m != c {
360                    return Some(m);
361                }
362            }
363        }
364    }
365    None
366}
367
368/// The (first) name of a character.
369pub fn name_of(c: char) -> Option<&'static str> {
370    SYMBOLS.iter().find(|(_, s)| *s == c).map(|(n, _)| *n)
371}

373#[cfg(test)]
374mod tests {
375    use super::*;
376
377    #[test]
378    fn names_resolve() {
379        assert_eq!(symbol("alpha"), Some('α'));
380        assert_eq!(symbol("doubleR"), Some('ℝ'));
381        assert_eq!(symbol("scriptL"), Some('ℒ'));
382        assert_eq!(symbol("nonesuch"), None);
383        assert_eq!(name_of('≤'), Some("le"));
384        // No name is listed twice.
385        let mut names: Vec<&str> = SYMBOLS.iter().map(|(n, _)| *n).collect();
386        names.sort_unstable();
387        let before = names.len();
388        names.dedup();
389        assert_eq!(before, names.len());
390    }
391}
