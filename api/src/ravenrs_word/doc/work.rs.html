1//! A count of the characters a paragraph's object lookups walk, the blocks
2//! [`crate::doc::Document::para_paths`] visits, and each block lookup by path with the table levels it
3//! descends, for the linear-time tests of the crates that use them: an input 8x the size must cost
4//! about 8x the steps, which a busy machine can't skew the way it skews a timing. Counted only in
5//! tests (this crate's, and other crates' through the `test-support` feature, which they turn on
6//! as a dev-dependency); in the app the count is nothing.
7
8#[cfg(any(test, feature = "test-support"))]
9thread_local! {
10    /// The steps counted on this thread.
11    static STEPS: std::cell::Cell<u64> = const { std::cell::Cell::new(0) };
12}
13
14/// Count `n` steps of a walk (nothing outside tests). Public for the walks other crates do over a
15/// paragraph's text themselves (Markdown and HTML comment export, for one).
16#[inline(always)]
17pub fn steps(n: usize) {
18    #[cfg(any(test, feature = "test-support"))]
19    STEPS.with(|s| s.set(s.get().saturating_add(n as u64)));
20    #[cfg(not(any(test, feature = "test-support")))]
21    let _ = n;
22}
23
24/// The steps `f` takes.
25#[cfg(any(test, feature = "test-support"))]
26pub fn count(f: impl FnOnce()) -> u64 {
27    let before = STEPS.with(std::cell::Cell::get);
28    f();
29    STEPS.with(std::cell::Cell::get).saturating_sub(before)
30}
