1//! Charts WordCraft makes and edits (Insert › Chart): the chart type, its data (categories and
2//! series), title, legend and data labels.
3//!
4//! A chart object ([`crate::doc::graphic::Graphic`] with `chart` set) keeps this model next to the
5//! items it is drawn with; the DOCX writer turns it into a chart part (`c:chartSpace`) with the
6//! data cached in the part itself, and the reader turns a chart part back into it when the part
7//! holds nothing this model leaves out. Charts from other programs stay as they were read.
8//!
9//! Every value from outside (commands, files) goes through [`ChartSpec::sanitize`]: series,
10//! points and text lengths are capped and values that aren't finite numbers become gaps.
11
12use serde::{Deserialize, Serialize};
13
14/// Most series in a chart.
15pub const MAX_SERIES: usize = 255;
16/// Most categories (points per series) in a chart.
17pub const MAX_POINTS: usize = 4000;
18/// Most cells (series × categories) in a chart: what the chart drawing reads at most.
19pub const MAX_CELLS: usize = 100_000;
20/// Longest title, series name or category label, characters.
21pub const MAX_TEXT: usize = 255;
22
23/// What a chart shows its data as.
24#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash, Serialize, Deserialize)]
25#[serde(rename_all = "camelCase")]
26pub enum ChartType {
27    /// Clustered columns.
28    #[default]
29    Column,
30    /// Stacked columns.
31    ColumnStacked,
32    /// Clustered horizontal bars.
33    Bar,
34    /// Stacked horizontal bars.
35    BarStacked,
36    Line,
37    Area,
38    Pie,
39    Doughnut,
40    /// XY scatter with markers: the categories are the x values.
41    Scatter,
42}
43
44impl ChartType {
45    pub const ALL: [ChartType; 9] = [
46        ChartType::Column,
47        ChartType::ColumnStacked,
48        ChartType::Bar,
49        ChartType::BarStacked,
50        ChartType::Line,
51        ChartType::Area,
52        ChartType::Pie,
53        ChartType::Doughnut,
54        ChartType::Scatter,
55    ];
56
57    /// The id commands use (`column`, `barStacked`…).
58    pub fn id(self) -> &'static str {
59        match self {
60            ChartType::Column => "column",
61            ChartType::ColumnStacked => "columnStacked",
62            ChartType::Bar => "bar",
63            ChartType::BarStacked => "barStacked",
64            ChartType::Line => "line",
65            ChartType::Area => "area",
66            ChartType::Pie => "pie",
67            ChartType::Doughnut => "doughnut",
68            ChartType::Scatter => "scatter",
69        }
70    }
71
72    /// The type an id names (any ASCII case).
73    pub fn from_id(id: &str) -> Option<ChartType> {
74        ChartType::ALL.into_iter().find(|t| t.id().eq_ignore_ascii_case(id.trim()))
75    }
76
77    /// The name shown for it.
78    pub fn label(self) -> &'static str {
79        match self {
80            ChartType::Column => "Clustered Column",
81            ChartType::ColumnStacked => "Stacked Column",
82            ChartType::Bar => "Clustered Bar",
83            ChartType::BarStacked => "Stacked Bar",
84            ChartType::Line => "Line",
85            ChartType::Area => "Area",
86            ChartType::Pie => "Pie",
87            ChartType::Doughnut => "Doughnut",
88            ChartType::Scatter => "Scatter",
89        }
90    }
91
92    /// Pie and doughnut charts show one series, a slice per category.
93    pub fn is_round(self) -> bool {
94        matches!(self, ChartType::Pie | ChartType::Doughnut)
95    }
96}

126/// One series: its name and a value per category (`None` = no value, a gap).
127#[derive(Clone, Debug, Default, PartialEq, Serialize, Deserialize)]
128#[serde(rename_all = "camelCase")]
129pub struct ChartSeries {
130    pub name: String,
131    #[serde(default)]
132    pub values: Vec<Option<f64>>,
133}

135/// A chart WordCraft can edit.
136#[derive(Clone, Debug, Default, PartialEq, Serialize, Deserialize)]
137#[serde(rename_all = "camelCase")]
138pub struct ChartSpec {
139    #[serde(rename = "type")]
140    pub kind: ChartType,
141    /// Category labels (scatter charts: the x values).
142    #[serde(default)]
143    pub categories: Vec<String>,
144    #[serde(default)]
145    pub series: Vec<ChartSeries>,
146    #[serde(default)]
147    pub title: Option<String>,
148    /// `None`: no legend.
149    #[serde(default)]
150    pub legend: Option<LegendPos>,
151    /// Show each point's value next to it.
152    #[serde(default)]
153    pub data_labels: bool,
154}
155
156impl ChartSpec {
157    /// A new chart of type `kind` with WordCraft's sample data: four quarters and three series
158    /// (one for pie and doughnut charts, x values for scatter charts), legend at the bottom.
159    pub fn sample(kind: ChartType) -> ChartSpec {
160        let (categories, series): (Vec<&str>, Vec<(&str, [f64; 4])>) = match kind {
161            ChartType::Pie | ChartType::Doughnut => (vec!["North", "East", "South", "West"], vec![("Share", [38.0, 24.0, 21.0, 17.0])]),
162            ChartType::Scatter => (vec!["1", "2", "3", "4"], vec![("Series 1", [2.1, 3.4, 3.0, 4.6]), ("Series 2", [1.2, 1.9, 2.8, 3.1])]),
163            _ => (
164                vec!["Q1", "Q2", "Q3", "Q4"],
165                vec![("Series 1", [3.2, 4.1, 3.6, 5.0]), ("Series 2", [2.0, 2.7, 3.9, 3.3]), ("Series 3", [1.5, 2.2, 2.4, 3.8])],
166            ),
167        };
168        ChartSpec {
169            kind,
170            categories: categories.into_iter().map(str::to_string).collect(),
171            series: series.into_iter().map(|(n, v)| ChartSeries { name: n.to_string(), values: v.into_iter().map(Some).collect() }).collect(),
172            title: None,
173            legend: Some(LegendPos::Bottom),
174            data_labels: false,
175        }
176    }
177
178    /// Number of categories: every series has as many values.
179    pub fn points(&self) -> usize {
180        self.categories.len()
181    }
182
183    /// Make the chart safe to store and draw: text cut to [`MAX_TEXT`] characters with control
184    /// characters turned into spaces and the ends trimmed (an empty title is none); at most
185    /// [`MAX_POINTS`] categories, [`MAX_SERIES`] series and [`MAX_CELLS`] cells; values that
186    /// aren't finite become gaps; every series as long as the categories (padded with gaps,
187    /// categories padded with empty labels to the longest series).
188    pub fn sanitize(&mut self) {
189        self.series.truncate(MAX_SERIES);
190        let longest = self.series.iter().map(|s| s.values.len()).max().unwrap_or(0);
191        let n = self.categories.len().max(longest).min(MAX_POINTS);
192        let keep = MAX_CELLS.checked_div(n).unwrap_or(MAX_SERIES).clamp(1, MAX_SERIES);
193        self.series.truncate(keep);
194        self.categories.truncate(n);
195        for c in &mut self.categories {
196            *c = clean_text(c);
197        }
198        self.categories.resize(n, String::new());
199        for s in &mut self.series {
200            s.name = clean_text(&s.name);
201            s.values.truncate(n);
202            for v in &mut s.values {
203                *v = v.filter(|x| x.is_finite());
204            }
205            s.values.resize(n, None);
206        }
207        self.title = self.title.as_deref().map(clean_text).filter(|t| !t.is_empty());
208    }
209
210    /// The chart after `sanitize`.
211    pub fn sanitized(mut self) -> ChartSpec {
212        self.sanitize();
213        self
214    }
215}
216
217/// `s` as chart text: control characters as spaces, ends trimmed, at most [`MAX_TEXT`] characters.
218pub fn clean_text(s: &str) -> String {
219    let s: String = s.chars().take(MAX_TEXT * 4).map(|c| if c.is_control() { ' ' } else { c }).collect();
220    s.trim().chars().take(MAX_TEXT).collect::<String>().trim_end().to_string()
221}

223#[cfg(test)]
224mod tests {
225    use super::*;
226
227    #[test]
228    fn sanitize_caps_sizes_and_drops_non_finite_values() {
229        let mut c = ChartSpec {
230            categories: vec!["a".into(); MAX_POINTS + 10],
231            series: (0..MAX_SERIES + 5)
232                .map(|i| ChartSeries { name: format!("{}\n{}", "x".repeat(400), i), values: vec![Some(f64::NAN), Some(f64::INFINITY), Some(1.0)] })
233                .collect(),
234            title: Some("  ".into()),
235            ..Default::default()
236        };
237        c.sanitize();
238        assert_eq!(c.points(), MAX_POINTS);
239        assert!(c.series.len() * c.points() <= MAX_CELLS);
240        assert!(!c.series.is_empty());
241        let s = &c.series[0];
242        assert_eq!(s.values.len(), MAX_POINTS);
243        assert_eq!(&s.values[..3], &[None, None, Some(1.0)]);
244        assert_eq!(s.name.chars().count(), MAX_TEXT);
245        assert!(!s.name.contains('\n'));
246        assert_eq!(c.title, None);
247    }
248
249    #[test]
250    fn short_categories_are_padded_to_the_longest_series() {
251        let c = ChartSpec {
252            categories: vec!["a".into()],
253            series: vec![ChartSeries { name: "s".into(), values: vec![Some(1.0), Some(2.0)] }],
254            ..Default::default()
255        }
256        .sanitized();
257        assert_eq!(c.categories, vec!["a".to_string(), String::new()]);
258        assert_eq!(ChartType::from_id("BARSTACKED"), Some(ChartType::BarStacked));
259        assert_eq!(LegendPos::from_id("top"), Some(LegendPos::Top));
260    }
