7/// How a section's text sits on its pages, top to bottom (`w:vAlign`, ECMA-376 §17.6.23,
8/// ST_VerticalJc). Table cells and text boxes use [`crate::doc::props::VAlign`], which has no
9/// Justified.
10#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash, Serialize, Deserialize, Default)]
11#[serde(rename_all = "camelCase")]
12pub enum PageVAlign {
13    #[default]
14    Top,
15    Center,
16    /// Justified (`both`): the room left on each page is shared out equally between the page's
17    /// paragraphs and tables, so the text runs from the top margin to the bottom one.
18    #[serde(alias = "both", alias = "justified")]
19    Justify,
20    Bottom,
21}
22
23impl PageVAlign {
24    /// From a command or file value: `top|center|justify|bottom` (also `both`, `justified`,
25    /// `middle`); anything else is Top.
26    pub fn parse(v: &str) -> PageVAlign {
27        match v {
28            "center" | "middle" => PageVAlign::Center,
29            "justify" | "justified" | "both" => PageVAlign::Justify,
30            "bottom" => PageVAlign::Bottom,
31            _ => PageVAlign::Top,
32        }
33    }
34}

360/// `n` in Thai digits, ๐ (U+0E50) to ๙ (#5757).
361fn thai_digits(n: u32) -> String {
362    n.to_string().chars().filter_map(|c| c.to_digit(10).and_then(|d| char::from_u32(0x0E50 + d))).collect()
363}

365/// Chicago Manual of Style note marks: *, †, ‡, §, then doubled (**, ††, …) and so on.
366fn chicago(n: u32) -> String {
367    const MARKS: [char; 4] = ['*', '\u{2020}', '\u{2021}', '\u{a7}'];
368    if n == 0 {
369        return String::new();
370    }
371    let i = (n - 1) as usize;
372    let k = (i / 4 + 1).min(10);
373    std::iter::repeat_n(MARKS.get(i % 4).copied().unwrap_or('*'), k).collect()
374}

376fn ordinal_suffix(n: u32) -> &'static str {
377    match (n % 10, n % 100) {
378        (_, 11..=13) => "th",
379        (1, _) => "st",
380        (2, _) => "nd",
381        (3, _) => "rd",
382        _ => "th",
383    }
384}

386/// Lowercase roman numerals (0 → empty; values above 3999 use repeated `m`).
387pub fn roman(mut n: u32) -> String {
388    const T: [(u32, &str); 13] = [
389        (1000, "m"),
390        (900, "cm"),
391        (500, "d"),
392        (400, "cd"),
393        (100, "c"),
394        (90, "xc"),
395        (50, "l"),
396        (40, "xl"),
397        (10, "x"),
398        (9, "ix"),
399        (5, "v"),
400        (4, "iv"),
401        (1, "i"),
402    ];
403    let mut s = String::new();
404    n = n.min(50_000);
405    for (v, r) in T {
406        while n >= v {
407            s.push_str(r);
408            n -= v;
409        }
410    }
411    s
412}

414/// Word-style letters: a..z, aa..zz, aaa.. (repeated letter, not base-26).
415pub fn letters(n: u32) -> String {
416    if n == 0 {
417        return String::new();
418    }
419    let n = n.min(780);
420    let k = (n - 1) / 26 + 1;
421    let c = (b'a' + ((n - 1) % 26) as u8) as char;
422    std::iter::repeat_n(c, k as usize).collect()
423}

449/// One letter of `alphabet`, repeated once per pass through it (as [`letters`]: a..z, aa..zz).
450fn repeated_letter(alphabet: &[char], n: u32) -> String {
451    let Some(&first) = alphabet.first() else { return n.to_string() };
452    if n == 0 {
453        return String::new();
454    }
455    let len = alphabet.len() as u32;
456    let n = n.min(len * 30);
457    let k = (n - 1) / len + 1;
458    let c = alphabet.get(((n - 1) % len) as usize).copied().unwrap_or(first);
459    std::iter::repeat_n(c, k as usize).collect()
460}

486/// English cardinal words, capitalised first letter (`Twenty-One`).
487pub fn cardinal_text(n: u32) -> String {
488    fn words(n: u32) -> String {
489        match n {
490            0..=19 => ONES.get(n as usize).copied().unwrap_or("").to_string(),
491            20..=99 => {
492                let t = TENS.get((n / 10) as usize).copied().unwrap_or("");
493                if n.is_multiple_of(10) { t.to_string() } else { format!("{t}-{}", ONES.get((n % 10) as usize).copied().unwrap_or("")) }
494            }
495            100..=999 => {
496                let h = format!("{} hundred", words(n / 100));
497                if n.is_multiple_of(100) { h } else { format!("{h} {}", words(n % 100)) }
498            }
499            _ => {
500                let th = format!("{} thousand", words(n / 1000));
501                if n.is_multiple_of(1000) { th } else { format!("{th} {}", words(n % 1000)) }
502            }
503        }
504    }
505    capitalize(&words(n.min(999_999)))
506}

508/// English ordinal words (`Twenty-First`).
509pub fn ordinal_text(n: u32) -> String {
510    let c = cardinal_text(n).to_lowercase();
511    let (head, last) = match c.rfind(['-', ' ']) {
512        Some(i) => (c.get(..=i).unwrap_or(""), c.get(i + 1..).unwrap_or("")),
513        None => ("", c.as_str()),
514    };
515    let last = match last {
516        "one" => "first".to_string(),
517        "two" => "second".to_string(),
518        "three" => "third".to_string(),
519        "five" => "fifth".to_string(),
520        "eight" => "eighth".to_string(),
521        "nine" => "ninth".to_string(),
522        "twelve" => "twelfth".to_string(),
523        l if l.ends_with('y') => format!("{}ieth", l.trim_end_matches('y')),
524        l => format!("{l}th"),
525    };
526    capitalize(&format!("{head}{last}"))
527}

529fn capitalize(s: &str) -> String {
530    let mut out = String::with_capacity(s.len());
531    let mut up = true;
532    for c in s.chars() {
533        if up {
534            out.extend(c.to_uppercase());
535        } else {
536            out.push(c);
537        }
538        up = c == '-' || c == ' ';
539    }
540    out
541}

715/// Smallest and largest line or character pitch the layout uses, points.
716pub const MIN_GRID_PITCH: f32 = 1.0;
717pub const MAX_GRID_PITCH: f32 = 1584.0;
718/// `w:charSpace` bound (4096ths of a point): ±1584 pt.
719pub const MAX_CHAR_SPACE: i32 = 1584 * 4096;
720
721/// The document grid (`w:docGrid`): lines per page and characters per line, as pitches.
722#[derive(Clone, Copy, Debug, PartialEq, Serialize, Deserialize)]
723#[serde(default, rename_all = "camelCase")]
724pub struct DocGrid {
725    #[serde(rename = "type")]
726    pub kind: DocGridType,
727    /// Distance between grid lines, points (`w:linePitch`, twips in the file).
728    pub line_pitch: f32,
729    /// Added to the default font size to give the character pitch, 4096ths of a point
730    /// (`w:charSpace`).
731    pub char_space: i32,
732}
733
734impl Default for DocGrid {
735    /// Word's grid for a new document: no grid, an 18 pt pitch kept for when one is turned on.
736    fn default() -> Self {
737        DocGrid { kind: DocGridType::Default, line_pitch: 18.0, char_space: 0 }
738    }
739}
740
741impl DocGrid {
742    /// The line pitch, clamped to something layout can use.
743    pub fn line_pitch(&self) -> f32 {
744        if self.line_pitch.is_finite() { self.line_pitch.clamp(MIN_GRID_PITCH, MAX_GRID_PITCH) } else { 18.0 }
745    }
746    /// The character pitch for a document whose default font is `font_size` points.
747    pub fn char_pitch(&self, font_size: f32) -> f32 {
748        let size = if font_size.is_finite() { font_size.clamp(MIN_GRID_PITCH, MAX_GRID_PITCH) } else { 10.5 };
749        (size + self.char_space.clamp(-MAX_CHAR_SPACE, MAX_CHAR_SPACE) as f32 / 4096.0).clamp(MIN_GRID_PITCH, MAX_GRID_PITCH)
750    }
751    /// The `w:charSpace` that gives `pitch` with the default font at `font_size` points.
752    pub fn char_space_for(pitch: f32, font_size: f32) -> i32 {
753        let pitch = if pitch.is_finite() { pitch.clamp(MIN_GRID_PITCH, MAX_GRID_PITCH) } else { font_size };
754        let size = if font_size.is_finite() { font_size.clamp(MIN_GRID_PITCH, MAX_GRID_PITCH) } else { 10.5 };
755        (((pitch - size) * 4096.0).round() as i32).clamp(-MAX_CHAR_SPACE, MAX_CHAR_SPACE)
756    }
757}

892impl SectionProps {
893    /// Join the section before this one into it, as when the break between them is deleted:
894    /// this section's setup stays, as in Word, and the headers and footers it leaves out (linked
895    /// to the previous section) are taken from `before`, so its text keeps showing them.
896    pub fn absorb(&mut self, before: &SectionProps) {
897        for (own, prev) in [(&mut self.headers, &before.headers), (&mut self.footers, &before.footers)] {
898            own.default = own.default.or(prev.default);
899            own.first = own.first.or(prev.first);
900            own.even = own.even.or(prev.even);
901        }
902    }
903    /// Width of the text area, points.
904    pub fn text_width(&self) -> f32 {
905        (self.page_w - self.margin_left - self.margin_right - self.gutter).max(18.0)
906    }
907    pub fn text_height(&self) -> f32 {
908        (self.page_h - self.margin_top - self.margin_bottom).max(18.0)
909    }
910    /// Swap to landscape/portrait (keeps margins relative to the paper as Word does: rotates them).
911    pub fn set_landscape(&mut self, on: bool) {
912        if on == self.landscape {
913            return;
914        }
915        self.landscape = on;
916        std::mem::swap(&mut self.page_w, &mut self.page_h);
917        let (t, b, l, r) = (self.margin_top, self.margin_bottom, self.margin_left, self.margin_right);
918        if on {
919            self.margin_top = l;
920            self.margin_bottom = r;
921            self.margin_left = b;
922            self.margin_right = t;
923        } else {
924            self.margin_left = t;
925            self.margin_right = b;
926            self.margin_bottom = l;
927            self.margin_top = r;
928        }
929    }
930    /// The line grid's pitch, points, when the section has a line grid (any grid but "no grid").
931    pub fn grid_line_pitch(&self) -> Option<f32> {
932        self.doc_grid.filter(|g| g.kind != DocGridType::Default).map(|g| g.line_pitch())
933    }
934    /// The character grid's pitch, points, when the section has one (`linesAndChars`,
935    /// `snapToChars`); `font_size` is the default font size (Normal style).
936    pub fn grid_char_pitch(&self, font_size: f32) -> Option<f32> {
937        self.doc_grid.filter(|g| matches!(g.kind, DocGridType::LinesAndChars | DocGridType::SnapToChars)).map(|g| g.char_pitch(font_size))
938    }
939    /// Lines per page the line pitch gives (at least 1).
940    pub fn grid_lines_per_page(&self) -> u32 {
941        let pitch = self.doc_grid.unwrap_or_default().line_pitch();
942        ((self.text_height() / pitch + 0.001).floor() as u32).max(1)
943    }
944    /// Characters per line the character pitch gives (at least 1).
945    pub fn grid_chars_per_line(&self, font_size: f32) -> u32 {
946        let pitch = self.doc_grid.unwrap_or_default().char_pitch(font_size);
947        ((self.text_width() / pitch + 0.001).floor() as u32).max(1)
948    }
949    /// Column (x offset from the text area's left, width) for each column, in reading order:
950    /// in a right-to-left section (`rtl`, `w:bidi`) the first column stands at the right edge
951    /// and the columns (with the space after each) run leftwards.
952    pub fn column_boxes(&self) -> Vec<(f32, f32)> {
953        let tw = self.text_width();
954        let n = self.columns.count.clamp(1, 45) as usize;
955        let boxes: Vec<(f32, f32)> = if !self.columns.widths.is_empty() && self.columns.widths.len() == n {
956            let mut x = 0.0;
957            self.columns
958                .widths
959                .iter()
960                .map(|(w, sp)| {
961                    let b = (x, w.max(18.0));
962                    x += w + sp;
963                    b
964                })
965                .collect()
966        } else {
967            let space = self.columns.space.max(0.0);
968            let w = ((tw - space * (n as f32 - 1.0)) / n as f32).max(18.0);
969            (0..n).map(|i| (i as f32 * (w + space), w)).collect()
970        };
971        if self.rtl { boxes.into_iter().map(|(x, w)| (tw - x - w, w)).collect() } else { boxes }
972    }
973}
