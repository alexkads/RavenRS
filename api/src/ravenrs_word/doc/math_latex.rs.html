1//! LaTeX ↔ math tree (Word's LaTeX equation input and conversion).
2//!
3//! Input covers the commands people type in equations: `\frac`, `\sqrt[n]{}`, scripts, big
4//! operators with limits, `\left…\middle…\right`, matrix and `cases`/`aligned` environments,
5//! accents, `\overbrace`/`\underbrace`, `\text`/`\mathrm`/`\mathbf`/`\mathbb`…, function names,
6//! spacing and every named symbol in [`crate::doc::math_symbols`]. Unknown commands stay as text.

14#[derive(Clone, Debug, PartialEq)]
15enum Tok {
16    Cmd(String),
17    Open,
18    Close,
19    Sup,
20    Sub,
21    Amp,
22    Row,
23    Ch(char),
24}

26/// Commands whose `{…}` argument is text, where spaces are part of the content.
27const TEXT_COMMANDS: &[&str] = &["text", "textrm", "mbox", "textnormal", "textbf", "textit"];
28
29fn tokenize(s: &str) -> Vec<Tok> {
30    let cs: Vec<char> = s.chars().take(20_000).collect();
31    let mut out = Vec::new();
32    let mut i = 0;
33    // A text command was just read: its `{` starts text, where spaces count (#448).
34    let mut text_next = false;
35    // Brace depth inside a text argument.
36    let mut text_depth: Option<u32> = None;
37    while let Some(&c) = cs.get(i) {
38        i += 1;
39        if text_depth.is_some() && matches!(c, ' ' | '\t' | '\n' | '\r') {
40            // A run of white space is one space, as in LaTeX.
41            if out.last() != Some(&Tok::Ch(' ')) {
42                out.push(Tok::Ch(' '));
43            }
44            continue;
45        }
46        let was_text_next = std::mem::take(&mut text_next);
47        match c {
48            '\\' => {
49                let start = i;
50                while cs.get(i).is_some_and(char::is_ascii_alphabetic) {
51                    i += 1;
52                }
53                if i == start {
54                    match cs.get(i) {
55                        Some('\\') => {
56                            i += 1;
57                            out.push(Tok::Row);
58                        }
59                        Some(&c) => {
60                            i += 1;
61                            out.push(Tok::Cmd(c.to_string()));
62                        }
63                        None => {}
64                    }
65                } else {
66                    let name: String = cs.get(start..i).unwrap_or(&[]).iter().collect();
67                    text_next = text_depth.is_none() && TEXT_COMMANDS.contains(&name.as_str());
68                    out.push(Tok::Cmd(name));
69                }
70            }
71            '{' => {
72                text_depth = match text_depth {
73                    Some(d) => Some(d.saturating_add(1)),
74                    None if was_text_next => Some(0),
75                    None => None,
76                };
77                out.push(Tok::Open);
78            }
79            '}' => {
80                text_depth = match text_depth {
81                    Some(0) | None => None,
82                    Some(d) => Some(d.saturating_sub(1)),
83                };
84                out.push(Tok::Close);
85            }
86            '^' => out.push(Tok::Sup),
87            '_' => out.push(Tok::Sub),
88            '&' => out.push(Tok::Amp),
89            // White space between a text command and its `{` doesn't end the wait.
90            ' ' | '\t' | '\n' | '\r' => text_next = was_text_next,
91            '~' => out.push(Tok::Ch('\u{A0}')),
92            c => out.push(Tok::Ch(c)),
93        }
94    }
95    out
96}

98/// Parse LaTeX math into nodes. Never fails.
99pub fn parse_latex(s: &str) -> Arg {
100    let mut p = Lp { t: tokenize(s), i: 0, depth: 0 };
101    let mut out = Vec::new();
102    while p.i < p.t.len() {
103        out.extend(p.seq(false));
104        // A stray `}` or `&`/`\\` at the top level.
105        match p.t.get(p.i) {
106            Some(Tok::Amp) => out.push(run("&")),
107            Some(Tok::Row) => {}
108            Some(Tok::Close) => {}
109            _ => {}
110        }
111        p.i += 1;
112    }
113    let out = merge_runs(out);
114    if out.iter().any(|n| matches!(n, MNode::Run(r) if !r.nor && r.text.contains('#'))) {
115        return vec![MNode::EqArr { rows: vec![out] }];
116    }
117    out
118}

120fn run(t: &str) -> MNode {
121    MNode::Run(MRun::new(t))

124struct Lp {

126    i: usize,

130impl Lp {
131    fn peek(&self) -> Option<&Tok> {
132        self.t.get(self.i)
133    }

135    /// A sequence up to `}` (in a group), `&`, `\\`, `\right`, `\middle` or `\end`.
136    fn seq(&mut self, in_group: bool) -> Arg {
137        self.depth += 1;
138        let mut out: Arg = Vec::new();
139        if self.depth > MAX_DEPTH {
140            self.i = self.t.len();
141            self.depth -= 1;
142            return out;
143        }
144        while let Some(tok) = self.peek().cloned() {
145            match tok {
146                Tok::Close if in_group => break,
147                Tok::Close | Tok::Amp | Tok::Row => break,
148                Tok::Cmd(ref c) if matches!(c.as_str(), "right" | "middle" | "end") => break,
149                Tok::Sup | Tok::Sub => {
150                    self.i += 1;
151                    let s = self.arg();
152                    if out.last().is_some_and(too_deep) {
153                        // Too deep to wrap (`x^1^1^1…`): the operator is dropped, the script kept.
154                        out.extend(s);
155                    } else {
156                        let base = out.pop();
157                        out.push(script(base, tok == Tok::Sup, s));
158                    }
159                }
160                Tok::Cmd(ref c) if c == "limits" || c == "nolimits" || c == "displaystyle" || c == "textstyle" => {
161                    self.i += 1;
162                    if c == "nolimits"
163                        && let Some(MNode::Nary { lim_loc, .. }) = out.last_mut()
164                    {
165                        *lim_loc = Some(crate::doc::math::LimLoc::SubSup);
166                    }
167                    if c == "limits"
168                        && let Some(MNode::Nary { lim_loc, .. }) = out.last_mut()
169                    {
170                        *lim_loc = Some(crate::doc::math::LimLoc::UndOvr);
171                    }
172                }
173                _ => {
174                    let a = self.atom();
175                    out.extend(a);
176                }
177            }
178        }
179        self.depth -= 1;
180        // Big operators take the following term as their operand.
181        nary_operands(out, self.depth)
182    }

184    /// A command argument: a `{group}` or one token.
185    fn arg(&mut self) -> Arg {
186        match self.peek() {
187            Some(Tok::Open) => {
188                self.i += 1;
189                let g = self.seq(true);
190                if self.peek() == Some(&Tok::Close) {
191                    self.i += 1;
192                }
193                g
194            }
195            Some(_) => self.atom(),
196            None => Vec::new(),
197        }
198    }
199
200    /// `[optional]` argument.
201    fn opt(&mut self) -> Option<Arg> {
202        if self.peek() != Some(&Tok::Ch('[')) {
203            return None;
204        }
205        self.i += 1;
206        let mut toks = Vec::new();
207        let mut level = 0i32;
208        while let Some(t) = self.peek().cloned() {
209            self.i += 1;
210            match t {
211                Tok::Ch(']') if level == 0 => break,
212                Tok::Open => level += 1,
213                Tok::Close => level -= 1,
214                _ => {}
215            }
216            toks.push(t);
217        }
218        let mut p = Lp { t: toks, i: 0, depth: self.depth };
219        Some(p.seq(false))
220    }
221
222    /// The text of a `{…}` argument, verbatim.
223    fn raw_arg(&mut self) -> String {
224        let mut s = String::new();
225        if self.peek() != Some(&Tok::Open) {
226            if let Some(Tok::Ch(c)) = self.peek().cloned() {
227                self.i += 1;
228                s.push(c);
229            }
230            return s;
231        }
232        self.i += 1;
233        let mut level = 0i32;
234        // The previous token was a control word (`\backslash{}`: the empty braces end its name).
235        let mut after_word = false;
236        while let Some(t) = self.peek().cloned() {
237            self.i += 1;
238            let was_word = std::mem::replace(&mut after_word, matches!(&t, Tok::Cmd(c) if c.chars().all(|c| c.is_ascii_alphabetic())));
239            match t {
240                Tok::Close if level == 0 => break,
241                Tok::Open if was_word && self.peek() == Some(&Tok::Close) => self.i += 1,
242                Tok::Open => {
243                    level += 1;
244                    s.push('{');
245                }
246                Tok::Close => {
247                    level -= 1;
248                    s.push('}');
249                }
250                Tok::Ch(c) => s.push(c),
251                Tok::Cmd(c) => {
252                    if c.chars().all(|c| c.is_ascii_alphabetic()) {
253                        match symbol(&c) {
254                            Some(sym) => s.push(sym),
255                            None => s.push_str(&c),
256                        }
257                    } else {
258                        s.push_str(&c);
259                    }
260                    if c == " " {
261                        s.push(' ');
262                    }
263                }
264                Tok::Sup => s.push('^'),
265                Tok::Sub => s.push('_'),
266                Tok::Amp => s.push('&'),
267                Tok::Row => s.push('\n'),
268            }
269        }
270        s
271    }

273    /// A delimiter after `\left`, `\right`, `\middle`, `\big…`.
274    fn delim(&mut self) -> Option<char> {
275        let t = self.peek().cloned()?;
276        self.i += 1;
277        match t {
278            Tok::Ch('.') => None,
279            Tok::Ch(c) => Some(c),
280            Tok::Cmd(c) => match c.as_str() {
281                "{" => Some('{'),
282                "}" => Some('}'),
283                "|" => Some('‖'),
284                other => symbol(other),
285            },
286            _ => None,
287        }
288    }

290    fn atom(&mut self) -> Arg {
291        if self.depth > MAX_DEPTH {
292            self.i = self.t.len();
293            return Vec::new();
294        }
295        self.depth += 1;
296        let out = self.atom_inner();
297        self.depth -= 1;
298        out
299    }
300
301    fn atom_inner(&mut self) -> Arg {
302        let Some(t) = self.peek().cloned() else { return Vec::new() };
303        self.i += 1;
304        match t {
305            Tok::Open => {
306                let g = self.seq(true);
307                if self.peek() == Some(&Tok::Close) {
308                    self.i += 1;
309                }
310                g
311            }
312            Tok::Ch(c) => vec![run(&c.to_string())],
313            Tok::Close | Tok::Amp | Tok::Row | Tok::Sup | Tok::Sub => Vec::new(),
314            Tok::Cmd(c) => self.command(&c),
315        }
316    }

318    fn command(&mut self, c: &str) -> Arg {
319        let styled = |a: Arg, sty: Option<MSty>, scr: MScr, nor: bool| -> Arg { restyle(a, sty, scr, nor) };
320        let one = |n: MNode| vec![n];
321        match c {
322            "frac" | "dfrac" | "tfrac" | "cfrac" => {
323                let num = self.arg();
324                let den = self.arg();
325                one(MNode::Frac { kind: FracKind::Bar, num, den })
326            }
327            "genfrac" => {
328                let beg = delim_arg(&self.arg());
329                let end = delim_arg(&self.arg());
330                let thickness = self.arg();
331                let _style = self.arg();
332                let num = self.arg();
333                let den = self.arg();
334                let kind = if is_zero_length(&thickness) { FracKind::NoBar } else { FracKind::Bar };
335                let frac = MNode::Frac { kind, num, den };
336                if beg.is_none() && end.is_none() {
337                    one(frac)
338                } else {
339                    one(MNode::Delim { beg, end, sep: None, grow: true, shp_match: false, elems: vec![vec![frac]] })
340                }
341            }
342            "binom" | "dbinom" | "tbinom" => {
343                let n = self.arg();
344                let k = self.arg();
345                one(MNode::Delim {
346                    beg: Some('('),
347                    end: Some(')'),
348                    sep: None,
349                    grow: true,
350                    shp_match: false,
351                    elems: vec![vec![MNode::Frac { kind: FracKind::NoBar, num: n, den: k }]],
352                })
353            }
354            "sqrt" => {
355                let deg = self.opt().unwrap_or_default();
356                let e = self.arg();
357                one(MNode::Rad { deg_hide: deg.is_empty(), deg, e })
358            }
359            "left" => {
360                let beg = self.delim();
361                let mut elems = vec![self.seq(false)];
362                let mut sep = None;
363                let mut end = None;
364                while let Some(Tok::Cmd(k)) = self.peek().cloned() {
365                    self.i += 1;
366                    if k == "middle" {
367                        sep = self.delim().or(Some('|'));
368                        elems.push(self.seq(false));
369                    } else {
370                        if k == "right" {
371                            end = self.delim();
372                        }
373                        break;
374                    }
375                }
376                one(MNode::Delim { beg, end, sep, grow: true, shp_match: false, elems })
377            }
378            "bigl" | "Bigl" | "biggl" | "Biggl" | "bigr" | "Bigr" | "biggr" | "Biggr" | "big" | "Big" | "bigg" | "Bigg" => {
379                self.delim().map(|d| vec![run(&d.to_string())]).unwrap_or_default()
380            }
381            "begin" => {
382                let env = self.raw_arg();
383                self.environment(env.trim())
384            }
385            "text" | "textrm" | "mbox" | "textnormal" => {
386                let t = self.raw_arg();
387                one(MNode::Run(MRun { text: t, nor: true, sty: Some(MSty::Plain), ..Default::default() }))
388            }
389            "textbf" => {
390                let t = self.raw_arg();
391                one(MNode::Run(MRun { text: t, nor: true, sty: Some(MSty::Bold), ..Default::default() }))
392            }
393            "textit" => {
394                let t = self.raw_arg();
395                one(MNode::Run(MRun { text: t, nor: true, sty: Some(MSty::Italic), ..Default::default() }))
396            }
397            "operatorname" => {
398                let starred = self.peek() == Some(&Tok::Ch('*'));
399                if starred {
400                    self.i += 1;
401                }
402                let t = self.raw_arg();
403                let name = vec![MNode::Run(MRun::plain(t))];
404                if starred && self.peek() == Some(&Tok::Sub) {
405                    self.i += 1;
406                    let lim = self.arg();
407                    vec![MNode::Lim { upper: false, e: name, lim }]
408                } else {
409                    self.apply(name)
410                }
411            }
412            "tag" => {
413                let starred = self.peek() == Some(&Tok::Ch('*'));
414                if starred {
415                    self.i += 1;
416                }
417                let t = self.raw_arg();
418                let num_str = if starred { format!("#{t}") } else { format!("# ({t})") };
419                let mut r = MRun::new(num_str);
420                r.lit = false;
421                vec![MNode::Run(r)]
422            }
423            "mathrm" | "rm" => {
424                let a = self.arg();
425                styled(a, Some(MSty::Plain), MScr::Roman, false)
426            }
427            "mathbf" | "bf" => {
428                let a = self.arg();
429                styled(a, Some(MSty::Bold), MScr::Roman, false)
430            }
431            "mathit" => {
432                let a = self.arg();
433                styled(a, Some(MSty::Italic), MScr::Roman, false)
434            }
435            "boldsymbol" | "bm" => {
436                let a = self.arg();
437                styled(a, Some(MSty::BoldItalic), MScr::Roman, false)
438            }
439            "mathbb" => {
440                let a = self.arg();
441                styled(a, None, MScr::DoubleStruck, false)
442            }
443            "mathcal" | "mathscr" => {
444                let a = self.arg();
445                styled(a, None, MScr::Script, false)
446            }
447            "mathfrak" => {
448                let a = self.arg();
449                styled(a, None, MScr::Fraktur, false)
450            }
451            "mathsf" => {
452                let a = self.arg();
453                styled(a, None, MScr::SansSerif, false)
454            }
455            "mathtt" => {
456                let a = self.arg();
457                styled(a, None, MScr::Monospace, false)
458            }
459            "overline" => one(MNode::Bar { top: true, e: self.arg() }),
460            "underline" => one(MNode::Bar { top: false, e: self.arg() }),
461            "boxed" | "fbox" => one(MNode::BorderBox { hide: [false; 4], strike: [false; 4], e: self.arg() }),
462            "phantom" => one(MNode::Phant { show: false, zero_wid: false, zero_asc: false, zero_desc: false, e: self.arg() }),
463            "overbrace" | "underbrace" | "overparen" | "underparen" | "overbracket" | "underbracket" => {
464                let top = c.starts_with("over");
465                let chr = match c {
466                    "overbrace" => '⏞',
467                    "underbrace" => '⏟',
468                    "overparen" => '⏜',
469                    "underparen" => '⏝',
470                    "overbracket" => '⎴',
471                    _ => '⎵',
472                };
473                let e = self.arg();
474                let g = MNode::GroupChr { chr, top, e };
475                let want = if top { Tok::Sup } else { Tok::Sub };
476                if self.peek() == Some(&want) {
477                    self.i += 1;
478                    let lim = self.arg();
479                    one(MNode::Lim { upper: top, e: vec![g], lim })
480                } else {
481                    one(g)
482                }
483            }
484            "stackrel" | "overset" => {
485                let over = self.arg();
486                let base = self.arg();
487                one(MNode::Lim { upper: true, e: base, lim: over })
488            }
489            "underset" => {
490                let under = self.arg();
491                let base = self.arg();
492                one(MNode::Lim { upper: false, e: base, lim: under })
493            }
494            "hat" | "widehat" | "tilde" | "widetilde" | "dot" | "ddot" | "dddot" | "check" | "breve" | "acute" | "grave" | "vec" | "bar"
495            | "overrightarrow" | "overleftarrow" | "overleftrightarrow" | "mathring" => {
496                let chr = match c {
497                    "hat" | "widehat" => '\u{302}',
498                    "tilde" | "widetilde" => '\u{303}',
499                    "dot" => '\u{307}',
500                    "ddot" => '\u{308}',
501                    "dddot" => '\u{20DB}',
502                    "check" => '\u{30C}',
503                    "breve" => '\u{306}',
504                    "acute" => '\u{301}',
505                    "grave" => '\u{300}',
506                    "bar" => '\u{305}',
507                    "overleftarrow" => '\u{20D6}',
508                    "overleftrightarrow" => '\u{20E1}',
509                    "mathring" => '\u{30A}',
510                    _ => '\u{20D7}',
511                };
512                one(MNode::Acc { chr, e: self.arg() })
513            }
514            "," | "thinspace" => vec![run("\u{2009}")],
515            ":" | ">" | "medspace" => vec![run("\u{205F}")],
516            ";" | "thickspace" => vec![run("\u{2005}")],
517            "!" | "negthinspace" => Vec::new(),
518            " " => vec![run(" ")],
519            "qquad" => vec![run("\u{2003}\u{2003}")],
520            "{" | "}" | "%" | "$" | "#" | "&" | "_" => {
521                // Escaped characters (a literal `#` must not number the equation).
522                let mut r = MRun::new(c);
523                r.lit = c == "#";
524                vec![MNode::Run(r)]
525            }
526            "|" => vec![run("‖")],
527            name if FUNCTION_NAMES.contains(&name) => {
528                let plain = vec![MNode::Run(MRun::plain(name))];
529                let limits = matches!(name, "lim" | "liminf" | "limsup" | "max" | "min" | "sup" | "inf" | "det" | "gcd" | "Pr");
530                let named = if limits && self.peek() == Some(&Tok::Sub) {
531                    self.i += 1;
532                    let lim = self.arg();
533                    vec![MNode::Lim { upper: false, e: plain, lim }]
534                } else if matches!(self.peek(), Some(Tok::Sup) | Some(Tok::Sub)) {
535                    let sup = self.peek() == Some(&Tok::Sup);
536                    self.i += 1;
537                    let s = self.arg();
538                    vec![script(Some(MNode::Run(MRun::plain(name))), sup, s)]
539                } else {
540                    plain
541                };
542                self.apply(named)
543            }
544            name => match symbol(name) {
545                Some(ch) if is_nary_char(ch) => vec![MNode::Nary {
546                    chr: ch,
547                    lim_loc: None,
548                    grow: false,
549                    sub_hide: true,
550                    sup_hide: true,
551                    sub: Vec::new(),
552                    sup: Vec::new(),
553                    e: Vec::new(),
554                }],
555                Some(ch) => vec![run(&ch.to_string())],
556                None => vec![run(&format!("\\{name}"))],
557            },
558        }
559    }

561    /// A function name applied to the next atom (with its scripts).
562    fn apply(&mut self, name: Arg) -> Arg {
563        let mut e = self.atom();
564        while let Some(t @ (Tok::Sup | Tok::Sub)) = self.peek().cloned() {
565            self.i += 1;
566            let s = self.arg();
567            if any_too_deep(&e) {
568                e.extend(s);
569                continue;
570            }
571            let base = if e.len() == 1 { e.pop() } else { Some(group(std::mem::take(&mut e))) };
572            e.push(script(base, t == Tok::Sup, s));
573        }
574        if e.is_empty() {
575            return name;
576        }
577        vec![MNode::Func { name, e }]
578    }

580    fn environment(&mut self, env: &str) -> Arg {
581        let base = env.trim_end_matches('*');
582        let mut col_jc = Vec::new();
583        if base == "array" || base == "tabular" {
584            let _ = self.opt();
585            let spec = self.raw_arg();
586            for ch in spec.chars() {
587                match ch {
588                    'l' | 'L' => col_jc.push(ColJc::Left),
589                    'r' | 'R' => col_jc.push(ColJc::Right),
590                    'c' | 'C' => col_jc.push(ColJc::Center),
591                    _ => {}
592                }
593            }
594        } else if base == "aligned" {
595            let _ = self.opt();
596        }
597        // Rows of `&`-separated cells up to `\end{…}`.
598        let mut rows: Vec<Vec<Arg>> = vec![vec![]];
599        loop {
600            let cell = self.seq(false);
601            if let Some(r) = rows.last_mut() {
602                r.push(cell);
603            }
604            match self.peek().cloned() {
605                Some(Tok::Amp) => self.i += 1,
606                Some(Tok::Row) => {
607                    self.i += 1;
608                    rows.push(vec![]);
609                }
610                Some(Tok::Cmd(k)) if k == "end" => {
611                    self.i += 1;
612                    let _ = self.raw_arg();
613                    break;
614                }
615                Some(Tok::Close) => self.i += 1,
616                Some(_) => self.i += 1,
617                None => break,
618            }
619        }
620        // A trailing `\\` leaves an empty last row.
621        if rows.len() > 1 && rows.last().is_some_and(|r| r.iter().all(|c| c.is_empty())) {
622            rows.pop();
623        }
624        let delims = match base {
625            "pmatrix" => Some((Some('('), Some(')'))),
626            "bmatrix" => Some((Some('['), Some(']'))),
627            "Bmatrix" => Some((Some('{'), Some('}'))),
628            "vmatrix" => Some((Some('|'), Some('|'))),
629            "Vmatrix" => Some((Some('‖'), Some('‖'))),
630            _ => None,
631        };
632        let wrap = |inner: MNode, beg: Option<char>, end: Option<char>| MNode::Delim {
633            beg,
634            end,
635            sep: None,
636            grow: true,
637            shp_match: false,
638            elems: vec![vec![inner]],
639        };
640        match base {
641            "matrix" | "pmatrix" | "bmatrix" | "Bmatrix" | "vmatrix" | "Vmatrix" | "smallmatrix" | "array" | "tabular" => {
642                let m = MNode::Matrix { rows, col_jc };
643                match delims {
644                    Some((b, e)) => vec![wrap(m, b, e)],
645                    None => vec![m],
646                }
647            }
648            "cases" | "dcases" => {
649                let eq = MNode::EqArr { rows: rows.into_iter().map(join_cells).collect() };
650                vec![wrap(eq, Some('{'), None)]
651            }
652            "rcases" => {
653                let eq = MNode::EqArr { rows: rows.into_iter().map(join_cells).collect() };
654                vec![wrap(eq, None, Some('}'))]
655            }
656            _ => vec![MNode::EqArr { rows: rows.into_iter().map(join_cells).collect() }],
657        }
658    }
659}

661/// Cells of an aligned row joined with `&` alignment marks.
662fn join_cells(cells: Vec<Arg>) -> Arg {
663    let mut out = Vec::new();
664    for (i, c) in cells.into_iter().enumerate() {
665        if i > 0 {
666            out.push(run("&"));
667        }
668        out.extend(c);
669    }
670    merge_runs(out)
671}

673fn group(a: Arg) -> MNode {
674    MNode::Delim { beg: None, end: None, sep: None, grow: true, shp_match: false, elems: vec![a] }

677fn script(base: Option<MNode>, sup: bool, s: Arg) -> MNode {
678    match base {
679        Some(MNode::Nary { chr, lim_loc, grow, sub_hide, sup_hide, sub, sup: sp, e }) if e.is_empty() => {
680            if sup {
681                MNode::Nary { chr, lim_loc, grow, sub_hide, sup_hide: false, sub, sup: s, e }
682            } else {
683                MNode::Nary { chr, lim_loc, grow, sub_hide: false, sup_hide, sub: s, sup: sp, e }
684            }
685        }
686        Some(MNode::Script { kind: ScriptKind::Sub, base, sub, .. }) if sup => MNode::Script { kind: ScriptKind::SubSup, base, sub, sup: s },
687        Some(MNode::Script { kind: ScriptKind::Sup, base, sup: sp, .. }) if !sup => MNode::Script { kind: ScriptKind::SubSup, base, sub: s, sup: sp },
688        b => {
689            let base = b.map(|b| vec![b]).unwrap_or_default();
690            if sup {
691                MNode::Script { kind: ScriptKind::Sup, base, sub: Vec::new(), sup: s }
692            } else {
693                MNode::Script { kind: ScriptKind::Sub, base, sub: s, sup: Vec::new() }
694            }
695        }
696    }
697}

699/// Big operators without an operand take the atoms after them up to a relation or binary operator.
700/// `depth` is how deep the operators already sit; past [`WRAP_DEPTH`] they stay without operand
701/// (`\sum\sum\sum…` must not nest once per operator).
702fn nary_operands(a: Arg, depth: usize) -> Arg {
703    let mut out: Arg = Vec::with_capacity(a.len());
704    let mut it = a.into_iter().peekable();
705    while let Some(n) = it.next() {
706        match n {
707            MNode::Nary { chr, lim_loc, grow, sub_hide, sup_hide, sub, sup, e } if e.is_empty() && depth < WRAP_DEPTH => {
708                let mut body = Vec::new();
709                while let Some(next) = it.peek() {
710                    let stop = match next {
711                        MNode::Run(r) => r.text.chars().next().is_some_and(|c| {
712                            matches!(math_class(c), crate::doc::math::MClass::Rel | crate::doc::math::MClass::Bin | crate::doc::math::MClass::Punct)
713                                || c == '&'
714                        }),
715                        _ => false,
716                    };
717                    if stop {
718                        break;
719                    }
720                    if let Some(x) = it.next() {
721                        body.push(x);
722                    }
723                }
724                let body = merge_runs(nary_operands(body, depth + 1));
725                if any_too_deep(&body) {
726                    // No room to wrap the operand: it follows the operator.
727                    out.push(MNode::Nary { chr, lim_loc, grow, sub_hide, sup_hide, sub, sup, e });
728                    out.extend(body);
729                } else {
730                    out.push(MNode::Nary { chr, lim_loc, grow, sub_hide, sup_hide, sub, sup, e: body });
731                }
732            }
733            other => out.push(other),
734        }
735    }
736    out
737}

739/// Set a style on every run inside.
740fn restyle(a: Arg, sty: Option<MSty>, scr: MScr, nor: bool) -> Arg {
741    a.into_iter()
742        .map(|n| match n {
743            MNode::Run(mut r) => {
744                if sty.is_some() {
745                    r.sty = sty;
746                }
747                if scr != MScr::Roman {
748                    r.scr = scr;
749                }
750                r.nor |= nor;
751                MNode::Run(r)
752            }
753            other => map_args(other, &|a| restyle(a, sty, scr, nor)),
754        })
755        .collect()
756}

783fn chars(text: &str, s: &mut String) {
784    for c in text.chars() {
785        match c {
786            '{' | '}' | '%' | '$' | '#' | '&' | '_' => {
787                s.push('\\');
788                s.push(c);
789            }
790            '\\' => s.push_str("\\backslash "),
791            '−' => s.push('-'),
792            c if c.is_ascii() => s.push(c),
793            c => match name_of(c) {
794                Some(n) if !matches!(n, "sqrt" | "matrix" | "of" | "above" | "below" | "box" | "rect") => {
795                    s.push('\\');
796                    s.push_str(n);
797                    s.push(' ');
798                }
799                _ => s.push(c),
800            },
801        }
802    }
803}
804
805/// Text inside `\text{…}`: only what LaTeX would read as markup is escaped.
806fn text_chars(text: &str, s: &mut String) {
807    for c in text.chars() {
808        match c {
809            '{' | '}' | '%' | '$' | '#' | '&' | '_' | '~' => {
810                s.push('\\');
811                s.push(c);
812            }
813            // `{}` ends the command name without adding a space (spaces count in text).
814            '\\' => s.push_str("\\backslash{}"),
815            c => s.push(c),
816        }
817    }
818}

820/// The delimiter of a `\genfrac` argument: one character, or none for `{}` and anything else.
821fn delim_arg(a: &[MNode]) -> Option<char> {
822    let [MNode::Run(r)] = a else { return None };
823    let mut cs = r.text.chars();
824    match (cs.next(), cs.next()) {
825        (Some(c), None) if c != '.' => Some(c),
826        _ => None,
827    }
828}

830/// Whether a `\genfrac` thickness is a zero length such as `0pt`, `0.0pt` or `0 em`. An empty
831/// thickness means the default rule and anything that is not a plain number and unit is not zero.
832fn is_zero_length(a: &[MNode]) -> bool {
833    let mut text = String::new();
834    for n in a {
835        let MNode::Run(r) = n else { return false };
836        text.extend(r.text.chars().filter(|c| !c.is_whitespace() && !matches!(c, '\u{2009}' | '\u{205F}' | '\u{2005}')));
837    }
838    let number = text.trim_end_matches(|c: char| c.is_ascii_alphabetic());
839    let unit = &text[number.len()..];
840    !number.is_empty()
841        && matches!(unit, "pt" | "em" | "ex" | "mm" | "cm" | "in" | "bp" | "pc" | "sp" | "mu")
842        && number.parse::<f64>().is_ok_and(|v| v == 0.0)
843}

845fn delim_tex(c: Option<char>) -> String {
846    match c {
847        None => ".".into(),
848        Some('{') => "\\{".into(),
849        Some('}') => "\\}".into(),
850        Some('‖') => "\\|".into(),
851        Some(c) if c.is_ascii() => c.to_string(),
852        Some(c) => name_of(c).map(|n| format!("\\{n} ")).unwrap_or_else(|| c.to_string()),
853    }
854}

856fn group_chr_tex(chr: char, top: bool) -> &'static str {
857    match chr {
858        '⏜' => "\\overparen",
859        '⏝' => "\\underparen",
860        '⎴' => "\\overbracket",
861        '⎵' => "\\underbracket",
862        _ if top => "\\overbrace",
863        _ => "\\underbrace",
864    }
865}

867fn node(node: &MNode, output: &mut String, depth: usize) {
868    match node {
869        MNode::Run(run) => run_node(run, output),
870        MNode::Frac { kind, num, den } => fraction_node(*kind, num, den, output, depth),
871        MNode::Script { kind, base, sub, sup } => script_node(*kind, base, sub, sup, output, depth),
872        MNode::Rad { deg, deg_hide, e } => radical_node(deg, *deg_hide, e, output, depth),
873        MNode::Nary { .. } => nary_node(node, output, depth),
874        MNode::Delim { beg, end, sep, elems, .. } => delimiter_node(*beg, *end, *sep, elems, output, depth),
875        MNode::Func { name, e } => {
876            arg(name, output, depth);
877            braced(e, output, depth);
878        }
879        MNode::Lim { upper, e, lim } => limit_node(*upper, e, lim, output, depth),
880        MNode::Acc { chr, e } => accent_node(*chr, e, output, depth),
881        MNode::Bar { top, e } => {
882            output.push_str(if *top { "\\overline" } else { "\\underline" });
883            braced(e, output, depth);
884        }
885        MNode::BorderBox { e, .. } => {
886            output.push_str("\\boxed");
887            braced(e, output, depth);
888        }
889        MNode::Boxed { e } => braced(e, output, depth),
890        MNode::GroupChr { chr, top, e } => {
891            output.push_str(group_chr_tex(*chr, *top));
892            braced(e, output, depth);
893        }
894        MNode::EqArr { rows } => equation_array_node(rows, output, depth),
895        MNode::Matrix { rows, .. } => matrix_node(rows, output, depth),
896        MNode::Phant { e, .. } => {
897            output.push_str("\\phantom");
898            braced(e, output, depth);
899        }
900    }
901}

903fn run_node(run: &MRun, output: &mut String) {
904    let (open, close) = if run.nor {
905        ("\\text{", "}")
906    } else {
907        match (run.sty, run.scr) {
908            (_, MScr::DoubleStruck) => ("\\mathbb{", "}"),
909            (_, MScr::Script) => ("\\mathcal{", "}"),
910            (_, MScr::Fraktur) => ("\\mathfrak{", "}"),
911            (_, MScr::SansSerif) => ("\\mathsf{", "}"),
912            (_, MScr::Monospace) => ("\\mathtt{", "}"),
913            (Some(MSty::Bold), _) => ("\\mathbf{", "}"),
914            (Some(MSty::BoldItalic), _) => ("\\boldsymbol{", "}"),
915            (Some(MSty::Plain), _) if run.text.chars().any(char::is_alphabetic) => {
916                if FUNCTION_NAMES.contains(&run.text.as_str()) {
917                    output.push('\\');
918                    output.push_str(&run.text);
919                    output.push(' ');
920                    return;
921                }
922                ("\\mathrm{", "}")
923            }
924            _ => ("", ""),
925        }
926    };
927    output.push_str(open);
928    if run.nor {
929        text_chars(&run.text, output)
930    } else {
931        chars(&run.text, output)
932    }
933    output.push_str(close);
934}

936fn fraction_node(kind: FracKind, numerator: &[MNode], denominator: &[MNode], output: &mut String, depth: usize) {
937    match kind {
938        FracKind::NoBar => {
939            output.push_str("\\genfrac{}{}{0pt}{}");
940            braced(numerator, output, depth);
941            braced(denominator, output, depth);
942        }
943        FracKind::Linear | FracKind::Skewed => {
944            braced(numerator, output, depth);
945            output.push('/');
946            braced(denominator, output, depth);
947        }
948        FracKind::Bar => {
949            output.push_str("\\frac");
950            braced(numerator, output, depth);
951            braced(denominator, output, depth);
952        }
953    }
954}

956fn script_node(kind: ScriptKind, base: &[MNode], sub: &[MNode], sup: &[MNode], output: &mut String, depth: usize) {
957    if kind == ScriptKind::Pre {
958        output.push_str("{}_");
959        braced(sub, output, depth);
960        output.push('^');
961        braced(sup, output, depth);
962        braced(base, output, depth);
963        return;
964    }
965    braced(base, output, depth);
966    if matches!(kind, ScriptKind::Sub | ScriptKind::SubSup) {
967        output.push('_');
968        braced(sub, output, depth);
969    }
970    if matches!(kind, ScriptKind::Sup | ScriptKind::SubSup) {
971        output.push('^');
972        braced(sup, output, depth);
973    }
974}

976fn radical_node(degree: &[MNode], degree_hidden: bool, expression: &[MNode], output: &mut String, depth: usize) {
977    output.push_str("\\sqrt");
978    if !degree_hidden && !degree.is_empty() {
979        output.push('[');
980        arg(degree, output, depth);
981        output.push(']');
982    }
983    braced(expression, output, depth);
984}

986fn nary_node(node: &MNode, output: &mut String, depth: usize) {
987    let MNode::Nary { chr, lim_loc, sub_hide, sup_hide, sub, sup, e, .. } = node else { return };
988    match name_of(*chr) {
989        Some(name) => {
990            output.push('\\');
991            output.push_str(name);
992        }
993        None => output.push(*chr),
994    }
995    match lim_loc {
996        Some(crate::doc::math::LimLoc::UndOvr) => output.push_str("\\limits"),
997        Some(crate::doc::math::LimLoc::SubSup) if !crate::doc::math::is_integral(*chr) => output.push_str("\\nolimits"),
998        _ => {}
999    }
1000    if !sub_hide && !sub.is_empty() {
1001        output.push('_');
1002        braced(sub, output, depth);
1003    }
1004    if !sup_hide && !sup.is_empty() {
1005        output.push('^');
1006        braced(sup, output, depth);
1007    }
1008    output.push(' ');
1009    braced(e, output, depth);
1010}

1012fn delimiter_node(begin: Option<char>, end: Option<char>, separator: Option<char>, elements: &[Arg], output: &mut String, depth: usize) {
1013    output.push_str("\\left");
1014    output.push_str(&delim_tex(begin));
1015    let middle = separator.map(|c| format!("\\middle{}", delim_tex(Some(c)))).unwrap_or_else(|| "\\middle|".to_string());
1016    for (index, element) in elements.iter().enumerate() {
1017        if index > 0 {
1018            output.push_str(&middle);
1019        }
1020        arg(element, output, depth);
1021    }
1022    output.push_str("\\right");
1023    output.push_str(&delim_tex(end));
1024}

1026fn limit_node(upper: bool, expression: &[MNode], limit: &[MNode], output: &mut String, depth: usize) {
1027    let simple = matches!(expression, [MNode::Run(run)] if run.sty == Some(MSty::Plain) && FUNCTION_NAMES.contains(&run.text.as_str()));
1028    if simple && !upper {
1029        arg(expression, output, depth);
1030        output.push('_');
1031        braced(limit, output, depth);
1032    } else if let [MNode::GroupChr { chr, top, e: inner }] = expression {
1033        output.push_str(group_chr_tex(*chr, *top));
1034        braced(inner, output, depth);
1035        output.push(if upper { '^' } else { '_' });
1036        braced(limit, output, depth);
1037    } else {
1038        output.push_str(if upper { "\\overset" } else { "\\underset" });
1039        braced(limit, output, depth);
1040        braced(expression, output, depth);
1041    }
1042}

1044fn accent_node(character: char, expression: &[MNode], output: &mut String, depth: usize) {
1045    let command = match character {
1046        '\u{302}' => "hat",
1047        '\u{303}' => "tilde",
1048        '\u{307}' => "dot",
1049        '\u{308}' => "ddot",
1050        '\u{20DB}' => "dddot",
1051        '\u{30C}' => "check",
1052        '\u{306}' => "breve",
1053        '\u{301}' => "acute",
1054        '\u{300}' => "grave",
1055        '\u{305}' | '\u{AF}' => "bar",
1056        '\u{20D6}' => "overleftarrow",
1057        '\u{20E1}' => "overleftrightarrow",
1058        _ => "vec",
1059    };
1060    output.push('\\');
1061    output.push_str(command);
1062    braced(expression, output, depth);
1063}

1065fn equation_array_node(rows: &[Arg], output: &mut String, depth: usize) {
1066    if rows.len() == 1
1067        && let Some(row) = rows.first()
1068        && crate::doc::math::has_number_mark(row)
1069    {
1070        arg(row, output, depth);
1071        return;
1072    }
1073    output.push_str("\\begin{aligned}");
1074    for (index, row) in rows.iter().enumerate() {
1075        if index > 0 {
1076            output.push_str("\\\\");
1077        }
1078        for child_node in row {
1079            match child_node {
1080                MNode::Run(run) if !run.lit && run.text.contains('&') => {
1081                    for (index, part) in run.text.split('&').enumerate() {
1082                        if index > 0 {
1083                            output.push('&');
1084                        }
1085                        let mut piece = run.clone();
1086                        piece.text = part.to_string();
1087                        run_node(&piece, output);
1088                    }
1089                }
1090                other => node(other, output, depth + 1),
1091            }
1092        }
1093    }
1094    output.push_str("\\end{aligned}");
1095}

1097fn matrix_node(rows: &[Vec<Arg>], output: &mut String, depth: usize) {
1098    output.push_str("\\begin{matrix}");
1099    for (row_index, row) in rows.iter().enumerate() {
1100        if row_index > 0 {
1101            output.push_str("\\\\");
1102        }
1103        for (column, cell) in row.iter().enumerate() {
1104            if column > 0 {
1105                output.push('&');
1106            }
1107            arg(cell, output, depth);
1108        }
1109    }
1110    output.push_str("\\end{matrix}");
1111}
