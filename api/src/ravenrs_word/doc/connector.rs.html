1//! Connectors and the shape properties beyond fill and outline: a shape's drawing id, its text
2//! warp, a rounded rectangle's corner, and a connector's ends.
3//!
4//! A connector (DrawingML `straightConnector1`, `bentConnector3`, `curvedConnector3`) is a line
5//! from its frame's top-left to its bottom-right corner (flips swap the corners). Each end may be
6//! glued to a connection site of another shape (`a:stCxn` / `a:endCxn`, ECMA-376 §20.1.2.2.36 and
7//! §20.1.2.2.13: the shape's drawing id and a site index). When a glued shape moves, [`route`]
8//! gives the connector's new frame from its ends.
9

16/// DrawingML outline ends (ST_LineCap); missing and unknown values use flat ends.
17#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Serialize, Deserialize)]
18#[serde(rename_all = "camelCase")]
19pub enum LineCap {
20    #[default]
21    Flat,
22    Square,
23    Round,
24}
25
26impl LineCap {
27    /// Read an ST_LineCap value; absent or unrecognized values use the flat default.
28    pub fn from_ooxml(value: &str) -> Self {
29        match value {
30            "sq" => Self::Square,
31            "rnd" => Self::Round,
32            _ => Self::Flat,
33        }
34    }
35
36    /// Return the canonical DrawingML spelling used when saving the outline.
37    pub fn ooxml(self) -> &'static str {
38        match self {
39            Self::Flat => "flat",
40            Self::Square => "sq",
41            Self::Round => "rnd",
42        }
43    }
44
45    /// Whether the cap can be omitted from older-compatible serialized shape properties.
46    pub fn is_flat(&self) -> bool {
47        *self == Self::Flat
48    }
49}

51/// An arrowhead's width across its line or length along it (ST_LineEndWidth and
52/// ST_LineEndLength, ECMA-376 §20.1.10.39–40): 2, 3 or 5 times the line's width, as Word draws it.
53#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Serialize, Deserialize)]
54#[serde(rename_all = "camelCase")]
55pub enum ArrowSize {
56    Small,
57    #[default]
58    Medium,
59    Large,
60}
61
62impl ArrowSize {
63    /// Read a DrawingML size (`sm`, `med`, `lg`); absent or unknown values are medium.
64    pub fn from_ooxml(value: &str) -> Self {
65        match value {
66            "sm" => Self::Small,
67            "lg" => Self::Large,
68            _ => Self::Medium,
69        }
70    }
71
72    /// The DrawingML spelling.
73    pub fn ooxml(self) -> &'static str {
74        match self {
75            Self::Small => "sm",
76            Self::Medium => "med",
77            Self::Large => "lg",
78        }
79    }
80
81    /// Read a VML width (`narrow`, `medium`, `wide`) or length (`short`, `medium`, `long`).
82    pub fn from_vml(value: &str) -> Self {
83        match value {
84            "narrow" | "short" => Self::Small,
85            "wide" | "long" => Self::Large,
86            _ => Self::Medium,
87        }
88    }
89
90    /// How many [`arrow_unit`]s the arrowhead measures.
91    pub fn factor(self) -> f32 {

93            Self::Small => 2.0,
94            Self::Medium => 3.0,
95            Self::Large => 5.0,
96        }
97    }

162    /// The arrowhead's length along its line and width across it, for a line `line_width` wide.
163    pub fn dims(self, line_width: f32) -> (f32, f32) {
164        let unit = arrow_unit(line_width);
165        (self.length.factor() * unit, self.width.factor() * unit)

169/// What an arrowhead's size is a multiple of: the line's width, but at least 2 points (Word draws
170/// a thinner line's arrowheads as for a 2-point line).
171pub fn arrow_unit(line_width: f32) -> f32 {
172    let w = if line_width.is_finite() { line_width.clamp(0.25, 200.0) } else { 0.75 };
173    w.max(2.0)
174}

228/// How a horizontal rule sits on its line (VML `o:hrpct`, `o:hralign`), as Word draws one:
229/// across `pct` percent of the line's width (the paragraph's, inside its indents), or at the
230/// shape's own width when `pct` is 0; `align` places it when it is narrower than the line.
231#[derive(Clone, Copy, Debug, PartialEq, Default, Serialize, Deserialize)]
232#[serde(default, rename_all = "camelCase")]
233pub struct HorizontalRule {
234    pub pct: f32,
235    pub align: crate::doc::props::Align,
236    /// Word's standard look (`o:hrstd`), kept for saving.
237    pub standard: bool,
238}

240impl HorizontalRule {
241    /// The rule's width on a line `line` wide, for a shape `own` wide.
242    pub fn width(&self, line: f32, own: f32) -> f32 {
243        let line = line.max(0.0);
244        if self.pct.is_finite() && self.pct > 0.0 { line * (self.pct.min(100.0) / 100.0) } else { own.clamp(0.0, line) }
245    }
246
247    /// Where a rule `w` wide starts on a line from `x`, `line` wide.
248    pub fn left(&self, x: f32, line: f32, w: f32) -> f32 {
249        match self.align {
250            crate::doc::props::Align::Center => x + (line - w) / 2.0,
251            crate::doc::props::Align::Right => x + line - w,
252            _ => x,
253        }
254    }

270/// The largest rounded-rectangle corner: half the shorter side, which makes its ends round.
271pub const MAX_CORNER: u32 = 50_000;
272
273/// A rounded rectangle's corner radius without its own: 16% of the shorter side.
274pub const DEFAULT_CORNER: f32 = 0.16;
275
276/// The corner radius, in points, of a `w` × `h` rounded rectangle with `corner` (see
277/// [`ShapeExtra::corner`]).
278pub fn corner_radius(corner: Option<u32>, w: f32, h: f32) -> f32 {
279    let side = crate::geom::finite(w).min(crate::geom::finite(h)).max(0.0);
280    side * corner.map_or(DEFAULT_CORNER, |c| c.min(MAX_CORNER) as f32 / 100_000.0)
281}

305/// The connection sites of a `kind` shape `w` × `h`, in its own unrotated frame, by index. The
306/// order follows the preset shapes' connection lists (for a rectangle: top, left, bottom, right).
307pub fn sites(kind: ShapeKind, w: f32, h: f32) -> Vec<(f32, f32)> {
308    let (w, h) = (crate::geom::finite(w), crate::geom::finite(h));
309    let (cx, cy) = (w / 2.0, h / 2.0);
310    match kind {
311        ShapeKind::Ellipse => {
312            // The four sides, and the points of the ellipse at 45° between them.
313            let (dx, dy) = (cx * std::f32::consts::FRAC_1_SQRT_2, cy * std::f32::consts::FRAC_1_SQRT_2);
314            vec![(cx, 0.0), (cx - dx, cy - dy), (0.0, cy), (cx - dx, cy + dy), (cx, h), (cx + dx, cy + dy), (w, cy), (cx + dx, cy - dy)]
315        }
316        ShapeKind::Triangle => vec![(cx, 0.0), (cx / 2.0, cy), (0.0, h), (cx, h), (w, h), (cx + cx / 2.0, cy)],
317        // The corners, the legs' midpoints and the hypotenuse's.
318        ShapeKind::RightTriangle => vec![(0.0, 0.0), (0.0, cy), (0.0, h), (cx, h), (w, h), (cx, cy)],
319        ShapeKind::Line | ShapeKind::StraightConnector | ShapeKind::ElbowConnector | ShapeKind::CurvedConnector => vec![(0.0, 0.0), (w, h)],
320        _ => vec![(cx, 0.0), (0.0, cy), (cx, h), (w, cy)],
321    }
322}

324/// Site `site` of a `kind` shape with frame `rect` (x, y, w, h) on the page, turned and flipped by
325/// `spin` about its centre. `None` for a site the shape doesn't have.
326pub fn site_point(kind: ShapeKind, rect: [f32; 4], spin: crate::geom::Spin, site: u32) -> Option<(f32, f32)> {
327    let [x, y, w, h] = rect;
328    let (sx, sy) = *sites(kind, w, h).get(site as usize)?;
329    let (cx, cy) = (x + w / 2.0, y + h / 2.0);
330    let (px, py) = spin.apply(cx, cy, x + sx, y + sy);
331    (px.is_finite() && py.is_finite()).then_some((px, py))
332}

334/// The site of a `kind` shape (frame `rect`, `spin`) nearest to page point `at`: its index, point
335/// and distance.
336pub fn nearest_site(kind: ShapeKind, rect: [f32; 4], spin: crate::geom::Spin, at: (f32, f32)) -> Option<(u32, (f32, f32), f32)> {
337    let n = sites(kind, rect[2], rect[3]).len() as u32;
338    (0..n)
339        .filter_map(|i| site_point(kind, rect, spin, i).map(|p| (i, p, ((p.0 - at.0).powi(2) + (p.1 - at.1).powi(2)).sqrt())))
340        .min_by(|a, b| a.2.total_cmp(&b.2))
341}

343/// A connector's frame: top-left corner, size and flips.
344#[derive(Clone, Copy, Debug, PartialEq)]
345pub struct Frame {
346    pub x: f32,
347    pub y: f32,
348    pub w: f32,
349    pub h: f32,
350    pub flip_h: bool,
351    pub flip_v: bool,
352}

354/// The frame of a connector from `start` to `end`.
355pub fn route(start: (f32, f32), end: (f32, f32)) -> Frame {
356    let f = crate::geom::finite;
357    let (x0, y0, x1, y1) = (f(start.0), f(start.1), f(end.0), f(end.1));
358    Frame { x: x0.min(x1), y: y0.min(y1), w: (x1 - x0).abs(), h: (y1 - y0).abs(), flip_h: x1 < x0, flip_v: y1 < y0 }
359}

361/// A connector's start and end from its frame (the inverse of [`route`]).
362pub fn ends(x: f32, y: f32, w: f32, h: f32, flip_h: bool, flip_v: bool) -> ((f32, f32), (f32, f32)) {
363    let (xa, xb) = if flip_h { (x + w, x) } else { (x, x + w) };
364    let (ya, yb) = if flip_v { (y + h, y) } else { (y, y + h) };
365    ((xa, ya), (xb, yb))
366}

368/// The path of a connector `kind` in a `w` × `h` frame at the origin, from its top-left to its
369/// bottom-right corner (flips are the frame's turn): straight, an elbow through the middle (out
370/// horizontally, across, in horizontally), or an S-curve. `None` for other kinds.
371pub fn connector_segs(kind: ShapeKind, w: f32, h: f32) -> Option<Vec<PathSeg>> {
372    let (w, h) = (crate::geom::finite(w), crate::geom::finite(h));
373    let mx = w / 2.0;
374    Some(match kind {
375        ShapeKind::StraightConnector => vec![PathSeg::Move(0.0, 0.0), PathSeg::Line(w, h)],
376        ShapeKind::ElbowConnector => vec![PathSeg::Move(0.0, 0.0), PathSeg::Line(mx, 0.0), PathSeg::Line(mx, h), PathSeg::Line(w, h)],
377        ShapeKind::CurvedConnector => vec![PathSeg::Move(0.0, 0.0), PathSeg::Cubic(mx, 0.0, mx, h, w, h)],
378        _ => return None,
379    })
380}

382/// The direction (unit vector) a line or connector `kind` arrives at its end (`at_end`) or
383/// leaves its start, in its unflipped `w` × `h` frame: where an arrowhead points.
384pub fn end_direction(kind: ShapeKind, w: f32, h: f32, at_end: bool) -> (f32, f32) {
385    let (dx, dy) = match kind {
386        ShapeKind::ElbowConnector | ShapeKind::CurvedConnector => {
387            if w.abs() > 0.01 || h.abs() < 0.01 {
388                (1.0, 0.0)
389            } else {
390                (0.0, 1.0)
391            }
392        }
393        _ => (w, h),
394    };
395    let len = (dx * dx + dy * dy).sqrt();
396    let (ux, uy) = if len > 1e-6 && len.is_finite() { (dx / len, dy / len) } else { (1.0, 0.0) };
397    if at_end { (ux, uy) } else { (-ux, -uy) }
398}

400/// Re-route a group's glued connectors. Resolved ends use the shapes' connection sites in child
401/// coordinates; free or unresolved ends keep their displayed position after child scaling and spin.
402/// Returns whether any moved. Non-groups and connectors needing a non-invertible or display-capped
403/// fallback are unchanged.
404pub fn reroute_group(group: &mut InlineObject) -> bool {
405    let InlineObject::Group { w, h, ch_w, ch_h, children, .. } = group else { return false };
406    let fin = crate::geom::finite;
407    // Match group_rects: invalid child extents use unit scale; invalid display extents become zero.
408    let sx = if fin(*ch_w) > 0.0 { fin(*w) / *ch_w } else { 1.0 };
409    let sy = if fin(*ch_h) > 0.0 { fin(*h) / *ch_h } else { 1.0 };
410    let shapes: Vec<(u32, ShapeKind, [f32; 4], crate::geom::Spin)> = children
411        .iter()
412        .filter_map(|c| match &c.obj {
413            InlineObject::Shape { kind, w, h, float, extra, .. } if extra.id != 0 => Some((extra.id, *kind, [c.x, c.y, *w, *h], float.spin())),
414            _ => None,
415        })
416        .collect();
417    if shapes.is_empty() {
418        return false;
419    }
420    let find = |e: Option<ConnEnd>| {
421        let e = e?;
422        let (_, kind, rect, spin) = shapes.iter().find(|s| s.0 == e.id)?;
423        site_point(*kind, *rect, *spin, e.site)
424    };
425    let mut moved = false;
426    for c in children.iter_mut() {
427        let InlineObject::Shape { kind, w, h, float, extra, .. } = &mut c.obj else { continue };
428        if !kind.is_connector() || (extra.start.is_none() && extra.end.is_none()) {
429            continue;
430        }
431        let (s, e, fallback) = match (find(extra.start), find(extra.end)) {
432            (Some(s), Some(e)) => (s, e, false),
433            (s, e) => {
434                if !sx.is_finite() || !sy.is_finite() || sx <= 0.0 || sy <= 0.0 {
435                    continue;
436                }
437                // Rendering scales the rectangle before spinning it. Undo only the group scale
438                // so route can rebuild an unrotated child frame without moving fallback ends.
439                let (x, y) = (fin(c.x) * sx, fin(c.y) * sy);
440                let (cw, ch) = ((fin(*w) * sx).clamp(0.0, 4000.0), (fin(*h) * sy).clamp(0.0, 4000.0));
441                let (cx, cy) = (x + cw / 2.0, y + ch / 2.0);
442                let spin = float.spin();
443                let s0 = spin.apply(cx, cy, x, y);
444                let e0 = spin.apply(cx, cy, x + cw, y + ch);
445                (s.unwrap_or((s0.0 / sx, s0.1 / sy)), e.unwrap_or((e0.0 / sx, e0.1 / sy)), true)
446            }
447        };
448        if ![s.0, s.1, e.0, e.1].into_iter().all(f32::is_finite) {
449            continue;
450        }
451        let f = route(s, e);
452        if ![f.x, f.y, f.w, f.h].into_iter().all(f32::is_finite) {
453            continue;
454        }
455        // group_rects caps displayed child sizes. A larger fallback span cannot be represented
456        // by an unrotated frame, so keep the original instead of moving its free/unresolved end.
457        if fallback && (f.w * sx > 4000.0 || f.h * sy > 4000.0) {
458            continue;
459        }
460        let same = (f.x - c.x).abs() < 0.01 && (f.y - c.y).abs() < 0.01 && (f.w - *w).abs() < 0.01 && (f.h - *h).abs() < 0.01;
461        if same && f.flip_h == float.flip_h && f.flip_v == float.flip_v && float.rot == 0.0 {
462            continue;
463        }
464        c.x = f.x;
465        c.y = f.y;
466        *w = f.w;
467        *h = f.h;
468        float.rot = 0.0;
469        float.flip_h = f.flip_h;
470        float.flip_v = f.flip_v;
471        moved = true;
472    }
473    moved
474}

533    /// One rerouting case: the free end settles and the attached end keeps its site.
534    fn assert_transformed_reroute(kind: ShapeKind, rot: f32, flip_h: bool, flip_v: bool, free: Option<ConnEnd>, free_start: bool) {
535        let attached = Some(ConnEnd { id: 7, site: 3 });
536        let (start, end) = if free_start { (free, attached) } else { (attached, free) };
537        let mut obj = shape(kind, 40.0, 20.0, ShapeExtra { start, end, ..Default::default() });
538        let InlineObject::Shape { float, .. } = &mut obj else { panic!() };
539        float.rot = rot;
540        float.flip_h = flip_h;
541        float.flip_v = flip_v;
542        // Frame (10, 30, 40, 20), centre (30, 40): mirror, then rotate
543        // clockwise in y-down group space. Do not use the routing transform
544        // to compute the expectation.
545        let sign = if free_start { -1.0 } else { 1.0 };
546        let dx = sign * if flip_h { -20.0 } else { 20.0 };
547        let dy = sign * if flip_v { -10.0 } else { 10.0 };
548        let (sin, cos) = rot.to_radians().sin_cos();
549        let expected = (30.0 + dx * cos - dy * sin, 40.0 + dx * sin + dy * cos);
550        let kids = vec![
551            GroupChild { x: 100.0, y: 200.0, obj: shape(ShapeKind::Rectangle, 40.0, 20.0, ShapeExtra { id: 7, ..Default::default() }) },
552            GroupChild { x: 10.0, y: 30.0, obj },
553        ];
554        let mut group = group(kids);
555        assert!(reroute_group(&mut group));
556        let InlineObject::Group { children, .. } = &group else { panic!() };
557        let c = &children[1];
558        let InlineObject::Shape { w, h, float, .. } = &c.obj else { panic!() };
559        let (start, end) = ends(c.x, c.y, *w, *h, float.flip_h, float.flip_v);
560        let (free, attached) = if free_start { (start, end) } else { (end, start) };
561        assert!(
562            (free.0 - expected.0).abs() < 0.001 && (free.1 - expected.1).abs() < 0.001,
563            "{kind:?}, {rot}, {flip_h}, {flip_v}, {free_start}: {free:?} != {expected:?}"
564        );
565        assert_eq!(attached, (140.0, 210.0));
566        assert_eq!(float.rot, 0.0);
567        assert!(!reroute_group(&mut group), "rerouting must settle");
568    }

631    fn rerouting_preserves_fallback_when_group_display_cap_prevents_route() {
632        for free_start in [false, true] {
633            let attached = Some(ConnEnd { id: 7, site: 3 });
634            let (start, end) = if free_start { (None, attached) } else { (attached, None) };
635            let mut obj = shape(ShapeKind::StraightConnector, 40.0, 0.0, ShapeExtra { start, end, ..Default::default() });
636            let InlineObject::Shape { float, .. } = &mut obj else { panic!() };
637            float.flip_h = free_start;
638            let mut group = group(vec![
639                GroupChild { x: 0.0, y: 0.0, obj: shape(ShapeKind::Rectangle, 40.0, 20.0, ShapeExtra { id: 7, ..Default::default() }) },
640                GroupChild { x: 3000.0, y: 10.0, obj },
641            ]);
642            let InlineObject::Group { w, .. } = &mut group else { panic!() };
643            *w = 400.0;
644            let visual_free = |group: &InlineObject| {
645                let rects = group.group_rects(0.0, 0.0, 400.0, 300.0);
646                let ([x, y, w, h], obj) = rects[1];
647                let spin = obj.frame().unwrap().2.spin();
648                let (px, py) = if free_start { (x, y) } else { (x + w, y + h) };
649                spin.apply(x + w / 2.0, y + h / 2.0, px, py)
650            };
651            let before = visual_free(&group);
652            assert_eq!(before, (6080.0, 10.0));
653            let moved = reroute_group(&mut group);
654            assert_eq!(visual_free(&group), before, "the rebuilt display frame cannot span both endpoints");
655            assert!(!moved, "an unrepresentable fallback route must leave the connector unchanged");
656            let InlineObject::Group { children, .. } = &group else { panic!() };
657            let c = &children[1];
658            let InlineObject::Shape { w, h, float, .. } = &c.obj else { panic!() };
659            assert_eq!((c.x, c.y, *w, *h, float.rot, float.flip_h), (3000.0, 10.0, 40.0, 0.0, 0.0, free_start));
660            assert!(!reroute_group(&mut group), "repeated rerouting must not shrink the connector");
661        }

665    fn rerouting_group_handles_degenerate_dimensions_without_nonfinite_writes() {
666        for dims in [
667            [0.0, 300.0, 200.0, 300.0],
668            [200.0, 0.0, 200.0, 300.0],
669            [-200.0, 300.0, 200.0, 300.0],
670            [f32::NAN, 300.0, 200.0, 300.0],
671            [200.0, f32::INFINITY, 200.0, 300.0],
672            [f32::MAX, 300.0, f32::MIN_POSITIVE, 300.0],
673            [f32::MIN_POSITIVE, 300.0, f32::MAX, 300.0],
674            [20000.0, f32::MIN_POSITIVE, 200.0, 300.0],
675            [200.0, 300.0, 0.0, 0.0],
676            [200.0, 300.0, f32::NAN, f32::INFINITY],
677            [200.0, 300.0, -200.0, -300.0],
678        ] {
679            let mut obj =
680                shape(ShapeKind::StraightConnector, 40.0, 20.0, ShapeExtra { start: Some(ConnEnd { id: 7, site: 3 }), ..Default::default() });
681            let InlineObject::Shape { float, .. } = &mut obj else { panic!() };
682            float.rot = 90.0;
683            let mut group = group(vec![
684                GroupChild { x: 100.0, y: 200.0, obj: shape(ShapeKind::Rectangle, 40.0, 20.0, ShapeExtra { id: 7, ..Default::default() }) },
685                GroupChild { x: 10.0, y: 30.0, obj },
686            ]);
687            let InlineObject::Group { w, h, ch_w, ch_h, .. } = &mut group else { panic!() };
688            (*w, *h, *ch_w, *ch_h) = (dims[0], dims[1], dims[2], dims[3]);
689            let moved = reroute_group(&mut group);
690            // group_rects treats invalid child extents as unit scale; display scales that
691            // cannot be inverted (or whose inverse overflows) must not change the connector.
692            assert_eq!(moved, dims[2] <= 0.0 || !dims[2].is_finite(), "{dims:?}");
693            let InlineObject::Group { children, .. } = &group else { panic!() };
694            let c = &children[1];
695            let InlineObject::Shape { w, h, float, .. } = &c.obj else { panic!() };
696            assert!([c.x, c.y, *w, *h, float.rot].into_iter().all(f32::is_finite), "{dims:?}");
697            assert!(!reroute_group(&mut group), "rerouting must settle: {dims:?}");
698        }
699        assert!(!reroute_group(&mut shape(ShapeKind::Rectangle, 40.0, 20.0, ShapeExtra::default())));
700    }

702    #[test]
703    fn group_connectors_follow_their_shapes() {
704        let a = ShapeExtra { id: 7, ..Default::default() };
705        let b = ShapeExtra { id: 9, ..Default::default() };
706        let link = ShapeExtra { start: Some(ConnEnd { id: 7, site: 3 }), end: Some(ConnEnd { id: 9, site: 2 }), ..Default::default() };
707        let mut kids = vec![
708            GroupChild { x: 0.0, y: 0.0, obj: shape(ShapeKind::Rectangle, 20.0, 10.0, a) },
709            GroupChild { x: 100.0, y: 50.0, obj: shape(ShapeKind::Ellipse, 20.0, 10.0, b) },
710            GroupChild { x: 0.0, y: 0.0, obj: shape(ShapeKind::ElbowConnector, 1.0, 1.0, link) },
711        ];
712        // Fully resolved endpoints must still come from their sites, regardless of the
713        // connector's old rotation and flips.
714        let InlineObject::Shape { float, .. } = &mut kids[2].obj else { panic!() };
715        float.rot = 90.0;
716        float.flip_h = true;
717        float.flip_v = true;
718        let mut group = group(kids);
719        assert!(reroute_group(&mut group));
720        // From the rectangle's right side (20, 5) to the ellipse's left side (100, 55).
721        let InlineObject::Group { children, .. } = &group else { panic!() };
722        let c = &children[2];
723        let InlineObject::Shape { w, h, float, .. } = &c.obj else { panic!() };
724        assert_eq!((c.x, c.y, *w, *h, float.flip_h, float.flip_v), (20.0, 5.0, 80.0, 50.0, false, false));
725        assert!(!reroute_group(&mut group), "settled");
726        // A fully resolved connector does not need an invertible group scale.
727        for (width, height) in [(400.0, 150.0), (0.0, 0.0), (f32::NAN, f32::INFINITY)] {
728            let mut scaled = group.clone();
729            let InlineObject::Group { w, h, children, .. } = &mut scaled else { panic!() };
730            (*w, *h) = (width, height);
731            let InlineObject::Shape { float, .. } = &mut children[2].obj else { panic!() };
732            float.rot = 90.0;
733            assert!(reroute_group(&mut scaled));
734            let InlineObject::Group { children, .. } = &scaled else { panic!() };
735            let c = &children[2];
736            let InlineObject::Shape { w, h, float, .. } = &c.obj else { panic!() };
737            assert_eq!(ends(c.x, c.y, *w, *h, float.flip_h, float.flip_v), ((20.0, 5.0), (100.0, 55.0)));
738            assert!(!reroute_group(&mut scaled));
739        }
740        // A connection to a missing id or site keeps that end where it was.
741        let InlineObject::Group { children, .. } = &mut group else { panic!() };
742        if let InlineObject::Shape { extra, .. } = &mut children[2].obj {
743            extra.end = Some(ConnEnd { id: 404, site: 0 });
744            extra.start = Some(ConnEnd { id: 7, site: 99 });
745        }
746        assert!(!reroute_group(&mut group));
747    }
