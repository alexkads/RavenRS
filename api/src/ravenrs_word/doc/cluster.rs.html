1//! Text clusters for complex scripts (Myanmar, Khmer, Tai Tham, the Indic scripts): where the
2//! caret may stop, what Backspace removes, and where Myanmar text may wrap.
3//!
4//! Extended grapheme clusters (UAX #29) split some syllables that a shaper draws as one unit: a
5//! Myanmar consonant stacked under a virama (`မ္ဘ`), the Myanmar vowel signs U+102B/U+102C and
6//! the visarga U+1038 (excluded from the grapheme `SpacingMark` class), Khmer subscript
7//! consonants after the coeng. A caret between those would sit inside a glyph cluster, so the
8//! caret steps and layout clusters both use [`joins_cluster`] on top of the grapheme rules.
9
10use unicode_segmentation::UnicodeSegmentation;
11
12/// Invisible stackers: the next consonant is drawn under this one (Myanmar virama, Khmer coeng,
13/// Tai Tham sakot).
14fn is_stacker(c: char) -> bool {
15    matches!(u32::from(c), 0x1039 | 0x17D2 | 0x1A60)
16}
17
18/// Dependent vowel signs, medials, tone marks and other combining signs of Myanmar, Khmer and
19/// Tai Tham: never the start of a cluster.
20fn is_dependent(c: char) -> bool {
21    matches!(u32::from(c),
22        0x102B..=0x103E
23        | 0x1056..=0x1059
24        | 0x105E..=0x1060
25        | 0x1062..=0x1064
26        | 0x1067..=0x106D
27        | 0x1071..=0x1074
28        | 0x1082..=0x108D
29        | 0x108F
30        | 0x109A..=0x109D
31        | 0xA9E5
32        | 0xAA7B..=0xAA7D
33        | 0x17B4..=0x17D1
34        | 0x17D3
35        | 0x17DD
36        | 0x1A55..=0x1A5E
37        | 0x1A61..=0x1A7C
38        | 0x1A7F)
39}

41/// No caret stop between `prev` and `next` although a grapheme boundary falls there: `next` is a
42/// sign that belongs to the syllable before it, or `prev` stacks the next consonant.
43pub fn joins_cluster(prev: char, next: char) -> bool {
44    is_stacker(prev) || is_dependent(next)
45}
46
47/// Is `off` a cluster boundary of `text` (0 and `text.len()` always are)?
48pub fn is_cluster_boundary(text: &str, off: usize) -> bool {
49    if off == 0 || off >= text.len() {
50        return true;
51    }
52    if !text.is_char_boundary(off) {
53        return false;
54    }
55    let mut g = unicode_segmentation::GraphemeCursor::new(off, text.len(), true);
56    if !g.is_boundary(text, 0).unwrap_or(true) {
57        return false;
58    }
59    let prev = text.get(..off).and_then(|t| t.chars().next_back());
60    let next = text.get(off..).and_then(|t| t.chars().next());
61    !matches!((prev, next), (Some(p), Some(n)) if joins_cluster(p, n))
62}
63
64/// Byte offsets where clusters start (graphemes, joined per [`joins_cluster`]), in order; the
65/// first is 0 for non-empty text.
66pub fn cluster_starts(text: &str) -> Vec<usize> {
67    // ASCII: every character starts a grapheme except LF after CR, and none joins (UAX #29).
68    if text.is_ascii() {
69        let b = text.as_bytes();
70        return (0..b.len()).filter(|&i| !(i > 0 && b.get(i - 1) == Some(&b'\r') && b.get(i) == Some(&b'\n'))).collect();
71    }
72    let mut out: Vec<usize> = Vec::new();
73    let mut prev: Option<char> = None;
74    for (i, g) in text.grapheme_indices(true) {
75        let first = g.chars().next();
76        let joined = matches!((prev, first), (Some(p), Some(n)) if joins_cluster(p, n));
77        if !joined {
78            out.push(i);
79        }
80        prev = g.chars().next_back();
81    }
82    out
83}

85/// `off` clamped to `text` and moved back to a character boundary.
86fn char_floor(text: &str, off: usize) -> usize {
87    let mut o = off.min(text.len());
88    while o > 0 && !text.is_char_boundary(o) {
89        o -= 1;
90    }
91    o
92}
93
94/// The previous cluster boundary before `off` (0 at the start).
95pub fn prev_cluster(text: &str, off: usize) -> usize {
96    let mut off = char_floor(text, off);
97    loop {
98        let mut c = unicode_segmentation::GraphemeCursor::new(off, text.len(), true);
99        off = c.prev_boundary(text, 0).ok().flatten().unwrap_or(0);
100        if is_cluster_boundary(text, off) {
101            return off;
102        }
103    }
104}
105
106/// The next cluster boundary after `off` (`text.len()` at the end).
107pub fn next_cluster(text: &str, off: usize) -> usize {
108    let mut off = char_floor(text, off);
109    loop {
110        let mut c = unicode_segmentation::GraphemeCursor::new(off, text.len(), true);
111        off = c.next_boundary(text, 0).ok().flatten().unwrap_or(text.len());
112        if is_cluster_boundary(text, off) {
113            return off;
114        }
115    }
116}

118/// Characters Backspace removes individually rather than with their whole cluster: Arabic
119/// combining marks and characters of the Brahmic scripts of South and South-East Asia.
120pub fn backspaces_by_char(c: char) -> bool {
121    matches!(u32::from(c),
122        0x0610..=0x061A // Arabic nonspacing marks (Unicode 17.0)
123        | 0x064B..=0x065F
124        | 0x0670
125        | 0x06D6..=0x06DC
126        | 0x06DF..=0x06E4
127        | 0x06E7..=0x06E8
128        | 0x06EA..=0x06ED
129        | 0x0897..=0x089F
130        | 0x08CA..=0x08E1
131        | 0x08E3..=0x08FF
132        | 0x10EFA..=0x10EFF
133        | 0x0900..=0x0DFF // Devanagari … Sinhala
134        | 0x0E00..=0x0EFF // Thai, Lao
135        | 0x0F00..=0x0FFF // Tibetan
136        | 0x1000..=0x109F // Myanmar
137        | 0x1780..=0x17FF // Khmer
138        | 0x1A20..=0x1AAF // Tai Tham
139        | 0x1B00..=0x1B7F // Balinese
140        | 0xA980..=0xA9FF // Javanese, Myanmar Extended-B
141        | 0xAA60..=0xAA7F) // Myanmar Extended-A
142}

144/// A Myanmar character that starts a syllable: a consonant (with the Mon, Shan and Karen ones),
145/// an independent vowel, the great sa.
146fn myanmar_syllable_start(c: char) -> bool {
147    matches!(u32::from(c), 0x1000..=0x102A | 0x103F | 0x1050..=0x1055 | 0x105A..=0x105D | 0x1061 | 0x1065..=0x1066 | 0x106E..=0x1070 | 0x1075..=0x1081 | 0x108E)
148}

150/// Line-break opportunities inside Myanmar text, which has no spaces between words: before each
151/// syllable (UAX #14 leaves class SA to a dictionary; Word breaks Myanmar between syllables). A
152/// consonant starts a syllable unless it is stacked (after U+1039) or a final (followed, perhaps
153/// after a medial, by the asat U+103A, the virama U+1039 or the dot below U+1037). Byte offsets,
154/// sorted; only between two Myanmar characters.
155pub fn myanmar_breaks(text: &str) -> Vec<usize> {
156    let chars: Vec<(usize, char)> = text.char_indices().collect();
157    let myanmar = |c: char| matches!(u32::from(c), 0x1000..=0x109F | 0xA9E0..=0xA9FF | 0xAA60..=0xAA7F);
158    let mut out = Vec::new();
159    for k in 1..chars.len() {
160        let (Some(&(i, c)), Some(&(_, prev))) = (chars.get(k), chars.get(k - 1)) else { continue };
161        if !myanmar(prev) || !myanmar_syllable_start(c) || is_stacker(prev) {
162            continue;
163        }
164        // Skip one medial ya/ha, then look for a killer.
165        let mut j = k + 1;
166        if chars.get(j).is_some_and(|&(_, n)| matches!(n, '\u{103B}' | '\u{103E}')) {
167            j += 1;
168        }
169        if chars.get(j).is_some_and(|&(_, n)| matches!(n, '\u{1037}' | '\u{1039}' | '\u{103A}')) {
170            continue;
171        }
172        out.push(i);
173    }
174    out
175}

181    /// "မြန်မာစာ" (Myanmar script): မြ | န် | မာ | စာ.
182    const MYANMAR: &str = "\u{1019}\u{103C}\u{1014}\u{103A}\u{1019}\u{102C}\u{1005}\u{102C}";
183
184    fn split<'a>(s: &'a str, at: &[usize]) -> Vec<&'a str> {
185        let mut v = Vec::new();
186        for (n, &a) in at.iter().enumerate() {
187            let b = at.get(n + 1).copied().unwrap_or(s.len());
188            v.push(&s[a..b]);
189        }
190        v
191    }

193    #[test]
194    fn myanmar_vowel_signs_stay_in_their_syllable() {
195        // Graphemes alone would cut "မာ" before the aa sign U+102C.
196        assert_eq!(split(MYANMAR, &cluster_starts(MYANMAR)), vec!["မြ", "န်", "မာ", "စာ"]);
197        // The prevowel ေ (U+1031) follows its consonant in memory: one cluster, as is visarga း.
198        assert_eq!(split("ကြေး", &cluster_starts("ကြေး")), vec!["ကြေး"]);
199        // Stacked consonants (virama U+1039) and kinzi join the syllable they hang from.
200        let stacked = "\u{1019}\u{1039}\u{1018}\u{102C}"; // မ္ဘာ
201        assert_eq!(cluster_starts(stacked), vec![0]);
202        let kinzi = "\u{1021}\u{1004}\u{103A}\u{1039}\u{1002}\u{101C}\u{102D}\u{1015}\u{103A}"; // အင်္ဂလိပ်
203        assert_eq!(split(kinzi, &cluster_starts(kinzi)), vec!["အ", "င်္ဂ", "လိ", "ပ်"]);
204    }

206    /// The ASCII shortcut gives the grapheme rules' answer.
207    #[test]
208    fn ascii_cluster_starts_match_the_graphemes() {
209        let graphemes = |t: &str| -> Vec<usize> { t.grapheme_indices(true).map(|(i, _)| i).collect() };
210        let all: String = (0u8..128).map(char::from).collect();
211        for t in ["", "a", "Hello, world.", "\r\n", "a\r\nb\n\rc\r\r\n\n", "\t \u{7f}x\u{0}", all.as_str()] {
212            assert_eq!(cluster_starts(t), graphemes(t), "{t:?}");
213        }
214        for a in 0u8..128 {
215            for b in 0u8..128 {
216                let t = format!("{}{}", char::from(a), char::from(b));
217                assert_eq!(cluster_starts(&t), graphemes(&t), "{t:?}");
218            }
219        }
220    }

222    #[test]
223    fn khmer_subscripts_and_indic_conjuncts_are_one_cluster() {
224        // ខ្មែរ: ខ + coeng + ម + ែ, then រ.
225        assert_eq!(split("ខ្មែរ", &cluster_starts("ខ្មែរ")), vec!["ខ្មែ", "រ"]);
226        // Devanagari conjuncts are grapheme clusters already.
227        assert_eq!(split("क्षि हि", &cluster_starts("क्षि हि")), vec!["क्षि", " ", "हि"]);
228        // Latin is untouched: one cluster per letter, combining accents included.
229        assert_eq!(split("e\u{301}a", &cluster_starts("e\u{301}a")), vec!["e\u{301}", "a"]);
230    }

232    #[test]
233    fn caret_steps_skip_over_whole_clusters() {
234        let s = MYANMAR;
235        assert_eq!(next_cluster(s, 0), 6);
236        assert_eq!(next_cluster(s, 12), 18, "past the aa sign, not before it");
237        assert_eq!(prev_cluster(s, 18), 12);
238        assert_eq!(prev_cluster(s, 0), 0);
239        assert_eq!(next_cluster(s, s.len()), s.len());
240        assert!(!is_cluster_boundary(s, 15) && is_cluster_boundary(s, 12));
241        // Hostile offsets: past the end, inside a character.
242        assert_eq!(next_cluster(s, 1_000), s.len());
243        assert!(!is_cluster_boundary(s, 1));
244        assert_eq!(next_cluster(s, 1), 6);
245        assert_eq!(prev_cluster(s, 13), 6);
246        assert_eq!(prev_cluster("", 5), 0);
247        assert_eq!(next_cluster("", 0), 0);
248    }

250    #[test]
251    fn myanmar_wraps_between_syllables() {
252        // မြန်မာစာ: not before န (န် is a final, killed by the asat), so before မာ and စာ.
253        assert_eq!(myanmar_breaks(MYANMAR), vec![12, 18]);
254        // Never before a stacked consonant or one followed by medial + asat.
255        assert!(myanmar_breaks("\u{1019}\u{1039}\u{1018}\u{102C}").is_empty());
256        assert!(myanmar_breaks("ကျွန်").is_empty(), "န် is the final of ကျွန်");
257        // Not between Latin and Myanmar (UAX #14 handles that) or inside Latin.
258        assert!(myanmar_breaks("abcက").is_empty());
259        assert!(myanmar_breaks("").is_empty());
260    }
