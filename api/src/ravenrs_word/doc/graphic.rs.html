10/// A chart, SmartArt diagram or native ink preview, as items ready to draw (and, for an editable chart or SmartArt
11/// graphic, its model).
12#[derive(Clone, Debug, Default, PartialEq, Serialize, Deserialize)]
13#[serde(rename_all = "camelCase")]
14pub struct Graphic {
15    pub kind: GraphicKind,
16    /// What to draw, in points from the object's top-left corner.
17    #[serde(default)]
18    pub items: Vec<GraphicItem>,
19    /// The size (points) the items were built for. Drawn scaled to the object's rectangle when
20    /// that differs (layout shrinks oversized inline objects); 0 means the rectangle's own size.
21    #[serde(default)]
22    pub w: f32,
23    #[serde(default)]
24    pub h: f32,
25    /// Where the graphic was read from, so saving writes the original drawing itself back (not
26    /// only these items). `None` for graphics made some other way.
27    #[serde(default, skip_serializing_if = "Option::is_none")]
28    pub source: Option<std::sync::Arc<Embedded>>,
29    /// A SmartArt graphic WordCraft can edit (one it made, or one read back exactly as it wrote
30    /// it): saving writes the diagram parts from it, and the items are drawn from it. See
31    /// [`crate::doc::smart_art`].
32    #[serde(default, skip_serializing_if = "Option::is_none")]
33    pub smart_art: Option<std::sync::Arc<crate::doc::smart_art::SmartArtSpec>>,
34    /// A chart WordCraft can edit (one it made, or one read back that holds nothing more): saving
35    /// writes the chart part from it, and the items are drawn from it. See [`crate::doc::chart`].
36    #[serde(default, skip_serializing_if = "Option::is_none")]
37    pub chart: Option<std::sync::Arc<crate::doc::chart::ChartSpec>>,
38}

40/// What an object read from a file needs to be written back as it was: its own markup in the
41/// source format and the package parts that markup refers to. Never edited; WordCraft only moves
42/// and resizes the object around it. The parts themselves (bytes, content type and their own
43/// relationships) are kept in [`crate::doc::Document::passthrough`] under their part names.
44#[derive(Clone, Debug, Default, PartialEq, Serialize, Deserialize)]
45pub struct Embedded {
46    /// The object's markup as a standalone XML element with its namespace declarations (DOCX:
47    /// the `a:graphic` of a chart, SmartArt diagram or native ink, an ink `mc:AlternateContent`,
48    /// the `w:object` of an OLE object, or the `w:pict` of a signature line).
49    pub xml: String,
50    /// The relationships it uses, by the ids in `xml` (or in its parts, for a SmartArt drawing).
51    pub refs: Vec<EmbeddedRef>,
52}
53
54impl Embedded {
55    /// The root element of an ODF embedded object's markup: its files are kept in
56    /// [`crate::doc::Document::passthrough`] for the ODT writer; DOCX shows it as its picture.
57    pub const ODF_OBJECT: &'static str = "<wordcraft:odf-object";
58
59    /// Is it an ODF embedded object (a spreadsheet in an ODT), not Word markup?
60    pub fn is_odf_object(&self) -> bool {
61        self.xml.trim_start().starts_with(Self::ODF_OBJECT)
62    }
63}

65/// One relationship an [`Embedded`] object uses.
66#[derive(Clone, Debug, Default, PartialEq, Serialize, Deserialize)]
67pub struct EmbeddedRef {
68    /// The relationship id the markup uses.
69    pub id: String,
70    /// Relationship type (a URI).
71    pub kind: String,
72    /// The part name (a `Document::passthrough` key), or the URL of an external target.
73    pub target: String,
74    pub external: bool,
75}

77#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash, Serialize, Deserialize)]
78#[serde(rename_all = "camelCase")]
79pub enum GraphicKind {
80    #[default]
81    Chart,
82    Diagram,
83    /// Word's native InkML, kept opaque; displayed as its fallback picture when available.
84    Ink,
85}
86
87impl GraphicKind {
88    /// What the graphic is, in words: the alt text of one that has none of its own.
89    pub fn noun(self) -> &'static str {
90        match self {
91            GraphicKind::Chart => "chart",
92            GraphicKind::Diagram => "diagram",
93            GraphicKind::Ink => "ink",
94        }
95    }
96}

98#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
99#[serde(tag = "type", rename_all = "camelCase")]
100pub enum GraphicItem {
101    /// A preset shape filling `rect` ([x, y, w, h]).
102    Shape {
103        rect: [f32; 4],
104        kind: ShapeKind,
105        fill: Option<Rgb>,
106        stroke: Option<Rgb>,
107        #[serde(default)]
108        stroke_width: f32,
109    },
110    /// A path: polygons, pie slices, line series.
111    Path {
112        segs: Vec<PathSeg>,
113        fill: Option<Rgb>,
114        stroke: Option<Rgb>,
115        #[serde(default)]
116        stroke_width: f32,
117    },
118    /// One line of text in `rect`, aligned horizontally by `align`, centred vertically.
119    Text {
120        rect: [f32; 4],
121        text: String,
122        size: f32,
123        color: Rgb,
124        #[serde(default)]
125        bold: bool,
126        #[serde(default)]
127        align: TextAlign,
128        #[serde(default)]
129        font: Option<String>,
130    },
131    /// A picture (`Document::media` key) stretched into `rect`.
132    Image { rect: [f32; 4], media: String },
133}
