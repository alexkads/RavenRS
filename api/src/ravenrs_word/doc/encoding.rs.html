1//! Windows-1252 (the byte encoding legacy Word text pieces use) and the other single-byte Windows
2//! code pages RTF files name with `\ansicpg` or a font's `\fcharset` (Cyrillic 1251, Central
3//! European 1250, Greek, Turkish, Hebrew, Arabic, Baltic, Vietnamese, Thai), Mac OS Roman
4//! (10000, RTF `\mac`), and the East Asian multibyte code pages (932 Shift-JIS, 936 GBK, 949 UHC,
5//! 950 Big5), whose bytes decode as runs rather than one at a time.
6

13/// One Windows-1252 byte as a char.
14pub fn cp1252(b: u8) -> char {
15    if (0x80..0xA0).contains(&b) {
16        CP1252.get((b - 0x80) as usize).and_then(|u| char::from_u32(u32::from(*u))).unwrap_or('\u{FFFD}')
17    } else {
18        b as char
19    }
20}

227/// The OEM code pages [`oem_byte`] decodes.
228pub const OEM_CODE_PAGES: [u16; 6] = [437, 737, 850, 852, 857, 866];
229
230/// One byte of an OEM code page in [`OEM_CODE_PAGES`] (any other code page reads as 437) as a
231/// char, with the graphic glyphs for 0x01..=0x1F and 0x7F (⌂); 0x00 stays NUL.
232pub fn oem_byte(code_page: u16, b: u8) -> char {
233    let table: &[u16; 128] = match code_page {
234        737 => &CP737,
235        850 => &CP850,
236        852 => &CP852,
237        857 => &CP857,
238        866 => &CP866,
239        _ => &CP437,
240    };
241    let u = match b {
242        0x01..=0x1F => OEM_GLYPHS.get(usize::from(b - 1)).copied(),
243        0x7F => Some(0x2302),
244        0x80..=0xFF => table.get(usize::from(b - 0x80)).copied(),
245        _ => Some(u16::from(b)),
246    };
247    u.and_then(|u| char::from_u32(u32::from(u))).unwrap_or('\u{FFFD}')
248}

250/// The single-byte Windows code pages [`windows_byte`] decodes (1252 included).
251pub const SINGLE_BYTE_CODE_PAGES: [u16; 10] = [874, 1250, 1251, 1252, 1253, 1254, 1255, 1256, 1257, 1258];
252
253/// Mac OS Roman, RTF's `\mac` character set.
254pub const MAC_ROMAN_CODE_PAGE: u16 = 10000;
255
256/// One byte of a single-byte code page as a char: a Windows one in [`SINGLE_BYTE_CODE_PAGES`] or
257/// Mac OS Roman ([`MAC_ROMAN_CODE_PAGE`]); any other code page reads as Windows-1252.
258pub fn windows_byte(code_page: u16, b: u8) -> char {
259    let table: &[u16; 128] = match code_page {
260        874 => &CP874,
261        1250 => &CP1250,
262        1251 => &CP1251,
263        1253 => &CP1253,
264        1254 => &CP1254,
265        1255 => &CP1255,
266        1256 => &CP1256,
267        1257 => &CP1257,
268        1258 => &CP1258,
269        MAC_ROMAN_CODE_PAGE => &MAC_ROMAN,
270        _ => return cp1252(b),
271    };
272    if b < 0x80 {
273        return b as char;
274    }
275    table.get(usize::from(b - 0x80)).and_then(|u| char::from_u32(u32::from(*u))).unwrap_or('\u{FFFD}')
276}

278/// Whether [`code_page_byte`] decodes `code_page`: a Windows one ([`SINGLE_BYTE_CODE_PAGES`]), an
279/// OEM (DOS) one ([`OEM_CODE_PAGES`]) or Mac OS Roman. Document formats (RTF) name all three.
280pub fn is_byte_code_page(code_page: u16) -> bool {
281    SINGLE_BYTE_CODE_PAGES.contains(&code_page) || OEM_CODE_PAGES.contains(&code_page) || code_page == MAC_ROMAN_CODE_PAGE
282}
283
284/// A byte of 8-bit document text in `code_page` (see [`is_byte_code_page`]) as a char. Below 0x80
285/// it is ASCII in every code page (text, so an OEM page's glyphs for control codes don't apply).
286pub fn code_page_byte(code_page: u16, b: u8) -> char {
287    if b >= 0x80 && OEM_CODE_PAGES.contains(&code_page) { oem_byte(code_page, b) } else { windows_byte(code_page, b) }
288}

290/// The East Asian multibyte code pages a document format names with `\ansicpg` or a font's `\cpg`:
291/// 932 Shift-JIS (Japanese), 936 GBK (Simplified Chinese), 949 UHC (Korean) and 950 Big5
292/// (Traditional Chinese).
293pub const MULTIBYTE_CODE_PAGES: [u16; 4] = [932, 936, 949, 950];
294
295/// Whether `code_page` is one of the East Asian multibyte code pages ([`MULTIBYTE_CODE_PAGES`]):
296/// its bytes decode as a run, a lead byte together with its trail byte.
297pub fn is_multibyte_code_page(code_page: u16) -> bool {
298    multibyte_encoding_name(code_page).is_some()
299}
300
301/// The WHATWG encoding name of an East Asian multibyte code page (see [`MULTIBYTE_CODE_PAGES`]),
302/// for `encoding_rs::Encoding::for_label`; `None` for the single-byte code pages, whose bytes
303/// decode one at a time, and anything else.
304pub fn multibyte_encoding_name(code_page: u16) -> Option<&'static str> {
305    Some(match code_page {
306        932 => "shift_jis",
307        936 => "gbk",
308        949 => "euc-kr",
309        950 => "big5",
310        _ => return None,
311    })
312}

314/// Whether a document format can name `code_page` for its text: a single-byte Windows, DOS or Mac
315/// one ([`is_byte_code_page`]) or an East Asian multibyte one ([`is_multibyte_code_page`]).
316pub fn is_text_code_page(code_page: u16) -> bool {
317    is_byte_code_page(code_page) || is_multibyte_code_page(code_page)
318}

320/// The code page of an RTF or Word font charset (`\fcharset`): 204 Russian → 1251, 238 Eastern
321/// European → 1250, 77 Mac → 10000, 254 PC / 255 OEM → 437, 128 Japanese → 932, 129 Korean → 949,
322/// 134 Simplified Chinese → 936, 136 Traditional Chinese → 950… `None` for the ANSI and default
323/// charsets (the document's code page applies) and symbol fonts.
324pub fn charset_code_page(charset: i32) -> Option<u16> {
325    Some(match charset {
326        77 => MAC_ROMAN_CODE_PAGE,
327        128 => 932,
328        129 => 949,
329        134 => 936,
330        136 => 950,
331        161 => 1253,
332        162 => 1254,
333        163 => 1258,
334        177 => 1255,
335        178 => 1256,
336        186 => 1257,
337        204 => 1251,
338        222 => 874,
339        238 => 1250,
340        254 | 255 => 437,
341        _ => return None,
342    })
