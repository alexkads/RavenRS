15/// A copied piece of a document: blocks, where the first and last paragraphs may be partial.
16#[derive(Clone, Debug, PartialEq, Default, Serialize, Deserialize)]
17pub struct Fragment {
18    pub blocks: Vec<Block>,
19    /// Copies of the text box stories its shapes show, by the source document's part id. Pasting
20    /// gives each box a new story made from these.
21    #[serde(default, skip_serializing_if = "BTreeMap::is_empty")]
22    pub parts: BTreeMap<u32, Part>,
23    /// Media items referenced by objects in this fragment.
24    #[serde(default, skip_serializing_if = "BTreeMap::is_empty", with = "crate::doc::byte_store")]
25    pub media: BTreeMap<String, Arc<Vec<u8>>>,
26}

28impl Fragment {
29    pub fn plain_text(&self) -> String {
30        self.to_document().plain_text(crate::doc::StoryRef::Body)
31    }
32    /// The fragment as the body of a new document (default styles).
33    pub fn to_document(&self) -> Document {
34        let mut d = Document::new();
35        d.body = self.blocks.iter().cloned().map(Arc::new).collect();
36        d.parts = self.parts.clone();
37        d.media = self.media.clone();
38        d
39    }
40    /// Whether the fragment holds an equation (in tables too), which a copy also puts on the
41    /// system clipboard as MathML (#816).
42    pub fn has_equation(&self) -> bool {
43        let mut found = false;
44        for b in &self.blocks {
45            each_para(b, 0, &mut |p| found |= p.objects.iter().any(|o| matches!(o, InlineObject::Equation { .. })));
46        }
47        found
48    }
49    /// The equation (its linear format and whether it is a display equation) when the fragment
50    /// is that one equation and nothing else but white space and empty paragraphs: a copy of it
51    /// also goes on the system clipboard in the MathML formats that equation editors read (#1297).
52    pub fn sole_equation(&self) -> Option<(&str, bool)> {
53        let mut found = None;
54        for b in &self.blocks {
55            let Block::Para(p) = b else { return None };
56            for c in p.text.chars().filter(|c| !c.is_whitespace()) {
57                if c != crate::doc::para::OBJ || found.is_some() {
58                    return None;
59                }
60                // The paragraph's only visible character, so its first object.
61                match p.objects.first() {
62                    Some(InlineObject::Equation { linear, display, .. }) => found = Some((linear.as_str(), *display)),
63                    _ => return None,
64                }
65            }
66        }
67        found
68    }
69    /// The text a copy puts on the clipboard: [`Self::plain_text`] with symbol-font characters
70    /// (Wingdings…) as their Unicode equivalents.
71    pub fn clipboard_text(&self) -> String {
72        self.to_document().shown_text(crate::doc::StoryRef::Body)
73    }
74    /// Add `piece` at the end, followed by a paragraph mark unless it already ends with one: how
75    /// Word's Spike collects what it cuts. The piece's text box stories get ids after this
76    /// fragment's, so pieces copied from different documents (or after an id was freed and reused)
77    /// never share one.
78    pub fn append_piece(&mut self, mut piece: Fragment) {
79        let Some(last) = piece.blocks.last() else { return };
80        let ends_with_mark = piece.blocks.len() >= 2 && matches!(last, Block::Para(p) if p.is_empty());
81        if !ends_with_mark {
82            let mut mark = Paragraph::new();
83            if let Block::Para(p) = last {
84                mark.props = p.props.clone();
85            }
86            piece.blocks.push(Block::Para(mark));
87        }
88        let next = self.parts.keys().next_back().map_or(1, |k| k.saturating_add(1));
89        let ids: BTreeMap<u32, u32> = piece.parts.keys().zip(next..=u32::MAX).map(|(old, new)| (*old, new)).collect();
90        let renumber = |b: &mut Block| {
91            each_para_mut(b, 0, &mut |p| {
92                let mut changed = false;
93                for o in p.objects.iter_mut() {
94                    for slot in o.text_box_slots() {
95                        if let Some(new) = slot.and_then(|old| ids.get(&old)) {
96                            *slot = Some(*new);
97                            changed = true;
98                        }
99                    }
100                }
101                if changed {
102                    p.touch();
103                }
104            })
105        };
106        piece.blocks.iter_mut().for_each(renumber);
107        for (old, mut part) in std::mem::take(&mut piece.parts) {
108            part.blocks.iter_mut().map(Arc::make_mut).for_each(renumber);
109            if let Some(new) = ids.get(&old) {
110                self.parts.insert(*new, part);
111            }
112        }
113        // The empty paragraph after the last mark is where the next piece starts.
114        if self.blocks.len() >= 2 && matches!(self.blocks.last(), Some(Block::Para(p)) if p.is_empty()) {
115            self.blocks.pop();
116        }
117        self.blocks.append(&mut piece.blocks);
118    }
119    pub fn from_text(s: &str) -> Fragment {
120        let lines: Vec<&str> = s.split('\n').collect();
121        Fragment {
122            blocks: lines.iter().map(|l| Block::Para(Paragraph::with_text(l.trim_end_matches('\r'), CharProps::default()))).collect(),
123            ..Default::default()
124        }
125    }

150/// How many tables deep `t` goes, itself included: 1 when its cells hold no table (without
151/// recursion: a pasted fragment may be deep).
152fn table_height(t: &Table) -> usize {
153    let mut most = 1;
154    let mut stack: Vec<(&Block, usize)> = cell_blocks(t).map(|b| (b, 1)).collect();
155    while let Some((b, d)) = stack.pop() {
156        if let Block::Table(inner) = b {
157            let d = d.saturating_add(1);
158            most = most.max(d);
159            stack.extend(cell_blocks(inner).map(|cb| (cb, d)));
160        }
161    }
162    most
163}

165/// `b` as blocks nesting at most `room` tables deep: a table that would go deeper is replaced by
166/// its paragraphs (tables in its cells too, document order), as the DOCX reader does.
167fn fit_tables(b: Block, room: usize, out: &mut Vec<Block>) {
168    match b {
169        Block::Table(t) if room == 0 => {
170            let mut stack: Vec<&Block> = cell_blocks(&t).rev().collect();
171            while let Some(cb) = stack.pop() {
172                match cb {
173                    Block::Para(p) => out.push(Block::Para(p.clone())),
174                    Block::Table(inner) => stack.extend(cell_blocks(inner).rev()),
175                }
176            }
177        }
178        Block::Table(mut t) if table_height(&t) > room => {
179            for cell in t.rows.iter_mut().flat_map(|r| r.cells.iter_mut()) {
180                let mut kept = Vec::new();
181                for cb in std::mem::take(&mut cell.blocks) {
182                    fit_tables(Arc::unwrap_or_clone(cb), room.saturating_sub(1), &mut kept);
183                }
184                cell.blocks = kept.into_iter().map(Arc::new).collect();
185            }
186            out.push(Block::Table(t));
187        }
188        b => out.push(b),
189    }
190}

194/// Most text box stories one paste may create (hostile documents can fan out).
195const MAX_PASTED_BOXES: usize = 4096;
196/// Most text box content (estimated bytes, see [`block_cost`]) one paste may copy: thousands of
197/// boxes can show the same story, and each needs a story of its own. Blocks without text boxes
198/// are shared between those stories, not copied, so they are not charged per box.
199const MAX_PASTED_BOX_BYTES: usize = 32 << 20;

201/// What a paste may still spend on text box stories.
202struct BoxBudget {
203    stories: usize,
204    bytes: usize,
205    /// The source stories being copied, innermost last: a box inside one of them showing it
206    /// again is a cycle.
207    active: Vec<u32>,
208    /// Each source story's cost ([`block_cost`]), worked out once per paste: a story too big for
209    /// what is left is not walked again for every box that shows it.
210    costs: BTreeMap<u32, usize>,
211    /// Boxes left without a story because the paste's budget couldn't hold theirs.
212    skipped: usize,
213    /// A box the fragment has no story for may show this document's story of that id (a
214    /// fragment copied from this document without its parts). Off for another document's
215    /// fragment, whose ids mean nothing here.
216    local: bool,
217}

221        BoxBudget { stories: MAX_PASTED_BOXES, bytes: MAX_PASTED_BOX_BYTES, active: Vec::new(), costs: BTreeMap::new(), skipped: 0, local: true }
222    }

225/// Rough size of a block's content, and whether it holds text boxes (and so must be copied).
226fn block_cost(b: &Block) -> (usize, bool) {
227    let (mut cost, mut boxes) = (std::mem::size_of::<Block>(), false);
228    each_para(b, 0, &mut |p| {
229        let items = p
230            .runs
231            .len()
232            .saturating_mul(std::mem::size_of::<Run>())
233            .saturating_add(p.objects.len().saturating_mul(std::mem::size_of::<InlineObject>()));
234        let objects = p.objects.iter().fold(0usize, |n, o| n.saturating_add(object_cost(o)));
235        cost = cost.saturating_add(p.text.len()).saturating_add(items).saturating_add(objects);
236        // The formatting a clone copies too: style ids, font names and the like in the runs,
237        // the mark and the paragraph's properties.
238        let runs = p.runs.iter().filter(|r| !r.props.is_empty()).fold(0usize, |n, r| n.saturating_add(serialized_len(&r.props)));
239        let props = serialized_len(&p.props).saturating_add(serialized_len(&p.mark)).saturating_add(p.section.as_ref().map_or(0, serialized_len));
240        cost = cost.saturating_add(runs).saturating_add(props);
241        boxes |= box_ids(p).next().is_some();
242    });
243    (cost, boxes)
244}

246/// Rough heap bytes a clone of `o` copies beyond its own storage: its strings and lists, and its
247/// group members however deep. What it shares through `Arc` (pictures' embedded objects, charts,
248/// freeforms, wrap polygons, gradients) is not copied, so not counted.
249fn object_cost(o: &InlineObject) -> usize {
250    let float = |f: &crate::doc::para::Float| f.alt.text.len().saturating_add(f.alt.title.len());
251    let mut cost = 0usize;
252    let mut todo = vec![o];
253    while let Some(o) = todo.pop() {
254        let add = match o {
255            InlineObject::Image { media, alt, float: f, .. } => media.len().saturating_add(alt.len()).saturating_add(float(f)),
256            InlineObject::Graphic { alt, float: f, .. } => alt.len().saturating_add(float(f)),
257            InlineObject::Shape { float: f, extra, .. } => {
258                let warp = extra.warp.as_ref().map_or(0, |w| {
259                    let adj = w.adj.iter().map(|(name, _)| std::mem::size_of::<(String, i64)>().saturating_add(name.len()));
260                    adj.fold(w.preset.len(), usize::saturating_add)
261                });
262                float(f).saturating_add(warp)
263            }
264            InlineObject::Group { float: f, children, .. } => {
265                todo.extend(children.iter().map(|c| &c.obj));
266                children.len().saturating_mul(std::mem::size_of::<crate::doc::para::GroupChild>()).saturating_add(float(f))
267            }
268            InlineObject::Field { instr, result, form, .. } => instr.len().saturating_add(result.len()).saturating_add(form.len()),
269            InlineObject::FieldStart { instr, .. } => instr.len(),
270            InlineObject::ControlStart { control } => {
271                std::mem::size_of::<crate::doc::control::ContentControl>().saturating_add(serialized_len(control))
272            }
273            InlineObject::NoteRef { custom, .. } => custom.len(),
274            InlineObject::BookmarkStart { name } | InlineObject::BookmarkEnd { name } => name.len(),
275            InlineObject::PermStart { group, editor, .. } => group.len().saturating_add(editor.len()),
276            InlineObject::Equation { linear, math, .. } => linear.len().saturating_add(math.heap_bytes()),
277            InlineObject::Ruby { base, ruby, lang, props, .. } => {
278                [base.len(), ruby.len(), lang.len(), std::mem::size_of::<CharProps>(), serialized_len(props)]
279                    .into_iter()
280                    .fold(0, usize::saturating_add)
281            }
282            InlineObject::Opaque { format, xml, text } => format.len().saturating_add(xml.len()).saturating_add(text.len()),
283            InlineObject::FieldEnd
284            | InlineObject::ControlEnd
285            | InlineObject::CommentStart { .. }
286            | InlineObject::CommentEnd { .. }
287            | InlineObject::PermEnd { .. }
288            | InlineObject::PositionalTab { .. } => 0,
289        };
290        cost = cost.saturating_add(add);
291    }
292    cost
293}

295/// Length of `v` written as JSON: at least the bytes of every string and list it owns. For the
296/// nested formatting and control types, which share nothing through `Arc`.
297fn serialized_len<T: Serialize>(v: &T) -> usize {
298    struct Count(usize);
299    impl std::io::Write for Count {
300        fn write(&mut self, b: &[u8]) -> std::io::Result<usize> {
301            self.0 = self.0.saturating_add(b.len());
302            Ok(b.len())
303        }
304        fn flush(&mut self) -> std::io::Result<()> {
305            Ok(())
306        }
307    }
308    let mut n = Count(0);
309    let _ = serde_json::to_writer(&mut n, v);
310    n.0
311}

313/// Call `f` on every paragraph of a block, in document order, however deep its tables nest. The
314/// walk keeps its own stack instead of recursing, so it is complete: a text box it doesn't reach
315/// is shown nowhere, which pruning relies on (#1339). `_depth` is unused (callers keep passing it).
316pub(crate) fn each_para<'a>(b: &'a Block, _depth: usize, f: &mut dyn FnMut(&'a Paragraph)) {
317    // Most blocks are paragraphs, which need no stack.
318    if let Block::Para(p) = b {
319        return f(p);
320    }
321    let mut todo = vec![b];
322    while let Some(b) = todo.pop() {
323        match b {
324            Block::Para(p) => f(p),
325            Block::Table(t) => {
326                // Last first, so they come off the stack in document order.
327                let cells = t.rows.iter().rev().flat_map(|r| r.cells.iter().rev());
328                todo.extend(cells.flat_map(|c| c.blocks.iter().rev()).map(|cb| &**cb));
329            }
330        }
331    }
332}

336    if let Block::Para(p) = b {
337        return f(p);
338    }
339    let mut todo = vec![b];
340    while let Some(b) = todo.pop() {
341        match b {
342            Block::Para(p) => f(p),
343            Block::Table(t) => {
344                let cells = t.rows.iter_mut().rev().flat_map(|r| r.cells.iter_mut().rev());
345                todo.extend(cells.flat_map(|c| c.blocks.iter_mut().rev()).map(Arc::make_mut));
346            }
347        }
348    }
349}

351/// Whether a paragraph anywhere in `b` (also inside tables) uses paragraph or character style
352/// `id` (tables to 16 levels deep). Table styles are the engine's `table.deleteStyle`'s job.
353fn uses_style(b: &Block, id: &str, depth: usize) -> bool {
354    let is = |s: &Option<String>| s.as_deref() == Some(id);
355    match b {
356        Block::Para(p) => is(&p.props.style) || is(&p.mark.style) || p.runs.iter().any(|r| is(&r.props.style)),
357        Block::Table(t) if depth < 16 => {
358            t.rows.iter().flat_map(|r| r.cells.iter()).flat_map(|c| c.blocks.iter()).any(|cb| uses_style(cb, id, depth + 1))
359        }
360        Block::Table(_) => false,
361    }
362}

364/// Point every paragraph and run in `b` that uses style `from` at `to`; returns how many
365/// paragraphs changed.
366fn restyle_block(b: &mut Block, from: &str, to: &Option<String>, depth: usize) -> usize {
367    let fix = |s: &mut Option<String>| {
368        if s.as_deref() == Some(from) {
369            s.clone_from(to);
370            true
371        } else {
372            false
373        }
374    };
375    match b {
376        Block::Para(p) => {
377            let mut changed = fix(&mut p.props.style);
378            changed |= fix(&mut p.mark.style);
379            for r in &mut p.runs {
380                changed |= fix(&mut r.props.style);
381            }
382            if changed {
383                p.normalize();
384                p.touch();
385            }
386            usize::from(changed)
387        }
388        Block::Table(t) => {
389            let mut n = 0;
390            if depth < 16 {
391                for cell in t.rows.iter_mut().flat_map(|r| r.cells.iter_mut()) {
392                    for cb in &mut cell.blocks {
393                        if uses_style(cb, from, depth + 1) {
394                            n += restyle_block(Arc::make_mut(cb), from, to, depth + 1);
395                        }
396                    }
397                }
398            }
399            n
400        }
401    }
402}

404/// What [`Document::adopt_with`] renumbers in another document's content: media keys, lists,
405/// note and comment stories, comment ids, text boxes (one it doesn't bring is dropped), revisions
406/// (after ours, at `rev`; an id past the `revs` it brings is dropped) and styles (those kept under new ids; unstyled paragraphs take
407/// `default_style` when it's given).
408struct Relink<'a> {
409    media: &'a BTreeMap<String, String>,
410    nums: &'a BTreeMap<u32, u32>,
411    notes: &'a BTreeMap<u32, u32>,
412    comments: &'a BTreeMap<u32, u32>,
413    boxes: &'a BTreeMap<u32, Part>,
414    /// Added to permission range ids: ours come first.
415    perms: u32,
416    rev: u32,
417    revs: u32,
418    styles: &'a BTreeMap<String, String>,
419    default_style: Option<&'a str>,
420}
421
422impl Relink<'_> {
423    fn block(&self, b: &mut Block) {
424        // Its own stack, like `each_para_mut`: tables however deeply nested are relinked (#1339).
425        let mut todo = vec![b];
426        while let Some(b) = todo.pop() {
427            match b {
428                Block::Para(p) => self.para(p),
429                Block::Table(t) => {
430                    self.style(&mut t.props.style);
431                    self.shift_change(&mut t.props.fmt_change);
432                    if let Some(c) = t.props.fmt_change.as_mut() {
433                        self.style(&mut c.old.style);
434                    }
435                    for row in t.rows.iter_mut() {
436                        self.shift_change(&mut row.props.fmt_change);
437                        for cell in row.cells.iter_mut() {
438                            self.shift_change(&mut cell.props.fmt_change);
439                            todo.extend(cell.blocks.iter_mut().map(Arc::make_mut));
440                        }
441                    }
442                }
443            }
444        }
445    }
446
447    fn style(&self, id: &mut Option<String>) {
448        if let Some(new) = id.as_ref().and_then(|i| self.styles.get(i)) {
449            *id = Some(new.clone());
450        }
451    }
452
453    /// A paragraph's style: unstyled ones take `default_style` when it's given.
454    fn para_style(&self, id: &mut Option<String>) {
455        match (&*id, self.default_style) {
456            (None, Some(d)) => *id = Some(d.to_string()),
457            _ => self.style(id),
458        }
459    }
460
461    fn num(&self, n: &mut Option<crate::doc::props::NumRef>) {
462        if let Some(n) = n.as_mut()
463            && n.num != 0
464        {
465            n.num = self.nums.get(&n.num).copied().unwrap_or(0);
466        }
467    }
468
469    /// Our id for the other document's revision `r`, if it has one.
470    fn rev_id(&self, r: u32) -> Option<u32> {
471        if r < self.revs { self.rev.checked_add(r) } else { None }
472    }
473
474    fn shift_change<T>(&self, c: &mut Option<Box<crate::doc::props::PropChange<T>>>) {
475        match (c.as_ref().and_then(|c| self.rev_id(c.rev)), c.as_mut()) {
476            (Some(v), Some(x)) => x.rev = v,
477            _ => *c = None,
478        }
479    }
480
481    fn para(&self, p: &mut Paragraph) {
482        self.num(&mut p.props.numbering);
483        self.para_style(&mut p.props.style);
484        for r in &mut p.runs {
485            self.style(&mut r.props.style);
486        }
487        self.style(&mut p.mark.style);
488        for c in p.runs.iter_mut().map(|r| &mut r.props).chain([&mut p.mark]) {
489            c.ins = c.ins.and_then(|r| self.rev_id(r));
490            c.del = c.del.and_then(|r| self.rev_id(r));
491            self.shift_change(&mut c.fmt_change);
492            // A tracked change's old formatting names styles and lists as the live one does.
493            if let Some(f) = c.fmt_change.as_mut() {
494                self.style(&mut f.old.style);
495            }
496        }
497        self.shift_change(&mut p.props.fmt_change);
498        if let Some(f) = p.props.fmt_change.as_mut() {
499            self.para_style(&mut f.old.style);
500            self.num(&mut f.old.numbering);
501        }
502        if let Some(s) = p.section.as_deref_mut() {
503            self.shift_change(&mut s.fmt_change);
504        }
505        match (p.props.num_change.as_ref().and_then(|c| self.rev_id(c.rev)), p.props.num_change.as_mut()) {
506            (Some(v), Some(c)) => c.rev = v,
507            _ => p.props.num_change = None,
508        }
509        for o in &mut p.objects {
510            self.object(o, 0);
511        }
512        p.touch();
513    }

515    fn object(&self, o: &mut InlineObject, depth: usize) {
516        // An id that names nothing here becomes one that names nothing at all.
517        let map = |m: &BTreeMap<u32, u32>, id: &mut u32| *id = m.get(id).copied().unwrap_or(u32::MAX);
518        match o {
519            InlineObject::Image { media, .. } => {
520                if let Some(k) = self.media.get(media.as_str()) {
521                    *media = k.clone();
522                }
523            }
524            InlineObject::Graphic { graphic, .. } => {
525                let mapped = |m: &str| self.media.get(m).filter(|k| k.as_str() != m);
526                if graphic.items.iter().any(|i| matches!(i, crate::doc::graphic::GraphicItem::Image { media, .. } if mapped(media).is_some())) {
527                    for i in &mut Arc::make_mut(graphic).items {
528                        if let crate::doc::graphic::GraphicItem::Image { media, .. } = i
529                            && let Some(k) = self.media.get(media.as_str())
530                        {
531                            *media = k.clone();
532                        }
533                    }
534                }
535            }
536            InlineObject::Group { children, .. } if depth < MAX_BOX_DEPTH => children.iter_mut().for_each(|c| self.object(&mut c.obj, depth + 1)),
537            InlineObject::NoteRef { id, .. } => map(self.notes, id),
538            InlineObject::CommentStart { id } | InlineObject::CommentEnd { id } => map(self.comments, id),
539            InlineObject::PermStart { id, .. } | InlineObject::PermEnd { id } => *id = id.saturating_add(self.perms),
540            _ => {}
541        }
542        for story in o.text_box_slots() {
543            if story.is_some_and(|id| !self.boxes.contains_key(&id)) {
544                *story = None;
545            }
546        }
547    }
548}

609    /// Convert notes between footnotes and endnotes: footnotes (`foot`) become endnotes and
610    /// endnotes (`end`) footnotes, both at once to swap them. Each note's kind changes, as do its
611    /// reference marks wherever they are (the body, text boxes, the note's own mark) and the
612    /// note styles (Footnote Text and Footnote Reference to their endnote twins and back).
613    /// Returns how many notes changed.
614    pub fn convert_notes(&mut self, foot: bool, end: bool) -> usize {
615        use crate::doc::para::NoteKind;
616        let flips: std::collections::HashSet<u32> = self
617            .parts
618            .iter()
619            .filter(|(_, p)| (foot && p.kind == PartKind::Footnote) || (end && p.kind == PartKind::Endnote))
620            .map(|(id, _)| *id)
621            .collect();
622        if flips.is_empty() {
623            return 0;
624        }
625        let swap = |s: &mut Option<String>| {
626            let to = match s.as_deref() {
627                Some("FootnoteText") => "EndnoteText",
628                Some("EndnoteText") => "FootnoteText",
629                Some("FootnoteReference") => "EndnoteReference",
630                Some("EndnoteReference") => "FootnoteReference",
631                _ => return,
632            };
633            *s = Some(to.to_string());
634        };
635        let refers = |p: &Paragraph| p.objects.iter().any(|o| matches!(o, InlineObject::NoteRef { id, .. } if flips.contains(id)));
636        // In a paragraph: flip the converted notes' marks and their runs' reference style; in a
637        // converted note's own story also the paragraph style.
638        let convert = |p: &mut Paragraph, in_note: bool| {
639            if !in_note && !refers(p) {
640                return;
641            }
642            let offs = p.object_offsets();
643            let mut at = Vec::new();
644            for (k, o) in p.objects.iter_mut().enumerate() {
645                if let InlineObject::NoteRef { kind, id, .. } = o
646                    && flips.contains(id)
647                {
648                    *kind = if *kind == NoteKind::Footnote { NoteKind::Endnote } else { NoteKind::Footnote };
649                    at.extend(offs.get(k).copied());
650                }
651            }
652            // Reference offsets and run boundaries are ordered; do not restart at the first
653            // reference for every run, including runs after the last reference.
654            let mut at = at.into_iter().peekable();
655            let mut start = 0usize;
656            for r in &mut p.runs {
657                crate::doc::work::steps(1);
658                let end = start.saturating_add(r.len);
659                while at.peek().is_some_and(|&o| o < start) {
660                    crate::doc::work::steps(1);
661                    at.next();
662                }
663                if at.peek().is_some_and(|&o| o < end) {
664                    swap(&mut r.props.style);
665                }
666                start = end;
667            }
668            if in_note {
669                swap(&mut p.props.style);
670                swap(&mut p.mark.style);
671            }
672            p.touch();
673        };
674        let touches = |b: &Block| {
675            let mut any = false;
676            each_para(b, 0, &mut |p| any |= refers(p));
677            any
678        };
679        for b in self.body.iter_mut() {
680            if touches(b) {
681                each_para_mut(Arc::make_mut(b), 0, &mut |p| convert(p, false));
682            }
683        }
684        for (id, part) in self.parts.iter_mut() {
685            let own = flips.contains(id);
686            if own {
687                part.kind = if part.kind == PartKind::Footnote { PartKind::Endnote } else { PartKind::Footnote };
688            }
689            for b in part.blocks.iter_mut() {
690                if own || touches(b) {
691                    each_para_mut(Arc::make_mut(b), 0, &mut |p| convert(p, own));
692                }
693            }
694        }
695        flips.len()
696    }

785    /// Delete `a..b`. Returns the position where the deletion collapsed to. On error the document
786    /// is unchanged: both ends are checked before anything is deleted.
787    pub fn delete_range(&mut self, a: &Pos, b: &Pos) -> Result<Pos> {
788        let (a, b) = order(a, b);
789        if a == b {
790            return Ok(a);
791        }
792        if a.story != b.story {
793            return Err(DocError::Invalid("range spans stories".into()));
794        }
795        // Check both ends before the first edit, so an error never leaves the document half-deleted.
796        for p in [&a, &b] {
797            match self.para(p.story, &p.path) {
798                Some(para) if para.clamp(p.off) == p.off => {}
799                Some(_) => return Err(DocError::BadOffset(p.off)),
800                None => return Err(DocError::BadPath(p.path.to_string())),
801            }
802        }
803        // Content control markers whose partner is outside the range stay (the control is
804        // emptied, not broken).
805        let mut kept = self.kept_control_markers(&a, &b);
806        // The walk below groups `kept` by paragraph (`chunk_by`) and looks offsets up
807        // (`binary_search`), so it must be sorted; `kept_control_markers` returns it so, and this
808        // keeps a change there from quietly losing markers.
809        if !kept.is_sorted() {
810            kept.sort();
811        }
812        let kept_in = |path: &Path| -> Vec<usize> { kept.iter().filter(|(p, _)| p == path).map(|(_, o)| *o).collect() };
813        if a.path == b.path {
814            self.para_mut(a.story, &a.path)?.delete_except(a.off, b.off, &kept_in(&a.path))?;
815            return Ok(a);
816        }
817        if a.path.parent() == b.path.parent() {
818            // Same container: trim the ends, drop the blocks between, join.
819            let (ia, ib) = (a.path.last(), b.path.last());
820            // Removing an already-empty paragraph must not impose its style or mark formatting
821            // on the following content. Test before trimming: deleting actual heading text is
822            // still an ordinary merge that keeps the first paragraph's properties.
823            let empty_start = self.para(a.story, &a.path).is_some_and(Paragraph::is_empty);
824            // Kept markers of the paragraphs in between move to the join.
825            let mut moved: Vec<(InlineObject, CharProps)> = Vec::new();
826            // `kept` is sorted: one walk of each paragraph's objects and runs (#3614).
827            for group in kept.chunk_by(|x, y| x.0 == y.0) {
828                let Some((path, _)) = group.first() else { continue };
829                if *path == a.path || *path == b.path {
830                    continue;
831                }
832                let Some(p) = self.para(a.story, path) else { continue };
833                let mut runs = p.run_ranges().peekable();
834                for (off, o) in p.objects_with_offsets() {
835                    if group.binary_search_by_key(&off, |(_, k)| *k).is_err() {
836                        continue;
837                    }
838                    while runs.next_if(|(r, _)| r.end <= off).is_some() {}
839                    let props = runs.peek().filter(|(r, _)| r.contains(&off)).map_or(&p.mark, |(_, c)| *c);
840                    moved.push((o.clone(), props.clone()));
841                }
842            }
843            {
844                let pa = self.para_mut(a.story, &a.path)?;
845                let end = pa.len();
846                pa.delete_except(a.off, end, &kept_in(&a.path))?;
847                for (o, props) in moved {
848                    let at = pa.len();
849                    pa.insert_object(at, o, &props)?;
850                }
851            }
852            let mut tail = {
853                let pb = self.para_mut(b.story, &b.path)?;
854                pb.delete_except(0, b.off, &kept_in(&b.path))?;
855                pb.clone()
856            };
857            let bl = self.container_mut(a.story, &a.path)?;
858            // Section breaks on the marks in between go with their paragraphs.
859            let between: Vec<_> = bl
860                .get(ia + 1..ib)
861                .unwrap_or_default()
862                .iter()
863                .filter_map(|b| match &**b {
864                    Block::Para(p) => p.section.clone(),
865                    Block::Table(_) => None,
866                })
867                .collect();
868            if ib < bl.len() && ia < ib {
869                bl.drain(ia + 1..=ib);
870            }
871            // The deleted marks include `a`'s, so a section break there goes too: the joined
872            // paragraph ends where `b`'s mark did, and its text takes the following section's
873            // setup, as in Word (#2127).
874            let sect = tail.section.take();
875            let pa = self.para_mut(a.story, &a.path)?;
876            let keep_props = if empty_start { tail.props.clone() } else { pa.props.clone() };
877            let keep_mark = empty_start.then(|| tail.mark.clone());
878            pa.append(tail);
879            pa.props = keep_props;
880            if let Some(mark) = keep_mark {
881                pa.mark = mark;
882            }
883            let before = std::mem::replace(&mut pa.section, sect);
884            if a.story == crate::doc::StoryRef::Body && a.path.0.len() == 1 {
885                // Nearest first, as "link to previous" resolved them before the delete.
886                let following = self.section_mut(a.path.last());
887                for s in between.iter().rev().chain(before.iter()) {
888                    following.absorb(s);
889                }
890            }
891            return Ok(a);
892        }
893        // Different containers (range enters or leaves a table): clear text in every paragraph
894        // inside, then remove the tables it covers whole (below).
895        // Whether `b` is at its paragraph's end, before that paragraph is trimmed.
896        let b_at_end = self.para(b.story, &b.path).is_some_and(|p| p.len() == b.off);
897        let paths = self.paths_between(&a, &b);
898        for p in paths.iter().rev() {
899            let (from, to) = {
900                let Some(para) = self.para(a.story, p) else { continue };
901                let from = if *p == a.path { a.off } else { 0 };
902                let to = if *p == b.path { b.off } else { para.len() };
903                (from, to)
904            };
905            self.para_mut(a.story, p)?.delete_except(from, to, &kept_in(p))?;
906        }
907        // Tables the range covers whole, in the innermost container holding both ends, go (#1198,
908        // #1844): those strictly between the ends, the one `a` is in when it starts at that
909        // table's very start, and the one `b` is in when it ends at that table's very end. A range
910        // that is one table and nothing else only empties it (Word's Delete keeps the table).
911        let common = a.path.0.iter().zip(&b.path.0).take_while(|(x, y)| x == y).count();
912        let level = common - common % 3;
913        let (Some(&ia), Some(&ib)) = (a.path.0.get(level), b.path.0.get(level)) else { return Ok(a) };
914        let (ia, ib) = (ia as usize, ib as usize);
915        if ia >= ib {
916            return Ok(a);
917        }
918        let prefix = a.path.0.get(..level).unwrap_or_default().to_vec();
919        let block_path = |i: usize| {
920            let mut v = prefix.clone();
921            v.push(i as u32);
922            Path(v)
923        };
924        // Paragraphs inside the block at index `i` of the container, in document order.
925        let paths = self.para_paths(a.story);
926        let inside = |i: usize| -> Vec<&Path> {
927            let pre = block_path(i);
928            paths.iter().filter(|q| q.0.len() > pre.0.len() && q.0.starts_with(&pre.0)).collect()
929        };
930        // A table with cells deeper than the paragraph walk goes has paragraphs `inside` never
931        // lists, so where it starts or ends isn't known: it is never taken as covered.
932        let walked = |i: usize| self.container(a.story, &block_path(i)).and_then(|bl| bl.get(i)).is_some_and(|b| !deeper_than_walk(b, level / 3));
933        let a_covers = a.path.0.len() > level + 1 && a.off == 0 && walked(ia) && inside(ia).first() == Some(&&a.path);
934        let b_covers = b.path.0.len() > level + 1 && walked(ib) && inside(ib).last() == Some(&&b.path) && b_at_end;
935        let first = if a_covers { ia } else { ia + 1 };
936        let last = if b_covers { ib } else { ib - 1 };
937        let mut removed = 0;
938        {
939            let bl = self.container_mut(a.story, &block_path(ia))?;
940            for i in (first..=last.min(bl.len().saturating_sub(1))).rev() {
941                if matches!(bl.get(i).map(|b| &**b), Some(Block::Table(_))) {
942                    bl.remove(i);
943                    removed += 1;
944                }
945            }
946            if removed == 0 {
947                return Ok(a);
948            }
949            // A story or cell always ends with a paragraph.
950            if bl.is_empty() || matches!(bl.last().map(|b| &**b), Some(Block::Table(_))) {
951                bl.push(para_block(Paragraph::new()));
952            }
953        }
954        // The paragraphs left between the ends join, as in a range within one container.
955        let at = if a_covers { Pos { story: a.story, path: block_path(ia), off: 0 } } else { a.clone() };
956        let end = if b_covers {
957            Pos { story: a.story, path: block_path((ib + 1).saturating_sub(removed)), off: 0 }
958        } else {
959            Pos { story: a.story, path: block_path(ib.saturating_sub(removed)), off: 0 }
960        };
961        let paras = |p: &Pos| p.path.0.len() == level + 1 && self.para(p.story, &p.path).is_some();
962        if paras(&at) && paras(&end) && at.path <= end.path {
963            return self.delete_range(&at, &end);
964        }
965        if self.para(at.story, &at.path).is_some() {
966            return Ok(at);
967        }
968        Ok(self.clamp(&at))
969    }

971    /// Apply `f` to the character formatting of `a..b`. A collapsed range changes nothing (the
972    /// engine keeps "pending" formatting for the caret instead).
973    pub fn format_range(&mut self, a: &Pos, b: &Pos, f: &dyn Fn(&mut CharProps)) -> Result<()> {
974        let (a, b) = order(a, b);
975        for p in self.paths_between(&a, &b) {
976            let para = self.para_mut(a.story, &p)?;
977            let from = if p == a.path { a.off } else { 0 };
978            let to = if p == b.path { b.off } else { para.len() };
979            para.format(from, to, f)?;
980            // A fully selected paragraph also gets the mark formatted (Word does), but a hyperlink
981            // ends before the mark: a linked mark would carry the link into every paragraph Enter
982            // makes from it (issue #413).
983            if to == para.len() && (from == 0 || p != a.path) {
984                let style = para.mark.style.clone();
985                f(&mut para.mark);
986                if para.mark.link.take().is_some() {
987                    para.mark.style = style.filter(|s| s != "Hyperlink");
988                }
989            }
990        }
991        Ok(())
992    }

1049    /// The table whose whole text `a..b` covers when the range runs across its cells: from the
1050    /// start of its first paragraph to the end of its last. Of the tables holding both ends, the
1051    /// outermost one covered whole (a table nested in a cell when the range covers just it).
1052    /// `None` for a range inside one cell, or one that doesn't cover a whole table.
1053    pub fn whole_table(&self, a: &Pos, b: &Pos) -> Option<Path> {
1054        let (a, b) = order(a, b);
1055        if a.story != b.story || a.off != 0 || a.path.parent() == b.path.parent() {
1056            return None;
1057        }
1058        let mut paths: Option<Vec<Path>> = None;
1059        // Table block paths are the prefixes of a cell paragraph's path one block deep, then
1060        // every three steps (block, row, cell) deeper.
1061        let mut n = 1usize;
1062        while n.saturating_add(3) <= a.path.0.len() {
1063            let pre = a.path.0.get(..n)?;
1064            if !b.path.0.starts_with(pre) {
1065                return None;
1066            }
1067            let tp = Path(pre.to_vec());
1068            if self.table(a.story, &tp).is_some() {
1069                let paths = paths.get_or_insert_with(|| self.para_paths(a.story));
1070                let mut inside = paths.iter().filter(|p| p.0.starts_with(pre));
1071                if let Some(first) = inside.next() {
1072                    let last = inside.next_back().unwrap_or(first);
1073                    let end = self.para(a.story, last).map_or(0, Paragraph::len);
1074                    if *first == a.path && *last == b.path && b.off >= end {
1075                        return Some(tp);
1076                    }
1077                }
1078            }
1079            n = n.saturating_add(3);
1080        }
1081        None
1082    }

1107    /// [`Self::adopt_with`], its permission range ids from at least `perms` on (and past ours);
1108    /// `perms` moves past them.
1109    fn adopt_in(&mut self, mut src: Document, keep_styles: bool, perms: &mut u32) -> Fragment {
1110        let mut renamed = BTreeMap::new();
1111        let mut default_style = None;
1112        if keep_styles {
1113            if src.styles.get("Normal").is_none() {
1114                Arc::make_mut(&mut src.styles).upsert(crate::doc::styles::Style { id: "Normal".into(), name: "Normal".into(), ..Default::default() });
1115            }
1116            // Ids given out here are taken too: two styles never get the same new one.
1117            let mut given = std::collections::HashSet::new();
1118            let room = src.styles.styles.len().saturating_mul(2).saturating_add(self.styles.styles.len()).saturating_add(1);
1119            for st in &src.styles.styles {
1120                if self.styles.get(&st.id).is_some() {
1121                    let fresh = (1..=room)
1122                        .map(|n| format!("{}{n}", st.id))
1123                        .find(|id| self.styles.get(id).is_none() && src.styles.get(id).is_none() && !given.contains(id));
1124                    if let Some(fresh) = fresh {
1125                        given.insert(fresh.clone());
1126                        renamed.insert(st.id.clone(), fresh);
1127                    }
1128                }
1129            }
1130            default_style = Some(renamed.get("Normal").cloned().unwrap_or_else(|| "Normal".into()));
1131        }
1132        let restyle = |id: &mut Option<String>| {
1133            if let Some(new) = id.as_ref().and_then(|i| renamed.get(i)) {
1134                *id = Some(new.clone());
1135            }
1136        };
1137        let mut media = BTreeMap::new();
1138        for (key, data) in &src.media {
1139            let ext = key.rsplit_once('.').map(|(_, e)| e).unwrap_or("png");
1140            media.insert(key.clone(), self.add_media(data.to_vec(), ext));
1141        }
1142        let next_abs = self.numbering.abstracts.iter().map(|a| a.id.saturating_add(1)).max().unwrap_or(0);
1143        let mut abs_map = BTreeMap::new();
1144        for (i, a) in src.numbering.abstracts.iter().enumerate() {
1145            let id = next_abs.saturating_add(i as u32);
1146            abs_map.insert(a.id, id);
1147            let mut a = crate::doc::numbering::AbstractNum { id, ..a.clone() };
1148            a.levels.iter_mut().for_each(|l| restyle(&mut l.style));
1149            Arc::make_mut(&mut self.numbering).abstracts.push(a);
1150        }
1151        let next_num = self.numbering.nums.iter().map(|n| n.id.saturating_add(1)).max().unwrap_or(1).max(1);
1152        let mut nums = BTreeMap::new();
1153        for (i, n) in src.numbering.nums.iter().enumerate() {
1154            let id = next_num.saturating_add(i as u32);
1155            nums.insert(n.id, id);
1156            let abstract_id = abs_map.get(&n.abstract_id).copied().unwrap_or(n.abstract_id);
1157            let mut n = crate::doc::numbering::Num { id, abstract_id, ..n.clone() };
1158            n.level_overrides.iter_mut().for_each(|(_, l)| restyle(&mut l.style));
1159            Arc::make_mut(&mut self.numbering).nums.push(n);
1160        }
1161        let renumber = |n: &mut Option<crate::doc::props::NumRef>| {
1162            if let Some(n) = n.as_mut()
1163                && n.num != 0
1164            {
1165                n.num = nums.get(&n.num).copied().unwrap_or(0);
1166            }
1167        };
1168        // What a style in the other document doesn't set comes from its defaults there, and
1169        // Word's own (no space, single lines) where those don't say.
1170        let mut para_defaults = src.styles.default_para.clone();
1171        para_defaults.space_before.get_or_insert(0.0);
1172        para_defaults.space_after.get_or_insert(0.0);
1173        para_defaults.line_spacing.get_or_insert(crate::doc::props::LineSpacing::Multiple(1.0));
1174        for st in &src.styles.styles {
1175            let mut st = st.clone();
1176            // Our defaults stay ours.
1177            st.default = false;
1178            if keep_styles {
1179                if let Some(new) = renamed.get(&st.id) {
1180                    st.name = format!("{} ({new})", st.name);
1181                    st.id = new.clone();
1182                }
1183                restyle(&mut st.based_on);
1184                restyle(&mut st.next);
1185                restyle(&mut st.linked);
1186                // A character style's runs get the defaults through their paragraph's style.
1187                if st.based_on.is_none() && st.kind != crate::doc::styles::StyleKind::Character {
1188                    let mut chr = src.styles.default_chr.clone();
1189                    chr.overlay(&st.chr);
1190                    st.chr = chr;
1191                    let mut para = para_defaults.clone();
1192                    para.overlay(&st.para);
1193                    st.para = para;
1194                }
1195            }
1196            if self.styles.get(&st.id).is_none() {
1197                renumber(&mut st.para.numbering);
1198                Arc::make_mut(&mut self.styles).upsert(st);
1199            }
1200        }
1201        let rev = u32::try_from(self.revisions.len()).unwrap_or(u32::MAX);
1202        self.revisions.extend(src.revisions.iter().cloned());
1203        // Notes and comments get stories (and comments ids) of their own here; their text is
1204        // relinked below, once every new id is known.
1205        let mut notes = BTreeMap::new();
1206        for (id, part) in src.parts.iter().filter(|(_, p)| matches!(p.kind, PartKind::Footnote | PartKind::Endnote)) {
1207            notes.insert(*id, self.add_part(part.kind, Vec::new()));
1208        }
1209        let mut comments = BTreeMap::new();
1210        let mut next_comment = self.comments.keys().next_back().map_or(0, |k| k.saturating_add(1));
1211        for (id, c) in &src.comments {
1212            if src.parts.get(&c.part).is_some_and(|p| p.kind == PartKind::Comment) {
1213                notes.insert(c.part, self.add_part(PartKind::Comment, Vec::new()));
1214                comments.insert(*id, next_comment);
1215                next_comment = next_comment.saturating_add(1);
1216            }
1217        }
1218        let boxes: BTreeMap<u32, Part> = src.parts.iter().filter(|(_, p)| p.kind == PartKind::TextBox).map(|(k, p)| (*k, p.clone())).collect();
1219        // Its permission ranges are numbered after ours, so no end of theirs closes one of ours.
1220        let first_perm = perm_end(self.body.iter().chain(self.parts.values().flat_map(|p| p.blocks.iter()))).max(*perms);
1221        *perms = first_perm.saturating_add(perm_end(src.body.iter().chain(src.parts.values().flat_map(|p| p.blocks.iter()))));
1222        let link = Relink {
1223            media: &media,
1224            nums: &nums,
1225            notes: &notes,
1226            comments: &comments,
1227            boxes: &boxes,
1228            perms: first_perm,
1229            rev,
1230            revs: u32::try_from(src.revisions.len()).unwrap_or(u32::MAX),
1231            styles: &renamed,
1232            default_style: default_style.as_deref(),
1233        };
1234        let mut parts = boxes.clone();
1235        for part in parts.values_mut() {
1236            part.blocks.iter_mut().for_each(|b| link.block(Arc::make_mut(b)));
1237        }
1238        // A note's or comment's text boxes get stories here now; the body's go with the fragment.
1239        let mut budget = BoxBudget::new();
1240        for (old, new) in &notes {
1241            let Some(part) = src.parts.get(old) else { continue };
1242            let mut blocks: Vec<Block> = part.blocks.iter().map(|b| (**b).clone()).collect();
1243            blocks.iter_mut().for_each(|b| link.block(b));
1244            self.adopt_text_boxes(&mut blocks, &parts, 0, &mut budget);
1245            if let Some(p) = self.parts.get_mut(new) {
1246                p.blocks = blocks.into_iter().map(Arc::new).collect();
1247            }
1248        }
1249        for (id, c) in &src.comments {
1250            if let (Some(new), Some(part)) = (comments.get(id), notes.get(&c.part)) {
1251                let parent = c.parent.and_then(|p| comments.get(&p).copied());
1252                // Its durable id is unique in its own document only: the writer gives the copy one.
1253                self.comments.insert(*new, crate::doc::Comment { parent, part: *part, durable_id: None, ..c.clone() });
1254            }
1255        }
1256        let mut blocks: Vec<Block> = src.body.iter().map(|b| (**b).clone()).collect();
1257        blocks.iter_mut().for_each(|b| link.block(b));
1258        Fragment { blocks, parts, media: self.media.clone() }
1259    }

1292    /// Media items referenced by objects in `blocks`, nested text box stories included.
1293    pub fn media_of_blocks(&self, blocks: &[Block]) -> BTreeMap<String, Arc<Vec<u8>>> {
1294        let mut keys = std::collections::BTreeSet::new();
1295        /// Collect media keys from an inline object recursively.
1296        fn collect_object(o: &InlineObject, keys: &mut std::collections::BTreeSet<String>, depth: usize) {
1297            match o {
1298                InlineObject::Image { media, .. } => {
1299                    keys.insert(media.clone());
1300                }
1301                InlineObject::Graphic { graphic, .. } => {
1302                    for item in &graphic.items {
1303                        if let crate::doc::graphic::GraphicItem::Image { media, .. } = item {
1304                            keys.insert(media.clone());
1305                        }
1306                    }
1307                }
1308                InlineObject::Group { children, .. } if depth < MAX_BOX_DEPTH => {
1309                    for child in children {
1310                        collect_object(&child.obj, keys, depth + 1);
1311                    }
1312                }
1313                _ => {}
1314            }
1315        }
1316        let mut collect_para = |p: &Paragraph| {
1317            for o in &p.objects {
1318                collect_object(o, &mut keys, 0);
1319            }
1320        };
1321        for b in blocks {
1322            each_para(b, 0, &mut collect_para);
1323        }
1324        for part in self.text_box_parts(blocks).values() {
1325            for b in &part.blocks {
1326                each_para(b, 0, &mut collect_para);
1327            }
1328        }
1329        let mut out = BTreeMap::new();
1330        for k in keys {
1331            if let Some(data) = self.media.get(&k) {
1332                out.insert(k, data.clone());
1333            }
1334        }
1335        out
1336    }

1338    fn copy_blocks(&self, a: &Pos, b: &Pos) -> Vec<Block> {
1339        let (a, b) = order(a, b);
1340        let mut out = Vec::new();
1341        if a.story != b.story {
1342            return out;
1343        }
1344        if a.path.parent() == b.path.parent() {
1345            let Some(bl) = self.container(a.story, &a.path) else { return out };
1346            for i in a.path.last()..=b.path.last() {
1347                let Some(blk) = bl.get(i) else { break };
1348                match &**blk {
1349                    Block::Para(p) => {
1350                        let mut p = p.clone();
1351                        let to = if i == b.path.last() { b.off } else { p.len() };
1352                        let from = if i == a.path.last() { a.off } else { 0 };
1353                        let _ = p.delete(to.min(p.len()), p.len());
1354                        let _ = p.delete(0, from.min(p.len()));
1355                        if i == b.path.last() && to < self.para(a.story, &b.path).map(Paragraph::len).unwrap_or(0) {
1356                            p.section = None;
1357                        }
1358                        out.push(Block::Para(p));
1359                    }
1360                    Block::Table(t) => out.push(Block::Table(t.clone())),
1361                }
1362            }
1363            return out;
1364        }
1365        // Across the cells of a whole table: the table itself, not its cells' paragraphs (#1744).
1366        if let Some(tp) = self.whole_table(&a, &b)
1367            && let Some(blk) = self.block(a.story, &tp)
1368        {
1369            out.push(blk.clone());
1370            return out;
1371        }
1372        for p in self.paths_between(&a, &b) {
1373            if let Some(para) = self.para(a.story, &p) {
1374                let mut q = para.clone();
1375                let to = if p == b.path { b.off } else { q.len() };
1376                let from = if p == a.path { a.off } else { 0 };
1377                let _ = q.delete(to.min(q.len()), q.len());
1378                let _ = q.delete(0, from.min(q.len()));
1379                q.section = None;
1380                out.push(Block::Para(q));
1381            }
1382        }
1383        out
1384    }

1386    /// Give every text box in `blocks` a new story of its own, copied from `parts` (or, for a

1390        let mut any = false;
1391        for b in blocks.iter() {

1393        }
1394        if !any {
1395            return;
1396        }
1397        for b in blocks.iter_mut() {
1398            each_para_mut(b, 0, &mut |p| {

1400                for o in p.objects.iter_mut() {
1401                    for story in o.text_box_slots() {

1403                            *story = self.adopt_text_box(old, parts, depth, budget);
1404                            changed = true;

1407                }
1408                if changed {
1409                    p.touch();
1410                }

1415    /// A new text box story copied from `old`; its own boxes get new stories too. Blocks without
1416    /// text boxes are shared with the source (copy-on-write), the others copied. A box showing a
1417    /// story it is inside, or whose story the paste's budget can't hold, gets none (CWE-400);
1418    /// only that box: a later box with a smaller story is still copied while the budget lasts.
1419    fn adopt_text_box(&mut self, old: u32, parts: &BTreeMap<u32, Part>, depth: usize, budget: &mut BoxBudget) -> Option<u32> {
1420        if depth >= MAX_BOX_DEPTH || budget.active.contains(&old) {
1421            return None;
1422        }
1423        if budget.stories == 0 {
1424            budget.skipped = budget.skipped.saturating_add(1);
1425            return None;
1426        }
1427        let src = parts.get(&old).or_else(|| self.parts.get(&old).filter(|p| budget.local && p.kind == PartKind::TextBox))?;
1428        // Charge what the copy really allocates before making any of it: the block slots, and the
1429        // blocks holding text boxes (deep-copied below). The rest are shared `Arc`s, however many
1430        // boxes show them, so they cost nothing more. Worked out once per paste.
1431        let cost = *budget.costs.entry(old).or_insert_with(|| {
1432            let slots = src.blocks.len().saturating_mul(std::mem::size_of::<Arc<Block>>());
1433            src.blocks.iter().fold(slots, |a, b| match block_cost(b) {
1434                (c, true) => a.saturating_add(c),
1435                (_, false) => a,
1436            })
1437        });
1438        let Some(left) = budget.bytes.checked_sub(cost) else {
1439            budget.skipped = budget.skipped.saturating_add(1);
1440            return None;
1441        };
1442        budget.bytes = left;
1443        budget.stories -= 1;
1444        let src_blocks = src.blocks.clone();
1445        // The copy keeps its text direction and alignment, not its link (a copy stands alone).
1446        let body = crate::doc::TextBody { next: None, ..src.body };
1447        budget.active.push(old);
1448        let mut blocks: Blocks = Vec::with_capacity(src_blocks.len());
1449        for b in src_blocks {
1450            let mut boxes = false;
1451            each_para(&b, 0, &mut |p| boxes |= box_ids(p).next().is_some());
1452            if boxes {
1453                let mut copy = (*b).clone();
1454                self.adopt_text_boxes(std::slice::from_mut(&mut copy), parts, depth + 1, budget);
1455                blocks.push(Arc::new(copy));
1456            } else {
1457                blocks.push(b);
1458            }
1459        }
1460        budget.active.pop();
1461        let id = self.try_add_part(PartKind::TextBox, blocks).ok()?;
1462        if let Some(p) = self.parts.get_mut(&id) {
1463            p.body = body;
1464        }
1465        Some(id)
1466    }

1484    /// Insert a fragment at `pos`; returns the position after the pasted content.
1485    pub fn insert_fragment(&mut self, pos: &Pos, frag: &Fragment) -> Result<Pos> {
1486        let mut media_remap = BTreeMap::new();
1487        for (k, v) in &frag.media {
1488            if let Some(existing) = self.media.get(k) {
1489                if existing.as_slice() != v.as_slice() {
1490                    let ext = k.rsplit_once('.').map_or("png", |(_, ext)| ext);
1491                    let new_key = self.add_media((**v).clone(), ext);
1492                    media_remap.insert(k.clone(), new_key);
1493                }
1494            } else {
1495                self.media.insert(k.clone(), v.clone());
1496            }
1497        }
1498        // Tables that would nest deeper than the limit here become their paragraphs (#1596).
1499        let room = MAX_TABLE_DEPTH.saturating_sub(pos.path.depth());
1500        let mut blocks = Vec::with_capacity(frag.blocks.len());
1501        for b in frag.blocks.iter().cloned() {
1502            fit_tables(b, room, &mut blocks);
1503        }
1504        if blocks.is_empty() {
1505            return Ok(pos.clone());
1506        }
1507        if self.para_at(pos).is_none() {
1508            return Err(DocError::BadPath(pos.path.to_string()));
1509        }
1510        let mut parts = frag.parts.clone();
1511        if !media_remap.is_empty() {
1512            for b in &mut blocks {
1513                remap_media_block(b, &media_remap);
1514            }
1515            for part in parts.values_mut() {
1516                for b in &mut part.blocks {
1517                    remap_media_block(Arc::make_mut(b), &media_remap);
1518                }
1519            }
1520        }
1521        // Pasted text boxes get stories of their own (never shared with the copy's source).
1522        let mut budget = BoxBudget::new();
1523        self.adopt_text_boxes(&mut blocks, &parts, 0, &mut budget);
1524        if budget.skipped > 0 {
1525            // Once per paste, not once per box (review of #3874).
1526            log::warn!(
1527                "paste: {} text boxes are left empty; their stories are past what one paste copies ({MAX_PASTED_BOXES} stories, {MAX_PASTED_BOX_BYTES} bytes)",
1528                budget.skipped
1529            );
1530        }
1531        // Single paragraph: inline insert keeping runs.
1532        if blocks.len() == 1
1533            && let Some(Block::Para(src)) = blocks.first()
1534        {
1535            let src = src.clone();
1536            let target = self.para_mut(pos.story, &pos.path)?;
1537            let tail = target.split_off(pos.off)?;
1538            let added = src.len();
1539            let keep = target.props.clone();
1540            let sect = tail.section.clone();
1541            let mut src = src;
1542            src.section = None;
1543            target.append(src);
1544            target.append(tail);
1545            target.props = keep;
1546            target.section = sect;
1547            return Ok(Pos { off: pos.off + added, ..pos.clone() });
1548        }
1549        // Multi-block: split, merge first into left part, last into right part.
1550        let after = self.split_paragraph(pos)?;
1551        let left = pos.path.clone();
1552        let first = blocks.remove(0);
1553        let last = blocks.pop();
1554        let mut idx = left.last() + 1;
1555        match first {
1556            Block::Para(p) => {
1557                let l = self.para_mut(pos.story, &left)?;
1558                let keep = l.props.clone();
1559                let mut p2 = p;
1560                let pprops = p2.props.clone();
1561                p2.section = None;
1562                l.append(p2);
1563                l.props = if keep.is_empty() { pprops } else { keep };
1564            }
1565            t @ Block::Table(_) => {
1566                self.insert_block(pos.story, &left.with_last(idx), t)?;
1567                idx += 1;
1568            }
1569        }
1570        for b in blocks {
1571            self.insert_block(pos.story, &left.with_last(idx), b)?;
1572            idx += 1;
1573        }
1574        let right = after.path.with_last(idx);
1575        match last {
1576            Some(Block::Para(p)) => {
1577                let r = self.para_mut(pos.story, &right)?;
1578                let tail = std::mem::replace(r, p.clone());
1579                let off = r.len();
1580                let sect = tail.section.clone();
1581                r.append(tail);
1582                r.section = sect;
1583                Ok(Pos { story: pos.story, path: right, off })
1584            }
1585            Some(t @ Block::Table(_)) => {
1586                self.insert_block(pos.story, &right, t)?;
1587                Ok(Pos { story: pos.story, path: right.with_last(idx + 1), off: 0 })
1588            }
1589            None => Ok(Pos { story: pos.story, path: right, off: 0 }),
1590        }
1591    }
