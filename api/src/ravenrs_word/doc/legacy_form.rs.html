1//! Legacy form fields (`FORMCHECKBOX`, `FORMTEXT`, `FORMDROPDOWN`): what their `w:ffData` says,
2//! read from the raw XML a [`crate::doc::InlineObject::Field`] keeps in its `form` (ECMA-376 Part 1
3//! §17.16.17). Only what layout and PDF export need; the XML itself stays the source of truth
4//! for saving.
5
6/// Most drop-down entries kept (Word allows 25).
7pub const MAX_ENTRIES: usize = 64;
8
9/// A legacy form field's kind and state.
10#[derive(Clone, Debug, PartialEq)]
11pub enum LegacyForm {
12    /// A check box: `checked` is its state (`w:checked`, else `w:default`), `size` its explicit
13    /// size in points (`w:size`), `None` when it follows the text (`w:sizeAuto`).
14    CheckBox { name: String, checked: bool, size: Option<f32> },
15    /// A text field; its text is the field's result.
16    Text { name: String, max_len: Option<u32> },
17    /// A drop-down list showing entry `selected` (`w:result`, else `w:default`, else the first).
18    DropDown { name: String, entries: Vec<String>, selected: usize },
19}
20
21impl LegacyForm {
22    /// The field's definition, or `None` when `ff_data` holds none of the three.
23    pub fn parse(ff_data: &str) -> Option<LegacyForm> {
24        let name = attr_of(ff_data, "w:name").unwrap_or_default();
25        if let Some(cb) = element(ff_data, "w:checkBox") {
26            let default = attr_of(cb, "w:default").is_some_and(|v| on(&v));
27            // `<w:checked/>` with no value means checked.
28            let checked = match tag(cb, "w:checked") {
29                Some(t) => attr_in(t).is_none_or(|v| on(&v)),
30                None => default,
31            };
32            let size =
33                attr_of(cb, "w:size").and_then(|v| v.trim().parse::<f32>().ok()).filter(|v| v.is_finite() && *v > 0.0).map(|v| (v / 2.0).min(1584.0));
34            return Some(LegacyForm::CheckBox { name, checked, size });
35        }
36        if let Some(dd) = element(ff_data, "w:ddList") {
37            let entries: Vec<String> = tags(dd, "w:listEntry").filter_map(attr_in).take(MAX_ENTRIES).collect();
38            let pick = |n: &str| attr_of(dd, n).and_then(|v| v.trim().parse::<usize>().ok());
39            let selected = pick("w:result").or_else(|| pick("w:default")).filter(|i| *i < entries.len()).unwrap_or(0);
40            return Some(LegacyForm::DropDown { name, entries, selected });
41        }
42        if ff_data.contains("<w:textInput") {
43            let max_len =
44                element(ff_data, "w:textInput").and_then(|t| attr_of(t, "w:maxLength")).and_then(|v| v.trim().parse::<u32>().ok()).filter(|n| *n > 0);
45            return Some(LegacyForm::Text { name, max_len });
46        }
47        None
48    }
49
50    /// The text a drop-down shows (its selected entry); `None` for other kinds or an empty list.
51    pub fn shown_text(&self) -> Option<&str> {
52        match self {
53            LegacyForm::DropDown { entries, selected, .. } => entries.get(*selected).map(String::as_str),
54            _ => None,
55        }
56    }
57
58    /// The field's bookmark-style name (`w:name`), possibly empty.
59    pub fn name(&self) -> &str {
60        match self {
61            LegacyForm::CheckBox { name, .. } | LegacyForm::Text { name, .. } | LegacyForm::DropDown { name, .. } => name,
62        }
63    }
64}
65
66fn on(v: &str) -> bool {
67    !matches!(v.trim(), "0" | "false" | "off")
68}
69
70/// The element `name` in `xml` with its content: from its start tag to its end tag, or the
71/// self-closing tag alone.
72fn element<'a>(xml: &'a str, name: &str) -> Option<&'a str> {
73    let open = tag(xml, name)?;
74    let start = xml.find(open)?;
75    if open.ends_with("/>") {
76        return Some(open);
77    }
78    let close = format!("</{name}>");
79    let end = xml.get(start..)?.find(&close)? + start + close.len();
80    xml.get(start..end)
81}
82
83/// The first start (or empty) tag `<name ...>` in `xml`.
84fn tag<'a>(xml: &'a str, name: &str) -> Option<&'a str> {
85    tags(xml, name).next()
86}
87
88/// Every start (or empty) tag `<name ...>` in `xml`, in order.
89fn tags<'a>(xml: &'a str, name: &str) -> impl Iterator<Item = &'a str> + use<'a> {
90    let pat = format!("<{name}");
91    let mut from = 0;
92    std::iter::from_fn(move || {
93        loop {
94            let at = from + xml.get(from..)?.find(&pat)?;
95            let after = at + pat.len();
96            from = after;
97            // `<w:checkBox` must not match `<w:checkBoxes`: the name ends at a space, `/` or `>`.
98            if xml.get(after..).and_then(|s| s.chars().next()).is_some_and(|c| c == ' ' || c == '/' || c == '>') {
99                let end = after + xml.get(after..)?.find('>')? + 1;
100                return xml.get(at..end);
101            }
102        }
103    })
104}

106/// The `w:val` of the first tag `name` in `xml`.
107fn attr_of(xml: &str, name: &str) -> Option<String> {
108    tag(xml, name).and_then(attr_in)
109}
110
111/// The `w:val` of a single tag, XML entities resolved.
112fn attr_in(tag: &str) -> Option<String> {
113    let at = tag.find("w:val=")? + "w:val=".len();
114    let quote = tag.get(at..)?.chars().next()?;
115    if quote != '"' && quote != '\'' {
116        return None;
117    }
118    let rest = tag.get(at + 1..)?;
119    let end = rest.find(quote)?;
120    Some(unescape(rest.get(..end)?))
121}
122
123fn unescape(mut s: &str) -> String {
124    let mut out = String::with_capacity(s.len());
125    while let Some((text, rest)) = s.split_once('&') {
126        out.push_str(text);
127        if let Some(end) = rest.find(['&', ';'])
128            && rest.as_bytes()[end] == b';'
129        {
130            let entity = &rest[..end];
131            let decoded = match entity {
132                "lt" => Some('<'),
133                "gt" => Some('>'),
134                "quot" => Some('"'),
135                "apos" => Some('\''),
136                "amp" => Some('&'),
137                _ => entity.strip_prefix('#').and_then(|n| {
138                    let value = if let Some(hex) = n.strip_prefix('x') {
139                        (!hex.is_empty() && hex.bytes().all(|b| b.is_ascii_hexdigit())).then(|| u32::from_str_radix(hex, 16).ok()).flatten()
140                    } else {
141                        (!n.is_empty() && n.bytes().all(|b| b.is_ascii_digit())).then(|| n.parse::<u32>().ok()).flatten()
142                    }?;
143                    char::from_u32(value)
144                        .filter(|c| matches!(*c, '\t' | '\n' | '\r' | '\u{20}'..='\u{d7ff}' | '\u{e000}'..='\u{fffd}' | '\u{10000}'..='\u{10ffff}'))
145                }),
146            };
147            if let Some(c) = decoded {
148                out.push(c);
149                s = &rest[end + 1..];
150                continue;
151            }
152        }
153        out.push('&');
154        s = rest;
155    }
156    out.push_str(s);
157    out
158}

160#[cfg(test)]
161mod tests {
162    use super::*;
163
164    #[test]
165    fn drop_down_entities_decode_once_including_xml_whitespace() {
166        let xml = r#"<w:ffData><w:name w:val="N&#x61;me"/><w:ddList><w:listEntry w:val="A&#10;B&#9;C&#13;D&#x1F600;&amp;#10;&lt;&gt;&quot;&apos;"/></w:ddList></w:ffData>"#;
167        let f = LegacyForm::parse(xml).unwrap();
168        assert_eq!(f.name(), "Name");
169        assert_eq!(f.shown_text(), Some("A\nB\tC\rD😀&#10;<>\"'"));
170        let malformed = r#"<w:ffData><w:ddList><w:listEntry w:val="&#0;&#xD800;&#x110000;&#-1;&#xZZ;&unknown;&amp"/></w:ddList></w:ffData>"#;
171        assert_eq!(LegacyForm::parse(malformed).unwrap().shown_text(), Some("&#0;&#xD800;&#x110000;&#-1;&#xZZ;&unknown;&amp"));
172    }
173
174    #[test]
175    fn check_boxes_text_fields_and_drop_downs() {
176        let cb = |inner: &str| format!(r#"<w:ffData><w:name w:val="Box1"/><w:enabled/><w:checkBox>{inner}</w:checkBox></w:ffData>"#);
177        assert_eq!(
178            LegacyForm::parse(&cb(r#"<w:sizeAuto/><w:default w:val="0"/>"#)),
179            Some(LegacyForm::CheckBox { name: "Box1".into(), checked: false, size: None })
180        );
181        assert_eq!(
182            LegacyForm::parse(&cb(r#"<w:sizeAuto/><w:default w:val="1"/>"#)).map(|f| matches!(f, LegacyForm::CheckBox { checked: true, .. })),
183            Some(true)
184        );
185        // `w:checked` wins over `w:default`; without a value it means checked.
186        assert!(matches!(
187            LegacyForm::parse(&cb(r#"<w:default w:val="1"/><w:checked w:val="0"/>"#)),
188            Some(LegacyForm::CheckBox { checked: false, .. })
189        ));
190        assert!(matches!(LegacyForm::parse(&cb(r#"<w:default w:val="0"/><w:checked/>"#)), Some(LegacyForm::CheckBox { checked: true, .. })));
191        assert!(matches!(
192            LegacyForm::parse(&cb(r#"<w:size w:val="30"/><w:default w:val="0"/>"#)),
193            Some(LegacyForm::CheckBox { size: Some(15.0), .. })
194        ));
195        let dd = r#"<w:ffData><w:name w:val="List1"/><w:ddList><w:result w:val="1"/><w:listEntry w:val="Alpha"/><w:listEntry w:val="B &amp; C"/></w:ddList></w:ffData>"#;
196        let f = LegacyForm::parse(dd).unwrap();
197        assert_eq!(f.shown_text(), Some("B & C"));
198        assert_eq!(f.name(), "List1");
199        let dd0 = r#"<w:ffData><w:ddList><w:listEntry w:val="Only"/></w:ddList></w:ffData>"#;
200        assert_eq!(LegacyForm::parse(dd0).and_then(|f| f.shown_text().map(str::to_string)).as_deref(), Some("Only"));
201        let tx = r#"<w:ffData><w:name w:val="Name1"/><w:textInput><w:maxLength w:val="40"/></w:textInput></w:ffData>"#;
202        assert_eq!(LegacyForm::parse(tx), Some(LegacyForm::Text { name: "Name1".into(), max_len: Some(40) }));
203        assert_eq!(LegacyForm::parse(""), None);
204        assert_eq!(LegacyForm::parse("<w:ffData><w:name w:val=\"x\"/></w:ffData>"), None);
205    }
