9/// The same blocks: the same content. The blocks are compared by address first — the
10/// document is copy-on-write, so a block that changed was copied and has another address —
11/// and a block that was copied but holds the same content is compared by value.
12pub(crate) fn same_blocks(a: &Blocks, b: &Blocks) -> bool {
13    a.len() == b.len() && a.iter().zip(b).all(|(x, y)| Arc::ptr_eq(x, y) || same_block(x, y))
14}
15
16/// The same block by value. A table's cells compare their blocks by address first too: an edit in
17/// one cell copies the whole table, which mustn't mean comparing every cell by value.
18fn same_block(a: &Block, b: &Block) -> bool {
19    match (a, b) {
20        (Block::Table(x), Block::Table(y)) => same_table(x, y),
21        _ => a == b,
22    }
23}

25/// The same table. Every field is listed, so a new one does not build until someone decides
26/// whether a change to it counts.
27fn same_table(a: &Table, b: &Table) -> bool {
28    let Table { props, grid, rows, controls } = a;
29    let Table { props: other_props, grid: other_grid, rows: other_rows, controls: other_controls } = b;
30    props == other_props
31        && grid == other_grid
32        && controls == other_controls
33        && rows.len() == other_rows.len()
34        && rows.iter().zip(other_rows).all(|(x, y)| same_row(x, y))
35}
36
37fn same_row(a: &Row, b: &Row) -> bool {
38    let Row { props, cells, controls } = a;
39    let Row { props: other_props, cells: other_cells, controls: other_controls } = b;
40    props == other_props
41        && controls == other_controls
42        && cells.len() == other_cells.len()
43        && cells.iter().zip(other_cells).all(|(x, y)| {
44            let Cell { props, blocks, controls } = x;
45            let Cell { props: other_props, blocks: other_blocks, controls: other_controls } = y;
46            props == other_props && controls == other_controls && same_blocks(blocks, other_blocks)
47        })
48}

50/// The same building blocks, their content compared by address first.
51pub(crate) fn same_glossary(a: &[BuildingBlock], b: &[BuildingBlock]) -> bool {
52    a.len() == b.len()
53        && a.iter().zip(b).all(|(x, y)| {
54            let BuildingBlock { name, gallery, category, description, guid, style, types, behaviors, blocks } = x;
55            let BuildingBlock {
56                name: other_name,
57                gallery: other_gallery,
58                category: other_category,
59                description: other_description,
60                guid: other_guid,
61                style: other_style,
62                types: other_types,
63                behaviors: other_behaviors,
64                blocks: other_blocks,
65            } = y;
66            name == other_name
67                && gallery == other_gallery
68                && category == other_category
69                && description == other_description
70                && guid == other_guid
71                && style == other_style
72                && types == other_types
73                && behaviors == other_behaviors
74                && same_blocks(blocks, other_blocks)
75        })
76}

78/// The same parts: the same keys, and each part the same.
79pub(crate) fn same_parts(a: &BTreeMap<u32, Part>, b: &BTreeMap<u32, Part>) -> bool {

83/// The same part: the same kind, text body and blocks. Every field is listed, so a new one
84/// does not build until someone decides whether a change to it counts.
85fn same_part(a: &Part, b: &Part) -> bool {
86    let Part { kind, blocks, body } = a;
87    let Part { kind: other_kind, blocks: other_blocks, body: other_body } = b;
88    kind == other_kind && body == other_body && same_blocks(blocks, other_blocks)
89}
