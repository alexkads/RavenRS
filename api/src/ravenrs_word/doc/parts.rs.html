5/// A free id, `min` or more, for something new (a part, a list, a comment): one past `largest`, the
6/// largest id in use, so ids keep growing; once that would pass `u32::MAX` (a file can use any id),
7/// the lowest one `used` (every id in use) doesn't hold. `None` when every id from `min` up is taken.
8pub fn free_id(largest: Option<u32>, used: impl Iterator<Item = u32>, min: u32) -> Option<u32> {
9    let Some(largest) = largest else { return Some(min) };
10    if let Some(next) = largest.checked_add(1) {
11        return Some(next.max(min));
12    }
13    let mut ids: Vec<u32> = used.filter(|&i| i >= min).collect();
14    ids.sort_unstable();
15    ids.dedup();
16    let mut want = min;
17    for i in ids {
18        if i > want {
19            break;
20        }
21        want = want.checked_add(1)?;
22    }
23    Some(want)
24}

38/// A document's own separators and continuation notice for its footnotes or its endnotes
39/// (ECMA-376 §17.11: the `w:footnote`/`w:endnote` entries of type `separator`,
40/// `continuationSeparator` and `continuationNotice`). `None` = the default: a short line above
41/// the notes, a line across the column above notes continued from the page before, no notice.
42/// The lines themselves are [`InlineObject::note_separator`](crate::doc::para::InlineObject::note_separator) objects in the paragraphs.
43#[derive(Clone, Debug, PartialEq, Serialize, Deserialize, Default)]
44#[serde(default, rename_all = "camelCase")]
45pub struct NoteSeparators {
46    /// Above the notes on a page.
47    pub separator: Option<Blocks>,
48    /// Above the rest of a note continued from the page before.
49    pub continuation_separator: Option<Blocks>,
50    /// Below the part of a note that continues on the next page.
51    pub continuation_notice: Option<Blocks>,
52}
53
54impl NoteSeparators {
55    /// No separator of the document's own: the defaults apply.
56    pub fn is_empty(&self) -> bool {
57        self.separator.is_none() && self.continuation_separator.is_none() && self.continuation_notice.is_none()
58    }
59}

71/// How a text box shows its text (Shape Format › Text): which way it runs, where it sits between
72/// the box's top and bottom, and the box its overflow continues in.
73#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize, Deserialize, Default)]
74#[serde(default, rename_all = "camelCase")]
75pub struct TextBody {
76    pub vert: props::TextVert,
77    pub anchor: props::VAlign,
78    /// Create Link: the text box (another `TextBox` part, shown by its own shape) the text
79    /// continues in when it doesn't fit this one. The linked box shows the rest of this box's
80    /// story; its own blocks are unused. See [`Document::text_box_chains`](crate::doc::Document::text_box_chains).
81    pub next: Option<u32>,
82    /// Internal margins (left, top, right, bottom), EMU, when the file gives its own
83    /// (`wps:bodyPr` `lIns`… ECMA-376 Part 1 §21.1.2.1.1, or VML `v:textbox` `inset`);
84    /// `None` is Word's defaults.
85    pub inset: Option<[u32; 4]>,
86    /// Resize shape to fit text (`a:spAutoFit`, VML `mso-fit-shape-to-text`): Word grows the
87    /// box to its text, so none of the text is cut.
88    pub auto_fit: bool,
89    /// The colour the shape's style gives its text (`wps:style/a:fontRef`): what automatic text
90    /// colour means inside this box, instead of black. Text whose colour comes from its runs,
91    /// styles or the document defaults keeps it (Word, observed with synthetic documents).
92    #[serde(skip_serializing_if = "Option::is_none")]
93    pub auto_color: Option<props::Rgb>,
94}

96/// Word's default text box margins (left, top, right, bottom), EMU: 0.1" and 0.05".
97pub const TEXT_BOX_INSET: [u32; 4] = [91_440, 45_720, 91_440, 45_720];
98
99/// Widest text box margin kept, EMU (22", the largest page side).
100pub const MAX_TEXT_BOX_INSET: u32 = 20_116_800;
101
102impl TextBody {
103    pub fn is_default(&self) -> bool {
104        *self == TextBody::default()
105    }
106
107    /// The internal margins (left, top, right, bottom) in points.
108    pub fn insets_pt(&self) -> [f32; 4] {
109        self.inset.unwrap_or(TEXT_BOX_INSET).map(|v| v.min(MAX_TEXT_BOX_INSET) as f32 / 12_700.0)
110    }
111}

116/// A range that stays editable while the document is protected; see [`Document::perm_ranges`](crate::doc::Document::perm_ranges).
117#[derive(Clone, Debug, PartialEq, Eq)]
118pub struct PermRange {
119    pub id: u32,
120    /// Editing group (`everyone`, `editors`…); empty when the range names an editor.
121    pub group: String,
122    /// The one editor the range is for; empty for a group.
123    pub editor: String,
124    pub start: Pos,
125    pub end: Pos,
126}
127
128impl PermRange {
129    /// Can everyone edit it (rather than one person or group)?
130    pub fn everyone(&self) -> bool {
131        self.group.eq_ignore_ascii_case("everyone")
132    }
