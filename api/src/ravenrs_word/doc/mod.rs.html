1//! WordCraft document model.
2//!
3//! A [`Document`] holds the main story (`body`) and secondary stories (`parts`: headers,
4//! footers, notes, comments, text boxes). A story is a list of [`Block`]s: paragraphs and tables;
5//! table cells hold block lists of their own. Blocks, the style sheet and the list definitions
6//! are `Arc`-shared, so cloning a document is cheap (undo snapshots, handing a frozen copy to the
7//! renderer) and edits copy only what they touch (`Arc::make_mut`).
8//!
9//! Positions ([`Pos`]) name a story, a [`Path`] to a paragraph and a byte offset in it. Paths
10//! compare in document order.

98/// A document.
99#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
100#[serde(default, rename_all = "camelCase")]
101pub struct Document {
102    pub body: Blocks,
103    /// Properties of the last section (OOXML's body-level `w:sectPr`).
104    pub last_section: SectionProps,
105    pub parts: BTreeMap<u32, Part>,
106    /// Shared, like the blocks: an undo step copies it only when a command changes it (a full
107    /// copy is kilobytes a style, so typing into a document with hundreds of styles cost megabytes
108    /// a step).
109    pub styles: Arc<StyleSheet>,
110    /// Shared like [`Document::styles`].
111    pub numbering: Arc<Numbering>,
112    pub comments: BTreeMap<u32, Comment>,
113    pub revisions: Vec<Revision>,
114    pub settings: Settings,
115    pub core: CoreProps,
116    /// Bibliography sources (References › Manage Sources).
117    pub sources: Vec<Source>,
118    /// Custom document properties, in file order.
119    pub custom_props: Vec<CustomProp>,
120    /// The document's own footnote separators and continuation notice.
121    #[serde(skip_serializing_if = "NoteSeparators::is_empty")]
122    pub footnote_separators: NoteSeparators,
123    /// The document's own endnote separators and continuation notice.
124    #[serde(skip_serializing_if = "NoteSeparators::is_empty")]
125    pub endnote_separators: NoteSeparators,
126    /// Embedded media (images) by key. Native JSON writes each as base64 (`byte_store.rs`).
127    #[serde(skip_serializing_if = "BTreeMap::is_empty", with = "byte_store")]
128    pub media: BTreeMap<String, Arc<Vec<u8>>>,
129    /// Source-format parts we keep verbatim for round-trip (e.g. custom XML), by part name.
130    #[serde(skip_serializing_if = "BTreeMap::is_empty", with = "byte_store")]
131    pub passthrough: BTreeMap<String, Arc<Vec<u8>>>,
132    /// The document's own building blocks (its glossary document), in file order.
133    #[serde(skip_serializing_if = "Vec::is_empty")]
134    pub glossary: Vec<BuildingBlock>,
135    /// Lookup cache for [`Document::add_media`]; not part of the document (never saved, ignored
136    /// by `==`).
137    #[serde(skip)]
138    media_index: MediaIndex,
139    /// Markup kept from the file it was opened from, for writing back what isn't edited (see
140    /// [`kept`]); never saved, ignored by `==`.
141    #[serde(skip)]
142    pub kept: kept::Kept,
143    /// Pictures a reader dropped because their target is missing, external or link-only (strict
144    /// export reports them); never saved, never drawn, ignored by `==`.
145    #[serde(skip)]
146    pub missing_pictures: MissingPictures,
147}
