257
258impl Watermark {
259    /// The box the text fills on a page `page_w` × `page_h` (points): its own size, else half the
260    /// page diagonal across (half the page width when not diagonal), as high as the text's length
261    /// makes it. Centred on the margins and, when diagonal, turned 45° counterclockwise (Word's
262    /// `rotation:315`).
263    pub fn frame(&self, page_w: f32, page_h: f32) -> [f32; 2] {
264        if let Some([w, h]) = self.size.filter(|[w, h]| w.is_finite() && h.is_finite() && *w > 0.0 && *h > 0.0) {
265            return [w.min(4000.0), h.min(4000.0)];
266        }
267        let (pw, ph) = (crate::geom::finite(page_w).clamp(72.0, 4000.0), crate::geom::finite(page_h).clamp(72.0, 4000.0));
268        let across = if self.diagonal { (pw * pw + ph * ph).sqrt() } else { pw };
269        let width = (across * 0.5).clamp(36.0, 2000.0);
270        let chars = self.text.chars().take(200).count().max(1) as f32;
271        [width, (width / (0.55 * chars)).clamp(12.0, 400.0)]
272    }
273}
