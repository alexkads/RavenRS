25    pub format: NumFormat,

27    pub text: String,

41    #[serde(skip_serializing_if = "Option::is_none")]

43    /// The tab stop that follows the number (Word's "Add tab stop at"), points from the
44    /// paragraph's left edge.
45    #[serde(skip_serializing_if = "Option::is_none")]
46    pub tab: Option<f32>,
47    /// Show all levels as decimal (legal numbering).
48    pub legal: bool,
49    /// Paragraph style linked to this level (e.g. Heading 1 for outline numbering).
50    pub style: Option<String>,

112/// Built-in list kinds for the Bullets / Numbering / Multilevel galleries.
113#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash, Serialize, Deserialize)]
114#[serde(rename_all = "camelCase")]
115pub enum ListKind {
116    /// • ○ ▪ repeating.
117    Bullet,
118    /// A specific bullet character.
119    BulletChar(char),
120    /// 1. a. i.
121    Numbered,
122    /// 1) a) i)
123    NumberedParen,
124    /// I. A. 1.
125    Outline,
126    /// A. B. C.
127    UpperLetter,
128    /// a) b) c)
129    LowerLetter,
130    /// a. b. c.
131    LowerLetterDot,
132    /// i. ii. iii.
133    LowerRoman,
134    /// 1. 1.1. 1.1.1.
135    Legal,
136}
137
138impl ListKind {
139    pub fn parse(s: &str) -> Option<ListKind> {
140        Some(match s {
141            "bullet" => ListKind::Bullet,
142            "numbered" | "number" => ListKind::Numbered,
143            "numberedParen" => ListKind::NumberedParen,
144            "outline" => ListKind::Outline,
145            "upperLetter" => ListKind::UpperLetter,
146            "lowerLetter" => ListKind::LowerLetter,
147            "lowerLetterDot" => ListKind::LowerLetterDot,
148            "lowerRoman" => ListKind::LowerRoman,
149            "legal" | "multilevel" => ListKind::Legal,
150            s if s.chars().count() == 1 => ListKind::BulletChar(s.chars().next()?),
151            _ => return None,
152        })
153    }
154    pub fn is_bullet(self) -> bool {
155        matches!(self, ListKind::Bullet | ListKind::BulletChar(_))
156    }
157}

159/// The bullet a level of a plain bulleted list shows: • ○ ▪ repeating (`level` is zero-based).
160pub fn default_bullet(level: usize) -> char {
161    ['•', '○', '▪'][level % 3]
162}
163
164fn bullet_level(i: usize, c: char) -> Level {
165    Level {
166        format: NumFormat::Bullet,
167        text: c.to_string(),
168        indent: 36.0 * (i as f32 + 1.0),
169        hanging: 18.0,
170        chr: CharProps {
171            font: match c {
172                '•' | '▪' | '○' => Some("Arial".into()),
173                // Plain text characters (a dash bullet) use the paragraph's font.
174                c if c.is_ascii() || c == '–' => None,
175                _ => Some("Segoe UI Symbol".into()),
176            },
177            ..CharProps::default()
178        },
179        ..Level::default()
180    }
181}

183/// Nine levels for a list kind.
184pub fn levels_for(kind: ListKind) -> Vec<Level> {
185    (0..9)
186        .map(|i| {
187            let ind = 36.0 * (i as f32 + 1.0);
188            let cyc = |a: NumFormat, b: NumFormat, c: NumFormat| match i % 3 {
189                0 => a,
190                1 => b,
191                _ => c,
192            };
193            let lvl = |format: NumFormat, text: String| Level { format, text, indent: ind, hanging: 18.0, ..Level::default() };
194            let n = i + 1;
195            match kind {
196                ListKind::Bullet => bullet_level(i, default_bullet(i)),
197                ListKind::BulletChar(c) => bullet_level(i, if i == 0 { c } else { default_bullet(i) }),
198                ListKind::Numbered => lvl(cyc(NumFormat::Decimal, NumFormat::LowerLetter, NumFormat::LowerRoman), format!("%{n}.")),
199                ListKind::NumberedParen => lvl(cyc(NumFormat::Decimal, NumFormat::LowerLetter, NumFormat::LowerRoman), format!("%{n})")),
200                ListKind::UpperLetter => lvl(cyc(NumFormat::UpperLetter, NumFormat::LowerLetter, NumFormat::LowerRoman), format!("%{n}.")),
201                ListKind::LowerLetter => lvl(cyc(NumFormat::LowerLetter, NumFormat::LowerRoman, NumFormat::Decimal), format!("%{n})")),
202                ListKind::LowerLetterDot => lvl(cyc(NumFormat::LowerLetter, NumFormat::LowerRoman, NumFormat::Decimal), format!("%{n}.")),
203                ListKind::LowerRoman => lvl(cyc(NumFormat::LowerRoman, NumFormat::LowerLetter, NumFormat::Decimal), format!("%{n}.")),
204                ListKind::Outline => {
205                    let f = [NumFormat::UpperRoman, NumFormat::UpperLetter, NumFormat::Decimal, NumFormat::LowerLetter, NumFormat::LowerRoman]
206                        .get(i)
207                        .copied()
208                        .unwrap_or(NumFormat::Decimal);
209                    lvl(f, format!("%{n}."))
210                }
211                ListKind::Legal => {
212                    let text: String = (1..=n).map(|k| format!("%{k}.")).collect();
213                    Level { format: NumFormat::Decimal, text, indent: 18.0 + 25.0 * i as f32, hanging: 18.0 + 7.0 * i as f32, ..Level::default() }
214                }
215            }
216        })
217        .collect()
218}

220impl Numbering {
221    pub fn num(&self, id: u32) -> Option<&Num> {
222        self.nums.iter().find(|n| n.id == id)
223    }
224    pub fn abstract_of(&self, num: u32) -> Option<&AbstractNum> {
225        let n = self.num(num)?;
226        self.abstracts.iter().find(|a| a.id == n.abstract_id)
227    }
228    /// Whether a paragraph in list `num` at `level` gets a label ([`Counters::next_label`] gives
229    /// one), without counting it.
230    pub fn shows_label(&self, num: u32, level: u8) -> bool {
231        self.abstract_of(num).is_some() && self.level(num, level.min(8)).is_some()
232    }
233    pub fn level(&self, num: u32, level: u8) -> Option<&Level> {
234        if let Some((_, l)) = self.num(num)?.level_overrides.iter().find(|(i, _)| *i == level) {
235            return Some(l);
236        }
237        self.abstract_of(num)?.levels.get(level as usize)
238    }
239    /// Create a new list of `kind`; returns the `num` id. List ids run out only past `u32::MAX`
240    /// lists, more than memory holds; then nothing is added and the id is 0, "no list".
241    pub fn add_list(&mut self, kind: ListKind) -> u32 {
242        self.try_add_list(kind).unwrap_or(0)
243    }
244    /// [`Numbering::add_list`], or an error, adding nothing, when list ids are all taken.
245    pub fn try_add_list(&mut self, kind: ListKind) -> crate::doc::Result<u32> {
246        self.try_add_levels(None, levels_for(kind))
247    }
248    /// A new list definition (`name`, `levels`) and a `num` for it; returns the `num` id. Both ids
249    /// are [`free_id`]s, so the largest ids a file can hold don't make them collide with its lists;
250    /// an error, adding nothing, when they are all taken.
251    pub fn try_add_levels(&mut self, name: Option<String>, levels: Vec<Level>) -> crate::doc::Result<u32> {
252        let aid = free_id(self.abstracts.iter().map(|a| a.id).max(), self.abstracts.iter().map(|a| a.id), 0);
253        let (Some(aid), Some(nid)) = (aid, self.free_num_id()) else { return Err(DocError::Invalid("no free list id".into())) };
254        self.abstracts.push(AbstractNum { id: aid, name, levels, ..Default::default() });
255        self.nums.push(Num { id: nid, abstract_id: aid, ..Default::default() });
256        Ok(nid)
257    }
258    /// A `num` id no list has (`num` 0 means "no list"), `None` when they are all taken.
259    fn free_num_id(&self) -> Option<u32> {
260        free_id(self.nums.iter().map(|n| n.id).max(), self.nums.iter().map(|n| n.id), 1)
261    }
262    /// A new `num` that restarts numbering of an existing list (`None`: no such list, or list ids
263    /// are all taken).
264    pub fn restart(&mut self, num: u32) -> Option<u32> {
265        let source = self.num(num)?;
266        let a = source.abstract_id;
267        let level_overrides = source.level_overrides.clone();
268        let nid = self.free_num_id()?;
269        // Every level starts again at its own effective start, so a level this list defines itself keeps its start.
270        let levels = self.abstracts.iter().find(|x| x.id == a).map_or(0, |x| x.levels.len());
271        let start_overrides = (0..levels).filter_map(|i| self.level(num, i as u8).map(|l| (i as u8, l.start))).collect();
272        self.nums.push(Num { id: nid, abstract_id: a, start_overrides, level_overrides });
273        Some(nid)
274    }
275    /// Find an existing list whose first level matches `kind`, so consecutive bullet clicks reuse it.
276    pub fn find_kind(&self, kind: ListKind) -> Option<u32> {
277        let want = levels_for(kind);
278        let first = want.first()?;
279        self.nums
280            .iter()
281            .find(|n| self.abstract_of(n.id).and_then(|a| a.levels.first()).is_some_and(|l| l.format == first.format && l.text == first.text))
282            .map(|n| n.id)
283    }
284}

286/// Running list counters while walking a document in order.
287#[derive(Default, Debug, Clone)]
288pub struct Counters {
289    /// Per abstract list: current value per level.
290    state: HashMap<u32, [u32; 9]>,
291    /// Which levels of which nums have started: a num's start override for a level applies
292    /// once, when that level of that num is first used (ECMA-376 Part 1, 17.9.27 startOverride).
293    seen: HashMap<(u32, u8), bool>,
294    /// Per abstract list: whether each level has counted yet (a value of 0 can be a real count when a
295    /// level starts at 0).
296    flags: HashMap<u32, [LevelFlags; 9]>,
297    /// The last explicitly rendered label and template at each abstract-list level.
298    labels: HashMap<(u32, u8), CountedLabel>,
299}

301#[derive(Debug, Clone)]
302struct CountedLabel {
303    label: Option<String>,
304    levels: Vec<usize>,
305}

307/// Whether a level has counted yet, and the value its next item takes if a start override says so.
308#[derive(Default, Debug, Clone, Copy)]
309struct LevelFlags {
310    started: bool,
311    first: Option<u32>,
312}
313
314impl Counters {
315    /// The current value of each level of `num`'s list (0 for a level not counted yet).
316    pub fn values(&self, numbering: &Numbering, num: u32) -> Option<[u32; 9]> {
317        let abs = numbering.abstract_of(num)?;
318        Some(self.state.get(&abs.id).copied().unwrap_or([0; 9]))
319    }
320
321    /// Advance the counter for `num`/`level` and return the label (`"1."`, `"a)"`, `"•"`).
322    pub fn next_label(&mut self, numbering: &Numbering, num: u32, level: u8) -> Option<(String, Level)> {
323        let n = numbering.num(num)?;
324        let abs = numbering.abstract_of(num)?;
325        let lv = (level as usize).min(8);
326        // This list's own definition of a level wins over its abstract list's.
327        let level_def = |k: usize| numbering.level(num, k as u8);
328        // A "none" level (Word's numFmt none) still counts and restarts the levels below it; only
329        // its number reads as nothing, so its literal text ("RESPONSE:") shows (#4985).
330        let def = level_def(lv)?.clone();
331        let st = self.state.entry(abs.id).or_insert([0; 9]);
332        let fl = self.flags.entry(abs.id).or_default();
333        // Only this level's override, not the num's others: the counters are shared by every
334        // num of the abstract list, so restarting its other levels here would renumber
335        // paragraphs of the other nums. A heading style can use one num for levels 1-4 and
336        // another, with start overrides on every level, for 5-9; its first level-5 heading
337        // must not restart the chapter count.
338        if let std::collections::hash_map::Entry::Vacant(e) = self.seen.entry((num, lv as u8)) {
339            e.insert(true);
340            if let Some((_, start)) = n.start_overrides.iter().find(|(l, _)| *l as usize == lv)
341                && let (Some(s), Some(f)) = (st.get_mut(lv), fl.get_mut(lv))
342            {
343                *s = 0;
344                *f = LevelFlags { started: false, first: Some(*start) };
345            }
346        }
347        // A level above that hasn't counted yet counts its first number here, as in Word: after
348        // 1.1 and 1.2 with no level-1 item before them, the next level-1 item is 2, whether or not
349        // the deeper level's text shows it.
350        for (k, (v, f)) in st.iter_mut().zip(fl.iter_mut()).enumerate().take(lv) {
351            if f.started {
352                continue;
353            }
354            let start = match self.seen.entry((num, k as u8)) {
355                std::collections::hash_map::Entry::Vacant(e) => {
356                    e.insert(true);
357                    n.start_overrides.iter().find(|(l, _)| *l as usize == k).map(|(_, s)| *s)
358                }
359                std::collections::hash_map::Entry::Occupied(_) => None,
360            };
361            *v = start.or(f.first).unwrap_or_else(|| level_def(k).map_or(1, |l| l.start));
362            *f = LevelFlags { started: true, first: None };
363        }
364        if let (Some(v), Some(f)) = (st.get_mut(lv), fl.get_mut(lv)) {
365            *v = if f.started { v.saturating_add(1) } else { f.first.unwrap_or(def.start) };
366            *f = LevelFlags { started: true, first: None };
367        }
368        // Deeper levels restart, unless they never restart or only restart after a level above
369        // this one (Word's lvlRestart).
370        for (d, (deeper, f)) in st.iter_mut().zip(fl.iter_mut()).enumerate().skip(lv + 1) {
371            let restarts = level_def(d).is_none_or(|l| l.restart && l.restart_after.is_none_or(|k| lv < k as usize));
372            if restarts {
373                *deeper = 0;
374                *f = LevelFlags::default();
375                self.labels.remove(&(abs.id, d as u8));
376            }
377        }
378        let vals = *st;
379        let started = *fl;
380        let numeric = def.format != NumFormat::Bullet;
381        let label = if numeric { render_label(numbering, num, &def, &vals, &started) } else { def.text.clone() };
382        // A "none" level shows its literal text, but has no number of its own for a reference to
383        // show (#4985): it is kept as one that shows none.
384        let counted = if def.format == NumFormat::None {
385            CountedLabel { label: None, levels: Vec::new() }
386        } else {
387            CountedLabel { label: numeric.then(|| label.clone()), levels: template_levels(&def.text) }
388        };
389        self.labels.insert((abs.id, lv as u8), counted);
390        Some((label, def))
391    }
392    /// Which levels of `num`'s list have counted a paragraph yet (a level never used has no
393    /// number of its own to show in a reference).
394    pub fn started(&self, numbering: &Numbering, num: u32) -> Option<[bool; 9]> {
395        let abs = numbering.abstract_of(num)?;
396        Some(std::array::from_fn(|k| self.labels.get(&(abs.id, k as u8)).is_some_and(|last| last.label.is_some())))
397    }
398
399    /// The label `level` of `num`'s list shows at this point of the walk, without counting: what
400    /// the last paragraph at that level showed (`"14."` while walking clause 14's sub-clauses).
401    /// `None` for a bullet or "none" level, or a level the list doesn't define.
402    pub fn current_label(&self, numbering: &Numbering, num: u32, level: u8) -> Option<String> {
403        let abs = numbering.abstract_of(num)?;
404        if let Some(last) = self.labels.get(&(abs.id, level.min(8))) {
405            return last.label.clone();
406        }
407        // A deeper item can count a level implicitly without rendering a paragraph there.
408        let def = numbering.level(num, level.min(8))?;
409        if matches!(def.format, NumFormat::None | NumFormat::Bullet) {
410            return None;
411        }
412        let vals = self.state.get(&abs.id).copied().unwrap_or([0; 9]);
413        let started = self.flags.get(&abs.id).copied().unwrap_or_default();
414        Some(render_label(numbering, num, def, &vals, &started))
415    }
416
417    /// Levels shown by the last rendered label, or the current definition for an implicit level.
418    pub fn current_template_levels(&self, numbering: &Numbering, num: u32, level: u8) -> Vec<usize> {
419        if let Some(abs) = numbering.abstract_of(num)
420            && let Some(last) = self.labels.get(&(abs.id, level.min(8)))
421        {
422            return last.levels.clone();
423        }
424        numbering.level(num, level.min(8)).map(|def| template_levels(&def.text)).unwrap_or_default()
425    }
426}

428/// The zero-based levels a label template shows (`"%1.%2"` → `[0, 1]`, `"(%3)"` → `[2]`).
429pub fn template_levels(text: &str) -> Vec<usize> {
430    let mut out = Vec::new();
431    let mut chars = text.chars().take(MAX_LEVEL_TEXT).peekable();
432    while let Some(c) = chars.next() {
433        if c == '%'
434            && let Some(d) = chars.peek().and_then(|d| d.to_digit(10))
435        {
436            chars.next();
437            let k = (d as usize).saturating_sub(1).min(8);
438            if !out.contains(&k) {
439                out.push(k);
440            }
441        }
442    }
443    out
444}

446/// Render a list label without advancing counters, retaining the existing hostile-input caps.
447fn render_label(numbering: &Numbering, num: u32, def: &Level, vals: &[u32; 9], started: &[LevelFlags; 9]) -> String {
448    let level_def = |k: usize| numbering.level(num, k as u8);
449    let mut label = String::new();
450    // Word's level text is short; .docx and the lists command already cap it at 64.
451    let mut chars = def.text.chars().take(MAX_LEVEL_TEXT).peekable();
452    while let Some(c) = chars.next() {
453        if c == '%'
454            && let Some(d) = chars.peek().and_then(|d| d.to_digit(10))
455        {
456            chars.next();
457            let k = (d as usize).saturating_sub(1).min(8);
458            let val = vals.get(k).copied().unwrap_or(0);
459            let counted = started.get(k).is_some_and(|f| f.started);
460            let val = if counted { val } else { level_def(k).map(|l| l.start).unwrap_or(1) };
461            // Legal numbering (w:isLgl) shows every level in Arabic digits, whatever its own
462            // format (二 of a Chinese outline becomes 2 in "2.1"); a "none" level stays empty.
463            let fmt = match level_def(k).map(|l| l.format).unwrap_or(NumFormat::Decimal) {
464                f @ (NumFormat::DecimalZero | NumFormat::None) => f,
465                _ if def.legal => NumFormat::Decimal,
466                f => f,
467            };
468            label.push_str(&fmt.format(val));
469            if label.len() > MAX_LABEL_BYTES {
470                break;
471            }
472        } else {
473            label.push(c);
474        }
475    }
476    label
477}

479/// Longest level text (in characters) expanded into a list label.
480const MAX_LEVEL_TEXT: usize = 64;
481/// Labels stop growing once they pass this many bytes.
482const MAX_LABEL_BYTES: usize = 256;

585    fn a_start_override_restarts_only_its_own_level_of_a_shared_list() {
586        // One abstract list numbers the headings through two nums: one for levels 1-4, and one
587        // restarting every level (startOverride 1 on 0..8) for levels 5-9, first used deep in
588        // chapter 2. Chapter 3 is still chapter 3, and the deep heading counts from 1.
589        let mut n = Numbering::default();
590        let chapters = n.add_list(ListKind::Legal);
591        let abs = n.abstract_of(chapters).map(|a| a.id).unwrap();
592        let deep = chapters + 1;
593        n.nums.push(Num { id: deep, abstract_id: abs, start_overrides: (0..9).map(|l| (l, 1)).collect(), ..Num::default() });
594        let mut c = Counters::default();
595        let mut l = |num, lv| c.next_label(&n, num, lv).unwrap().0;
596        assert_eq!(l(chapters, 0), "1.");
597        assert_eq!(l(chapters, 0), "2.");
598        assert_eq!(l(chapters, 1), "2.1.");
599        assert_eq!(l(chapters, 2), "2.1.1.");
600        assert_eq!(l(chapters, 3), "2.1.1.1.");
601        assert_eq!(l(deep, 4), "2.1.1.1.1.", "the deep num restarts only level 5");
602        assert_eq!(l(deep, 4), "2.1.1.1.2.");
603        assert_eq!(l(chapters, 0), "3.", "and not the chapter count");
604        assert_eq!(l(chapters, 1), "3.1.");

629    #[test]
630    fn a_deeper_item_first_counts_the_levels_above_it() {
631        // Measured in Word: a level above that hasn't counted yet takes its first number at the
632        // first deeper item, whether or not that item's text shows it.
633        let labels_with = |starts: [u32; 3], texts: [&str; 3], overrides: &[(u8, u32)], seq: &[u8]| {
634            let mut n = Numbering::default();
635            let id = n.add_list(ListKind::Legal);
636            if let Some(a) = n.abstracts.first_mut() {
637                for (l, (s, t)) in a.levels.iter_mut().zip(starts.into_iter().zip(texts)) {
638                    l.start = s;
639                    l.text = t.into();
640                }
641            }
642            if let Some(num) = n.nums.iter_mut().find(|x| x.id == id) {
643                num.start_overrides = overrides.to_vec();
644            }
645            let mut c = Counters::default();
646            seq.iter().map(|lv| c.next_label(&n, id, *lv).unwrap().0).collect::<Vec<_>>()
647        };
648        let labels = |starts: [u32; 3], texts: [&str; 3], seq: &[u8]| labels_with(starts, texts, &[], seq);
649        let std = ["%1.", "%1.%2.", "%1.%2.%3."];
650        assert_eq!(labels([1, 1, 1], std, &[1, 1, 0, 0]), ["1.1.", "1.2.", "2.", "3."]);
651        assert_eq!(labels([10, 1, 1], std, &[1, 1, 1, 1, 0, 0]), ["10.1.", "10.2.", "10.3.", "10.4.", "11.", "12."]);
652        assert_eq!(labels([1, 1, 1], ["%1.", "(%2)", "%1.%2.%3."], &[1, 1, 0, 0]), ["(1)", "(2)", "2.", "3."]);
653        assert_eq!(labels([1, 1, 1], std, &[2, 1, 0, 1]), ["1.1.1.", "1.2.", "2.", "2.1."]);
654        assert_eq!(labels([1, 1, 1], ["%1.", "%1.%2.", "(%3)"], &[2, 1, 0]), ["(1)", "1.2.", "2."]);
655        assert_eq!(labels([1, 1, 1], std, &[0, 2, 1, 0]), ["1.", "1.1.1.", "1.2.", "2."]);
656        // The num's own start for the level above counts in its place, as the level's start does.
657        assert_eq!(labels_with([1, 1, 1], std, &[(0, 10)], &[1, 1, 0]), ["10.1.", "10.2.", "11."]);
658        assert_eq!(labels_with([1, 1, 1], std, &[(0, 5)], &[1, 0, 1]), ["5.1.", "6.", "6.1."]);
659        // A level-1 item after the first deeper ones restarts level 2 below it, as usual.
660        assert_eq!(labels([1, 1, 1], std, &[1, 1, 0, 1, 1, 0, 2]), ["1.1.", "1.2.", "2.", "2.1.", "2.2.", "3.", "3.1.1."]);
661    }
