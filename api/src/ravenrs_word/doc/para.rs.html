1//! Paragraphs: text with formatting runs and inline objects.
2//!
3//! A paragraph's text is one `String`. Formatting is a list of [`Run`]s whose byte lengths sum to
4//! the text length. Inline objects (pictures, fields, note references, bookmarks, comment
5//! anchors…) sit in the text as U+FFFC; the k-th U+FFFC is `objects[k]`. Special characters:
6//! `\t` tab, `\n` manual line break, U+000C page break, U+000E column break.

26/// A paragraph.
27#[derive(Clone, Debug, Serialize, Deserialize)]
28pub struct Paragraph {
29    pub text: String,
30    pub runs: Vec<Run>,
31    #[serde(default, skip_serializing_if = "ParaProps::is_empty")]
32    pub props: ParaProps,

99    /// The text with objects removed or replaced by their text (fields → result).
100    pub fn plain_text(&self) -> String {
101        let mut out = String::with_capacity(self.text.len());
102        let mut k = 0;
103        for c in self.text.chars() {
104            if c == OBJ {
105                if let Some(o) = self.objects.get(k) {
106                    out.push_str(o.plain_text());
107                }
108                k += 1;
109            } else {
110                out.push(c);
111            }
112        }
113        out
114    }
115    /// [`Self::plain_text`] as it reads on screen: characters in runs set in a symbol font
116    /// (Symbol, Wingdings…) become their standard Unicode equivalents
117    /// ([`crate::doc::symbol_fonts`]). For copying as text and plain-text export; the stored text is
118    /// unchanged.
119    pub fn shown_text(&self) -> String {
120        let mut out = String::with_capacity(self.text.len());
121        let mut k = 0;
122        let mut runs = self.run_ranges().peekable();
123        for (i, c) in self.text.char_indices() {
124            while runs.peek().is_some_and(|(r, _)| r.end <= i) {
125                runs.next();
126            }
127            if c == OBJ {
128                if let Some(o) = self.objects.get(k) {
129                    out.push_str(o.plain_text());
130                }
131                k += 1;
132                continue;
133            }
134            let font = runs.peek().filter(|(r, _)| r.contains(&i)).and_then(|(_, p)| p.shown_font());
135            out.push(font.and_then(|f| crate::doc::symbol_fonts::symbol_char(f, c)).unwrap_or(c));
136        }
137        out
138    }
139    /// [`Self::plain_text`] without tracked deletions: the text as it reads with every change accepted.
140    pub fn final_text(&self) -> String {
141        self.text_without(&self.deleted_ranges())
142    }
143    /// The byte ranges of the tracked deletions.
144    pub fn deleted_ranges(&self) -> Vec<std::ops::Range<usize>> {
145        self.run_ranges().filter(|(_, c)| c.del.is_some()).map(|(r, _)| r).collect()

147    /// [`Self::plain_text`] without the text in the byte ranges `dropped` (in any order, overlapping
148    /// or not).
149    pub fn text_without(&self, dropped: &[std::ops::Range<usize>]) -> String {
150        let mut sorted = dropped.to_vec();
151        sorted.sort_unstable_by_key(|r| r.start);
152        let mut next = sorted.iter().peekable();
153        let mut out = String::with_capacity(self.text.len());
154        let mut k = 0;
155        for (i, c) in self.text.char_indices() {
156            // By start: once the ranges that end by `i` are skipped, `i` is dropped if the next has begun.
157            while next.next_if(|r| r.end <= i).is_some() {}
158            let keep = next.peek().is_none_or(|r| r.start > i);
159            if c == OBJ {
160                if keep && let Some(o) = self.objects.get(k) {
161                    out.push_str(o.plain_text());
162                }
163                k += 1;
164            } else if keep {
165                out.push(c);
166            }
167        }
168        out
169    }

171    fn check(&self, off: usize) -> Result<()> {
172        if off > self.text.len() || !self.text.is_char_boundary(off) {
173            return Err(DocError::BadOffset(off));
174        }
175        Ok(())
176    }

178    /// Clamp `off` to the text and back to a char boundary.
179    pub fn clamp(&self, off: usize) -> usize {
180        let mut o = off.min(self.text.len());
181        while o > 0 && !self.text.is_char_boundary(o) {
182            o -= 1;
183        }
184        o
185    }

187    /// Index of the run containing byte `off` and the offset where that run starts. At a run
188    /// boundary the earlier run wins (typing continues the formatting to the left).
189    fn run_at(&self, off: usize) -> Option<(usize, usize)> {
190        let mut start = 0;
191        for (i, r) in self.runs.iter().enumerate() {
192            let end = start + r.len;
193            if off < end || (off == end && off > start) {
194                return Some((i, start));
195            }
196            start = end;
197        }
198        None
199    }

201    /// Formatting in effect at `off` (what typing there would get): the run to the left, or the
202    /// first run at offset 0, or the paragraph mark when empty.
203    pub fn props_at(&self, off: usize) -> &CharProps {
204        if off == 0 {
205            return self.runs.first().map(|r| &r.props).unwrap_or(&self.mark);
206        }
207        self.run_at(off).and_then(|(i, _)| self.runs.get(i)).map(|r| &r.props).unwrap_or(&self.mark)
208    }

210    /// Formatting of the character starting at `off`.
211    pub fn props_of_char(&self, off: usize) -> &CharProps {
212        let mut start = 0;
213        for r in &self.runs {
214            if off < start + r.len {
215                return &r.props;
216            }
217            start += r.len;
218        }
219        &self.mark
220    }

262    /// Insert plain text (U+FFFC is dropped) at `off` with `props`. Returns the inserted length.
263    pub fn insert_text(&mut self, off: usize, s: &str, props: &CharProps) -> Result<usize> {
264        self.check(off)?;
265        let clean: std::borrow::Cow<str> = if s.contains(OBJ) { s.chars().filter(|c| *c != OBJ).collect::<String>().into() } else { s.into() };
266        if clean.is_empty() {
267            return Ok(0);
268        }
269        self.text.insert_str(off, &clean);
270        self.insert_run(off, clean.len(), props.clone());
271        self.touch();
272        Ok(clean.len())
273    }

275    /// Insert an object at `off`.
276    pub fn insert_object(&mut self, off: usize, obj: InlineObject, props: &CharProps) -> Result<()> {
277        self.check(off)?;
278        let k = self.objects_before(off);
279        self.text.insert(off, OBJ);
280        self.objects.insert(k.min(self.objects.len()), obj);
281        self.insert_run(off, OBJ.len_utf8(), props.clone());
282        self.touch();
283        Ok(())
284    }

286    fn insert_run(&mut self, off: usize, len: usize, props: CharProps) {
287        // Split the run at `off`, insert, merge neighbours.
288        let mut start = 0;
289        let mut idx = self.runs.len();
290        for (i, r) in self.runs.iter().enumerate() {
291            if off <= start {
292                idx = i;
293                break;
294            }
295            if off < start + r.len {
296                // split
297                let left = off - start;
298                let right = r.len - left;
299                let p = r.props.clone();
300                if let Some(r) = self.runs.get_mut(i) {
301                    r.len = left;
302                }
303                self.runs.insert(i + 1, Run { len: right, props: p });
304                idx = i + 1;
305                break;
306            }
307            start += r.len;
308        }
309        self.runs.insert(idx.min(self.runs.len()), Run { len, props });
310        self.normalize();
311    }

313    /// Merge adjacent equal runs and drop empty ones.
314    pub fn normalize(&mut self) {
315        // Runs must end on character boundaries and cover the text exactly (never expected, but
316        // files lie): clamp each end to the text, push it past a split character, pad the tail.
317        let n = self.text.len();
318        let mut end = 0usize;
319        let mut out: Vec<Run> = Vec::with_capacity(self.runs.len());
320        for r in self.runs.drain(..) {
321            let mut e = end.saturating_add(r.len).min(n);
322            while !self.text.is_char_boundary(e) {
323                e += 1;
324            }
325            if e == end {
326                continue;
327            }
328            match out.last_mut() {
329                Some(l) if l.props == r.props => l.len += e - end,
330                _ => out.push(Run { len: e - end, props: r.props }),
331            }
332            end = e;
333        }
334        if end < n {
335            match out.last_mut() {
336                Some(l) => l.len += n - end,
337                None => out.push(Run { len: n, props: self.mark.clone() }),
338            }
339        }
340        self.runs = out;
341    }

343    /// Delete `a..b` (bytes). Objects inside are removed.
344    pub fn delete(&mut self, a: usize, b: usize) -> Result<()> {
345        let (a, b) = (a.min(b), a.max(b));
346        self.check(a)?;
347        self.check(b)?;
348        if a == b {
349            return Ok(());
350        }
351        if self.runs.is_empty() || a == 0 {
352            // Keep the formatting of what's deleted for the mark when emptying.
353            if b == self.text.len() {
354                // Formatting only: the mark is a tracked change of its own or it is not, whatever
355                // revision the deleted text was part of.
356                let revisions = (self.mark.ins, self.mark.del, self.mark.fmt_change.clone());
357                let mut mark = self.props_of_char(a).clone();
358                (mark.ins, mark.del, mark.fmt_change) = revisions;
359                self.mark = mark;
360            }
361        }
362        let k0 = self.objects_before(a);
363        let k1 = self.objects_before(b);
364        if k1 > k0 && k1 <= self.objects.len() {
365            self.objects.drain(k0..k1);
366        }
367        self.text.replace_range(a..b, "");
368        let mut start = 0;
369        for r in &mut self.runs {
370            let (rs, re) = (start, start + r.len);
371            start = re;
372            let ov = re.min(b).saturating_sub(rs.max(a));
373            r.len -= ov.min(r.len);
374        }
375        self.normalize();
376        self.touch();
377        Ok(())
378    }

380    /// Delete `a..b` except the one-character objects at the offsets in `keep` (control
381    /// markers a deletion leaves in place).
382    pub fn delete_except(&mut self, a: usize, b: usize, keep: &[usize]) -> Result<()> {
383        let (a, b) = (a.min(b), a.max(b));
384        let mut keep: Vec<usize> =
385            keep.iter().copied().filter(|k| *k >= a && k + OBJ.len_utf8() <= b && self.text.get(*k..).is_some_and(|t| t.starts_with(OBJ))).collect();
386        if keep.is_empty() {
387            return self.delete(a, b);
388        }
389        keep.sort_unstable();
390        keep.dedup();
391        // Pieces between the kept objects, last first so earlier offsets stay valid.
392        let mut pieces = Vec::with_capacity(keep.len() + 1);
393        let mut from = a;
394        for k in &keep {
395            pieces.push((from, *k));
396            from = k + OBJ.len_utf8();
397        }
398        pieces.push((from, b));
399        for (x, y) in pieces.into_iter().rev() {
400            if y > x {
401                self.delete(x, y)?;
402            }
403        }
404        Ok(())
405    }

407    /// Apply `f` to the formatting of `a..b`.
408    pub fn format(&mut self, a: usize, b: usize, f: &dyn Fn(&mut CharProps)) -> Result<()> {
409        let (a, b) = (a.min(b), a.max(b));
410        self.check(a)?;
411        self.check(b)?;
412        if a == b {
413            return Ok(());
414        }
415        let mut out = Vec::with_capacity(self.runs.len() + 2);
416        let mut start = 0;
417        for r in self.runs.drain(..) {
418            let (rs, re) = (start, start + r.len);
419            start = re;
420            let cut = [rs, a.clamp(rs, re), b.clamp(rs, re), re];
421            for w in cut.windows(2) {
422                let (x, y) = (w[0], w[1]);
423                if y <= x {
424                    continue;
425                }
426                let mut p = r.props.clone();
427                if x >= a && y <= b {
428                    f(&mut p);
429                }
430                out.push(Run { len: y - x, props: p });
431            }
432        }
433        self.runs = out;
434        self.normalize();
435        self.touch();
436        Ok(())
437    }

439    /// Split at `off`: `self` keeps `..off`, the returned paragraph gets `off..` with the same
440    /// paragraph properties (the section break, if any, moves to the second half).
441    pub fn split_off(&mut self, off: usize) -> Result<Paragraph> {
442        self.check(off)?;
443        let k = self.objects_before(off);
444        let tail_text = self.text.split_off(off);
445        let tail_objs = if k <= self.objects.len() { self.objects.split_off(k) } else { Vec::new() };
446        let mut tail_runs = Vec::new();
447        let mut start = 0;
448        let mut keep = Vec::new();
449        for r in self.runs.drain(..) {
450            let (rs, re) = (start, start + r.len);
451            start = re;
452            if re <= off {
453                keep.push(r);
454            } else if rs >= off {
455                tail_runs.push(r);
456            } else {
457                keep.push(Run { len: off - rs, props: r.props.clone() });
458                tail_runs.push(Run { len: re - off, props: r.props });
459            }
460        }
461        self.runs = keep;
462        // The new paragraph's mark takes the formatting at the split, but never a hyperlink.
463        let carry = self.props_at(off).unlinked();
464        let mut tail = Paragraph {
465            text: tail_text,
466            runs: tail_runs,
467            props: self.props.clone(),
468            mark: if self.text.is_empty() && off == 0 { self.mark.clone() } else { carry },
469            objects: tail_objs,
470            section: self.section.take(),
471            rev: next_rev(),
472        };
473        if tail.text.is_empty() {
474            tail.mark = self.props_at(off).unlinked();
475        }
476        self.normalize();
477        tail.normalize();
478        self.touch();
479        Ok(tail)
480    }

513    /// Start of the word at or before `off` (Ctrl+Left).
514    pub fn word_start(&self, off: usize) -> usize {
515        let off = self.clamp(off);
516        let Some(before) = self.text.get(..off) else { return 0 };
517        let mut idx = off;
518        let mut seen_word = false;
519        for (i, c) in before.char_indices().rev() {
520            let w = crate::doc::bidi::is_word_char(c);
521            if w {
522                seen_word = true;
523                idx = i;
524            } else if seen_word {
525                break;
526            } else {
527                idx = i;
528            }
529        }
530        idx
531    }
532    /// End of the word after `off`, including trailing spaces (Ctrl+Right, like Word).
533    pub fn word_end(&self, off: usize) -> usize {
534        let off = self.clamp(off);
535        let Some(after) = self.text.get(off..) else { return self.text.len() };
536        let it = after.char_indices().peekable();
537        let mut end = self.text.len();
538        let first_word = after.chars().next().is_some_and(|c| crate::doc::bidi::is_word_char(c) && c != '\'');
539        let mut in_space = false;
540        for (i, c) in it {
541            let w = crate::doc::bidi::is_word_char(c);
542            if c == ' ' || c == NBSP {
543                in_space = true;
544                continue;
545            }
546            if in_space || (first_word && !w) || (!first_word && i > 0) {
547                end = off + i;
548                break;
549            }
550        }
551        end
552    }
553    /// The word around `off` (double-click): (start, end) without trailing space.
554    pub fn word_at(&self, off: usize) -> (usize, usize) {
555        let off = self.clamp(off);
556        let is_w = |c: char| crate::doc::bidi::is_word_char(c);
557        let mut a = off;
558        for (i, c) in self.text.get(..off).unwrap_or("").char_indices().rev() {
559            if !is_w(c) {
560                break;
561            }
562            a = i;
563        }
564        let mut b = off;
565        for (i, c) in self.text.get(off..).unwrap_or("").char_indices() {
566            if !is_w(c) {
567                if i == 0 && a == off {
568                    b = off + c.len_utf8();
569                }
570                break;
571            }
572            b = off + i + c.len_utf8();
573        }
574        // Word also selects the trailing spaces.
575        let mut e = b;
576        for (i, c) in self.text.get(b..).unwrap_or("").char_indices() {
577            if c != ' ' {
578                break;
579            }
580            e = b + i + 1;
581        }
582        (a, e)
583    }
584    /// Sentence around `off` (Ctrl+click): ends after `.`, `!`, `?` and following spaces.
585    pub fn sentence_at(&self, off: usize) -> (usize, usize) {
586        let off = self.clamp(off);
587        let t = &self.text;
588        let mut a = 0;
589        let mut prev_end = false;
590        for (i, c) in t.char_indices() {
591            if i > off {
592                break;
593            }
594            if prev_end && c != ' ' {
595                a = i;
596            }
597            if matches!(c, '.' | '!' | '?') {
598                prev_end = true;
599            } else if c != ' ' {
600                prev_end = false;
601            }
602        }
603        let mut b = t.len();
604        // A caret on the space right after a sentence's end belongs to that sentence; a terminator
605        // directly followed by text ("Hello.World") doesn't end the sentence the caret is in.
606        let mut ended = t.get(..off).unwrap_or("").ends_with(['.', '!', '?']) && t.get(off..).unwrap_or("").starts_with(' ');
607        for (i, c) in t.get(off..).unwrap_or("").char_indices() {
608            if ended && c != ' ' {
609                b = off + i;
610                break;
611            }
612            if matches!(c, '.' | '!' | '?') {
613                ended = true;
614            }
615        }
616        (a, b)
617    }
618}
