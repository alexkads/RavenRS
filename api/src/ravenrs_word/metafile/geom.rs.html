1//! Path operations in `f64` and the shape builders that turn WMF/EMF rectangles, ellipses and arcs
2//! into cubic Béziers (the classic 4/3·tan(Δ/4) approximation, at most 90° per piece).
3
4use std::f64::consts::{FRAC_PI_2, PI, TAU};
5
6pub(crate) type P = (f64, f64);
7
8#[derive(Clone, Copy, Debug, PartialEq)]
9pub(crate) enum Op {
10    M(P),
11    L(P),
12    C(P, P, P),
13    Z,
14}
15
16pub(crate) fn rect(l: f64, t: f64, r: f64, b: f64) -> Vec<Op> {
17    vec![Op::M((l, t)), Op::L((r, t)), Op::L((r, b)), Op::L((l, b)), Op::Z]
18}

20/// Corner radii are half the EMF/WMF corner ellipse size, clamped so the corners never overlap.
21pub(crate) fn round_rect(l: f64, t: f64, r: f64, b: f64, ew: f64, eh: f64) -> Vec<Op> {
22    // Reversed corners (right < left, bottom < top) describe the same rectangle.
23    let (l, r) = (l.min(r), l.max(r));
24    let (t, b) = (t.min(b), t.max(b));
25    let rx = (ew / 2.0).abs().min((r - l) / 2.0);
26    let ry = (eh / 2.0).abs().min((b - t) / 2.0);
27    let mut ops = vec![Op::M((l + rx, t)), Op::L((r - rx, t))];
28    ops.extend(arc(r - rx, t + ry, rx, ry, -FRAC_PI_2, FRAC_PI_2));
29    ops.push(Op::L((r, b - ry)));
30    ops.extend(arc(r - rx, b - ry, rx, ry, 0.0, FRAC_PI_2));
31    ops.push(Op::L((l + rx, b)));
32    ops.extend(arc(l + rx, b - ry, rx, ry, FRAC_PI_2, FRAC_PI_2));
33    ops.push(Op::L((l, t + ry)));
34    ops.extend(arc(l + rx, t + ry, rx, ry, PI, FRAC_PI_2));
35    ops.push(Op::Z);
36    ops
37}

39pub(crate) fn ellipse(l: f64, t: f64, r: f64, b: f64) -> Vec<Op> {
40    let (cx, cy, rx, ry) = center(l, t, r, b);
41    let mut ops = vec![Op::M((cx + rx, cy))];
42    ops.extend(arc(cx, cy, rx, ry, 0.0, TAU));
43    ops.push(Op::Z);
44    ops
45}

47/// Which closed or open shape an ARC/CHORD/PIE record draws.
48#[derive(Clone, Copy, PartialEq, Eq)]
49pub(crate) enum ArcKind {
50    Arc,
51    Chord,
52    Pie,
53}
54
55/// ARC, CHORD and PIE: the arc runs counter-clockwise on screen from the start point to the end point.
56pub(crate) fn arc_shape(l: f64, t: f64, r: f64, b: f64, s: P, e: P, kind: ArcKind) -> Vec<Op> {
57    let (cx, cy, rx, ry) = center(l, t, r, b);
58    let ang = |p: P| ((p.1 - cy) / ry.max(f64::MIN_POSITIVE)).atan2((p.0 - cx) / rx.max(f64::MIN_POSITIVE));
59    let (a0, a1) = (ang(s), ang(e));
60    let mut d = (a0 - a1).rem_euclid(TAU);
61    if d < 1e-12 {
62        d = TAU;
63    }
64    let sweep = -d;
65    let start = (cx + rx * a0.cos(), cy + ry * a0.sin());
66    let mut ops = if kind == ArcKind::Pie { vec![Op::M((cx, cy)), Op::L(start)] } else { vec![Op::M(start)] };
67    ops.extend(arc(cx, cy, rx, ry, a0, sweep));
68    match kind {
69        ArcKind::Arc => {}
70        ArcKind::Chord => ops.push(Op::Z),
71        ArcKind::Pie => ops.extend([Op::L((cx, cy)), Op::Z]),
72    }
73    ops
74}

76fn center(l: f64, t: f64, r: f64, b: f64) -> (f64, f64, f64, f64) {
77    ((l + r) / 2.0, (t + b) / 2.0, (r - l).abs() / 2.0, (b - t).abs() / 2.0)
78}

80/// Cubic pieces along an ellipse from parameter `a0` over signed `sweep` radians. The current point
81/// must already be at `(cx + rx cos a0, cy + ry sin a0)`.
82fn arc(cx: f64, cy: f64, rx: f64, ry: f64, a0: f64, sweep: f64) -> Vec<Op> {
83    let n = ((sweep.abs() / FRAC_PI_2).ceil() as usize).clamp(1, 8);
84    let step = sweep / n as f64;
85    let k = 4.0 / 3.0 * (step / 4.0).tan();
86    let pt = |t: f64| (cx + rx * t.cos(), cy + ry * t.sin());
87    let mut ops = Vec::with_capacity(n);
88    for i in 0..n {
89        let t0 = a0 + step * i as f64;
90        let t1 = t0 + step;
91        let (p0, p3) = (pt(t0), pt(t1));
92        let c1 = (p0.0 - k * rx * t0.sin(), p0.1 + k * ry * t0.cos());
93        let c2 = (p3.0 + k * rx * t1.sin(), p3.1 - k * ry * t1.cos());
94        ops.push(Op::C(c1, c2, p3));
95    }
96    ops
97}

99/// Consecutive subpaths: `counts[i]` points each, taken in order from `pts`. Counts that run past the
100/// point list are cut short rather than failing the whole record.
101pub(crate) fn subpaths(pts: &[P], counts: &[usize], close: bool) -> Vec<Op> {
102    // The exact size first: empty groups add no ops, so a long count list of zeros (file data)
103    // mustn't reserve room the canvas budget never sees.
104    let (mut n, mut at) = (0usize, 0usize);
105    for &c in counts {
106        let end = at.saturating_add(c).min(pts.len());
107        if end > at {
108            n = n.saturating_add(end - at + usize::from(close));
109        }
110        at = end;
111    }
112    let mut ops = Vec::with_capacity(n);
113    let mut at = 0usize;
114    for &c in counts {
115        let end = at.saturating_add(c).min(pts.len());
116        let Some((first, rest)) = pts.get(at..end).and_then(<[_]>::split_first) else {
117            at = end;
118            continue;
119        };
120        ops.push(Op::M(*first));
121        ops.extend(rest.iter().map(|p| Op::L(*p)));
122        if close {
123            ops.push(Op::Z);
124        }
125        at = end;
126    }
127    ops
128}
