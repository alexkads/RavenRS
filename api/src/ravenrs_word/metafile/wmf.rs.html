1//! WMF reader (MS-WMF). A file is an optional 22-byte placeable header, a 9-word standard header, then
2//! records of (size in 16-bit words, function, parameters). Parameters are read at byte 6 of each record.
3

10const PLACEABLE_KEY: u32 = 0x9AC6_CDD7;
11const DEFAULT_INCH: f64 = 1440.0;
12
13pub(crate) fn is_wmf(b: Bytes) -> bool {
14    if b.u32(0) == Some(PLACEABLE_KEY) {
15        return true;
16    }
17    matches!(b.u16(0), Some(1 | 2)) && b.u16(2) == Some(9) && matches!(b.u16(4), Some(0x100 | 0x300))
18}

20/// Reads all records. Returns the drawing state and the picture frame, or an error when no record
21/// could be read at all.
22pub(crate) fn parse(input: Bytes) -> Result<(Canvas, Option<Frame>), Error> {
23    let placeable = input.u32(0) == Some(PLACEABLE_KEY);
24    // Placeable bounds: logical corners (left, top, right, bottom) and their size in points.
25    let (hdr, bounds) = if placeable {
26        let [l, t, r, b] = [6, 8, 10, 12].map(|o| input.i16(o).map(f64::from));
27        let (Some(l), Some(t), Some(r), Some(b)) = (l, t, r, b) else { return Err(Error::BadHeader) };
28        let inch = input.u16(14).filter(|v| *v > 0).map_or(DEFAULT_INCH, f64::from);
29        let (w, h) = (r - l, b - t);
30        let bounds = (w > 0.0 && h > 0.0).then_some(([l, t, r, b], (w * 72.0 / inch, h * 72.0 / inch)));
31        (22, bounds)
32    } else {
33        (0, None)
34    };
35    let hsize = input.u16(hdr + 2).ok_or(Error::BadHeader)? as usize;
36    if hsize < 9 {
37        return Err(Error::BadHeader);
38    }
39    let mut pos = hdr.checked_add(hsize.checked_mul(2).ok_or(Error::BadHeader)?).ok_or(Error::BadHeader)?;
40    // Played into a target like Word does: the placeable bounds are the first window, the target (the
41    // picture size) the viewport. A plain WMF's target is its first window.
42    let mut cv = Canvas::new();
43    if let Some((c, size)) = bounds {
44        cv.play_into(Some([c[0], c[1], c[2] - c[0], c[3] - c[1]]), size);
45    }
46    let mut player_viewport = bounds.is_some();
47    let mut parsed = 0usize;
48    while !cv.full() {
49        let (Some(words), Some(func)) = (input.u32(pos), pos.checked_add(4).and_then(|o| input.u16(o))) else { break };
50        let Some(len) = (words as usize).checked_mul(2).filter(|l| *l >= 6) else { break };
51        let Some(rec) = input.slice(pos, len) else { break };
52        if func == 0 {
53            break;
54        }
55        record(&mut cv, func, Bytes(rec));
56        if func == 0x020C && cv.window_set && !player_viewport && !cv.viewport_set {
57            let we = cv.map.window_ext();
58            cv.play_into(None, (we.0.abs(), we.1.abs()));
59            player_viewport = true;
60        }
61        parsed += 1;
62        let Some(next) = pos.checked_add(len) else { break };
63        pos = next;
64    }
65    if parsed == 0 {
66        return Err(Error::NoRecords);
67    }
68    let frame = match bounds {
69        // Played into its target: the frame is the target.
70        Some((_, size)) if !cv.viewport_set && cv.map.uses_extents() => Some(Frame { rect: [0.0, 0.0, size.0, size.1], size: Some(size) }),
71        // The file set its own viewport, so the drawing is in its device units and the logical bounds
72        // go through the final mapping.
73        Some((c, size)) => {
74            let (a, b) = (cv.map.device((c[0], c[1])), cv.map.device((c[2], c[3])));
75            let rect = [a.0.min(b.0), a.1.min(b.1), (b.0 - a.0).abs(), (b.1 - a.1).abs()];
76            (rect[2] > 0.0 && rect[3] > 0.0).then_some(Frame { rect, size: Some(size) })
77        }
78        None if player_viewport && !cv.viewport_set && cv.map.uses_extents() => {
79            let ve = cv.map.viewport_ext();
80            Some(Frame { rect: [0.0, 0.0, ve.0, ve.1], size: None })
81        }
82        None if cv.window_set => Some(Frame { rect: cv.map.window_rect(), size: None }),
83        None => None,
84    };
85    Ok((cv, frame))
86}

88fn record(cv: &mut Canvas, func: u16, r: Bytes) {
89    if cv.full() {
90        return;
91    }
92    let _ = match func {
93        0x0103 => r.u16(6).map(|m| cv.set_map_mode(u32::from(m))),
94        0x0106 => r.u16(6).map(|m| cv.set_polyfill(u32::from(m))),
95        0x0104 => r.u16(6).map(|m| cv.set_rop2(u32::from(m))),
96        // Y is stored before X in these records.
97        0x020B => xy16(r, 6).map(|p| cv.set_window_org(p)),
98        0x020C => xy16(r, 6).map(|p| cv.set_window_ext(p)),
99        0x020D => xy16(r, 6).map(|p| cv.set_viewport_org(p)),
100        0x020E => xy16(r, 6).map(|p| cv.set_viewport_ext(p)),
101        0x001E => {
102            cv.save_dc();
103            Some(())
104        }
105        0x0127 => r.i16(6).map(|n| cv.restore_dc(i32::from(n))),
106        0x0214 => xy16(r, 6).map(|p| cv.move_to(p)),
107        0x0213 => xy16(r, 6).map(|p| cv.line_to(p)),
108        0x0325 => counted_points(r).map(|pts| cv.poly(&pts, false, false, true)),
109        0x0324 => counted_points(r).map(|pts| cv.poly(&pts, true, true, true)),
110        0x0538 => polypolygon(r).map(|ops| cv.shape(ops, true, true)),
111        0x041B => rect16(r, 6).map(|[l, t, rr, b]| cv.shape(geom::rect(l, t, rr, b), true, true)),
112        0x061C => {
113            let [ew, eh] = [r.i16(8), r.i16(6)];
114            let (Some(ew), Some(eh)) = (ew, eh) else { return };
115            rect16(r, 10).map(|[l, t, rr, b]| {
116                cv.shape(geom::round_rect(l, t, rr, b, f64::from(ew), f64::from(eh)), true, true);
117            })
118        }
119        0x0418 => rect16(r, 6).map(|[l, t, rr, b]| cv.shape(geom::ellipse(l, t, rr, b), true, true)),
120        0x0817 | 0x081A | 0x0830 => arc16(r).map(|(bounds, s, e)| {
121            let kind = match func {
122                0x0817 => ArcKind::Arc,
123                0x081A => ArcKind::Pie,
124                _ => ArcKind::Chord,
125            };
126            let [l, t, rr, b] = bounds;
127            cv.shape(geom::arc_shape(l, t, rr, b, s, e, kind), kind != ArcKind::Arc, true);
128        }),
129        0x02FA => create_pen(cv, r),
130        0x02FC => create_brush(cv, r),
131        // Fonts, palettes, pattern brushes and regions take a handle slot too, so later SELECTOBJECT
132        // handles count them; selecting one draws nothing.
133        0x02FB | 0x00F7 | 0x01F9 | 0x0142 | 0x06FF => {
134            cv.add_object(None, Obj::Other);
135            Some(())
136        }
137        0x012D => r.u16(6).map(|h| cv.select(wmf_handle(h))),
138        0x01F0 => r.u16(6).map(|h| cv.delete_object(u32::from(h))),
139        // STRETCHDIB: rop(4) usage(2) srcH srcW ySrc xSrc destH destW yDest xDest, then the DIB.
140        0x0F43 => {
141            let rop = r.u32(6).unwrap_or(0);
142            dib_blit(cv, r, 28, rop, [i16s(r, 26), i16s(r, 24), i16s(r, 22), i16s(r, 20)], [i16s(r, 18), i16s(r, 16), i16s(r, 14), i16s(r, 12)])
143        }
144        0x0B41 => dib_stretchblt(cv, r),
145        0x0940 => dib_bitblt(cv, r),
146        _ => Some(()),
147    };
148}

150/// A 16-bit WMF handle as a canvas object index: stock objects are `0x8000 | index`, which the
151/// canvas reads with the 32-bit stock flag.
152fn wmf_handle(h: u16) -> u32 {
153    let h = u32::from(h);
154    if h & 0x8000 != 0 { (h & 0x7FFF) | 0x8000_0000 } else { h }
155}

157/// An i16 field as f64; 0 past the end.
158fn i16s(r: Bytes, off: usize) -> f64 {
159    r.i16(off).map_or(0.0, f64::from)
160}

162/// CREATEPENINDIRECT: style, width (x, y), colour. Only the width's x component is used.
163fn create_pen(cv: &mut Canvas, r: Bytes) -> Option<()> {
164    let style = r.u16(6)?;
165    let width = r.i16(8)?;
166    let color = r.u32(12)?;
167    cv.add_object(None, Obj::Pen(Pen { style: u32::from(style), width: f64::from(width), color: colorref(color) }));
168    Some(())
169}
170
171/// CREATEBRUSHINDIRECT: style, colour, hatch (hatch is not drawn).
172fn create_brush(cv: &mut Canvas, r: Bytes) -> Option<()> {
173    let style = r.u16(6)?;
174    let color = r.u32(8)?;
175    cv.add_object(None, Obj::Brush(Brush { style: u32::from(style), color: colorref(color) }));
176    Some(())
177}

179/// (x, y) from a Y-then-X pair stored at `off`.
180fn xy16(r: Bytes, off: usize) -> Option<P> {
181    Some((f64::from(r.i16(off + 2)?), f64::from(r.i16(off)?)))
182}
183
184/// Rectangle stored as bottom, right, top, left (the order MS-WMF uses for RECTANGLE and ELLIPSE).
185fn rect16(r: Bytes, off: usize) -> Option<[f64; 4]> {
186    let (b, rr, t, l) = (r.i16(off)?, r.i16(off + 2)?, r.i16(off + 4)?, r.i16(off + 6)?);
187    Some([f64::from(l), f64::from(t), f64::from(rr), f64::from(b)])
188}
189
190/// ARC, PIE and CHORD: end Y, end X, start Y, start X, then the bounding rectangle (bottom, right, top, left).
191fn arc16(r: Bytes) -> Option<([f64; 4], P, P)> {
192    let end = xy16(r, 6)?;
193    let start = xy16(r, 10)?;
194    let bounds = rect16(r, 14)?;
195    Some((bounds, start, end))
196}

198/// A count of 16-bit points, then the points (POLYLINE, POLYGON).
199fn counted_points(r: Bytes) -> Option<Vec<P>> {
200    let n = r.u16(6)? as usize;
201    r.tail(8)?.points(0, n, false)
202}
203
204/// POLYPOLYGON: polygon count, one 16-bit point count per polygon, then all points.
205fn polypolygon(r: Bytes) -> Option<Vec<geom::Op>> {
206    let n = r.u16(6)? as usize;
207    let mut counts = Vec::with_capacity(n);
208    for i in 0..n {
209        counts.push(r.u16(i.checked_mul(2)?.checked_add(8)?)? as usize);
210    }
211    let total = counts.iter().try_fold(0usize, |a, c| a.checked_add(*c))?;
212    if total > MAX_PATH_OPS {
213        return None;
214    }
215    let pts = r.tail(n.checked_mul(2)?.checked_add(8)?)?.points(0, total, false)?;
216    Some(geom::subpaths(&pts, &counts, true))
217}

219/// Whether a DIBBITBLT or DIBSTRETCHBLT record carries a bitmap: without one it is exactly
220/// `(function >> 8) + 3` words long (MS-WMF 2.3.1.2, 2.3.1.3).
221fn has_bitmap(func: u16, r: Bytes) -> bool {
222    r.0.len() / 2 > usize::from(func >> 8) + 3
223}
224
225/// META_DIBBITBLT: rop(4), ySrc, xSrc, then height, width, yDest, xDest and the DIB at 22. The
226/// bitmap-less form has a reserved word before the height, so its fields sit 2 bytes later; it, or
227/// a source-less raster operation, fills the destination.
228fn dib_bitblt(cv: &mut Canvas, r: Bytes) -> Option<()> {
229    let rop = r.u32(6)?;
230    if !has_bitmap(0x0940, r) {
231        cv.fill_rop([i16s(r, 22), i16s(r, 20), i16s(r, 18), i16s(r, 16)], rop);
232        return Some(());
233    }
234    let dst = [i16s(r, 20), i16s(r, 18), i16s(r, 16), i16s(r, 14)];
235    if Canvas::sourceless(rop) {
236        cv.fill_rop(dst, rop);
237        return Some(());
238    }
239    let src = [i16s(r, 12), i16s(r, 10), dst[2], dst[3]];
240    dib_blit(cv, r, 22, rop, dst, src)
241}

243/// META_DIBSTRETCHBLT: rop(4), srcH, srcW, ySrc, xSrc, then destH, destW, yDest, xDest and the DIB at
244/// 26. The bitmap-less form has a reserved word after xSrc, so the destination sits 2 bytes later;
245/// it, or a source-less raster operation, fills the destination.
246fn dib_stretchblt(cv: &mut Canvas, r: Bytes) -> Option<()> {
247    let rop = r.u32(6)?;
248    if !has_bitmap(0x0B41, r) {
249        cv.fill_rop([i16s(r, 26), i16s(r, 24), i16s(r, 22), i16s(r, 20)], rop);
250        return Some(());
251    }
252    let dst = [i16s(r, 24), i16s(r, 22), i16s(r, 20), i16s(r, 18)];
253    if Canvas::sourceless(rop) {
254        cv.fill_rop(dst, rop);
255        return Some(());
256    }
257    let src = [i16s(r, 16), i16s(r, 14), i16s(r, 12), i16s(r, 10)];
258    dib_blit(cv, r, 26, rop, dst, src)
259}

261/// A DIB at `at`, stretched from source rectangle `src` into `dst` (both in logical units, pixels for `src`).
262/// The raster operation `rop` decides how the pixels combine (see [`dib::prepare`]).
263fn dib_blit(cv: &mut Canvas, r: Bytes, at: usize, rop: u32, dst: [f64; 4], src: [f64; 4]) -> Option<()> {
264    let dib = r.tail(at)?;
265    let n = dib::info_len(dib)?;
266    let info = dib.slice(0, n)?;
267    let bits = dib.tail(n)?;
268    let (mut img, dst) = dib::decode_clip(Bytes(info), bits, false, src.map(|v| v as i64), dst)?;
269    dib::prepare(&mut img, dib::Blend::Rop(rop));
270    cv.bitmap(dst, img);
271    Some(())
