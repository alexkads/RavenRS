1//! EMF reader (MS-EMF). Records are (type, size in bytes) followed by a payload; offsets inside a
2//! record (bitmap info and bits) are relative to the record start, so each handler reads the record slice.

10const EMF_SIGNATURE: u32 = 0x464D_4520;
11/// Hundredths of a millimetre to points.
12const HMM_TO_PT: f64 = 72.0 / 2540.0;
13/// SRCCOPY, the plain copy of a bitmap.
14const SRCCOPY: u32 = 0x00CC_0020;
15/// LogPenEx pen type bits; PS_COSMETIC pens are one device pixel wide.
16const PS_TYPE_MASK: u32 = 0x000F_0000;
17const PS_COSMETIC: u32 = 0;
18/// EXTTEXTOUT options: the string holds glyph indices, and the dx array holds (x, y) pairs.
19const ETO_GLYPH_INDEX: u32 = 0x10;
20const ETO_PDY: u32 = 0x2000;
21
22pub(crate) fn is_emf(b: Bytes) -> bool {
23    b.u32(0) == Some(1) && b.u32(40) == Some(EMF_SIGNATURE)
24}

26/// The picture frame: `rclFrame` (0.01 mm) mapped to device units through `szlDevice` and
27/// `szlMillimeters`, sized in points. Without a usable frame, the device bounds stand in (1 unit = 1 point).
28pub(crate) fn parse(input: Bytes) -> Result<(Canvas, Option<Frame>), Error> {
29    if !is_emf(input) {
30        return Err(Error::BadHeader);
31    }
32    let hsize = input.u32(4).ok_or(Error::BadHeader)? as usize;
33    let bounds = rect32(input, 8).ok_or(Error::BadHeader)?;
34    let [l, t, r, b] = bounds;
35    let frame = frame_from_header(input).or_else(|| (r > l && b > t).then_some(Frame { rect: [l, t, r - l, b - t], size: None }));
36    let mut cv = Canvas::new();
37    if let Some(px_mm) = device_px_per_mm(input) {
38        cv.set_device_mm(px_mm);
39    }
40    let mut parsed = 0usize;
41    let mut pos = hsize.max(88);
42    while !cv.full() {
43        let (Some(typ), Some(size)) = (input.u32(pos), pos.checked_add(4).and_then(|o| input.u32(o))) else { break };
44        let size = size as usize;
45        if size < 8 || !size.is_multiple_of(4) {
46            break;
47        }
48        let Some(rec) = input.slice(pos, size) else { break };
49        if typ == 14 {
50            break;
51        }
52        record(&mut cv, typ, Bytes(rec));
53        parsed += 1;
54        let Some(next) = pos.checked_add(size) else { break };
55        pos = next;
56    }
57    if parsed == 0 {
58        return Err(Error::NoRecords);
59    }
60    Ok((cv, frame))
61}

63/// Device pixels per millimetre on each axis, from `szlDevice` (72) and `szlMillimeters` (80).
64fn device_px_per_mm(input: Bytes) -> Option<(f64, f64)> {
65    let (dw, dh) = (input.i32(72)?, input.i32(76)?);
66    let (mw, mh) = (input.i32(80)?, input.i32(84)?);
67    (dw > 0 && dh > 0 && mw > 0 && mh > 0).then(|| (f64::from(dw) / f64::from(mw), f64::from(dh) / f64::from(mh)))
68}

70/// The frame from `rclFrame` (offset 24), `szlDevice` (72) and `szlMillimeters` (80) in the header.
71pub(crate) fn frame_from_header(input: Bytes) -> Option<Frame> {
72    let [fl, ft, fr, fb] = rect32(input, 24)?;
73    if fr <= fl || fb <= ft {
74        return None;
75    }
76    // Device units per hundredth of a millimetre.
77    let (px_mm_x, px_mm_y) = device_px_per_mm(input)?;
78    let (sx, sy) = (px_mm_x / 100.0, px_mm_y / 100.0);
79    let rect = [fl * sx, ft * sy, (fr - fl) * sx, (fb - ft) * sy];
80    Some(Frame { rect, size: Some(((fr - fl) * HMM_TO_PT, (fb - ft) * HMM_TO_PT)) })
81}

83fn record(cv: &mut Canvas, typ: u32, r: Bytes) {
84    if cv.full() {
85        return;
86    }
87    let _ = match typ {
88        9 => xy32(r, 8).map(|p| cv.set_window_ext(p)),
89        10 => xy32(r, 8).map(|p| cv.set_window_org(p)),
90        11 => xy32(r, 8).map(|p| cv.set_viewport_ext(p)),
91        12 => xy32(r, 8).map(|p| cv.set_viewport_org(p)),
92        17 => r.u32(8).map(|m| cv.set_map_mode(m)),
93        19 => r.u32(8).map(|m| cv.set_polyfill(m)),
94        20 => r.u32(8).map(|m| cv.set_rop2(m)),
95        33 => {
96            cv.save_dc();
97            Some(())
98        }
99        34 => r.i32(8).map(|n| cv.restore_dc(n)),
100        35 => xform(r, 8).map(|x| cv.modify_world(x, 4)),
101        36 => xform(r, 8).zip(r.u32(32)).map(|(x, m)| cv.modify_world(x, m)),
102        37 => r.u32(8).map(|i| cv.select(i)),
103        38 => create_pen(cv, r),
104        95 => create_ext_pen(cv, r),
105        39 => create_brush(cv, r),
106        82 => create_font(cv, r),
107        22 => r.u32(8).map(|a| cv.set_text_align(a)),
108        24 => r.u32(8).map(|c| cv.set_text_color(colorref(c))),
109        84 => ext_text_out_w(cv, r),
110        40 => r.u32(8).map(|i| cv.delete_object(i)),
111        27 => xy32(r, 8).map(|p| cv.move_to(p)),
112        54 => xy32(r, 8).map(|p| cv.line_to(p)),
113        42 => rect32(r, 8).map(|[l, t, rr, b]| cv.shape(geom::ellipse(l, t, rr, b), true, true)),
114        43 => rect32(r, 8).map(|[l, t, rr, b]| cv.shape(geom::rect(l, t, rr, b), true, true)),
115        44 => {
116            let (cw, ch) = (r.i32(24), r.i32(28));
117            let (Some(cw), Some(ch)) = (cw, ch) else { return };
118            rect32(r, 8).map(|[l, t, rr, b]| {
119                cv.shape(geom::round_rect(l, t, rr, b, f64::from(cw), f64::from(ch)), true, true);
120            })
121        }
122        45..=47 => arc32(r).map(|(bounds, s, e)| {
123            let kind = match typ {
124                45 => ArcKind::Arc,
125                46 => ArcKind::Chord,
126                _ => ArcKind::Pie,
127            };
128            let [l, t, rr, b] = bounds;
129            cv.shape(geom::arc_shape(l, t, rr, b, s, e, kind), kind != ArcKind::Arc, true);
130        }),
131        4 | 87 => counted(r, typ == 4).map(|pts| cv.poly(&pts, false, false, true)),
132        3 | 86 => counted(r, typ == 3).map(|pts| cv.poly(&pts, true, true, true)),
133        6 | 89 => counted(r, typ == 6).map(|pts| cv.line_chain(&pts, false, true)),
134        // POLYBEZIER starts its own figure at the first point and leaves the current point alone;
135        // POLYBEZIERTO continues from the current point.
136        2 | 85 => counted(r, typ == 2).and_then(|pts| {
137            let from = *pts.first()?;
138            cv.bezier(from, pts.get(1..)?);
139            Some(())
140        }),
141        5 | 88 => counted(r, typ == 5).map(|pts| cv.bezier_to(&pts)),
142        7 | 90 => poly_groups(r, typ == 7).map(|(counts, pts)| cv.shape(geom::subpaths(&pts, &counts, false), false, true)),
143        8 | 91 => poly_groups(r, typ == 8).map(|(counts, pts)| cv.shape(geom::subpaths(&pts, &counts, true), true, true)),
144        59 => {
145            cv.begin_path();
146            Some(())
147        }
148        60 => {
149            cv.end_path();
150            Some(())
151        }
152        61 => {
153            cv.close_figure();
154            Some(())
155        }
156        62 => {
157            cv.paint_path(true, false);
158            Some(())
159        }
160        63 => {
161            cv.paint_path(true, true);
162            Some(())
163        }
164        64 => {
165            cv.paint_path(false, true);
166            Some(())
167        }
168        76 | 77 => bitblt(cv, r, typ == 77),
169        80 => set_dibits(cv, r),
170        81 => stretch_dibits(cv, r),
171        114 => alphablend(cv, r),
172        _ => Some(()),
173    };
174}

176/// Two little-endian i32 values at `off`, as a point.
177fn xy32(r: Bytes, off: usize) -> Option<P> {
178    Some((f64::from(r.i32(off)?), f64::from(r.i32(off + 4)?)))
179}
180
181/// RECTL (left, top, right, bottom) as i32 values widened to f64.
182fn rect32(r: Bytes, off: usize) -> Option<[f64; 4]> {
183    Some([f64::from(r.i32(off)?), f64::from(r.i32(off + 4)?), f64::from(r.i32(off + 8)?), f64::from(r.i32(off + 12)?)])
184}
185
186/// An i32 field as f64; 0 past the end.
187fn i32s(r: Bytes, off: usize) -> f64 {
188    r.i32(off).map_or(0.0, f64::from)
189}

191/// XFORM: six f32 values in M11, M12, M21, M22, Dx, Dy order.
192fn xform(r: Bytes, off: usize) -> Option<Xform> {
193    let mut x = [0.0f64; 6];
194    for (i, v) in x.iter_mut().enumerate() {
195        *v = f64::from(r.f32(off + i * 4)?);
196    }
197    Some(x)
198}

200/// ARC, CHORD, PIE: bounds at 8, start point at 24, end point at 32.
201fn arc32(r: Bytes) -> Option<([f64; 4], P, P)> {
202    Some((rect32(r, 8)?, xy32(r, 24)?, xy32(r, 32)?))
203}

205/// A point count from a record, rejected when it exceeds the path budget (before any allocation).
206fn budget(n: u32) -> Option<usize> {
207    let n = n as usize;
208    (n <= MAX_PATH_OPS).then_some(n)
209}
210
211/// Point count at 24, then the points (32-bit or 16-bit) at 28.
212fn counted(r: Bytes, wide: bool) -> Option<Vec<P>> {
213    let n = budget(r.u32(24)?)?;
214    r.tail(28)?.points(0, n, wide)
215}

217/// Subpath count at 24, total point count at 28, per-subpath counts at 32, then the points.
218fn poly_groups(r: Bytes, wide: bool) -> Option<(Vec<usize>, Vec<P>)> {
219    let n = budget(r.u32(24)?)?;
220    // The record must hold the n counts before anything is allocated for them.
221    let raw = r.slice(32, n.checked_mul(4)?)?;
222    let counts: Vec<usize> = raw.as_chunks::<4>().0.iter().map(|c| u32::from_le_bytes(*c) as usize).collect();
223    let total = counts.iter().try_fold(0usize, |a, c| a.checked_add(*c))?;
224    let total = budget(u32::try_from(total).ok()?)?;
225    let pts = r.tail(n.checked_mul(4)?.checked_add(32)?)?.points(0, total, wide)?;
226    Some((counts, pts))
227}

229/// EMR_CREATEPEN: handle, LogPen (style, width x/y, colour).
230fn create_pen(cv: &mut Canvas, r: Bytes) -> Option<()> {
231    let slot = r.u32(8)? as usize;
232    let style = r.u32(12)?;
233    let width = r.i32(16)?;
234    let color = r.u32(24)?;
235    cv.add_object(Some(slot), Obj::Pen(Pen { style, width: f64::from(width), color: colorref(color) }));
236    Some(())
237}

239/// EMR_EXTCREATEPEN: handle, bitmap offsets (unused), LogPenEx (style, width, brush style, colour).
240/// A cosmetic pen's width is in device pixels (always 1), so it is a hairline; a BS_NULL brush draws
241/// no line at all.
242fn create_ext_pen(cv: &mut Canvas, r: Bytes) -> Option<()> {
243    let slot = r.u32(8)? as usize;
244    let (style, width, brush, color) = (r.u32(28)?, r.u32(32)?, r.u32(36)?, r.u32(40)?);
245    let width = if style & PS_TYPE_MASK == PS_COSMETIC { 0.0 } else { f64::from(width) };
246    let style = if brush == BS_NULL { PS_NULL } else { style & PS_STYLE_MASK };
247    cv.add_object(Some(slot), Obj::Pen(Pen { style, width, color: colorref(color) }));
248    Some(())
249}

260/// EMR_EXTCREATEFONTINDIRECTW: handle, then a LogFont (height, width, escapement, orientation, weight,
261/// italic, ..., face name of 32 UTF-16 units at 40).
262fn create_font(cv: &mut Canvas, r: Bytes) -> Option<()> {
263    let slot = r.u32(8)? as usize;
264    let mut face = [0u16; 32];
265    for (i, u) in face.iter_mut().enumerate() {
266        *u = r.u16(40 + i * 2).unwrap_or(0);
267    }
268    let font = Font { height: r.i32(12)?, escapement: r.i32(20)?, weight: r.i32(28)?, italic: *r.0.get(32)? != 0, face };
269    cv.add_object(Some(slot), Obj::Font(font));
270    Some(())
271}

273/// EMR_EXTTEXTOUTW: an EmrText at 36 with the reference point, the character count at 44, the string
274/// offset at 48, the options at 52 and the dx array offset at 72 (offsets from the record start).
275/// Glyph-index strings are skipped: they cannot be mapped back to characters.
276fn ext_text_out_w(cv: &mut Canvas, r: Bytes) -> Option<()> {
277    let at = xy32(r, 36)?;
278    let n = r.u32(44)? as usize;
279    let (off, options, off_dx) = (r.u32(48)? as usize, r.u32(52)?, r.u32(72)? as usize);
280    if options & ETO_GLYPH_INDEX != 0 || n == 0 {
281        return None;
282    }
283    let units: Vec<u16> = r.slice(off, n.checked_mul(2)?)?.as_chunks::<2>().0.iter().map(|c| u16::from_le_bytes(*c)).collect();
284    // dx has one entry per UTF-16 unit (two with ETO_PDY); a character takes the sum of its units.
285    let step = if options & ETO_PDY != 0 { 8 } else { 4 };
286    let raw_dx: Vec<f64> = match r.slice(off_dx, n.checked_mul(step)?) {
287        Some(b) if off_dx != 0 => b.chunks_exact(step).filter_map(|c| Some(f64::from(i32::from_le_bytes(c.get(..4)?.try_into().ok()?)))).collect(),
288        _ => Vec::new(),
289    };
290    let mut text = String::new();
291    let mut dx = Vec::new();
292    let mut i = 0;
293    for ch in char::decode_utf16(units.iter().copied()) {
294        let c = ch.unwrap_or(char::REPLACEMENT_CHARACTER);
295        // A replacement stands for one unpaired unit.
296        let w = if c == char::REPLACEMENT_CHARACTER { 1 } else { c.len_utf16() };
297        if !raw_dx.is_empty() {
298            dx.push(raw_dx.get(i..i + w).map_or(0.0, |s| s.iter().sum()));
299        }
300        i += w;
301        text.push(c);
302    }
303    let text = text.trim_end_matches('\0').to_string();
304    dx.truncate(text.chars().count());
305    cv.text(at, text, &dx);
306    Some(())
307}

309/// BITBLT (76) and STRETCHBLT (77): dest at 24..40, rop at 40, source x/y at 44, bmi at 84, bits at
310/// 92; STRETCHBLT adds the source w/h at 100. A source-less rop fills the destination instead.
311fn bitblt(cv: &mut Canvas, r: Bytes, stretch: bool) -> Option<()> {
312    let dst = [i32s(r, 24), i32s(r, 28), i32s(r, 32), i32s(r, 36)];
313    let rop = r.u32(40)?;
314    if Canvas::sourceless(rop) {
315        cv.fill_rop(dst, rop);
316        return Some(());
317    }
318    let (sw, sh) = if stretch { (i32s(r, 100), i32s(r, 104)) } else { (dst[2], dst[3]) };
319    let src = [i32s(r, 44), i32s(r, 48), sw, sh];
320    blit(cv, r, dst, src, (84, 92), Blend::Rop(rop))
321}

323/// ALPHABLEND: laid out like STRETCHBLT, with a BLENDFUNCTION at 40. Constant alpha at 42 applies to
324/// every pixel; AlphaFormat at 43 with AC_SRC_ALPHA (1) also uses the per-pixel alpha.
325fn alphablend(cv: &mut Canvas, r: Bytes) -> Option<()> {
326    let dst = [i32s(r, 24), i32s(r, 28), i32s(r, 32), i32s(r, 36)];
327    let constant = *r.0.get(42)?;
328    let per_pixel = (*r.0.get(43)? & 0x01) != 0;
329    let src = [i32s(r, 44), i32s(r, 48), i32s(r, 100), i32s(r, 104)];
330    blit(cv, r, dst, src, (84, 92), Blend::Alpha { constant, per_pixel })
331}

333/// STRETCHDIBITS: dest x/y at 24, src x/y/w/h at 32..48, bmi offset/size at 48, bits at 56, rop at 68,
334/// dest w/h at 72.
335fn stretch_dibits(cv: &mut Canvas, r: Bytes) -> Option<()> {
336    let dst = [i32s(r, 24), i32s(r, 28), i32s(r, 72), i32s(r, 76)];
337    let src = [i32s(r, 32), i32s(r, 36), i32s(r, 40), i32s(r, 44)];
338    let rop = r.u32(68)?;
339    blit(cv, r, dst, src, (48, 56), Blend::Rop(rop))
340}

342/// SETDIBITSTODEVICE (MS-EMF 2.3.1.5): dest x/y at 24, src x/y at 32, size (cxSrc, cySrc) at 40, bmi
343/// at 48, bits at 56, start scan at 68. The image keeps its own size; the bits begin at scan line
344/// `iStartScan`, so the source y counts from there.
345fn set_dibits(cv: &mut Canvas, r: Bytes) -> Option<()> {
346    let (cx, cy) = (i32s(r, 40), i32s(r, 44));
347    if cx <= 0.0 || cy <= 0.0 {
348        return None;
349    }
350    let dst = [i32s(r, 24), i32s(r, 28), cx, cy];
351    let src = [i32s(r, 32), i32s(r, 36) - f64::from(r.u32(68)?), cx, cy];
352    blit(cv, r, dst, src, (48, 56), Blend::Rop(SRCCOPY))
353}

355/// Decodes the bitmap at `at` = (offset of the bmi offset/length pair, offset of the bits pair), both
356/// inside the record, and stretches the source rectangle `src` into `dst`. Both are `[x, y, w, h]`; a
357/// non-positive source size means the whole bitmap. `blend` decides how it combines.
358fn blit(cv: &mut Canvas, r: Bytes, dst: [f64; 4], src: [f64; 4], at: (usize, usize), blend: Blend) -> Option<()> {
359    let (bo, bl) = (r.u32(at.0)? as usize, r.u32(at.0 + 4)? as usize);
360    let (po, pl) = (r.u32(at.1)? as usize, r.u32(at.1 + 4)? as usize);
361    if pl == 0 {
362        return None;
363    }
364    let info = Bytes(r.slice(bo, bl)?);
365    let pixels = Bytes(r.slice(po, pl)?);
366    let (mut img, dst) = dib::decode_clip(info, pixels, blend.per_pixel(), src.map(|v| v as i64), dst)?;
367    dib::prepare(&mut img, blend);
368    cv.bitmap(dst, img);
369    Some(())
