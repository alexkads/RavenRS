1//! WordCraft metafiles: reads Windows Metafiles (WMF) and Enhanced Metafiles (EMF) into a neutral
2//! vector model of filled and stroked paths, RGBA bitmaps and text runs, in a picture frame with y down.
3//!
4//! The reader is pure Rust over byte slices (no file system), so it builds for wasm. Every read is
5//! bounds-checked and every count is capped; hostile input yields a shorter picture or an [`Error`],
6//! never a panic.
7//!
8//! Records follow MS-WMF and MS-EMF. Records outside the supported set are skipped silently.
9#![deny(clippy::unwrap_used, clippy::expect_used, clippy::panic, clippy::unimplemented, clippy::todo, clippy::unreachable)]
10#![forbid(unsafe_code)]

43/// A parsed metafile. Items use coordinates in `0..width` and `0..height`, y down.
44#[derive(Debug, Clone, PartialEq)]
45pub struct Picture {
46    /// Size of the drawing in its own units: points for placeable WMF and EMF (from the frame), device
47    /// or logical units otherwise.
48    pub width: f32,
49    pub height: f32,
50    pub items: Vec<Item>,
51}
52
53#[derive(Debug, Clone, PartialEq)]
54pub enum Item {
55    /// A filled and/or stroked path, in picture coordinates.
56    Path {
57        segs: Vec<Seg>,
58        fill: Option<[u8; 4]>,
59        /// Colour and width in picture units.
60        stroke: Option<([u8; 4], f32)>,
61        even_odd: bool,
62    },
63    /// A bitmap stretched into `rect` = `[x, y, w, h]` (picture units), RGBA8 rows top-down.
64    Bitmap { rect: [f32; 4], width: u32, height: u32, rgba: Vec<u8> },
65    /// A run of text; the renderers shape it with the named font (see [`Text::layout`]).
66    Text(Text),
67}

69/// A text run, in picture units.
70#[derive(Debug, Clone, PartialEq)]
71pub struct Text {
72    pub text: String,
73    /// The reference point that `halign` and `valign` refer to.
74    pub x: f32,
75    pub y: f32,
76    /// Font height: the em size, or the cell height (ascent + descent) when `cell` is set.
77    pub size: f32,
78    pub cell: bool,
79    pub font: String,
80    pub bold: bool,
81    pub italic: bool,
82    pub color: [u8; 4],
83    /// Baseline direction in radians, clockwise from +x (y down).
84    pub angle: f32,
85    pub halign: HAlign,
86    pub valign: VAlign,
87    /// Advance of each character along the baseline; empty when the font's own advances apply.
88    pub dx: Vec<f32>,
89}

125/// The picture frame in device units (`rect` = `[x, y, w, h]`) and, when the file states one, its
126/// size in picture units.
127pub(crate) struct Frame {
128    pub rect: [f64; 4],
129    pub size: Option<(f64, f64)>,
130}

132/// Parses a WMF (placeable or plain) or EMF file.
133pub fn parse(bytes: &[u8]) -> Result<Picture, Error> {
134    let b = Bytes(bytes);
135    let (cv, frame) = if emf::is_emf(b) {
136        emf::parse(b)?
137    } else if wmf::is_wmf(b) {
138        wmf::parse(b)?
139    } else {
140        return Err(Error::NotMetafile);
141    };
142    Ok(build(cv.items, frame))
143}

145/// True when the bytes start like a WMF (placeable or plain header) or an EMF.
146pub fn is_metafile(bytes: &[u8]) -> bool {
147    let b = Bytes(bytes);
148    emf::is_emf(b) || wmf::is_wmf(b)
149}

151/// Maps device-space output into the picture frame. Without a usable frame the frame is the bounding
152/// box of the drawing, so the origin is still the top-left of what was drawn.
153fn build(raw: Vec<Raw>, frame: Option<Frame>) -> Picture {
154    let frame = frame.filter(|f| f.rect.iter().all(|v| v.is_finite()) && f.rect[2] > 0.0 && f.rect[3] > 0.0);
155    let (rect, size) = match frame {
156        Some(f) => (f.rect, f.size.unwrap_or((f.rect[2], f.rect[3]))),
157        None => match bbox(&raw) {
158            Some([x0, y0, x1, y1]) => ([x0, y0, x1 - x0, y1 - y0], (x1 - x0, y1 - y0)),
159            None => return Picture { width: 0.0, height: 0.0, items: Vec::new() },
160        },
161    };
162    let sx = if rect[2] > 0.0 { size.0 / rect[2] } else { 1.0 };
163    let sy = if rect[3] > 0.0 { size.1 / rect[3] } else { 1.0 };
164    let tf = move |p: P| ((p.0 - rect[0]) * sx, (p.1 - rect[1]) * sy);
165    let stroke_scale = (sx + sy) / 2.0;
166    // No stroke is wider than the picture's diagonal: anything wider is a mapping artefact.
167    let diag = size.0.hypot(size.1);
168    let items = raw.into_iter().filter_map(|r| match r {
169        Raw::Path { ops, fill, stroke, even_odd } => {
170            let segs = ops.iter().map(|op| seg(op, tf)).collect::<Option<Vec<Seg>>>()?;
171            if segs.len() < 2 {
172                return None;
173            }
174            let stroke = stroke.and_then(|(c, w)| Some((c, f((w * stroke_scale).min(diag))?)));
175            Some(Item::Path { segs, fill, stroke, even_odd })
176        }
177        Raw::Bitmap { rect: r, img } => {
178            let (x, y) = tf((r[0], r[1]));
179            let rect = [f(x)?, f(y)?, f(r[2] * sx)?, f(r[3] * sy)?];
180            Some(Item::Bitmap { rect, width: img.w as u32, height: img.h as u32, rgba: img.px })
181        }
182        Raw::Text(t) => text_item(t, tf, sx, sy),
183    });
184    let items = items.collect();
185    Picture { width: f(size.0).unwrap_or(0.0), height: f(size.1).unwrap_or(0.0), items }
186}

188/// A device-space text run mapped into the picture: the reference point through `tf`, lengths by the
189/// frame scale. TA_* flags become the alignment.
190fn text_item(t: RawText, tf: impl Fn(P) -> P, sx: f64, sy: f64) -> Option<Item> {
191    let (x, y) = tf(t.origin);
192    let halign = match t.align & 6 {
193        2 => HAlign::Right,
194        6 => HAlign::Center,
195        _ => HAlign::Left,
196    };
197    let valign = match t.align & 24 {
198        8 => VAlign::Bottom,
199        24 => VAlign::Baseline,
200        _ => VAlign::Top,
201    };
202    let dx = t.dx.iter().map(|d| f(d * sx.abs())).collect::<Option<Vec<f32>>>()?;
203    let size = f(t.size * sy.abs()).filter(|s| *s > 0.0)?;
204    Some(Item::Text(Text {
205        text: t.text,
206        x: f(x)?,
207        y: f(y)?,
208        size,
209        cell: t.cell,
210        font: t.font,
211        bold: t.bold,
212        italic: t.italic,
213        color: t.color,
214        angle: f(t.angle).unwrap_or(0.0),
215        halign,
216        valign,
217        dx,
218    }))
219}

221fn f(v: f64) -> Option<f32> {
222    let v = v as f32;
223    v.is_finite().then_some(v)
224}

226fn seg(op: &Op, tf: impl Fn(P) -> P) -> Option<Seg> {
227    let pt = |p: P| {
228        let (x, y) = tf(p);
229        Some((f(x)?, f(y)?))
230    };
231    Some(match *op {
232        Op::M(p) => {
233            let (x, y) = pt(p)?;
234            Seg::Move(x, y)
235        }
236        Op::L(p) => {
237            let (x, y) = pt(p)?;
238            Seg::Line(x, y)
239        }
240        Op::C(a, b, c) => {
241            let (x1, y1) = pt(a)?;
242            let (x2, y2) = pt(b)?;
243            let (x3, y3) = pt(c)?;
244            Seg::Cubic(x1, y1, x2, y2, x3, y3)
245        }
246        Op::Z => Seg::Close,
247    })
248}

250/// Bounding box `[x0, y0, x1, y1]` over the finite points of the output.
251fn bbox(raw: &[Raw]) -> Option<[f64; 4]> {
252    let mut b: Option<[f64; 4]> = None;
253    let mut grow = |p: P| {
254        if p.0.is_finite() && p.1.is_finite() {
255            b = Some(match b {
256                None => [p.0, p.1, p.0, p.1],
257                Some([x0, y0, x1, y1]) => [x0.min(p.0), y0.min(p.1), x1.max(p.0), y1.max(p.1)],
258            });
259        }
260    };
261    for r in raw {
262        match r {
263            Raw::Path { ops, .. } => {
264                for op in ops {
265                    match *op {
266                        Op::M(p) | Op::L(p) => grow(p),
267                        Op::C(a, c, d) => {
268                            grow(a);
269                            grow(c);
270                            grow(d);
271                        }
272                        Op::Z => {}
273                    }
274                }
275            }
276            Raw::Bitmap { rect, .. } => {
277                grow((rect[0], rect[1]));
278                grow((rect[0] + rect[2], rect[1] + rect[3]));
279            }
280            Raw::Text(t) => grow(t.origin),
281        }
282    }
283    b
