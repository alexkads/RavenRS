1//! Device-independent bitmaps. BITMAPINFOHEADER DIBs at 1/4/8/16/24/32 bpp with BI_RGB or BI_BITFIELDS
2//! decode to RGBA8 rows, top-down. Compressed (RLE, JPEG, PNG) and V4/V5-only features are not decoded.
3
4use crate::metafile::{MAX_BITMAP_SIDE, MAX_PIXELS, bytes::Bytes};
5
6const BI_RGB: u32 = 0;
7const BI_BITFIELDS: u32 = 3;
8/// Most palette entries a DIB can carry (the colour table never exceeds 2^8).
9const MAX_PALETTE: usize = 256;
10const ROP_SRCAND: u32 = 0x0088_00C6;
11const ROP_SRCPAINT: u32 = 0x00EE_0086;
12const ROP_SRCINVERT: u32 = 0x0066_0046;
13
14/// A decoded picture, rows top-down.
15pub(crate) struct Rgba {
16    pub w: usize,
17    pub h: usize,
18    pub px: Vec<u8>,
19}
20
21/// How a decoded bitmap combines with the destination.
22#[derive(Clone, Copy)]
23pub(crate) enum Blend {
24    /// A raster operation (BITBLT, STRETCHDIB and friends).
25    Rop(u32),
26    /// EMR_ALPHABLEND: a constant alpha (0..=255), times the per-pixel alpha when `per_pixel` is set
27    /// (AC_SRC_ALPHA: the source colours are premultiplied by it).
28    Alpha { constant: u8, per_pixel: bool },
29}
30
31impl Blend {
32    /// Whether the decoder must keep the 32-bit alpha byte of each pixel.
33    pub fn per_pixel(self) -> bool {
34        matches!(self, Blend::Alpha { per_pixel: true, .. })
35    }
36}

38struct Header {
39    w: usize,
40    h: usize,
41    top_down: bool,
42    bpp: usize,
43    masks: [u32; 3],
44    pal_off: usize,
45    pal_n: usize,
46}
47
48impl Header {
49    fn read(info: Bytes) -> Option<Header> {
50        let size = info.u32(0)? as usize;
51        let w = info.i32(4)?;
52        let h = info.i32(8)?;
53        let bpp = info.u16(14)? as usize;
54        let comp = info.u32(16)?;
55        let clr_used = info.u32(32)? as usize;
56        if size < 40 || w <= 0 || h == 0 || !matches!(bpp, 1 | 4 | 8 | 16 | 24 | 32) || !matches!(comp, BI_RGB | BI_BITFIELDS) {
57            return None;
58        }
59        let masks = if comp == BI_BITFIELDS {
60            [info.u32(40)?, info.u32(44)?, info.u32(48)?]
61        } else if bpp == 16 {
62            [0x7C00, 0x03E0, 0x001F]
63        } else {
64            [0xFF_0000, 0xFF00, 0xFF]
65        };
66        let masks_len = if comp == BI_BITFIELDS && size == 40 { 12 } else { 0 };
67        let pal_off = size.checked_add(masks_len)?;
68        // Indexed formats take 2^bpp entries when clrUsed is 0. Deeper formats carry only the
69        // entries clrUsed names (WMF DIBs can have them), still capped.
70        let pal_n = match (bpp <= 8, clr_used) {
71            (true, 0) => 1usize << bpp,
72            (true, n) => n.min(1usize << bpp),
73            (false, n) => n.min(MAX_PALETTE),
74        };
75        Some(Header { w: w as usize, h: h.unsigned_abs() as usize, top_down: h < 0, bpp, masks, pal_off, pal_n })
76    }
77}

79/// Byte length of the info part (header, masks, palette) of a DIB, so the pixel bits follow it.
80pub(crate) fn info_len(dib: Bytes) -> Option<usize> {
81    let h = Header::read(dib)?;
82    h.pal_off.checked_add(h.pal_n.checked_mul(4)?)
83}

85/// Decodes the part of a DIB that the source rectangle `src` = `[x, y, w, h]` selects and returns it
86/// (top-down) with the part of the destination `dst` that it fills; see [`clip_rect`] for the
87/// coordinates. `info` is the info part (header plus masks and palette) and `bits` the pixel bits. The
88/// 32-bit alpha byte is kept only when `keep_alpha` is set (per-pixel alpha blends); otherwise pixels
89/// are opaque. Only the clipped rows and columns are decoded, so the work and the memory follow the
90/// crop, not the whole bitmap; the whole bitmap is still validated (size cap, palette, bits length).
91/// None as well when the clipped width or height is over [`MAX_BITMAP_SIDE`], which no renderer draws.
92pub(crate) fn decode_clip(info: Bytes, bits: Bytes, keep_alpha: bool, src: [i64; 4], dst: [f64; 4]) -> Option<(Rgba, [f64; 4])> {
93    let hd = Header::read(info)?;
94    let pixels = hd.w.checked_mul(hd.h)?;
95    if pixels == 0 || pixels > MAX_PIXELS {
96        return None;
97    }
98    let stride = hd.w.checked_mul(hd.bpp)?.checked_add(31)? / 32 * 4;
99    let data = bits.slice(0, stride.checked_mul(hd.h)?)?;
100    let pal = info.slice(hd.pal_off, hd.pal_n * 4)?;
101    let (clip, out) = clip_rect(hd.w, hd.h, !hd.top_down, src, dst)?;
102    // The renderers draw no bitmap with a side over the cap (`Item::take_pixels`); decoding one
103    // would only fill the parse's memory budget with pixels nobody draws.
104    let cap = MAX_BITMAP_SIDE as usize;
105    if clip.x1 - clip.x0 > cap || clip.r1 - clip.r0 > cap {
106        return None;
107    }
108    let (ow, oh) = (clip.x1 - clip.x0, clip.r1 - clip.r0);
109    let mut px = vec![0u8; ow.checked_mul(oh)?.checked_mul(4)?];
110    #[cfg(test)]
111    DECODED.with(|d| d.set(d.get() + ow * oh));
112    // The depth, palette and masks are the same for every pixel: pick the row decoder once.
113    let kind = RowKind::new(&hd, pal, keep_alpha);
114    for (y, line) in (clip.r0..clip.r1).zip(px.chunks_exact_mut(ow * 4)) {
115        let src_row = if hd.top_down { y } else { hd.h - 1 - y };
116        let row = data.get(src_row * stride..(src_row + 1) * stride)?;
117        kind.decode_row(row, clip.x0..clip.x1, line.as_chunks_mut::<4>().0)?;
118    }
119    Some((Rgba { w: ow, h: oh, px }, out))
120}

128/// One channel of a masked 16/32 bpp pixel, with the shift and scale worked out once per bitmap.
129/// Channels of up to 8 bits (every common 16 bpp layout) go through a table of their values, so a
130/// pixel costs no division.
131#[derive(Clone)]
132struct Channel {
133    mask: u32,
134    shift: u32,
135    max: u64,
136    table: [u8; 256],
137}
138
139impl Channel {
140    fn new(mask: u32) -> Channel {
141        let shift = if mask == 0 { 0 } else { mask.trailing_zeros() };
142        let max = u64::from(mask >> shift);
143        let mut table = [0u8; 256];
144        if mask != 0 && max <= 255 {
145            for (val, t) in table.iter_mut().enumerate().take(max as usize + 1) {
146                *t = ((val as u64 * 255 + max / 2) / max) as u8;
147            }
148        }
149        Channel { mask, shift, max, table }
150    }
151
152    /// The channel of `v` scaled to 8 bits; 0 when the mask is empty.
153    fn get(&self, v: u32) -> u8 {
154        if self.mask == 0 {
155            return 0;
156        }
157        let val = u64::from((v & self.mask) >> self.shift);
158        if self.max <= 255 {
159            return self.table[usize::from(val as u8)];
160        }
161        ((val * 255 + self.max / 2) / self.max) as u8
162    }
163}

165/// How the pixels of one DIB turn into RGBA, decided once from its header.
166enum RowKind {
167    /// 1, 4 or 8 bpp: `bpp` bits per index into a table of every possible index.
168    Indexed { bpp: usize, table: Box<[[u8; 4]; 256]> },
169    /// 24 bpp: BGR triples.
170    Bgr24,
171    /// 32 bpp with the default masks: BGRA quads, alpha kept or forced opaque.
172    Bgra32 { keep_alpha: bool },
173    /// 16 bpp, or 32 bpp with other masks.
174    Masked { bpp: usize, ch: Box<[Channel; 3]>, keep_alpha: bool },
175}
176
177impl RowKind {
178    fn new(hd: &Header, pal: &[u8], keep_alpha: bool) -> RowKind {
179        match hd.bpp {
180            1 | 4 | 8 => {
181                // An index past the palette reads as black, as in `palette`.
182                let mut table = Box::new([[0u8, 0, 0, 255]; 256]);
183                for (i, t) in table.iter_mut().enumerate().take(1 << hd.bpp) {
184                    let [r, g, b] = palette(pal, i);
185                    *t = [r, g, b, 255];
186                }
187                RowKind::Indexed { bpp: hd.bpp, table }
188            }
189            24 => RowKind::Bgr24,
190            32 if hd.masks == [0xFF_0000, 0xFF00, 0xFF] => RowKind::Bgra32 { keep_alpha },
191            _ => RowKind::Masked { bpp: hd.bpp, ch: Box::new(hd.masks.map(Channel::new)), keep_alpha },
192        }
193    }
194
195    /// Decodes the columns `cols` of one source row `row` (all of its bytes) into `out`. `None` when
196    /// the row is shorter than the columns need.
197    fn decode_row(&self, row: &[u8], cols: std::ops::Range<usize>, out: &mut [[u8; 4]]) -> Option<()> {
198        match self {
199            RowKind::Indexed { bpp: 8, table } => {
200                for (d, s) in out.iter_mut().zip(row.get(cols)?) {
201                    *d = table[usize::from(*s)];
202                }
203            }
204            RowKind::Indexed { bpp: 4, table } => {
205                for (d, x) in out.iter_mut().zip(cols) {
206                    let b = *row.get(x / 2)?;
207                    *d = table[usize::from(if x.is_multiple_of(2) { b >> 4 } else { b & 0x0F })];
208                }
209            }
210            RowKind::Indexed { table, .. } => {
211                for (d, x) in out.iter_mut().zip(cols) {
212                    let b = *row.get(x / 8)?;
213                    *d = table[usize::from((b >> (7 - x % 8)) & 1)];
214                }
215            }
216            RowKind::Bgr24 => {
217                let src = row.get(cols.start.checked_mul(3)?..cols.end.checked_mul(3)?)?;
218                for (d, s) in out.iter_mut().zip(src.as_chunks::<3>().0) {
219                    *d = [s[2], s[1], s[0], 255];
220                }
221            }
222            RowKind::Bgra32 { keep_alpha } => {
223                let src = row.get(cols.start.checked_mul(4)?..cols.end.checked_mul(4)?)?;
224                for (d, s) in out.iter_mut().zip(src.as_chunks::<4>().0) {
225                    *d = [s[2], s[1], s[0], if *keep_alpha { s[3] } else { 255 }];
226                }
227            }
228            RowKind::Masked { bpp: 16, ch, .. } => {
229                let [r, g, b] = &**ch;
230                let src = row.get(cols.start.checked_mul(2)?..cols.end.checked_mul(2)?)?;
231                for (d, s) in out.iter_mut().zip(src.as_chunks::<2>().0) {
232                    let v = u32::from(u16::from_le_bytes(*s));
233                    *d = [r.get(v), g.get(v), b.get(v), 255];
234                }
235            }
236            RowKind::Masked { ch, keep_alpha, .. } => {
237                let [r, g, b] = &**ch;
238                let src = row.get(cols.start.checked_mul(4)?..cols.end.checked_mul(4)?)?;
239                for (d, s) in out.iter_mut().zip(src.as_chunks::<4>().0) {
240                    let v = u32::from_le_bytes(*s);
241                    *d = [r.get(v), g.get(v), b.get(v), if *keep_alpha { (v >> 24) as u8 } else { 255 }];
242                }
243            }
244        }
245        Some(())
246    }
247}

249/// Palette entries are BGRX quads; an index past the table reads as black rather than failing the image.
250fn palette(pal: &[u8], i: usize) -> [u8; 3] {
251    pal.get(i * 4..i * 4 + 4).and_then(|c| <[u8; 4]>::try_from(c).ok()).map_or([0, 0, 0], |c| [c[2], c[1], c[0]])
252}

254/// The columns `x0..x1` and top-down rows `r0..r1` of a bitmap that a clipped source rectangle covers.
255struct Clip {
256    x0: usize,
257    x1: usize,
258    r0: usize,
259    r1: usize,
260}

262/// Clips a source rectangle `src` = `[x, y, w, h]` to a `w` x `h` bitmap and returns the covered pixels
263/// with the part of the destination `dst` that they fill. `src` is in DIB pixel coordinates (`y` counted
264/// from the DIB's origin, so bottom-up images count from the bottom); a non-positive source size means the
265/// whole bitmap. Clipping shrinks the destination by the same ratio and shifts it by the clipped-off part,
266/// so the rest of the image keeps its place. A negative destination extent is kept (the caller mirrors).
267/// `None` when nothing of the bitmap is covered.
268fn clip_rect(w: usize, h: usize, bottom_up: bool, src: [i64; 4], dst: [f64; 4]) -> Option<(Clip, [f64; 4])> {
269    let (iw, ih) = (w as i64, h as i64);
270    let [sx, sy, sw, sh] = if src[2] > 0 && src[3] > 0 { src } else { [0, 0, iw, ih] };
271    // The request's top row in top-down order.
272    let top = if bottom_up { ih.saturating_sub(sy).saturating_sub(sh) } else { sy };
273    let (x0, x1) = (sx.max(0), sx.saturating_add(sw).min(iw));
274    let (r0, r1) = (top.max(0), top.saturating_add(sh).min(ih));
275    if x1 <= x0 || r1 <= r0 {
276        return None;
277    }
278    let (kx, ky) = (dst[2] / sw as f64, dst[3] / sh as f64);
279    let out = [dst[0] + (x0 - sx) as f64 * kx, dst[1] + (r0 - top) as f64 * ky, (x1 - x0) as f64 * kx, (r1 - r0) as f64 * ky];
280    Some((Clip { x0: x0 as usize, x1: x1 as usize, r0: r0 as usize, r1: r1 as usize }, out))
281}

283/// Applies a blend to decoded pixels before they are drawn. The picture has no destination pixels, so
284/// raster operations are drawn as they look over white paper: SRCAND (destination AND source) leaves
285/// white source pixels out and draws the rest; SRCPAINT and SRCINVERT leave black ones out. Other
286/// operations draw the source as copied. Constant alpha scales every pixel's alpha, after a
287/// per-pixel (premultiplied) source is made straight.
288pub(crate) fn prepare(img: &mut Rgba, blend: Blend) {
289    match blend {
290        Blend::Rop(ROP_SRCAND) => clear_where(img, [255, 255, 255]),
291        Blend::Rop(ROP_SRCPAINT | ROP_SRCINVERT) => clear_where(img, [0, 0, 0]),
292        Blend::Rop(_) => {}
293        Blend::Alpha { constant, per_pixel } => {
294            for p in img.px.as_chunks_mut::<4>().0 {
295                if per_pixel {
296                    unpremultiply(p);
297                }
298                p[3] = (u16::from(p[3]) * u16::from(constant) / 255) as u8;
299            }
300        }
301    }
302}

304/// Makes the pixels of colour `rgb` transparent: where a raster operation keeps the destination.
305fn clear_where(img: &mut Rgba, rgb: [u8; 3]) {
306    for p in img.px.as_chunks_mut::<4>().0 {
307        if p[..3] == rgb {
308            p[3] = 0;
309        }
310    }
311}

313/// AC_SRC_ALPHA sources are premultiplied; the model holds straight RGBA, so divide the colour by the
314/// alpha (rounded, capped at 255). A transparent pixel keeps its colour bytes.
315fn unpremultiply(p: &mut [u8; 4]) {
316    let a = u32::from(p[3]);
317    if a == 0 || a == 255 {
318        return;
319    }
320    for c in &mut p[..3] {
321        *c = ((u32::from(*c) * 255 + a / 2) / a).min(255) as u8;
322    }
323}

325/// Mirrors a decoded image left-right and/or top-bottom. Rows come from the buffer as chunks, so a
326/// short buffer flips only the rows it has.
327pub(crate) fn flip(img: &mut Rgba, horizontal: bool, vertical: bool) {
328    let (w, h) = (img.w, img.h);
329    if w == 0 || h == 0 {
330        return;
331    }
332    let row = w * 4;
333    if horizontal {
334        for line in img.px.chunks_exact_mut(row) {
335            for x in 0..w / 2 {
336                for c in 0..4 {
337                    line.swap(x * 4 + c, (w - 1 - x) * 4 + c);
338                }
339            }
340        }
341    }
342    if vertical {
343        // Rows the buffer actually holds: a short buffer flips only those.
344        let n = h.min(img.px.len() / row);
345        for y in 0..n / 2 {
346            swap_rows(&mut img.px, y * row, (n - 1 - y) * row, row);
347        }
348    }
349}

351/// Swaps the `len` bytes at `a` with those at `b` (`a < b`); does nothing when either is out of range.
352fn swap_rows(px: &mut [u8], a: usize, b: usize, len: usize) {
353    let Some((head, tail)) = px.split_at_mut_checked(b) else { return };
354    if let (Some(top), Some(bottom)) = (head.get_mut(a..a.saturating_add(len)), tail.get_mut(..len)) {
355        top.swap_with_slice(bottom);
356    }
357}

364    fn decode_whole(info: Bytes, bits: Bytes, keep_alpha: bool) -> Option<Rgba> {
365        let hd = Header::read(info)?;
366        let pixels = hd.w.checked_mul(hd.h)?;
367        if pixels == 0 || pixels > MAX_PIXELS {
368            return None;
369        }
370        let stride = hd.w.checked_mul(hd.bpp)?.checked_add(31)? / 32 * 4;
371        let data = bits.slice(0, stride.checked_mul(hd.h)?)?;
372        let pal = info.slice(hd.pal_off, hd.pal_n * 4)?;

407    /// A BITMAPINFOHEADER DIB with a palette for indexed depths; returns (info, bits).
408    fn make_dib(w: usize, h: i32, bpp: usize, seed: &mut u64) -> (Vec<u8>, Vec<u8>) {
409        let mut rnd = || {
410            *seed ^= *seed << 13;
411            *seed ^= *seed >> 7;
412            *seed ^= *seed << 17;
413            *seed
414        };
415        let mut info = Vec::new();
416        info.extend(40u32.to_le_bytes());
417        info.extend((w as i32).to_le_bytes());
418        info.extend(h.to_le_bytes());
419        info.extend(1u16.to_le_bytes());
420        info.extend((bpp as u16).to_le_bytes());
421        info.extend([0u8; 24]);
422        if bpp <= 8 {
423            for _ in 0..(1usize << bpp) * 4 {
424                info.push(rnd() as u8);
425            }
426        }
427        let stride = (w * bpp).div_ceil(32) * 4;
428        let bits = (0..stride * h.unsigned_abs() as usize).map(|_| rnd() as u8).collect();
429        (info, bits)
430    }

432    /// The per-pixel decoder `decode_clip` used before the depth was chosen once per bitmap. Kept as
433    /// the reference the fast decoder must match byte for byte.
434    fn pixel(data: &[u8], row: usize, x: usize, hd: &Header, pal: &[u8], keep_alpha: bool) -> Option<[u8; 4]> {
435        DECODED.with(|d| d.set(d.get() + 1));
436        let [m_r, m_g, m_b] = hd.masks;
437        let [r, g, b] = match hd.bpp {
438            1 => {
439                let b = *data.get(row + x / 8)?;
440                palette(pal, usize::from((b >> (7 - x % 8)) & 1))
441            }
442            4 => {
443                let b = *data.get(row + x / 2)?;
444                palette(pal, usize::from(if x.is_multiple_of(2) { b >> 4 } else { b & 0x0F }))
445            }
446            8 => palette(pal, usize::from(*data.get(row + x)?)),
447            24 => {
448                let [b, g, r] = data.get(row + x * 3..row + x * 3 + 3)?.try_into().ok()?;
449                [r, g, b]
450            }
451            16 => {
452                let o = row + x * 2;
453                let v = u32::from(u16::from_le_bytes(data.get(o..o + 2)?.try_into().ok()?));
454                [field(v, m_r), field(v, m_g), field(v, m_b)]
455            }
456            _ => {
457                let o = row + x * 4;
458                let v = u32::from_le_bytes(data.get(o..o + 4)?.try_into().ok()?);
459                let a = if keep_alpha { (v >> 24) as u8 } else { 255 };
460                return Some([field(v, m_r), field(v, m_g), field(v, m_b), a]);
461            }
462        };
463        Some([r, g, b, 255])
464    }

466    /// Extracts one channel from a masked pixel value and scales it to 8 bits.
467    fn field(v: u32, mask: u32) -> u8 {
468        if mask == 0 {
469            return 0;
470        }
471        let shift = mask.trailing_zeros();
472        let max = u64::from(mask >> shift);
473        let val = u64::from((v & mask) >> shift);
474        ((val * 255 + max / 2) / max) as u8
475    }

477    /// `decode_clip` as it was: the same clip, then `pixel` for every output pixel.
478    fn decode_clip_reference(info: Bytes, bits: Bytes, keep_alpha: bool, src: [i64; 4], dst: [f64; 4]) -> Option<(Rgba, [f64; 4])> {
479        let hd = Header::read(info)?;
480        let pixels = hd.w.checked_mul(hd.h)?;
481        if pixels == 0 || pixels > MAX_PIXELS {
482            return None;
483        }
484        let stride = hd.w.checked_mul(hd.bpp)?.checked_add(31)? / 32 * 4;
485        let data = bits.slice(0, stride.checked_mul(hd.h)?)?;
486        let pal = info.slice(hd.pal_off, hd.pal_n * 4)?;
487        let (clip, out) = clip_rect(hd.w, hd.h, !hd.top_down, src, dst)?;
488        let (ow, oh) = (clip.x1 - clip.x0, clip.r1 - clip.r0);
489        let mut px = vec![0u8; ow.checked_mul(oh)?.checked_mul(4)?];
490        for (y, line) in (clip.r0..clip.r1).zip(px.chunks_exact_mut(ow * 4)) {
491            let src_row = if hd.top_down { y } else { hd.h - 1 - y };
492            let row = src_row * stride;
493            for (x, dst_px) in (clip.x0..clip.x1).zip(line.as_chunks_mut::<4>().0) {
494                *dst_px = pixel(data, row, x, &hd, pal, keep_alpha)?;
495            }
496        }
497        Some((Rgba { w: ow, h: oh, px }, out))
498    }

500    /// A DIB with chosen header fields: `masks` makes it BI_BITFIELDS (inline after a 40 byte
501    /// header, or inside a larger header), `clr_used` limits the palette, and the palette, when the
502    /// depth is indexed, has `pal_entries` entries (fewer than the depth names reads as black).
503    fn make_dib_with(
504        w: usize,
505        h: i32,
506        bpp: usize,
507        masks: Option<([u32; 3], usize)>,
508        clr_used: u32,
509        pal_entries: usize,
510        seed: &mut u64,
511    ) -> (Vec<u8>, Vec<u8>) {
512        let mut rnd = || {
513            *seed ^= *seed << 13;
514            *seed ^= *seed >> 7;
515            *seed ^= *seed << 17;
516            *seed
517        };
518        let size = masks.map_or(40, |(_, s)| s);
519        let mut info = Vec::new();
520        info.extend((size as u32).to_le_bytes());
521        info.extend((w as i32).to_le_bytes());
522        info.extend(h.to_le_bytes());
523        info.extend(1u16.to_le_bytes());
524        info.extend((bpp as u16).to_le_bytes());
525        info.extend(if masks.is_some() { 3u32 } else { 0 }.to_le_bytes());
526        info.extend([0u8; 12]);
527        info.extend(clr_used.to_le_bytes());
528        info.extend([0u8; 4]);
529        info.resize(size.max(40), 0);
530        if let Some((m, _)) = masks {
531            for (i, v) in m.iter().enumerate() {
532                let at = 40 + i * 4;
533                if size == 40 {
534                    info.extend(v.to_le_bytes());
535                } else if let Some(dst) = info.get_mut(at..at + 4) {
536                    dst.copy_from_slice(&v.to_le_bytes());
537                }
538            }
539        }
540        for _ in 0..pal_entries * 4 {
541            info.push(rnd() as u8);
542        }
543        let stride = (w * bpp).div_ceil(32) * 4;
544        let bits = (0..stride * h.unsigned_abs() as usize).map(|_| rnd() as u8).collect();
545        (info, bits)
546    }

548    fn assert_same(info: &[u8], bits: &[u8], what: &str) {
549        for keep_alpha in [false, true] {
550            for src in [[0, 0, 0, 0], [3, 1, 20, 3], [-4, -4, 100, 100], [36, 4, 5, 5], [1, 0, 1, 5], [8, 2, 17, 2], [0, 3, 37, 1]] {
551                let dst = [1.0, 2.0, 30.0, 40.0];
552                let old = decode_clip_reference(Bytes(info), Bytes(bits), keep_alpha, src, dst);
553                let new = decode_clip(Bytes(info), Bytes(bits), keep_alpha, src, dst);
554                match (old, new) {
555                    (None, None) => {}
556                    (Some((a, ad)), Some((b, bd))) => {
557                        assert_eq!((a.w, a.h, &a.px), (b.w, b.h, &b.px), "{what} alpha={keep_alpha} {src:?}");
558                        assert_eq!(ad.map(f64::to_bits), bd.map(f64::to_bits), "{what} alpha={keep_alpha} {src:?}");
559                    }
560                    _ => panic!("old and new disagree on None: {what} alpha={keep_alpha} {src:?}"),
561                }
562            }
563        }
564    }

566    #[test]
567    fn decode_clip_matches_the_per_pixel_reference_at_every_depth_palette_and_mask() {
568        let mut seed = 0x9E37_79B9_7F4A_7C15u64;
569        // Widths that leave padding bits in the row, so a row's last byte is only partly used.
570        for w in [1, 9, 37] {
571            for h in [6, -6] {
572                for bpp in [1usize, 4, 8] {
573                    let n = 1usize << bpp;
574                    // Full palette, clrUsed shorter than the depth (the rest reads black), a
575                    // palette cut short of what clrUsed names, and none at all.
576                    for (clr_used, entries) in [(0, n), (3, 3), (n as u32 + 50, n), (0, 1), (0, 0)] {
577                        let (info, bits) = make_dib_with(w, h, bpp, None, clr_used, entries, &mut seed);
578                        assert_same(&info, &bits, &format!("{bpp} bpp w={w} h={h} clrUsed={clr_used} palette={entries}"));
579                    }
580                }
581                let (info, bits) = make_dib_with(w, h, 24, None, 0, 0, &mut seed);
582                assert_same(&info, &bits, &format!("24 bpp w={w} h={h}"));
583                // A colour table on a deep DIB is carried but unused.
584                let (info, bits) = make_dib_with(w, h, 24, None, 5, 5, &mut seed);
585                assert_same(&info, &bits, &format!("24 bpp with palette w={w} h={h}"));
586                // Masks on depths that ignore them still move the palette and the bits.
587                for bpp in [1usize, 8, 24] {
588                    let n = if bpp == 24 { 0 } else { 1 << bpp };
589                    let (info, bits) = make_dib_with(w, h, bpp, Some(([0xF800, 0x07E0, 0x001F], 40)), 0, n, &mut seed);
590                    assert_same(&info, &bits, &format!("{bpp} bpp with masks w={w} h={h}"));
591                }
592                let masks16: [Option<[u32; 3]>; 5] = [
593                    None,
594                    Some([0xF800, 0x07E0, 0x001F]),
595                    Some([0x7C00, 0x03E0, 0x001F]),
596                    Some([0x0F00, 0x00F0, 0x000F]),
597                    Some([0xF800, 0x0000, 0x001F]),
598                ];
599                let masks32: [Option<[u32; 3]>; 6] = [
600                    None,
601                    Some([0xFF_0000, 0xFF00, 0xFF]),
602                    Some([0xFF, 0xFF00, 0xFF_0000]),
603                    Some([0x3FF0_0000, 0x000F_FC00, 0x0000_03FF]),
604                    Some([0xFFFF_FFFF, 0x0000_FF00, 0]),
605                    Some([0x00FF_FF00, 0x00FF_FF00, 0x0000_00FF]),
606                ];
607                for (bpp, sets) in [(16usize, masks16.as_slice()), (32, masks32.as_slice())] {
608                    for m in sets {
609                        // 40: the masks follow the header; 56 and 108: they are inside it.
610                        for hsize in [40usize, 56, 108] {
611                            let inline = m.map(|m| (m, hsize));
612                            let (info, bits) = make_dib_with(w, h, bpp, inline, 0, 0, &mut seed);
613                            assert_same(&info, &bits, &format!("{bpp} bpp masks={m:x?} header={hsize} w={w} h={h}"));
614                        }
615                    }
616                }
617            }
618        }
619    }

622    /// `cargo test -p wordcraft-metafile --release -- --ignored --nocapture dib_decode_speed`.
623    #[test]
624    #[ignore = "timing only"]
625    fn dib_decode_speed() {
626        let mut seed = 6u64;
627        for bpp in [1usize, 4, 8, 16, 24, 32] {
628            let (info, bits) = make_dib(8000, -5000, bpp, &mut seed);
629            for round in 0..4 {
630                let t = std::time::Instant::now();
631                let a = decode_clip_reference(Bytes(&info), Bytes(&bits), false, [0, 0, 8000, 5000], [0.0, 0.0, 1.0, 1.0]).map(|x| x.0.px);
632                let slow = t.elapsed().as_secs_f64() * 1e3;
633                let t = std::time::Instant::now();
634                let b = decode_clip(Bytes(&info), Bytes(&bits), false, [0, 0, 8000, 5000], [0.0, 0.0, 1.0, 1.0]).map(|x| x.0.px);
635                let fast = t.elapsed().as_secs_f64() * 1e3;
636                assert!(a == b);
637                eprintln!("{bpp:2} bpp round {round}: reference {slow:4.0} ms, decode_clip {fast:4.0} ms");
638            }
639        }
640    }
