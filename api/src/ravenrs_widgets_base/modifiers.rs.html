7/// A view wrapped in a box with layout and background modifiers; see [`ViewExt`].
8#[derive(Debug)]
9pub struct Styled<V> {
10    content: V,
11    style: LayoutStyle,
12    background: Option<Fill>,
13    radius: f32,
14    border: Option<(f32, Fill)>,
15    shadow: Option<Shadow>,
16}
17
18/// SwiftUI-style modifiers available on every view.
19///
20/// ```
21/// use ravenrs_style::Semantic;
22/// use ravenrs_widgets::{Text, ViewExt};
23///
24/// let chip = Text::new("New").padding((4.0, 10.0)).background(Semantic::Control).corner_radius(999.0);
25/// ```
26pub trait ViewExt: View + Sized {
27    /// Adds padding inside the box, in logical pixels.
28    fn padding(self, padding: impl Into<Insets>) -> Styled<Self> {
29        Styled::new(self).padding(padding)
30    }
31    /// Paints a background behind the content.
32    fn background(self, fill: impl Into<Fill>) -> Styled<Self> {
33        Styled::new(self).background(fill)
34    }
35    /// Fixes the width and/or height in logical pixels; `None` keeps the content size.
36    fn frame(self, width: Option<f32>, height: Option<f32>) -> Styled<Self> {
37        Styled::new(self).frame(width, height)
38    }
39    /// Fixes the width in logical pixels.
40    fn width(self, width: f32) -> Styled<Self> {
41        Styled::new(self).width(width)
42    }
43    /// Fixes the height in logical pixels.
44    fn height(self, height: f32) -> Styled<Self> {
45        Styled::new(self).height(height)
46    }
47    /// Takes a share of the free space in the parent stack.
48    fn grow(self, grow: f32) -> Styled<Self> {
49        Styled::new(self).grow(grow)
50    }
51    /// Stretches across the parent stack's cross axis.
52    fn stretch(self) -> Styled<Self> {
53        Styled::new(self).stretch()
54    }
55    /// Centers the content inside the box.
56    fn centered(self) -> Styled<Self> {
57        Styled::new(self).centered()
58    }
59    /// Shows `text` above the view while the pointer is over it; see [`Tooltip`].
60    fn tooltip(self, text: impl Into<String>) -> Tooltip<Self> {
61        Tooltip::new(self, text)
62    }
63    /// Draws the view and everything in it at this opacity, from `0.0` to `1.0`, easing to it
64    /// when it changes. Layout and input stay as they are; see [`Faded`].
65    fn opacity(self, opacity: f32) -> Faded<Self> {
66        Faded { content: self, opacity: opacity.clamp(0.0, 1.0) }
67    }
68    /// Tags the view with an id that tests and automation find it by, such as
69    /// `Button::new("Save").test_id("save")`. The id stays the same when the screen around the
70    /// view is reordered or wrapped, is not read by screen readers, and becomes `data-testid`
71    /// on the web; see [`BuildCx::test_id`] and [`TestId`].
72    /// With `e2e-source`, also keeps this call site's file, line and column for source queries.
73    #[cfg_attr(feature = "e2e-source", track_caller)]
74    fn test_id(self, id: impl Into<String>) -> TestId<Self> {
75        TestId {
76            content: self,
77            id: id.into(),
78            #[cfg(feature = "e2e-source")]
79            source: std::panic::Location::caller(),
80        }
81    }
82}

84/// A view tagged with an id for tests and automation; see [`ViewExt::test_id`].
85#[derive(Debug)]
86pub struct TestId<V> {
87    content: V,
88    id: String,
89    #[cfg(feature = "e2e-source")]
90    source: &'static std::panic::Location<'static>,
91}
92
93impl<V: View> View for TestId<V> {
94    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
95        let node = self.content.build(cx);
96        cx.test_id(node, &self.id);
97        #[cfg(feature = "e2e-source")]
98        cx.test_source(node, self.source);
99        node
100    }
101}

103/// A view drawn at an opacity, such as the last result a playground keeps while its code has
104/// problems; see [`ViewExt::opacity`]. Each shape fades on its own, so overlapping shapes show
105/// through each other while it is partly faded.
106#[derive(Debug)]
107pub struct Faded<V> {
108    content: V,
109    opacity: f32,
110}
111
112impl<V: View> View for Faded<V> {
113    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
114        let node = self.content.build(cx);
115        let effect = ravenrs_core::Effect { opacity: self.opacity, ..ravenrs_core::Effect::IDENTITY };
116        let transition = cx.theme().motion.fast;
117        cx.motion(node, crate::Motion::new().effect(effect, transition));
118        node
119    }
120}

124#[cfg_attr(feature = "live", ravenrs_live::expose)]
125impl<V: View> Styled<V> {
126    fn new(content: V) -> Self {
127        Self {
128            content,
129            style: LayoutStyle { align: Align::Stretch, ..LayoutStyle::stack(Axis::Vertical) },
130            background: None,
131            radius: 0.0,
132            border: None,
133            shadow: None,
134        }
135    }
136
137    /// Sets the padding inside the box, in logical pixels.
138    pub fn padding(mut self, padding: impl Into<Insets>) -> Self {
139        self.style.padding = padding.into();
140        self
141    }
142
143    /// Paints a background behind the content.
144    pub fn background(mut self, fill: impl Into<Fill>) -> Self {
145        self.background = Some(fill.into());
146        self
147    }
148
149    /// Rounds the box corners; radius in logical pixels.
150    pub fn corner_radius(mut self, radius: f32) -> Self {
151        self.radius = radius;
152        self
153    }
154
155    /// Draws a border of `width` logical pixels.
156    pub fn border(mut self, width: f32, fill: impl Into<Fill>) -> Self {
157        self.border = Some((width, fill.into()));
158        self
159    }
160
161    /// Casts a drop shadow behind the box.
162    pub fn shadow(mut self, shadow: Shadow) -> Self {
163        self.shadow = Some(shadow);
164        self
165    }
166
167    /// Fixes the width and/or height in logical pixels; `None` keeps the content size.
168    pub fn frame(mut self, width: Option<f32>, height: Option<f32>) -> Self {
169        self.style.width = width;
170        self.style.height = height;
171        self
172    }
173
174    /// Fixes the width in logical pixels.
175    pub fn width(mut self, width: f32) -> Self {
176        self.style.width = Some(width);
177        self
178    }
179
180    /// Fixes the height in logical pixels.
181    pub fn height(mut self, height: f32) -> Self {
182        self.style.height = Some(height);
183        self
184    }
185
186    /// Caps the width in logical pixels.
187    pub fn max_width(mut self, width: f32) -> Self {
188        self.style.max_width = Some(width);
189        self
190    }
191
192    /// Takes a share of the free space in the parent stack.
193    pub fn grow(mut self, grow: f32) -> Self {
194        self.style.grow = grow;
195        self.style.min_width.get_or_insert(0.0);
196        self.style.min_height.get_or_insert(0.0);
197        self
198    }
199
200    /// Stretches across the parent stack's cross axis.
201    pub fn stretch(mut self) -> Self {
202        self.style.align_self = Some(Align::Stretch);
203        self
204    }
205
206    /// Centers the content inside the box.
207    pub fn centered(mut self) -> Self {
208        self.style.align = Align::Center;
209        self.style.justify = ravenrs_layout::Justify::Center;
210        self
211    }
212}

214impl<V: View> View for Styled<V> {
215    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
216        let child = cx.child(&self.content);
217        let theme = *cx.theme();
218        let paint = if self.background.is_some() || self.border.is_some() || self.shadow.is_some() {
219            let fill = self.background.map_or(ravenrs_core::Color::TRANSPARENT, |f| f.resolve(&theme));
220            Paint::Quad(QuadPaint {
221                fill: fill.into(),
222                corner_radius: self.radius,
223                border: self.border.map(|(width, f)| Border { width, color: f.resolve(&theme) }),
224                shadow: self.shadow,
225            })
226        } else {
227            Paint::None
228        };
229        cx.container(self.style, &[child], paint)
230    }
