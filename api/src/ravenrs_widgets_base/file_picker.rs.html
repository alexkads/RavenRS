1//! The file chooser service and the files it brings, with no view.
2
3use std::fmt;
4use std::rc::Rc;
5
6/// A file brought into the app, whole, in memory.
7#[derive(Clone, PartialEq, Eq)]
8#[cfg_attr(feature = "live", ravenrs_live::expose(part = "fields"))]
9pub struct PickedFile {
10    /// Its name, with no folder. Never trust it as a path.
11    pub name: String,
12    /// Its size on disk. A file bigger than the request allowed comes with no bytes.
13    pub size: u64,
14    /// Its content.
15    pub bytes: Rc<[u8]>,
16}
17
18#[cfg_attr(feature = "live", ravenrs_live::expose)]
19impl PickedFile {
20    /// A file with this name and content.
21    pub fn new(name: impl Into<String>, bytes: impl Into<Rc<[u8]>>) -> Self {
22        let bytes = bytes.into();
23        Self { name: name.into(), size: bytes.len() as u64, bytes }
24    }
25
26    /// The part of the name after the last dot, in lower case.
27    pub fn extension(&self) -> Option<String> {
28        self.name.rsplit_once('.').map(|(_, extension)| extension.to_lowercase())
29    }
30}
31
32impl fmt::Debug for PickedFile {
33    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
34        f.debug_struct("PickedFile").field("name", &self.name).field("size", &self.size).finish()
35    }
36}

38/// What to ask of a [`FilePicker`].
39#[derive(Clone, Debug, Default, PartialEq, Eq)]
40#[non_exhaustive]
41#[cfg_attr(feature = "live", ravenrs_live::expose(part = "fields"))]
42pub struct FileRequest {
43    /// The extensions offered, such as `pdf`; empty offers every file.
44    pub extensions: Vec<String>,
45    /// Whether more than one can be chosen.
46    pub multiple: bool,
47    /// Files above this size are not read, and come with no bytes.
48    pub most_bytes: Option<u64>,
49}
50
51/// Asks the person for files. Platform layers install one with [`crate::Ui::set_file_picker`].
52pub trait FilePickerProvider {
53    /// Shows the system's chooser, and calls `done` with what was chosen, now or later
54    /// on the UI thread. Canceling gives an empty list.
55    fn pick(&self, request: &FileRequest, done: Box<dyn FnOnce(Vec<PickedFile>)>);
56
57    /// Hands a file over to the person: the system's dialog on a desktop, a download in a
58    /// browser. Nothing happens where there is no system to ask.
59    fn save(&self, _name: &str, _bytes: &[u8]) {}
60}
61
62/// A picker that chooses nothing, for where there is no system to ask (tests, headless use).
63#[derive(Clone, Copy, Debug, Default)]
64#[cfg_attr(feature = "live", ravenrs_live::expose(part = "fields"))]
65pub struct NoFilePicker;
66
67impl FilePickerProvider for NoFilePicker {
68    fn pick(&self, _request: &FileRequest, done: Box<dyn FnOnce(Vec<PickedFile>)>) {
69        done(Vec::new());
70    }
71}

73/// A shared handle to the file chooser, from [`crate::BuildCx::file_picker`].
74#[derive(Clone)]
75#[cfg_attr(feature = "live", ravenrs_live::expose(part = "fields"))]
76pub struct FilePicker(Rc<dyn FilePickerProvider>);
77
78#[cfg_attr(feature = "live", ravenrs_live::expose)]
79impl FilePicker {
80    /// Wraps a provider.
81    pub fn new(provider: impl FilePickerProvider + 'static) -> Self {
82        Self(Rc::new(provider))
83    }
84
85    /// Asks for files; `done` runs with what was chosen.
86    pub fn pick(&self, request: &FileRequest, done: impl FnOnce(Vec<PickedFile>) + 'static) {
87        self.0.pick(request, Box::new(done));
88    }
89
90    /// Hands a file over to the person.
91    pub fn save(&self, name: &str, bytes: &[u8]) {
92        self.0.save(name, bytes);
93    }
94}
95
96impl Default for FilePicker {
97    fn default() -> Self {
98        Self::new(NoFilePicker)
99    }
100}
101
102impl fmt::Debug for FilePicker {
103    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
104        f.write_str("FilePicker")
105    }
