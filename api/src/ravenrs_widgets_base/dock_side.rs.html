1//! [`DockSide`]: the places of a dock area, with their words in [`Labels`].
2
3use serde::{Deserialize, Serialize};
4
5use crate::labels::Labels;
6
7/// One of the five places a `DockArea` has.
8#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash, Serialize, Deserialize)]
9#[serde(rename_all = "lowercase")]
10#[non_exhaustive]
11#[cfg_attr(feature = "live", ravenrs_live::expose(part = "variants"))]
12pub enum DockSide {
13    /// The column on the left.
14    Left,
15    /// The column on the right.
16    Right,
17    /// The strip under the center, between the columns.
18    Bottom,
19    /// The strip above the center, between the columns.
20    Top,
21    /// What takes the room left.
22    Center,
23}
24
25#[cfg_attr(feature = "live", ravenrs_live::expose)]
26impl DockSide {
27    /// The four docks that collapse and have a size of their own.
28    #[doc(hidden)]
29    pub const DOCKS: [Self; 4] = [Self::Left, Self::Right, Self::Top, Self::Bottom];
30    /// All five, in the order they are searched.
31    #[doc(hidden)]
32    pub const ALL: [Self; 5] = [Self::Center, Self::Left, Self::Right, Self::Top, Self::Bottom];
33    /// All five, in the order a menu lists them.
34    #[doc(hidden)]
35    pub const ALL_IN_MENU: [Self; 5] = [Self::Top, Self::Left, Self::Right, Self::Bottom, Self::Center];
36}
37
38#[cfg_attr(feature = "live", ravenrs_live::expose)]
39impl Labels {
40    /// A place of a `DockArea`.
41    pub fn dock_place(&self, side: DockSide) -> &str {
42        match side {
43            DockSide::Left => &self.dock_left,
44            DockSide::Right => &self.dock_right,
45            DockSide::Bottom => &self.dock_bottom,
46            DockSide::Top => &self.dock_top,
47            _ => &self.dock_center,
48        }
49    }
50
51    /// A place in the middle of a sentence.
52    fn dock_place_lower(&self, side: DockSide) -> String {
53        self.dock_place(side).to_lowercase()
54    }
55
56    /// The name of the strip that collapses a dock.
57    pub fn dock_hide_name(&self, side: DockSide) -> String {
58        self.dock_hide.replace("{place}", &self.dock_place_lower(side))
59    }
60
61    /// The name of the strip that brings a dock back.
62    pub fn dock_show_name(&self, side: DockSide) -> String {
63        self.dock_show.replace("{place}", &self.dock_place_lower(side))
64    }
65
66    /// The name of the bar that resizes a dock.
67    pub fn dock_resize_name(&self, side: DockSide) -> String {
68        self.dock_resize.replace("{place}", &self.dock_place_lower(side))
69    }
70
71    /// The name of the button that hides a panel at the edge of its dock.
72    pub fn dock_auto_hide_name(&self, title: &str) -> String {
73        self.dock_auto_hide.replace("{title}", title)
74    }
75
76    /// The name of the button that puts a hidden panel back in its dock.
77    pub fn dock_pin_name(&self, title: &str) -> String {
78        self.dock_pin.replace("{title}", title)
79    }
80
81    /// A hidden panel on its rail, and what is said as it hides.
82    pub fn dock_hidden_name(&self, title: &str, side: DockSide) -> String {
83        self.dock_hidden.replace("{title}", title).replace("{place}", &self.dock_place_lower(side))
84    }
85
86    /// What is said as a hidden panel goes back in its dock.
87    pub fn dock_pinned_name(&self, title: &str) -> String {
88        self.dock_pinned.replace("{title}", title)
89    }
90
91    /// What is said as a panel floats, and the name of its box.
92    pub fn dock_floating_name(&self, title: &str) -> String {
93        self.dock_floating.replace("{title}", title)
94    }
95
96    /// The name of the button that puts a floating panel back.
97    pub fn dock_dock_back_name(&self, title: &str) -> String {
98        self.dock_dock_back.replace("{title}", title)
99    }
100
101    /// The name of the header that moves a floating panel.
102    pub fn dock_move_floating_name(&self, title: &str) -> String {
103        self.dock_move_floating.replace("{title}", title)
104    }
105
106    /// The name of an edge or a corner that resizes a floating panel.
107    pub fn dock_resize_floating_name(&self, title: &str) -> String {
108        self.dock_resize_floating.replace("{title}", title)
109    }
110
111    /// What is announced after a panel went to a place.
112    pub fn dock_moved_to(&self, side: DockSide) -> String {
113        self.dock_moved.replace("{place}", &self.dock_place_lower(side))
114    }
115}
