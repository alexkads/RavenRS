1//! [`CablePlug`]: a cable that hangs, swings and settles between any two places of
2//! the window: two widgets, a widget and the pointer, a widget and a point.
3//!
4//! ```ignore
5//! let button = cx.child(&Button::new("Exportar"));
6//! let tip = cx.child(&Text::new("Exporte em PDF aqui"));
7//! CablePlug::between(button, tip).build(cx);
8//! ```
9//!
10//! The runtime places the ends on the widgets' frames after every layout, runs the
11//! physics in the window's [`CableWorld`] (on the GPU when the platform gives one,
12//! otherwise on the CPU), paints only while some cable still moves and shows every
13//! cable at rest under reduced motion. Components that draw on their own canvas put
14//! [`Cable`]s in their scene instead and read the shapes back with [`Cables`].
15

25/// Where on a widget an end of a cable goes, and the way the cable leaves it.
26#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
27#[cfg_attr(feature = "live", ravenrs_live::expose(part = "variants"))]
28pub enum CableSide {
29    /// The middle of the left edge, leaving to the left.
30    Left,
31    /// The middle of the right edge, leaving to the right.
32    Right,
33    /// The middle of the top edge, leaving upward.
34    Top,
35    /// The middle of the bottom edge, leaving downward.
36    Bottom,
37    /// The center, leaving any way: a jack seen from the front.
38    #[default]
39    Center,
40}
41
42#[cfg_attr(feature = "live", ravenrs_live::expose)]
43impl CableSide {
44    pub(crate) fn end(self, frame: Rect) -> CableEnd {
45        let c = frame.center();
46        match self {
47            Self::Left => CableEnd::at(Point::new(frame.x(), c.y)).toward(-1.0, 0.0),
48            Self::Right => CableEnd::at(Point::new(frame.max_x(), c.y)).toward(1.0, 0.0),
49            Self::Top => CableEnd::at(Point::new(c.x, frame.y())).toward(0.0, -1.0),
50            Self::Bottom => CableEnd::at(Point::new(c.x, frame.max_y())).toward(0.0, 1.0),
51            Self::Center => CableEnd::at(c),
52        }
53    }
54}

56/// What an end of a [`CablePlug`] is held by.
57#[derive(Clone, Copy, Debug, PartialEq)]
58#[cfg_attr(feature = "live", ravenrs_live::expose(part = "variants"))]
59pub enum CableAnchor {
60    /// A widget built in this build, on one of its sides.
61    Node(NodeId, CableSide),
62    /// A point of the window.
63    Point(CableEnd),
64    /// The pointer, wherever it goes: a cable being carried.
65    Pointer,
66}

68/// A cable between two places of the window. Build it with `CablePlug::build` in the
69/// build of the view that owns it, after building what it ties.
70#[derive(Clone)]
71#[cfg_attr(feature = "live", ravenrs_live::expose(part = "fields"))]
72pub struct CablePlug {
73    pub(crate) key: Option<u64>,
74    pub(crate) from: CableAnchor,
75    pub(crate) to: CableAnchor,
76    pub(crate) via: Vec<Point>,
77    pub(crate) physics: CablePhysics,
78    pub(crate) look: CableLook,
79    pub(crate) release: CableRelease,
80    pub(crate) grow: bool,
81    pub(crate) on_click: Option<Rc<dyn Fn()>>,
82    pub(crate) on_drop: Option<Rc<dyn Fn(Point)>>,
83}

91#[cfg_attr(feature = "live", ravenrs_live::expose)]
92impl CablePlug {
93    fn new(from: CableAnchor, to: CableAnchor) -> Self {
94        Self {
95            key: None,
96            from,
97            to,
98            via: Vec::new(),
99            physics: CablePhysics::default(),
100            look: CableLook::default(),
101            release: CableRelease::Fall,
102            grow: true,
103            on_click: None,
104            on_drop: None,
105        }
106    }
107
108    /// A cable from the center of `a` to the center of `b`.
109    pub fn between(a: NodeId, b: NodeId) -> Self {
110        Self::new(CableAnchor::Node(a, CableSide::Center), CableAnchor::Node(b, CableSide::Center))
111    }
112
113    /// A cable out of `a` whose other end is in the hand: it follows the pointer
114    /// until it is tied with [`Self::to`] or the pointer lets go ([`Self::on_drop`]).
115    pub fn from(a: NodeId) -> Self {
116        Self::new(CableAnchor::Node(a, CableSide::Center), CableAnchor::Pointer)
117    }
118
119    /// A cable between any two anchors.
120    pub fn anchored(from: CableAnchor, to: CableAnchor) -> Self {
121        Self::new(from, to)
122    }
123
124    /// Ties the far end to `b`.
125    pub fn to(mut self, b: NodeId) -> Self {
126        self.to = CableAnchor::Node(b, CableSide::Center);
127        self
128    }
129
130    /// Ties the far end to a point of the window.
131    pub fn to_point(mut self, p: Point) -> Self {
132        self.to = CableAnchor::Point(CableEnd::at(p));
133        self
134    }
135
136    /// Lets the far end follow the pointer.
137    pub fn loose(mut self) -> Self {
138        self.to = CableAnchor::Pointer;
139        self
140    }
141
142    /// The sides of the two widgets the ends go on.
143    pub fn sides(mut self, from: CableSide, to: CableSide) -> Self {
144        if let CableAnchor::Node(n, _) = self.from {
145            self.from = CableAnchor::Node(n, from);
146        }
147        if let CableAnchor::Node(n, _) = self.to {
148            self.to = CableAnchor::Node(n, to);
149        }
150        self
151    }
152
153    /// Points the cable is held at on the way.
154    pub fn via(mut self, points: impl IntoIterator<Item = Point>) -> Self {
155        self.via = points.into_iter().collect();
156        self
157    }
158
159    /// Keeps the same cable across builds even when what it ties changes, so moving
160    /// its end makes it swing instead of a new cable showing up.
161    pub fn key(mut self, key: u64) -> Self {
162        self.key = Some(key);
163        self
164    }
165
166    /// Body color.
167    pub fn color(mut self, color: Color) -> Self {
168        self.look.color = color;
169        self
170    }
171
172    /// Width in logical pixels.
173    pub fn thickness(mut self, width: f32) -> Self {
174        self.look.thickness = width.max(0.5);
175        self
176    }
177
178    /// Strength of the highlight along it, `0` to `1`.
179    pub fn shine(mut self, shine: f32) -> Self {
180        self.look.shine = shine.clamp(0.0, 1.0);
181        self
182    }
183
184    /// Its surface: smooth, braided, coiled, striped or a picture of yours.
185    pub fn texture(mut self, texture: CableTexture) -> Self {
186        self.look.texture = texture;
187        self
188    }
189
190    /// Draws a plug at each attached end.
191    pub fn plugs(mut self, plugs: bool) -> Self {
192        self.look.plugs = plugs;
193        self
194    }
195
196    /// Color of the plugs.
197    pub fn plug_color(mut self, color: Color) -> Self {
198        self.look.plug_color = color;
199        self
200    }
201
202    /// A dashed cable: dash and gap lengths.
203    pub fn dash(mut self, dash: f32, gap: f32) -> Self {
204        self.look.dash = Some([dash, gap]);
205        self
206    }
207
208    /// How much longer than the path it is: the sag. `1.0` is taut.
209    pub fn slack(mut self, slack: f32) -> Self {
210        self.physics.slack = slack.max(1.0);
211        self
212    }
213
214    /// The whole physics at once: weight, damping, stiffness, points.
215    pub fn physics(mut self, physics: CablePhysics) -> Self {
216        self.physics = physics;
217        self
218    }
219
220    /// Downward pull, in logical pixels per second squared.
221    pub fn gravity(mut self, gravity: f32) -> Self {
222        self.physics.gravity = gravity;
223        self
224    }
225
226    /// What happens when it is no longer built: it falls (the default), it is pulled
227    /// back to its start, or it vanishes.
228    pub fn release(mut self, release: CableRelease) -> Self {
229        self.release = release;
230        self
231    }
232
233    /// Pulled back to its start when no longer built, like a switchboard cord.
234    pub fn retract(self, retract: bool) -> Self {
235        let release = if retract { CableRelease::Retract } else { CableRelease::Fall };
236        self.release(release)
237    }
238
239    /// A new cable starts taut and drops into its sag (the default); without it the
240    /// cable shows up at rest.
241    pub fn grow(mut self, grow: bool) -> Self {
242        self.grow = grow;
243        self
244    }
245
246    /// Called on a click on the cable.
247    pub fn on_click(mut self, handler: impl Fn() + 'static) -> Self {
248        self.on_click = Some(Rc::new(handler));
249        self
250    }
251
252    /// Called with where the pointer is when it lets go of a cable whose end it holds.
253    pub fn on_drop(mut self, handler: impl Fn(Point) + 'static) -> Self {
254        self.on_drop = Some(Rc::new(handler));
255        self
256    }
257}

294/// How far a point is from a line through `points`.
295pub fn distance_to_cable(points: &[Point], p: Point) -> f32 {
296    let mut best = f32::INFINITY;
297    for w in points.windows(2) {
298        let (a, b) = (w[0], w[1]);
299        let (dx, dy) = (b.x - a.x, b.y - a.y);
300        let length = dx * dx + dy * dy;
301        let t =
302            if length > 0.0 { (((p.x - a.x) * dx + (p.y - a.y) * dy) / length).clamp(0.0, 1.0) } else { 0.0 };
303        let (x, y) = (a.x + dx * t - p.x, a.y + dy * t - p.y);
304        best = best.min(x.hypot(y));
305    }
306    best
307}

309/// A cable built this build, waiting for the layout to place its ends.
310#[derive(Clone)]
311pub(crate) struct CableRequest {
312    pub plug: CablePlug,
313    /// The widget that built it, for an identity that lasts across builds.
314    pub owner: u64,
315    pub index: u64,
316}
317
318/// The id of a cable built by `owner`, the `index`th of its build, unless it has a key.
319pub(crate) fn cable_id(request: &CableRequest, from: Option<u64>, to: Option<u64>) -> u64 {
320    let mut h = request.owner ^ 0x6361_626c_6500_0000;
321    let mix = |h: u64, v: u64| {
322        let mut x = h ^ v.wrapping_mul(0x9e37_79b9_7f4a_7c15);
323        x = (x ^ (x >> 30)).wrapping_mul(0xbf58_476d_1ce4_e5b9);
324        x ^ (x >> 31)
325    };
326    match request.plug.key {
327        Some(key) => mix(h, key),
328        None => {
329            h = mix(h, from.unwrap_or(request.index));
330            mix(h, to.unwrap_or(0x0070_6f69_6e74_6572))
331        }
332    }
333}

335/// The group every cable built with [`CablePlug`] belongs to: the window's.
336pub(crate) const PLUG_GROUP: u64 = 0;

338/// The cable as the world takes it, once its ends are placed.
339pub(crate) fn place(request: &CableRequest, id: u64, from: CableEnd, to: CableEnd) -> Cable {
340    let plug = &request.plug;
341    Cable {
342        id,
343        group: PLUG_GROUP,
344        from,
345        to,
346        via: Arc::from(plug.via.as_slice()),
347        physics: plug.physics,
348        look: plug.look.clone(),
349        release: plug.release,
350        grow: plug.grow,
351        points: None,
352        view: ravenrs_core::cable::CableView::default(),
353    }
