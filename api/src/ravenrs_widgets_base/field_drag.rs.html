1//! A field of data carried by the pointer, from where data is looked at (a `DataInspector`) to
2//! where an expression is written (an `ExpressionEditor`, a field of a `ParameterForm`): what
3//! people see and hear while carrying it, and the expression that reads it.
4
5use crate::context::DragPayload;
6
7/// The kind of a [`DragPayload`] carrying a [`FieldDrag`].
8pub const FIELD_DRAG: &str = "field";
9
10/// A field of data being carried: its path as people read it and the expression that reads it.
11#[derive(Clone, Debug, PartialEq, Eq)]
12#[cfg_attr(feature = "live", ravenrs_live::expose(part = "fields"))]
13pub struct FieldDrag {
14    /// The path as people read it, as `cliente.nome` or `itens[0].preço`.
15    pub path: String,
16    /// The expression of the criteria language that reads it, as `[cliente.nome]`.
17    pub expression: String,
18}
19
20#[cfg_attr(feature = "live", ravenrs_live::expose)]
21impl FieldDrag {
22    /// The field at `path`, read by `expression`.
23    pub fn new(path: impl Into<String>, expression: impl Into<String>) -> Self {
24        Self { path: path.into(), expression: expression.into() }
25    }
26
27    /// What a drag of it carries.
28    pub fn payload(&self) -> DragPayload {
29        DragPayload::new(FIELD_DRAG, &self.path, self.clone())
30    }
31
32    /// The field a drag carries, when it carries one.
33    pub fn carried(payload: &DragPayload) -> Option<&FieldDrag> {
34        if payload.kind() == FIELD_DRAG { payload.data::<FieldDrag>() } else { None }
35    }
36}
37
38/// The expression of the criteria language that reads the field at these keys (a row of a
39/// list is its number): `["cliente", "nome"]` reads `[cliente.nome]`.
40pub fn field_expression(keys: &[String]) -> String {
41    format!("[{}]", keys.join("."))
42}
43
44#[cfg(test)]
45mod tests {
46    use super::*;
47
48    #[test]
49    fn a_field_travels_in_a_payload_and_comes_back_whole() {
50        let field =
51            FieldDrag::new("itens[0].preço", field_expression(&["itens".into(), "0".into(), "preço".into()]));
52        assert_eq!(field.expression, "[itens.0.preço]");
53        let payload = field.payload();
54        assert_eq!(payload.label(), "itens[0].preço");
55        assert_eq!(FieldDrag::carried(&payload), Some(&field));
56        assert_eq!(FieldDrag::carried(&DragPayload::new("other", "x", 1_u8)), None);
57    }
58}
