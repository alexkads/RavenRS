1//! Identities of the element tree: [`WidgetId`] lasts across builds, [`NodeId`] one build.
2
3/// Stable identity of a widget across rebuilds, derived from its position in the
4/// view tree (or from an explicit key, see [`BuildCx::keyed_child`](crate::BuildCx::keyed_child)). Hover,
5/// press and focus are remembered per `WidgetId`.
6#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
7#[cfg_attr(feature = "live", ravenrs_live::expose(part = "fields"))]
8pub struct WidgetId(u64);
9
10#[cfg_attr(feature = "live", ravenrs_live::expose)]
11impl WidgetId {
12    #[doc(hidden)]
13    pub const ROOT: Self = Self(0x5241_5645_4e52_5321);
14
15    #[doc(hidden)]
16    pub fn child(self, key: u64) -> Self {
17        let mut h = self.0 ^ key.wrapping_mul(0x9e37_79b9_7f4a_7c15);
18        h = (h ^ (h >> 30)).wrapping_mul(0xbf58_476d_1ce4_e5b9);
19        h = (h ^ (h >> 27)).wrapping_mul(0x94d0_49bb_1331_11eb);
20        Self(h ^ (h >> 31))
21    }
22
23    /// The raw identity, for derived ids such as accessibility node ids.
24    #[doc(hidden)]
25    pub fn raw(self) -> u64 {
26        self.0
27    }
28}
29

33#[cfg_attr(feature = "live", ravenrs_live::expose(part = "fields"))]
