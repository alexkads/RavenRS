12enum Content {
13    Static(String),
14    Dynamic(Rc<dyn Fn() -> String>),
15}

17/// Text in the theme's type scale. It wraps to the width its container offers.
18#[cfg_attr(feature = "live", ravenrs_live::expose(part = "fields"))]
19pub struct Text {
20    content: Content,
21    style: TextStyle,
22    size: Option<f32>,
23    weight: Option<Weight>,
24    foreground: Fill,
25    truncate: bool,
26    align: TextAlign,
27    #[cfg(feature = "e2e-source")]
28    source: &'static std::panic::Location<'static>,
29}
30
31impl fmt::Debug for Text {
32    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
33        let content = match &self.content {
34            Content::Static(s) => s.as_str(),
35            Content::Dynamic(_) => "<dynamic>",
36        };
37        f.debug_struct("Text").field("content", &content).field("style", &self.style).finish()
38    }
39}

41#[cfg_attr(feature = "live", ravenrs_live::expose)]
42impl Text {
43    /// Static text.
44    #[cfg_attr(feature = "e2e-source", track_caller)]
45    pub fn new(text: impl Into<String>) -> Self {
46        Self::with_content(Content::Static(text.into()))
47    }
48
49    /// Text derived from a state; the view updates when the state changes.
50    #[cfg_attr(feature = "e2e-source", track_caller)]
51    pub fn bound<T: 'static>(state: &State<T>, format: impl Fn(&T) -> String + 'static) -> Self {
52        let state = state.clone();
53        Self::with_content(Content::Dynamic(Rc::new(move || state.with(&format))))
54    }
55
56    /// Text computed by a closure on every build; any state it reads is tracked.
57    #[cfg_attr(feature = "e2e-source", track_caller)]
58    pub fn dynamic(f: impl Fn() -> String + 'static) -> Self {
59        Self::with_content(Content::Dynamic(Rc::new(f)))
60    }
61
62    #[cfg_attr(feature = "e2e-source", track_caller)]
63    fn with_content(content: Content) -> Self {
64        Self {
65            content,
66            style: TextStyle::Body,
67            size: None,
68            weight: None,
69            foreground: Semantic::Ink.into(),
70            truncate: false,
71            align: TextAlign::Start,
72            #[cfg(feature = "e2e-source")]
73            source: std::panic::Location::caller(),
74        }
75    }
76
77    /// A step of the type scale (`Body` by default).
78    pub fn style(mut self, style: TextStyle) -> Self {
79        self.style = style;
80        self
81    }
82
83    /// Overrides the style's size; line height scales with it.
84    pub fn font_size(mut self, size: f32) -> Self {
85        self.size = Some(size);
86        self
87    }
88
89    /// Overrides the style's weight.
90    pub fn font_weight(mut self, weight: Weight) -> Self {
91        self.weight = Some(weight);
92        self
93    }
94
95    /// Keeps the text on one line, ended with `…` when it does not fit. Screen
96    /// readers still get the whole text.
97    pub fn truncate(mut self) -> Self {
98        self.truncate = true;
99        self
100    }
101
102    /// Where the lines sit across the width the text gets: at the start (the default), in the
103    /// middle or at the end. With more than one line, each line moves on its own. A text that
104    /// takes only the width of its words has no room to move: stretch it, as a `VStack` does
105    /// with [`Align::Stretch`](ravenrs_layout::Align::Stretch), or give it a width.
106    pub fn align(mut self, align: TextAlign) -> Self {
107        self.align = align;
108        self
109    }
110
111    /// Text color, as a theme role (`Ink` by default) or a literal color.
112    pub fn foreground(mut self, fill: impl Into<Fill>) -> Self {
113        self.foreground = fill.into();
114        self
115    }
116}

118impl View for Text {
119    fn build(&self, cx: &mut BuildCx<'_>) -> NodeId {
120        let theme = cx.theme();
121        let base = theme.text(self.style);
122        let size = self.size.unwrap_or(base.size);
123        let attrs = TextAttrs {
124            size,
125            line_height: base.line_height * size / base.size,
126            weight: self.weight.unwrap_or(base.weight).0,
127            letter_spacing: base.letter_spacing,
128            family: None,
129        };
130        let color = self.foreground.resolve(theme);
131        let content = match &self.content {
132            Content::Static(s) => std::borrow::Cow::Borrowed(s.as_str()),
133            Content::Dynamic(f) => std::borrow::Cow::Owned(f()),
134        };
135        let node = if self.truncate {
136            cx.text_truncated(&content, &attrs, color, LayoutStyle::leaf())
137        } else {
138            cx.text(&content, &attrs, color, LayoutStyle::leaf())
139        };
140        if self.align != TextAlign::Start {
141            cx.text_align(node, self.align);
142        }
143        if let Some(level) = heading_level(self.style) {
144            cx.semantics(node, Semantics::new(ravenrs_core::Role::Heading).level(level));
145        }
146        #[cfg(feature = "e2e-source")]
147        cx.view_source(node, self.source);
148        node
149    }
150}

152/// The heading level screen readers announce for large text styles.
153#[doc(hidden)]
154pub fn heading_level(style: TextStyle) -> Option<u32> {
155    match style {
156        TextStyle::Display => Some(1),
157        TextStyle::Title => Some(2),
158        TextStyle::Headline => Some(3),
159        _ => None,
160    }
