12/// A single-line text input bound to a `String`.
13///
14/// Supports typing, deleting, moving by character, word and line, selecting
15/// with Shift, the mouse (drag, double-click) or Cmd/Ctrl+A, copy, cut and
16/// paste, and Enter (submit). Passwords can be pasted but not copied.
17///
18/// ```
19/// use ravenrs_reactive::State;
20/// use ravenrs_widgets::{IconName, TextField, Ui};
21///
22/// let query = State::new(String::new());
23/// let field = TextField::new(query.binding()).placeholder("Search").icon(IconName::Search).clearable();
24/// let mut ui = Ui::new(field, ravenrs_style::Theme::raven_dark());
25/// ui.update();
26/// ```

249
250    /// The selected range, if it is not empty.
251    pub fn selection(&self) -> Option<(usize, usize)> {
252        let a = self.anchor?;
253        (a != self.caret).then(|| (a.min(self.caret), a.max(self.caret)))
254    }

262            self.anchor = None;
263        }
264        self.caret = to;
265    }

268fn is_word(c: char) -> bool {
269    c.is_alphanumeric() || c == '_'
270}
271
272/// The start of the word before `at` (Alt/Ctrl+Left).
273fn word_start(chars: &[char], at: usize) -> usize {
274    let mut i = at.min(chars.len());
275    while i > 0 && !is_word(chars[i - 1]) {
276        i -= 1;
277    }
278    while i > 0 && is_word(chars[i - 1]) {
279        i -= 1;
280    }
281    i
282}
283
284/// The end of the word after `at` (Alt/Ctrl+Right).
285fn word_end(chars: &[char], at: usize) -> usize {
286    let mut i = at.min(chars.len());
287    while i < chars.len() && !is_word(chars[i]) {
288        i += 1;
289    }
290    while i < chars.len() && is_word(chars[i]) {
291        i += 1;
292    }
293    i
294}

296/// The word around `at`, for double-click selection.
297#[doc(hidden)]
298pub fn word_at(text: &str, at: usize) -> (usize, usize) {
299    let chars: Vec<char> = text.chars().collect();
300    let at = at.min(chars.len());
301    let on_word = |i: usize| chars.get(i).is_some_and(|c| is_word(*c));
302    if !on_word(at) && !(at > 0 && on_word(at - 1)) {
303        return (at, (at + 1).min(chars.len()));
304    }
305    let mut start = at;
306    while start > 0 && is_word(chars[start - 1]) {
307        start -= 1;
308    }
309    let mut end = at;
310    while end < chars.len() && is_word(chars[end]) {
311        end += 1;
312    }
313    (start, end)
314}

316/// Removes the selection, leaving the caret where it was. Returns false when
317/// nothing was selected.
318fn delete_selection(text: &mut String, cursor: &mut Cursor) -> bool {
319    let Some((a, b)) = cursor.selection() else { return false };
320    text.replace_range(byte_index(text, a)..byte_index(text, b), "");
321    *cursor = Cursor::at(a);
322    true
323}

325#[doc(hidden)]
326pub fn insert(text: &mut String, cursor: &mut Cursor, s: &str) {
327    delete_selection(text, cursor);
328    let at = byte_index(text, cursor.caret);
329    text.insert_str(at, s);
330    *cursor = Cursor::at(cursor.caret + s.chars().count());
331}

344/// Applies a key to `text` and the cursor: typing, deleting, moving (Shift
345/// extends the selection; Alt or Ctrl moves by word; Cmd moves to the ends),
346/// select all, and copy, cut and paste with Cmd or Ctrl. Returns false for
347/// keys the field does not handle, including a paste the clipboard cannot
348/// read synchronously (the browser then delivers it as typed text).
349#[doc(hidden)]
350pub fn edit(
351    text: &mut String,
352    cursor: &mut Cursor,
353    key: &Key,
354    modifiers: Modifiers,
355    clipboard: Option<ClipboardAccess<'_>>,
356) -> bool {
357    let chars: Vec<char> = text.chars().collect();
358    let len = chars.len();
359    cursor.caret = cursor.caret.min(len);
360    cursor.anchor = cursor.anchor.map(|a| a.min(len));
361    let shift = modifiers.shift;
362    let word = modifiers.alt || modifiers.ctrl;
363    let command = modifiers.ctrl || modifiers.meta;
364    let multiline = clipboard.as_ref().is_some_and(|access| access.multiline);
365    if command && let Key::Character(c) = key {
366        match c.to_ascii_lowercase().as_str() {
367            "a" => *cursor = Cursor { caret: len, anchor: Some(0) },
368            "c" | "x" => {
369                let Some(access) = clipboard else { return false };
370                let Some((a, b)) = cursor.selection() else { return true };
371                if access.secure {
372                    return true;
373                }
374                access.clipboard.write(&chars[a..b].iter().collect::<String>());
375                if c.eq_ignore_ascii_case("x") {
376                    delete_selection(text, cursor);
377                }
378            }
379            "v" => {
380                let Some(pasted) = clipboard.and_then(|access| access.clipboard.read()) else { return false };
381                if multiline {
382                    insert(text, cursor, &pasted.replace("\r\n", "\n").replace('\r', "\n"));
383                } else {
384                    // A single-line field keeps the first line.
385                    let line = pasted.lines().next().unwrap_or_default();
386                    insert(text, cursor, line);
387                }
388            }
389            _ => return false,
390        }
391        return true;
392    }
393    match key {
394        // Line breaks (from a browser paste, say) become spaces in one line.
395        Key::Character(s) if !command => {
396            insert(text, cursor, &s.replace("\r\n", " ").replace(['\n', '\r'], " "))
397        }
398        Key::Space if !command => insert(text, cursor, " "),
399        Key::Backspace | Key::Delete if delete_selection(text, cursor) => {}
400        Key::Backspace if cursor.caret > 0 => {
401            let from = if modifiers.meta {
402                0
403            } else if word {
404                word_start(&chars, cursor.caret)
405            } else {
406                cursor.caret - 1
407            };
408            text.replace_range(byte_index(text, from)..byte_index(text, cursor.caret), "");
409            *cursor = Cursor::at(from);
410        }
411        Key::Delete if cursor.caret < len => {
412            let to = if word { word_end(&chars, cursor.caret) } else { cursor.caret + 1 };
413            text.replace_range(byte_index(text, cursor.caret)..byte_index(text, to), "");
414            *cursor = Cursor::at(cursor.caret);
415        }
416        Key::Backspace | Key::Delete => {}
417        Key::ArrowLeft | Key::ArrowRight => {
418            let left = *key == Key::ArrowLeft;
419            let to = match (cursor.selection(), shift) {
420                // Without Shift, an arrow collapses the selection to that side.
421                (Some((a, b)), false) if !word && !modifiers.meta => {
422                    if left {
423                        a
424                    } else {
425                        b
426                    }
427                }
428                _ if modifiers.meta => {
429                    if left {
430                        0
431                    } else {
432                        len
433                    }
434                }
435                _ if word => {
436                    if left {
437                        word_start(&chars, cursor.caret)
438                    } else {
439                        word_end(&chars, cursor.caret)
440                    }
441                }
442                _ if left => cursor.caret.saturating_sub(1),
443                _ => (cursor.caret + 1).min(len),
444            };
445            cursor.move_to(to, shift);
446        }
447        Key::Home | Key::ArrowUp => cursor.move_to(0, shift),
448        Key::End | Key::ArrowDown => cursor.move_to(len, shift),
449        _ => return false,
450    }
451    true
452}

454/// How many steps a [`TextHistory`] keeps; older ones are let go.
455const HISTORY_LIMIT: usize = 200;
456
457/// One change to a text: at byte `at`, `removed` gave way to `inserted`, and the caret went
458/// from `before` to `after`. Only what changed is kept, so a long text costs little per step.
459#[derive(Clone, Debug, PartialEq, Eq)]
460struct Step {
461    at: usize,
462    removed: String,
463    inserted: String,
464    before: Cursor,
465    after: Cursor,
466}
467
468#[cfg_attr(feature = "live", ravenrs_live::expose)]
469impl Step {
470    /// What changed from `old` to `new`: what lies between their common start and end.
471    fn between(old: &str, new: &str, before: Cursor, after: Cursor) -> Self {
472        let mut start = old.bytes().zip(new.bytes()).take_while(|(a, b)| a == b).count();
473        while !old.is_char_boundary(start) || !new.is_char_boundary(start) {
474            start -= 1;
475        }
476        let room = old.len().min(new.len()) - start;
477        let mut end = old.bytes().rev().zip(new.bytes().rev()).take(room).take_while(|(a, b)| a == b).count();
478        while !old.is_char_boundary(old.len() - end) || !new.is_char_boundary(new.len() - end) {
479            end -= 1;
480        }
481        Self {
482            at: start,
483            removed: old[start..old.len() - end].to_owned(),
484            inserted: new[start..new.len() - end].to_owned(),
485            before,
486            after,
487        }
488    }
489
490    /// Takes in a step that comes right after this one in the same run: more letters of a
491    /// word, or more characters taken out by Backspace or Delete.
492    fn join(&mut self, next: Step) -> bool {
493        if self.removed.is_empty() && next.removed.is_empty() && next.at == self.at + self.inserted.len() {
494            self.inserted.push_str(&next.inserted);
495        } else if self.inserted.is_empty()
496            && next.inserted.is_empty()
497            && next.at + next.removed.len() == self.at
498        {
499            // Backspace: each character comes off before the last.
500            self.at = next.at;
501            self.removed.insert_str(0, &next.removed);
502        } else if self.inserted.is_empty() && next.inserted.is_empty() && next.at == self.at {
503            // Delete: each character comes off after the caret, which stays.
504            self.removed.push_str(&next.removed);
505        } else {
506            return false;
507        }
508        self.after = next.after;
509        true
510    }
511}

513/// What the step still open is made of, so the next key of the same kind joins it.
514#[derive(Clone, Copy, Debug, PartialEq, Eq)]
515enum Run {
516    /// Letters, digits and `_` typed one after another: a word.
517    Word,
518    /// Characters taken out one by one with Backspace or Delete.
519    Deleting,
520}

522/// What can be undone and redone in a text field, a text area or a code editor: Cmd/Ctrl+Z
523/// undoes, Cmd+Shift+Z or Ctrl+Y redoes.
524///
525/// A word typed letter by letter is one step, and so is each space or sign after it, each
526/// paste, cut and line break. Moving the caret closes the step being typed. When the app
527/// changes the text by itself, the next key starts a new step from what the app left, so an
528/// undo never jumps back past it; [`record_change`](Self::record_change) makes such a change
529/// a step of its own, to be undone too. It keeps the last 200 steps, and of each only what
530/// changed, so a long text costs little.
531///
532/// Each field keeps its own; give a `CodeEditor` one through its `history` to keep it while
533/// the editor comes and goes, or to undo from a button.
534///
535/// ```
536/// use ravenrs_widgets::{Cursor, TextHistory};
537///
538/// let mut history = TextHistory::new();
539/// let mut text = String::from("final");
540/// let mut cursor = Cursor::at(5);
541/// history.record_change("draft", &text, cursor);
542/// assert!(history.undo(&mut text, &mut cursor));
543/// assert_eq!(text, "draft");
544/// assert!(history.redo(&mut text, &mut cursor));
545/// assert_eq!(text, "final");
546/// ```
547#[derive(Clone, Debug, Default, PartialEq, Eq)]
548#[cfg_attr(feature = "live", ravenrs_live::expose(part = "fields"))]
549pub struct TextHistory {
550    undo: Vec<Step>,
551    redo: Vec<Step>,
552    /// The text as the last key left it, to notice a change made from outside.
553    known: Option<String>,
554    /// The kind of the step still open, which the next key of that kind joins.
555    open: Option<Run>,
556}
557
558#[cfg_attr(feature = "live", ravenrs_live::expose)]
559impl TextHistory {
560    /// Nothing to undo or redo.
561    pub fn new() -> Self {
562        Self::default()
563    }
564
565    /// Whether there is a step to undo.
566    pub fn can_undo(&self) -> bool {
567        !self.undo.is_empty()
568    }
569
570    /// Whether there is a step to redo.
571    pub fn can_redo(&self) -> bool {
572        !self.redo.is_empty()
573    }
574
575    /// Puts back the text and the caret as they were before the last step. False when there
576    /// is nothing to undo, or when the text is no longer the one the step was made on.
577    pub fn undo(&mut self, text: &mut String, cursor: &mut Cursor) -> bool {
578        let Some(step) = self.undo.pop() else { return false };
579        let end = step.at + step.inserted.len();
580        if text.get(step.at..end) != Some(step.inserted.as_str()) {
581            self.forget();
582            return false;
583        }
584        text.replace_range(step.at..end, &step.removed);
585        *cursor = step.before;
586        self.redo.push(step);
587        self.settle(text);
588        true
589    }
590
591    /// Does again the last step undone. False when there is nothing to redo.
592    pub fn redo(&mut self, text: &mut String, cursor: &mut Cursor) -> bool {
593        let Some(step) = self.redo.pop() else { return false };
594        let end = step.at + step.removed.len();
595        if text.get(step.at..end) != Some(step.removed.as_str()) {
596            self.forget();
597            return false;
598        }
599        text.replace_range(step.at..end, &step.inserted);
600        *cursor = step.after;
601        self.undo.push(step);
602        self.settle(text);
603        true
604    }
605
606    /// Takes in a change the app made by itself (replacing all, reloading from disk) as one
607    /// step: the text was `before` and is now `after`, and the caret was at `cursor`.
608    pub fn record_change(&mut self, before: &str, after: &str, cursor: Cursor) {
609        if before == after {
610            return;
611        }
612        self.push(Step::between(before, after, cursor, cursor));
613        self.redo.clear();
614        self.known = Some(after.to_owned());
615    }
616
617    /// Lets go of every step, as when the text is replaced by another.
618    pub fn clear(&mut self) {
619        *self = Self::default();
620    }
621
622    fn forget(&mut self) {
623        self.undo.clear();
624        self.redo.clear();
625        self.open = None;
626    }
627
628    fn settle(&mut self, text: &str) {
629        self.known = Some(text.to_owned());
630        self.open = None;
631    }
632
633    fn push(&mut self, step: Step) {
634        self.undo.push(step);
635        if self.undo.len() > HISTORY_LIMIT {
636            self.undo.remove(0);
637        }
638        self.open = None;
639    }
640
641    /// Applies a key with `apply`, keeping the step it makes; Cmd/Ctrl+Z, Cmd+Shift+Z and
642    /// Ctrl+Y undo and redo instead. Returns whether the key was handled.
643    #[doc(hidden)]
644    pub fn apply(
645        &mut self,
646        text: &mut String,
647        cursor: &mut Cursor,
648        key: &Key,
649        modifiers: Modifiers,
650        apply: impl FnOnce(&mut String, &mut Cursor) -> bool,
651    ) -> bool {
652        let command = modifiers.ctrl || modifiers.meta;
653        if command && let Key::Character(c) = key {
654            let c = c.to_ascii_lowercase();
655            if c == "z" && !modifiers.shift {
656                self.undo(text, cursor);
657                return true;
658            }
659            if (c == "z" && modifiers.shift) || (c == "y" && modifiers.ctrl && !modifiers.meta) {
660                self.redo(text, cursor);
661                return true;
662            }
663        }
664        if self.known.as_deref().is_some_and(|known| known != text.as_str()) {
665            // The app changed the text, and the steps kept are about another one.
666            self.forget();
667        }
668        let (old, before) = (text.clone(), *cursor);
669        if !apply(text, cursor) {
670            return false;
671        }
672        if *text == old {
673            if *cursor != before {
674                self.open = None;
675            }
676            self.known = Some(old);
677            return true;
678        }
679        let run = match key {
680            Key::Character(s) if !command && before.selection().is_none() && s.chars().all(is_word) => {
681                Some(Run::Word)
682            }
683            Key::Backspace | Key::Delete if !modifiers.alt && !command && before.selection().is_none() => {
684                Some(Run::Deleting)
685            }
686            _ => None,
687        };
688        let step = Step::between(&old, text, before, *cursor);
689        let joined = run.is_some()
690            && self.open == run
691            && self
692                .undo
693                .last_mut()
694                .is_some_and(|last| last.after.caret == before.caret && last.join(step.clone()));
695        if !joined {
696            self.push(step);
697        }
698        self.open = run;
699        self.redo.clear();
700        self.known = Some(text.clone());
701        true
702    }
703}

705/// The char index nearest to `x` (relative to the text's left edge), given
706/// the x position of every char boundary.
707#[doc(hidden)]
708pub fn index_at(boundaries: &[f32], x: f32) -> usize {
709    let after = boundaries.partition_point(|b| *b < x);
710    match after {
711        0 => 0,
712        n if n >= boundaries.len() => boundaries.len().saturating_sub(1),
713        n => {
714            if x - boundaries[n - 1] < boundaries[n] - x {
715                n - 1
716            } else {
717                n
718            }
719        }
720    }
721}

726        let cursor: State<Cursor> = cx.use_state(|| {
727            if self.select_on_open {
728                Cursor { caret: self.value.get().chars().count(), anchor: Some(0) }
729            } else {
730                Cursor::at(usize::MAX)
731            }
732        });
733        let focus_requested: State<bool> = cx.use_state(|| false);
734        let history: State<TextHistory> = cx.use_state(TextHistory::new);
735        // Which line sent before is shown, counted in the recall list; none while typing.
736        let recalled: State<Option<usize>> = cx.use_state(|| None);
737        let state = self.state;
738        if self.autofocus && !state.disabled && !*focus_requested.peek() {
739            focus_requested.set_if_changed(true);
740            cx.request_focus();
741        }
742        if self.keeps_focus && !state.disabled && cx.nothing_focused() {
743            cx.request_focus();
744        }
745        // Set by the clear button, which took the focus with its click.
746        let refocus: State<bool> = cx.use_state(|| false);
747        if *refocus.peek() {
748            refocus.set(false);
749            cx.request_focus();
750        }
751        let focused = cx.is_focused() && !state.disabled;
752        let had_focus: State<bool> = cx.use_state(|| false);
753        if *had_focus.peek() != focused {
754            had_focus.set(focused);
755            if !focused && let Some(left) = &self.on_leave {
756                left();
757            }
758        }
759        let hovered = cx.is_hovered() && !state.disabled;
760        let text = self.value.get();
761        let len = text.chars().count();
762        let cur = cursor.get();
763        let caret_at = cur.caret.min(len);
764        let visible = self.idle_text.as_ref().filter(|_| !focused).unwrap_or(&text);
765        let shown: String = if self.secure { "•".repeat(len) } else { visible.clone() };
766        let dense = matches!(self.size, ButtonSize::Small | ButtonSize::Mini);
767        let attrs = util::attrs(&theme, self.size.text_style(), None);
768
769        let mut children = Vec::new();
770        if let Some(icon) = self.icon {
771            let size = if self.size == ButtonSize::Mini { 12.0 } else { 16.0 };
772            children.push(util::icon(cx, icon, size, state.ink(theme.color(Semantic::InkMuted))));
773        }
774        // Keyed, so that the clear button coming and going does not move their state.
775        if let Some(prefix) = &self.prefix {
776            children.push(cx.keyed_child(1, prefix));
777        }
778        // Where each char boundary falls, for the caret, the selection and clicks.
779        let boundaries: Rc<Vec<f32>> = Rc::new(if shown.chars().count() <= 512 {
780            let chars: Vec<char> = shown.chars().collect();
781            (0..=chars.len())
782                .map(|i| cx.measure_text(&chars[..i].iter().collect::<String>(), &attrs).width)
783                .collect()
784        } else {
785            // Long text: one measurement, spread evenly.
786            let w = cx.measure_text(&shown, &attrs).width;
787            let n = shown.chars().count();
788            (0..=n).map(|i| w * i as f32 / n as f32).collect()
789        });
790        let caret_x = boundaries.get(caret_at).copied().unwrap_or(0.0);
791        let selection = cur.selection().map(|(a, b)| {
792            let at = |i: usize| boundaries.get(i.min(len)).copied().unwrap_or(0.0);
793            (at(a), at(b))
794        });
795        let highlight = theme.color(Semantic::Accent).with_alpha(if focused { 0.32 } else { 0.16 });
796        let content = if shown.is_empty() {
797            cx.text(
798                &self.placeholder,
799                &attrs,
800                state.ink(theme.color(Semantic::InkMuted)),
801                LayoutStyle::leaf(),
802            )
803        } else {
804            cx.text(&shown, &attrs, state.ink(theme.color(Semantic::Ink)), LayoutStyle::leaf())
805        };
806        let accent = theme.color(Semantic::Accent);
807        let line = attrs.line_height;
808        let end_aligned = self.end_aligned;
809        let text_width = boundaries.last().copied().unwrap_or(0.0);
810        // How far the text starts into an area of this width.
811        let inset = move |width: f32| if end_aligned { (width - text_width).max(0.0) } else { 0.0 };
812        let text_area = cx.container(
813            LayoutStyle {
814                grow: 1.0,
815                min_width: Some(0.0),
816                align: Align::Center,
817                justify: if end_aligned { Justify::End } else { Justify::Start },
818                overflow_hidden: true,
819                ..LayoutStyle::stack(Axis::Horizontal)
820            },
821            &[content],
822            Paint::Custom(Rc::new(move |f: Rect, scene| {
823                let y = f.center().y - line / 2.0 + 2.0;
824                let start = f.x() + inset(f.width());
825                if let Some((a, b)) = selection {
826                    scene.push_quad(
827                        Quad::new(Rect::new(start + a, y, b - a, line - 4.0), highlight).radius(2.0),
828                    );
829                }
830                if focused && selection.is_none() && state.accepts_input() {
831                    let x = (start + caret_x).min(f.max_x() - 1.5);
832                    scene.push_quad(Quad::new(Rect::new(x, y, 1.5, line - 4.0), accent).radius(0.75));
833                }
834            })),
835        );
836        cx.clip(text_area);
837        // Press places the caret, dragging selects, Shift-click extends.
838        if !state.disabled {
839            let (drag_cursor, mods, b) = (cursor.clone(), cx.modifiers(), boundaries.clone());
840            cx.on_drag(
841                text_area,
842                Rc::new(move |d| {
843                    let at = index_at(&b, d.position.x - d.frame.x() - inset(d.frame.width()));
844                    match d.phase {
845                        DragPhase::Start => drag_cursor.update(|c| {
846                            if mods.get().shift {
847                                c.move_to(at, true);
848                            } else {
849                                *c = Cursor { caret: at, anchor: Some(at) };
850                            }
851                        }),
852                        DragPhase::Move => drag_cursor.update(|c| c.move_to(at, true)),
853                        _ => {}
854                    }
855                }),
856            );
857            let (word_cursor, word_value) = (cursor.clone(), self.value.clone());
858            cx.on_double_click(
859                text_area,
860                Rc::new(move || {
861                    let caret = word_cursor.get().caret;
862                    let (a, b) = word_at(&word_value.get(), caret);
863                    word_cursor.set(Cursor { caret: b, anchor: Some(a) });
864                }),
865            );
866            cx.cursor(text_area, CursorIcon::Text);
867        }
868        children.push(text_area);
869        if self.clearable && !text.is_empty() && state.accepts_input() {
870            let (value, caret, refocus) = (self.value.clone(), cursor.clone(), refocus.clone());
871            let clear = IconButton::new(IconName::X)
872                .label(cx.labels().clear.clone())
873                .style(ButtonStyle::Ghost)
874                .size(ButtonSize::Mini)
875                .on_click(move || {
876                    value.set(String::new());
877                    caret.set(Cursor::at(usize::MAX));
878                    refocus.set(true);
879                });
880            children.push(cx.keyed_child(2, &clear));
881        }
882        if let Some(suffix) = &self.suffix {
883            children.push(cx.keyed_child(3, suffix));
884        }
885
886        let border = if focused {
887            Border { width: 1.5, color: accent }
888        } else {
889            let base = theme.color(Semantic::Border);
890            Border {
891                width: 1.0,
892                color: state.ink(if hovered { base.lerp(theme.color(Semantic::Ink), 0.25) } else { base }),
893            }
894        };
895        let style = LayoutStyle {
896            padding: Insets::symmetric(0.0, if dense && !self.flat { 8.0 } else { 12.0 }),
897            height: if self.flat { None } else { Some(self.size.height()) },
898            grow: if self.flat { 1.0 } else { 0.0 },
899            width: self.width,
900            min_width: Some(if dense { 0.0 } else { 160.0 }),
901            spacing: 8.0,
902            align: Align::Center,
903            align_self: Some(if self.width.is_some() { Align::Start } else { Align::Stretch }),
904            ..LayoutStyle::stack(Axis::Horizontal)
905        };
906        let paint = if self.flat {
907            let fill = if hovered && !focused {
908                theme.color(Semantic::Ink).with_alpha(0.05)
909            } else {
910                Color::TRANSPARENT
911            };
912            QuadPaint {
913                corner_radius: theme.radii.sm,
914                border: focused.then_some(border),
915                ..QuadPaint::new(fill)
916            }
917        } else {
918            QuadPaint {
919                corner_radius: theme.radii.md,
920                border: Some(border),
921                ..QuadPaint::new(state.fill(theme.color(Semantic::Control)))
922            }
923        };
924        let node = cx.container(style, &children, Paint::Quad(paint));
925        crate::motion::control(cx, node);
926        // The visible text is exposed as the value; the placeholder names the
927        // field until a label (a FormField, say) gives it a real name.
928        cx.semantics(text_area, Semantics::hidden());
929        let mut semantics = Semantics::new(Role::TextInput).value(shown);
930        if self.secure {
931            semantics = semantics.protected();
932        }
933        if !self.placeholder.is_empty() {
934            semantics = semantics.name(self.placeholder.clone());
935            semantics.fallback_name = true;
936        }
937        cx.semantics(node, state.semantics(semantics));
938        if state.disabled {
939            return node;
940        }
941
942        let (value, submit, cancel) = (self.value.clone(), self.on_submit.clone(), self.on_cancel.clone());
943        let (intercept, clipboard, secure) = (self.intercept.clone(), cx.clipboard(), self.secure);
944        let follows_end = self.caret_follows_end;
945        let recall = self.recall.clone();
946        cx.on_key(
947            node,
948            Rc::new(move |key, modifiers| {
949                if let Some(i) = &intercept
950                    && i(key, modifiers)
951                {
952                    return true;
953                }
954                if *key == Key::Escape
955                    && let Some(cancel) = &cancel
956                {
957                    cancel();
958                    return true;
959                }
960                if *key == Key::Enter {
961                    if let Some(sent) = recall.as_ref().filter(|_| state.accepts_input()) {
962                        let line = value.get();
963                        if !line.is_empty() && sent.peek().last() != Some(&line) {
964                            sent.update(|sent| sent.push(line));
965                        }
966                        recalled.set(None);
967                    }
968                    if let (Some(submit), true) = (&submit, state.accepts_input()) {
969                        submit(value.get());
970                    }
971                    return true;
972                }
973                if let Some(sent) = recall.as_ref().filter(|_| state.accepts_input())
974                    && matches!(key, Key::ArrowUp | Key::ArrowDown)
975                    && !modifiers.shift
976                {
977                    let count = sent.peek().len();
978                    let now = recalled.get();
979                    let next = match (key, now) {
980                        (_, _) if count == 0 => return true,
981                        (Key::ArrowUp, None) => Some(count - 1),
982                        (Key::ArrowUp, Some(at)) => Some(at.saturating_sub(1)),
983                        (_, Some(at)) if at + 1 < count => Some(at + 1),
984                        _ => None,
985                    };
986                    let line = next.and_then(|at| sent.peek().get(at).cloned()).unwrap_or_default();
987                    recalled.set(next);
988                    cursor.set(Cursor::at(line.chars().count()));
989                    value.set(line);
990                    return true;
991                }
992                let mut text = value.get();
993                let mut c = cursor.get();
994                let access = ClipboardAccess { clipboard: &clipboard, secure, multiline: false };
995                let handled = if state.accepts_input() {
996                    let mut handled = false;
997                    history.update(|steps| {
998                        handled = steps.apply(&mut text, &mut c, key, modifiers, |t, c| {
999                            edit(t, c, key, modifiers, Some(access))
1000                        });
1001                    });
1002                    handled
1003                } else {
1004                    edit(&mut text, &mut c, key, modifiers, Some(access))
1005                };
1006                if handled {
1007                    let changed = text != value.get();
1008                    if changed && !state.accepts_input() {
1009                        // Copying still works; the text stays as it was.
1010                        return true;
1011                    }
1012                    if follows_end && c.caret == text.chars().count() {
1013                        c.caret = usize::MAX;
1014                    }
1015                    cursor.set(c);
1016                    if changed {
1017                        value.set(text);
1018                    }
1019                }
1020                handled
1021            }),
1022        );
1023        cx.focusable(node, theme.radii.md);
1024        cx.cursor(node, CursorIcon::Text);
1025        node
1026    }
1027}
