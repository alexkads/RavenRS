13/// A stable number for a text key, for [`BuildCx::keyed_child`].
14#[doc(hidden)]
15pub fn hash_text(text: &str) -> u64 {
16    text.bytes().fold(0xcbf2_9ce4_8422_2325_u64, |h, b| (h ^ u64::from(b)).wrapping_mul(0x100_0000_01b3))
17}

25/// Whether an editor takes input. Disabled ones are out of the tab order and
26/// fade; read-only ones can still be focused and read, on a lighter fill.
27#[derive(Clone, Copy, Debug, Default)]
28#[doc(hidden)]
29pub struct EditorState {
30    pub disabled: bool,
31    pub read_only: bool,
32}
33
34#[cfg_attr(feature = "live", ravenrs_live::expose)]
35impl EditorState {
36    pub fn accepts_input(self) -> bool {
37        !self.disabled && !self.read_only
38    }
39
40    pub fn ink(self, color: Color) -> Color {
41        if self.disabled { faded(color) } else { color }
42    }
43
44    pub fn fill(self, color: Color) -> Color {
45        if self.disabled {
46            faded(color)
47        } else if self.read_only {
48            color.with_alpha(color.a * 0.5)
49        } else {
50            color
51        }
52    }
53
54    pub fn semantics(self, semantics: Semantics) -> Semantics {
55        semantics.disabled(self.disabled).read_only(self.read_only)
56    }
57}

96#[doc(hidden)]

104/// A few words in a box whose corner is at `at`, outlined in `tone`, which screen readers
105/// announce as a status: why a change was refused, or what to do next. `ink` is the color of the
106/// words and `fill` the color of the box, so each component keeps its own palette.
107#[doc(hidden)]
108#[allow(clippy::too_many_arguments, reason = "the parts of one box, each chosen by the caller")]
109pub fn bubble(
110    cx: &mut BuildCx<'_>,
111    icon_name: IconName,
112    tone: Color,
113    text: &str,
114    at: Point,
115    ink: Color,
116    fill: Color,
117) -> NodeId {
118    let theme = *cx.theme();
119    let words_attrs = attrs(&theme, TextStyle::Callout, None);
120    let glyph = icon(cx, icon_name, 16.0, tone);
121    let words =
122        cx.text(text, &words_attrs, ink, LayoutStyle { max_width: Some(280.0), ..LayoutStyle::leaf() });
123    let node = cx.container(
124        LayoutStyle {
125            absolute: Some((at.x, at.y)),
126            padding: ravenrs_core::Insets::symmetric(theme.spacing.s2, theme.spacing.s3),
127            spacing: theme.spacing.s2,
128            align: ravenrs_layout::Align::Start,
129            ..LayoutStyle::stack(ravenrs_layout::Axis::Horizontal)
130        },
131        &[glyph, words],
132        Paint::Quad(crate::context::QuadPaint {
133            corner_radius: theme.radii.md,
134            border: Some(ravenrs_core::Border { width: 1.0, color: tone }),
135            shadow: Some(theme.elevation.raised),
136            ..crate::context::QuadPaint::new(fill)
137        }),
138    );
139    cx.semantics(node, Semantics::new(ravenrs_core::Role::Status).name(text.to_owned()));
140    node
141}

143/// A pending focus move inside a group: which item takes focus on its next
144/// build. Requesting one also bumps a reactive tick, so the group rebuilds even
145/// when nothing else changed (arrow keys in a toolbar only move focus).
146#[derive(Clone, Debug)]
147#[doc(hidden)]
148pub struct FocusTarget {
149    target: std::rc::Rc<std::cell::Cell<Option<usize>>>,
150    tick: ravenrs_reactive::State<u32>,
151}
152
153#[cfg_attr(feature = "live", ravenrs_live::expose)]
154impl FocusTarget {
155    #[doc(hidden)]
156    pub fn new(initial: Option<usize>) -> Self {
157        Self {
158            target: std::rc::Rc::new(std::cell::Cell::new(initial)),
159            tick: ravenrs_reactive::State::new(0),
160        }
161    }
162
163    #[doc(hidden)]
164    pub fn request(&self, index: usize) {
165        self.target.set(Some(index));
166        self.tick.update(|t| *t = t.wrapping_add(1));
167    }
168
169    /// True once if focus should move to `index`; reading subscribes the build.
170    #[doc(hidden)]
171    pub fn take(&self, index: usize) -> bool {
172        let _ = self.tick.get();
173        let hit = self.target.get() == Some(index);
174        if hit {
175            self.target.set(None);
176        }
177        hit
178    }
179}

181/// Roving focus for item `index` of `count` in a row or column (radio buttons,
182/// tabs, toolbar buttons, menu items): arrow keys move focus to the neighbor,
183/// wrapping; Home and End jump to the ends.
184#[derive(Clone, Debug)]
185#[doc(hidden)]
186pub struct Roving {
187    #[doc(hidden)]
188    pub index: usize,
189    #[doc(hidden)]
190    pub count: usize,
191    #[doc(hidden)]
192    pub focus_to: FocusTarget,
193}

195/// The shared [`FocusTarget`] of a group; call once per build of the group.
196/// `initial` focuses that item when the group first appears (an opened menu).
197#[doc(hidden)]
198pub fn roving_cell(cx: &mut BuildCx<'_>, initial: Option<usize>) -> FocusTarget {
199    cx.use_state(|| FocusTarget::new(initial)).get()
200}

202#[cfg_attr(feature = "live", ravenrs_live::expose)]
203impl Roving {
204    /// Takes focus if an arrow key moved it here, and handles arrow keys on
205    /// `node`. `on_move` runs with the new index (to select it, for patterns
206    /// where selection follows focus).
207    #[doc(hidden)]
208    pub fn wire(&self, cx: &mut BuildCx<'_>, node: NodeId, on_move: Option<std::rc::Rc<dyn Fn(usize)>>) {
209        self.wire_with(cx, node, on_move, None);
210    }
211
212    /// Like [`Self::wire`], with `before` seeing keys first (a menubar title
213    /// opens its menu with Down; menu items switch menus with Left and Right).
214    #[doc(hidden)]
215    pub fn wire_with(
216        &self,
217        cx: &mut BuildCx<'_>,
218        node: NodeId,
219        on_move: Option<std::rc::Rc<dyn Fn(usize)>>,
220        before: Option<crate::KeyHandler>,
221    ) {
222        if self.focus_to.take(self.index) {
223            // By node: the focusable element may belong to a child widget (a button).
224            cx.request_focus_node(node);
225        }
226        let (index, count, focus_to) = (self.index, self.count, self.focus_to.clone());
227        cx.on_key(
228            node,
229            std::rc::Rc::new(move |key, modifiers| {
230                use ravenrs_core::Key;
231                if let Some(before) = &before
232                    && before(key, modifiers)
233                {
234                    return true;
235                }
236                if count == 0 {
237                    return false;
238                }
239                let target = match key {
240                    Key::ArrowRight | Key::ArrowDown => (index + 1) % count,
241                    Key::ArrowLeft | Key::ArrowUp => (index + count - 1) % count,
242                    Key::Home => 0,
243                    Key::End => count - 1,
244                    _ => return false,
245                };
246                focus_to.request(target);
247                if let Some(on_move) = &on_move {
248                    on_move(target);
249                }
250                true
251            }),
252        );
253    }
254}

256/// The outline of a five-pointed star, for what the icon font only has as a line.
257#[doc(hidden)]
258pub fn star_path(center: ravenrs_core::Point, radius: f32) -> Vec<ravenrs_core::PathElement> {
259    use ravenrs_core::{PathElement, Point};
260    let mut path = Vec::with_capacity(11);
261    for point in 0..10 {
262        let reach = if point % 2 == 0 { radius } else { radius * 0.45 };
263        let angle = -std::f32::consts::FRAC_PI_2 + point as f32 * std::f32::consts::PI / 5.0;
264        let at = Point::new(center.x + reach * angle.cos(), center.y + reach * angle.sin());
265        path.push(if point == 0 { PathElement::MoveTo(at) } else { PathElement::LineTo(at) });
266    }
267    path.push(PathElement::Close);
268    path
269}

290/// The outline of a dashed rounded rectangle, just inside `frame`: each corner in one piece,
291/// and on each side as many dashes as fit, evenly spread, so that no corner gets a stub.
292#[doc(hidden)]
293pub fn dashed_outline(
294    frame: ravenrs_core::Rect,
295    radius: f32,
296    width: f32,
297    dash: f32,
298    gap: f32,
299) -> Vec<ravenrs_core::PathElement> {
300    use ravenrs_core::{PathElement, Point};
301
302    let mut path = Vec::new();
303    if frame.width() <= 0.0 || frame.height() <= 0.0 {
304        return path;
305    }
306    let outer = radius.min(frame.width() / 2.0).min(frame.height() / 2.0).max(0.0);
307    let corner = outer.max(width);
308    let mut rectangle = |x: f32, y: f32, w: f32, h: f32| {
309        path.extend([
310            PathElement::MoveTo(Point::new(x, y)),
311            PathElement::LineTo(Point::new(x + w, y)),
312            PathElement::LineTo(Point::new(x + w, y + h)),
313            PathElement::LineTo(Point::new(x, y + h)),
314            PathElement::Close,
315        ]);
316    };
317    // Where the dashes of a side of this length start, and how far apart.
318    let dashes = |length: f32| {
319        let length = length.max(0.0);
320        let count = if length > 0.0 { ((length - gap) / (dash + gap)).round().max(1.0) } else { 0.0 };
321        let dash = dash.min(length);
322        let space = ((length - count * dash) / (count + 1.0)).max(0.0);
323        (0..count as u32).map(move |i| space + (dash + space) * i as f32).map(move |from| (from, dash))
324    };
325    let across = frame.width() - corner * 2.0;
326    for (from, long) in dashes(across) {
327        let x = frame.x() + corner + from;
328        rectangle(x, frame.y(), long, width);
329        rectangle(x, frame.max_y() - width, long, width);
330    }
331    let down = frame.height() - corner * 2.0;
332    for (from, long) in dashes(down) {
333        let y = frame.y() + corner + from;
334        rectangle(frame.x(), y, width, long);
335        rectangle(frame.max_x() - width, y, width, long);
336    }
337    // A quarter of a circle as a cubic curve.
338    const K: f32 = 0.552_284_8;
339    let centers = [
340        (frame.x() + corner, frame.y() + corner, -1.0, -1.0),
341        (frame.max_x() - corner, frame.y() + corner, 1.0, -1.0),
342        (frame.max_x() - corner, frame.max_y() - corner, 1.0, 1.0),
343        (frame.x() + corner, frame.max_y() - corner, -1.0, 1.0),
344    ];
345    for (cx, cy, sx, sy) in centers {
346        let at = |dx: f32, dy: f32| Point::new(cx + sx * dx, cy + sy * dy);
347        path.push(PathElement::MoveTo(at(corner, 0.0)));
348        path.push(PathElement::CubicTo(at(corner, corner * K), at(corner * K, corner), at(0.0, corner)));
349        let rest = corner - width;
350        path.push(PathElement::LineTo(at(0.0, rest)));
351        path.push(PathElement::CubicTo(at(rest * K, rest), at(rest, rest * K), at(rest, 0.0)));
352        path.push(PathElement::Close);
353    }
354    path
355}

357/// Paints a spinner of `size` at the center of the element's frame. Being only a paint, it
358/// can sit behind content that must keep its size, as in a loading button.
359#[doc(hidden)]
360pub fn spinner_paint(size: f32, color: Color) -> Paint {
361    Paint::Live(Rc::new(move |f: Rect, time, scene| {
362        let c = f.center();
363        let dot = (size * 0.18).max(2.0);
364        let r = size / 2.0 - dot / 2.0;
365        // One turn every 0.8 s; the brightness moves continuously around the ring.
366        let head = (time / 0.8).fract() as f32 * 8.0;
367        for i in 0..8 {
368            let a = i as f32 / 8.0 * std::f32::consts::TAU;
369            let age = (i as f32 - head).rem_euclid(8.0);
370            let alpha = 1.0 - age / 9.0;
371            let rect = Rect::new(c.x + r * a.cos() - dot / 2.0, c.y + r * a.sin() - dot / 2.0, dot, dot);
372            scene.push_quad(Quad::new(rect, color.with_alpha(color.a * alpha)).radius(dot / 2.0));
373        }
374    }))
375}

377/// `#RRGGBB`, or `#RRGGBBAA` when it is not opaque.
378#[doc(hidden)]
379pub fn to_hex(color: Color) -> String {
380    let byte = |channel: f32| (channel.clamp(0.0, 1.0) * 255.0).round() as u8;
381    let rgb = format!("#{:02X}{:02X}{:02X}", byte(color.r), byte(color.g), byte(color.b));
382    if byte(color.a) == 255 { rgb } else { format!("{rgb}{:02X}", byte(color.a)) }
383}

385/// The outline of a line through `points`: a stroke is a fill of its expanded outline.
386#[doc(hidden)]
387pub fn stroke_outline(points: &[Point], width: f32) -> Vec<PathElement> {
388    let half = width / 2.0;
389    let normal = |i: usize| {
390        let (a, b) = (points[i.saturating_sub(1)], points[(i + 1).min(points.len() - 1)]);
391        let (dx, dy) = (b.x - a.x, b.y - a.y);
392        let length = dx.hypot(dy).max(f32::EPSILON);
393        (-dy / length * half, dx / length * half)
394    };
395    let mut outline = Vec::with_capacity(points.len() * 2 + 2);
396    for (i, p) in points.iter().enumerate() {
397        let (nx, ny) = normal(i);
398        let at = Point::new(p.x + nx, p.y + ny);
399        outline.push(if i == 0 { PathElement::MoveTo(at) } else { PathElement::LineTo(at) });
400    }
401    for (i, p) in points.iter().enumerate().rev() {
402        let (nx, ny) = normal(i);
403        outline.push(PathElement::LineTo(Point::new(p.x - nx, p.y - ny)));
404    }
405    outline.push(PathElement::Close);
406    outline
407}

415    fn points(path: &[PathElement]) -> Vec<Point> {
416        path.iter()
417            .flat_map(|element| match *element {
418                PathElement::MoveTo(p) | PathElement::LineTo(p) => vec![p],
419                PathElement::CubicTo(a, b, c) => vec![a, b, c],
420                PathElement::Close => vec![],
421            })
422            .collect()
423    }

425    #[test]
426    fn dashes_stay_inside_the_frame_at_any_size() {
427        for (w, h) in [(400.0, 200.0), (30.0, 30.0), (9.0, 200.0), (1.0, 1.0), (0.5, 40.0)] {
428            let frame = Rect::new(10.0, 20.0, w, h);
429            let path = dashed_outline(frame, 8.0, 1.0, 6.0, 4.0);
430            assert!(!path.is_empty());
431            for p in points(&path) {
432                let inside = p.x >= frame.x() - 2.0
433                    && p.x <= frame.max_x() + 2.0
434                    && p.y >= frame.y() - 2.0
435                    && p.y <= frame.max_y() + 2.0;
436                assert!(inside, "{p:?} out of {frame:?}");
437            }
438        }
