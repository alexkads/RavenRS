1//! [`Summary`]: an aggregate of a column, with its words in [`Labels`].
2
3use crate::labels::Labels;
4
5/// An aggregate shown in group rows and in the footer.
6#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
7#[non_exhaustive]
8#[cfg_attr(feature = "live", ravenrs_live::expose(part = "variants"))]
9pub enum Summary {
10    /// Number of non-empty cells.
11    Count,
12    /// Sum of the numeric values.
13    Sum,
14    /// Mean of the numeric values.
15    Average,
16    /// Smallest numeric value.
17    Min,
18    /// Largest numeric value.
19    Max,
20}
21
22#[cfg_attr(feature = "live", ravenrs_live::expose)]
23impl Summary {
24    /// Short caption shown before the aggregate ("Sum", "Avg"…).
25    pub fn label(self) -> &'static str {
26        match self {
27            Summary::Count => "Count",
28            Summary::Sum => "Sum",
29            Summary::Average => "Avg",
30            Summary::Min => "Min",
31            Summary::Max => "Max",
32        }
33    }
34}
35
36#[cfg_attr(feature = "live", ravenrs_live::expose(part = "part3"))]
37impl Labels {
38    /// What a grid's footer calls this summary.
39    pub fn summary(&self, summary: Summary) -> &str {
40        let index = match summary {
41            Summary::Count => 0,
42            Summary::Sum => 1,
43            Summary::Average => 2,
44            Summary::Min => 3,
45            _ => 4,
46        };
47        &self.grid_summaries[index]
48    }
49}
