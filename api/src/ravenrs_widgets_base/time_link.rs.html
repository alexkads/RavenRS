1//! [`TimeLink`]: the clock and a frame, turned into an effect.
2
3use std::fmt;
4use std::rc::Rc;
5
6use ravenrs_core::{Effect, Rect};
7
8/// Turns the clock (seconds) and an element's frame into an [`Effect`], worked out
9/// on every painted frame while the element shows. See [`BuildCx::time_linked`](crate::BuildCx::time_linked).
10#[derive(Clone)]
11#[cfg_attr(feature = "live", ravenrs_live::expose(part = "fields"))]
12pub struct TimeLink(Rc<dyn Fn(f64, Rect) -> Effect>);
13
14#[cfg_attr(feature = "live", ravenrs_live::expose)]
15impl TimeLink {
16    /// The effect is `f(seconds, frame)`.
17    pub fn new(f: impl Fn(f64, Rect) -> Effect + 'static) -> Self {
18        Self(Rc::new(f))
19    }
20
21    /// Floats up and down by `distance` pixels every `period` seconds.
22    pub fn float(distance: f32, period: f32) -> Self {
23        Self::new(move |t, _| {
24            let phase = (t as f32 / period.max(0.01)) * std::f32::consts::TAU;
25            Effect::moved(0.0, phase.sin() * distance)
26        })
27    }
28
29    /// The effect at `seconds` for an element at `frame`.
30    pub fn effect_at(&self, seconds: f64, frame: Rect) -> Effect {
31        (self.0)(seconds, frame)
32    }
33}
34
35impl fmt::Debug for TimeLink {
36    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
37        f.write_str("TimeLink")
38    }
39}

41#[cfg(test)]
42mod tests {
43    use super::*;
44
45    #[test]
46    fn a_float_returns_to_its_start_each_period() {
47        let link = TimeLink::float(6.0, 2.0);
48        let frame = Rect::new(0.0, 0.0, 10.0, 10.0);
49        assert!(link.effect_at(0.0, frame).offset.y.abs() < 1e-5);
50        assert!((link.effect_at(0.5, frame).offset.y - 6.0).abs() < 1e-4);
51        assert!(link.effect_at(2.0, frame).offset.y.abs() < 1e-4);
52    }
53}
