1//! [`Motion`], how a node animates, as the build records it. The runtime that plays it
2//! lives in `motion`.
3
4use ravenrs_animation::Transition;
5use ravenrs_core::Effect;
6
7use crate::ids::{NodeId, WidgetId};
8
9/// How a node animates. Attach it with [`BuildCx::motion`](crate::BuildCx::motion).
10///
11/// ```
12/// use ravenrs_animation::Transition;
13/// use ravenrs_core::Effect;
14/// use ravenrs_widgets::Motion;
15///
16/// // Colors ease on hover; the node pops in and fades out.
17/// let motion = Motion::new()
18///     .paint(Transition::fast())
19///     .enter(Effect::hidden().with_scale(0.96), Transition::snappy())
20///     .exit(Effect::hidden(), Transition::fast());
21/// ```
22#[derive(Clone, Copy, Debug)]
23#[cfg_attr(feature = "live", ravenrs_live::expose(part = "fields"))]
24pub struct Motion {
25    #[doc(hidden)]
26    pub paint: Option<Transition>,
27    #[doc(hidden)]
28    pub layout: Option<Transition>,
29    #[doc(hidden)]
30    pub shape_only: bool,
31    #[doc(hidden)]
32    pub underlay: bool,
33    #[doc(hidden)]
34    pub shared: Option<(WidgetId, u64)>,
35    #[doc(hidden)]
36    pub surface_exit: bool,
37    #[doc(hidden)]
38    pub effect: (Effect, Transition),
39    #[doc(hidden)]
40    pub enter: Option<(Effect, Transition)>,
41    #[doc(hidden)]
42    pub exit: Option<(Effect, Transition)>,
43    #[doc(hidden)]
44    pub key: Option<u64>,
45    #[doc(hidden)]
46    pub anchored: bool,
47}
48
49impl Default for Motion {
50    fn default() -> Self {
51        Self::new()
52    }
53}
54
55#[cfg_attr(feature = "live", ravenrs_live::expose)]
56impl Motion {
57    /// Nothing animates yet.
58    pub const fn new() -> Self {
59        Self {
60            paint: None,
61            layout: None,
62            shape_only: false,
63            underlay: false,
64            shared: None,
65            surface_exit: false,
66            effect: (Effect::IDENTITY, Transition::Instant),
67            enter: None,
68            exit: None,
69            key: None,
70            anchored: false,
71        }
72    }
73
74    /// Eases this node's own fill, border, shadow, corner radius and text color
75    /// whenever a build changes them, like CSS `transition`. Colors travel
76    /// through OKLab. Gradients and `Paint::Custom` change at once.
77    pub const fn paint(mut self, transition: Transition) -> Self {
78        self.paint = Some(transition);
79        self
80    }
81
82    /// Glides the node from its old frame to its new one when layout moves it.
83    /// Its children travel with it. Input keeps using the final frame.
84    pub const fn layout(mut self, transition: Transition) -> Self {
85        self.layout = Some(transition);
86        self
87    }
88
89    /// Makes this node the same visual object as the node of a previous build
90    /// with the same `scope` and `tag`, wherever it is now: a selection marker
91    /// that slides from the old tab to the new one. `scope` is usually the
92    /// widget id of the common parent.
93    pub const fn shared_layout(mut self, scope: WidgetId, tag: u64, transition: Transition) -> Self {
94        self.shared = Some((scope, tag));
95        self.layout = Some(transition);
96        self
97    }
98
99    /// The exit fades this node as one surface: its content goes first, over the still
100    /// solid panel, and then the panel. Opacity applies to each shape on its own, so
101    /// fading all at once would show the panel through its fields, in a gray like metal.
102    /// The panel must be the first thing the node paints.
103    pub const fn surface_exit(mut self) -> Self {
104        self.surface_exit = true;
105        self
106    }
107
108    /// With [`Self::layout`], moves only the node's own shape. Its children stay
109    /// where layout put them.
110    pub const fn shape_only(mut self) -> Self {
111        self.shape_only = true;
112        self
113    }
114
115    /// Paints the node's own shape behind all its siblings, so a moving marker
116    /// slides under the labels it crosses. The node must be a direct child of
117    /// the container holding those siblings.
118    pub const fn underlay(mut self) -> Self {
119        self.underlay = true;
120        self
121    }
122
123    /// Eases the node and its subtree to this effect, such as a slight shrink
124    /// while pressed. Layout and hit testing are unaffected.
125    pub const fn effect(mut self, effect: Effect, transition: Transition) -> Self {
126        self.effect = (effect, transition);
127        self
128    }
129
130    /// When the node first appears, it starts from this effect.
131    pub const fn enter(mut self, from: Effect, transition: Transition) -> Self {
132        self.enter = Some((from, transition));
133        self
134    }
135
136    /// When the node stops being built, its last picture stays on screen and
137    /// plays this effect. The picture takes no input and is hidden from
138    /// assistive technology: for the app, the node is already gone. Reopening
139    /// during the exit reverses it from where it is.
140    pub const fn exit(mut self, to: Effect, transition: Transition) -> Self {
141        self.exit = Some((to, transition));
142        self
143    }
144
145    /// The node's identity among the motions of its view. Defaults to call
146    /// order, so use it for nodes built in a loop whose items come and go.
147    pub const fn key(mut self, key: u64) -> Self {
148        self.key = Some(key);
149        self
150    }
151
152    /// For the root of an overlay: scales around the point nearest its anchor,
153    /// and a vertical offset flips when the overlay opened on the other side.
154    pub const fn anchored(mut self) -> Self {
155        self.anchored = true;
156        self
157    }
158}

160/// The identity of an animated visual object.
161#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
162#[doc(hidden)]
163pub enum MotionKey {
164    Node(WidgetId, u64),
165    Shared(WidgetId, u64),
166}
