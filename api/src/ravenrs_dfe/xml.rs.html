1//! A small XML tree over `quick-xml`, keeping what canonicalization needs.
2//!
3//! Document type declarations are refused, so no entity can be defined and no external
4//! resource can be reached. Names stay as written, prefix included; only C14N resolves them.
5
6use quick_xml::Reader;
7use quick_xml::events::{BytesStart, Event};

9use crate::erro::DfeError;

11/// The deepest nesting accepted; no document of this layout comes close.
12const PROFUNDIDADE_MAXIMA: usize = 128;
13
14/// One node of an element.
15#[derive(Clone, Debug, PartialEq, Eq)]
16pub enum Node {
17    /// A child element.
18    Element(Element),
19    /// Character data, with references already resolved.
20    Text(String),
21    /// A processing instruction.
22    Pi {
23        /// Its target.
24        target: String,
25        /// Its data, without the spaces after the target.
26        data: String,
27    },
28}
29
30/// An element with its attributes as written, namespace declarations included.
31#[derive(Clone, Debug, Default, PartialEq, Eq)]
32pub struct Element {
33    /// The name, with its prefix if it has one.
34    pub name: String,
35    /// The attributes in document order, with their values resolved.
36    pub attrs: Vec<(String, String)>,
37    /// The children in document order.
38    pub children: Vec<Node>,
39}
40
41/// A whole document: what comes before the root, the root and what comes after.
42#[derive(Clone, Debug, PartialEq, Eq)]
43pub struct Document {
44    /// Processing instructions before the root.
45    pub prolog: Vec<Node>,
46    /// The root element.
47    pub root: Element,
48    /// Processing instructions after the root.
49    pub epilog: Vec<Node>,
50}
51
52impl Element {
53    /// The name without its prefix.
54    #[must_use]
55    pub fn local_name(&self) -> &str {
56        self.name.rsplit(':').next().unwrap_or(&self.name)
57    }
58
59    /// The value of an attribute, by the name as written.
60    #[must_use]
61    pub fn attr(&self, name: &str) -> Option<&str> {
62        self.attrs.iter().find(|(key, _)| key == name).map(|(_, value)| value.as_str())
63    }
64
65    /// The direct child elements.
66    pub fn elements(&self) -> impl Iterator<Item = &Element> {
67        self.children.iter().filter_map(|node| match node {
68            Node::Element(element) => Some(element),
69            _ => None,
70        })
71    }
72
73    /// The first direct child element with this local name.
74    #[must_use]
75    pub fn child(&self, local: &str) -> Option<&Element> {
76        self.elements().find(|element| element.local_name() == local)
77    }
78
79    /// The first descendant with this local name, this element included, depth first.
80    #[must_use]
81    pub fn find(&self, local: &str) -> Option<&Element> {
82        if self.local_name() == local {
83            return Some(self);
84        }
85        self.elements().find_map(|element| element.find(local))
86    }
87
88    /// All the text directly inside this element.
89    #[must_use]
90    pub fn text(&self) -> String {
91        let mut text = String::new();
92        for node in &self.children {
93            if let Node::Text(part) = node {
94                text.push_str(part);
95            }
96        }
97        text
98    }
99
100    /// The text of the first direct child with this local name.
101    #[must_use]
102    pub fn child_text(&self, local: &str) -> Option<String> {
103        self.child(local).map(Element::text)
104    }
105
106    /// The text of the first descendant with this local name.
107    #[must_use]
108    pub fn find_text(&self, local: &str) -> Option<String> {
109        self.find(local).map(Element::text)
110    }
111}

113/// Escapes text for an element's content in an XML this crate writes.
114#[must_use]
115pub fn escape_text(text: &str) -> String {
116    let mut out = String::with_capacity(text.len());
117    for caractere in text.chars() {
118        match caractere {
119            '&' => out.push_str("&amp;"),
120            '<' => out.push_str("&lt;"),
121            '>' => out.push_str("&gt;"),
122            other => out.push(other),
123        }
124    }
125    out
126}

128/// Escapes text for an attribute value in an XML this crate writes.
129#[must_use]
130pub fn escape_attr(text: &str) -> String {
131    let mut out = String::with_capacity(text.len());
132    for caractere in text.chars() {
133        match caractere {
134            '&' => out.push_str("&amp;"),
135            '<' => out.push_str("&lt;"),
136            '"' => out.push_str("&quot;"),
137            other => out.push(other),
138        }
139    }
140    out
141}

143/// Writes an element back as text, without any formatting of its own.
144#[must_use]
145pub fn to_string(element: &Element) -> String {
146    let mut out = String::new();
147    write_element(&mut out, element);
148    out
149}
150
151fn write_element(out: &mut String, element: &Element) {
152    out.push('<');
153    out.push_str(&element.name);
154    for (key, value) in &element.attrs {
155        out.push(' ');
156        out.push_str(key);
157        out.push_str("=\"");
158        out.push_str(&escape_attr(value));
159        out.push('"');
160    }
161    if element.children.is_empty() {
162        out.push_str("/>");
163        return;
164    }
165    out.push('>');
166    for node in &element.children {
167        match node {
168            Node::Element(child) => write_element(out, child),
169            Node::Text(text) => out.push_str(&escape_text(text)),
170            Node::Pi { target, data } => {
171                out.push_str("<?");
172                out.push_str(target);
173                if !data.is_empty() {
174                    out.push(' ');
175                    out.push_str(data);
176                }
177                out.push_str("?>");
178            }
179        }
180    }
181    out.push_str("</");
182    out.push_str(&element.name);
183    out.push('>');
184}

186/// Reads a whole document.
187pub fn parse_document(xml: &str) -> Result<Document, DfeError> {
188    let xml = xml.strip_prefix('\u{feff}').unwrap_or(xml);
189    let mut reader = Reader::from_str(xml);
190    let config = reader.config_mut();
191    config.trim_text(false);
192    config.check_end_names = true;
193    config.expand_empty_elements = false;
194
195    let mut stack: Vec<Element> = Vec::new();
196    let mut prolog = Vec::new();
197    let mut epilog = Vec::new();
198    let mut root = None;
199
200    loop {
201        let event = reader.read_event().map_err(|erro| xml_error(&erro.to_string()))?;
202        match event {
203            Event::Decl(decl) => {
204                if let Some(Ok(encoding)) = decl.encoding()
205                    && !encoding.eq_ignore_ascii_case(b"utf-8")
206                {
207                    return Err(xml_error("só UTF-8 é aceito"));
208                }
209            }
210            Event::DocType(_) => return Err(xml_error("declaração de tipo de documento recusada")),
211            Event::Start(start) => {
212                if stack.len() >= PROFUNDIDADE_MAXIMA {
213                    return Err(xml_error("profundidade máxima excedida"));
214                }
215                stack.push(start_element(&start)?);
216            }
217            Event::Empty(start) => {
218                let element = start_element(&start)?;
219                attach(element, &mut stack, &mut root)?;
220            }
221            Event::End(_) => {
222                let element = stack.pop().ok_or_else(|| xml_error("fechamento sem abertura"))?;
223                attach(element, &mut stack, &mut root)?;
224            }
225            Event::Text(text) => {
226                let decoded = text.decode().map_err(|erro| xml_error(&erro.to_string()))?;
227                push_text(&mut stack, &normalize_lines(&decoded), root.is_some())?;
228            }
229            Event::CData(data) => {
230                let decoded = data.decode().map_err(|erro| xml_error(&erro.to_string()))?;
231                push_text(&mut stack, &normalize_lines(&decoded), root.is_some())?;
232            }
233            Event::GeneralRef(reference) => {
234                let resolved = if reference.is_char_ref() {
235                    let caractere = reference
236                        .resolve_char_ref()
237                        .map_err(|erro| xml_error(&erro.to_string()))?
238                        .ok_or_else(|| xml_error("referência de caractere inválida"))?;
239                    caractere.to_string()
240                } else {
241                    let name = reference.decode().map_err(|erro| xml_error(&erro.to_string()))?;
242                    predefined(&name)?.to_string()
243                };
244                push_nonblank(&mut stack, &resolved)?;
245            }
246            Event::PI(pi) => {
247                let target = String::from_utf8_lossy(pi.target()).into_owned();
248                let data = String::from_utf8_lossy(pi.content()).trim_start().to_string();
249                let node = Node::Pi { target, data };
250                if let Some(parent) = stack.last_mut() {
251                    parent.children.push(node);
252                } else if root.is_some() {
253                    epilog.push(node);
254                } else {
255                    prolog.push(node);
256                }
257            }
258            Event::Comment(_) => {}
259            Event::Eof => break,
260        }
261    }
262    if !stack.is_empty() {
263        return Err(xml_error("elemento sem fechamento"));
264    }
265    let root = root.ok_or_else(|| xml_error("documento sem elemento raiz"))?;
266    Ok(Document { prolog, root, epilog })
267}

269/// Reads a document and keeps only its root element.
270pub fn parse(xml: &str) -> Result<Element, DfeError> {
271    parse_document(xml).map(|document| document.root)
272}

274fn xml_error(motivo: &str) -> DfeError {
275    DfeError::Xml(motivo.to_string())
276}
277
278fn attach(element: Element, stack: &mut [Element], root: &mut Option<Element>) -> Result<(), DfeError> {
279    if let Some(parent) = stack.last_mut() {
280        parent.children.push(Node::Element(element));
281        Ok(())
282    } else if root.is_none() {
283        *root = Some(element);
284        Ok(())
285    } else {
286        Err(xml_error("mais de um elemento raiz"))
287    }
288}

290/// Adds text from the document; outside the root it may only be blank.
291fn push_text(stack: &mut [Element], text: &str, _after_root: bool) -> Result<(), DfeError> {
292    if stack.is_empty() {
293        if text.chars().all(|c| matches!(c, ' ' | '\t' | '\n' | '\r')) {
294            return Ok(());
295        }
296        return Err(xml_error("texto fora do elemento raiz"));
297    }
298    push_nonblank(stack, text)
299}
300
301fn push_nonblank(stack: &mut [Element], text: &str) -> Result<(), DfeError> {
302    let Some(parent) = stack.last_mut() else {
303        return Err(xml_error("referência fora do elemento raiz"));
304    };
305    if let Some(Node::Text(last)) = parent.children.last_mut() {
306        last.push_str(text);
307    } else if !text.is_empty() {
308        parent.children.push(Node::Text(text.to_string()));
309    }
310    Ok(())
311}

313fn start_element(start: &BytesStart<'_>) -> Result<Element, DfeError> {
314    let name =
315        std::str::from_utf8(start.name().as_ref()).map_err(|erro| xml_error(&erro.to_string()))?.to_string();
316    let mut attrs = Vec::new();
317    for attribute in start.attributes().with_checks(true) {
318        let attribute = attribute.map_err(|erro| xml_error(&erro.to_string()))?;
319        let key = std::str::from_utf8(attribute.key.as_ref())
320            .map_err(|erro| xml_error(&erro.to_string()))?
321            .to_string();
322        let raw = std::str::from_utf8(&attribute.value).map_err(|erro| xml_error(&erro.to_string()))?;
323        attrs.push((key, attr_value(raw)?));
324    }
325    Ok(Element { name, attrs, children: Vec::new() })
326}

328/// Normalizes literal line ends to `\n`, the first step of every XML parser.
329fn normalize_lines(text: &str) -> String {
330    text.replace("\r\n", "\n").replace('\r', "\n")
331}
332
333/// Resolves an attribute value: literal white space becomes a space, references do not.
334fn attr_value(raw: &str) -> Result<String, DfeError> {
335    let normalized = normalize_lines(raw);
336    let mut out = String::with_capacity(normalized.len());
337    let mut rest = normalized.as_str();
338    while let Some(position) = rest.find(['&', '\t', '\n', '<']) {
339        out.push_str(&rest[..position]);
340        let tail = &rest[position..];
341        if tail.starts_with('<') {
342            return Err(xml_error("'<' em valor de atributo"));
343        }
344        if !tail.starts_with('&') {
345            out.push(' ');
346            rest = &tail[1..];
347            continue;
348        }
349        let end = tail.find(';').ok_or_else(|| xml_error("referência sem fim em atributo"))?;
350        let name = &tail[1..end];
351        if let Some(code) = name.strip_prefix('#') {
352            let value = match code.strip_prefix('x') {
353                Some(hex) => u32::from_str_radix(hex, 16),
354                None => code.parse(),
355            }
356            .map_err(|_| xml_error("referência de caractere inválida"))?;
357            let caractere = char::from_u32(value).ok_or_else(|| xml_error("caractere inválido"))?;
358            out.push(caractere);
359        } else {
360            out.push(predefined(name)?);
361        }
362        rest = &tail[end + 1..];
363    }
364    out.push_str(rest);
365    Ok(out)
366}
367
368fn predefined(name: &str) -> Result<char, DfeError> {
369    match name {
370        "amp" => Ok('&'),
371        "lt" => Ok('<'),
372        "gt" => Ok('>'),
373        "quot" => Ok('"'),
374        "apos" => Ok('\''),
375        _ => Err(xml_error("entidade desconhecida")),
376    }
377}

385        let root = parse("<a x=\"1\"><b>texto &amp; mais</b><c/></a>").expect("xml");
