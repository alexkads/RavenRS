ShortcutHintBar

Line of key hints under what they act on.

Example

The component itself, running in WebAssembly, beside its code in a playground. Try it.

Code

use ravenrs::prelude::*;

fn shortcut_hint_bar() -> impl View {
    ShortcutHintBar::new()
        .leading("3 of 40")
        .hint(["←", "→"], "move")
        .hint(["1", "5"], "rate")
        .hint(["P"], "pick")
}
