Components / TextField

TextField

Single-line text input with selection and clipboard.

Example

This is the component itself, running in WebAssembly. Try it.

Code

use ravenrs::prelude::*;

fn text_field() -> impl View {
    let name = State::new(String::new());
    let shown = name.clone();
    VStack::new((
        TextField::new(name.binding()).placeholder("Full name").icon(IconName::User),
        Text::dynamic(move || format!("Hello, {}", shown.get())).foreground(Semantic::InkMuted),
    ))
    .spacing(12.0)
    .align(Align::Stretch)
}
